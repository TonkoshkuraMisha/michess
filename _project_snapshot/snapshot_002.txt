async def _resolve_user(db: AsyncSession, username: str) -> User | None:
    champ_meta = get_champion_meta(username)
    if champ_meta is not None:
        champ_user = await ensure_champion_seeded(db, champ_meta["id"])
        if champ_user is not None:
            return champ_user

    result = await db.execute(select(User).where(User.username == username))
    user = result.scalars().first()
    if user:
        return user

    result = await db.execute(select(User).where(User.username.ilike(f"%{username}%")))
    return result.scalars().first()


async def _build_active_game_payload(db: AsyncSession, game_id: int, user_id: int) -> dict | None:
    """Формирует полный слепок активной партии для игрока (цвет, часы, FEN, PGN, версия стейта)."""
    from src.services.game_state import state_manager

    state = await state_manager.get_game_state(game_id)
    if not state or state.get("status") != "in_progress":
        return None

    white_id = int(state["white_id"])
    black_id = int(state["black_id"])
    if int(user_id) not in (white_id, black_id):
        return None

    white_user = await db.get(User, white_id)
    black_user = await db.get(User, black_id)

    now_ms = int(time.time() * 1000)
    elapsed = max(0, now_ms - int(state.get("last_move_at", now_ms)))
    turn = state.get("turn", "white")
    w_ms = int(state.get("white_time_ms", 180000))
    b_ms = int(state.get("black_time_ms", 180000))

    white_time = max(0, w_ms - elapsed) if turn in ("white", "w") else w_ms
    black_time = max(0, b_ms - elapsed) if turn in ("black", "b") else b_ms

    return {
        "active": True,
        "event": "match_found",
        "game_id": int(game_id),
        "color": "white" if int(user_id) == white_id else "black",
        "opponent_id": black_id if int(user_id) == white_id else white_id,
        "white_username": white_user.username if white_user else "Белые",
        "white_rating": white_user.rating if white_user else 1200,
        "black_username": black_user.username if black_user else "Черные",
        "black_rating": black_user.rating if black_user else 1200,
        "base_time_ms": int(state.get("base_time_ms", 180000)),
        "increment_ms": int(state.get("increment_ms", 0)),
        "white_time_ms": white_time,
        "black_time_ms": black_time,
        "fen": state.get("board_fen"),
        "pgn": state.get("pgn", ""),
        "state_version": int(state.get("state_version", 0)),
    }

@router.post("/matchmaking/join")
async def join_matchmaking_http(
        req: QueueJoinRequest,
        current_user: User = Depends(get_current_user),
        db: AsyncSession = Depends(get_db),
):
    """Публикует вызов в лобби по HTTP (или сразу запускает матч, если соперник уже ждёт)."""
    from src.services.matchmaker import MatchmakingError, matchmaker

    try:
        matched_gid = await matchmaker.join_queue(
            user_id=current_user.id,
            username=current_user.username,
            rating=current_user.rating,
            base_time_ms=req.base_time_ms,
            increment_ms=req.increment_ms,
            min_rating=req.min_rating,
            max_rating=req.max_rating,
            db=db,
        )
    except MatchmakingError as err:
        # Фронтенд дублирует действие по WS и HTTP: если WS уже стартовал партию, отдаём её.
        payload = await _active_game_payload_or_none(db, current_user.id) if err.code == "self_in_game" else None
        if payload:
            return payload
        raise HTTPException(status_code=409, detail=err.message)
    if matched_gid is not None:
        payload = await _build_active_game_payload(db, int(matched_gid), current_user.id)
        if payload:
            return payload
    return {"active": False, "queued": True}

@router.post("/matchmaking/accept/{target_user_id}")
async def accept_challenge_http(
        target_user_id: int,
        current_user: User = Depends(get_current_user),
        db: AsyncSession = Depends(get_db),
):
    """
    Принимает вызов соперника из лобби по HTTP и возвращает данные начавшейся партии.
    Безопасен к гонкам с WebSocket-сообщением accept_challenge.
    """
    from src.services.matchmaker import MatchmakingError, matchmaker
    from src.services.connection_manager import manager
    from src.services.game_state import state_manager

    if int(target_user_id) == current_user.id:
        raise HTTPException(status_code=400, detail="Нельзя принять собственный вызов")

    try:
        game_id = await matchmaker.accept_challenge(
            challenger_id=int(target_user_id),
            acceptor_id=current_user.id,
            connection_manager=manager,
            state_manager=state_manager,
            db=db,
        )
    except MatchmakingError as err:
        # Фронтенд дублирует действие по WS и HTTP: проигравший гонку запрос отдаёт уже
        # начавшуюся партию игрока, а не ошибку.
        if err.code in ("self_in_game", "challenge_gone", "players_busy"):
            payload = await _active_game_payload_or_none(db, current_user.id)
            if payload:
                return payload
        status_code = 404 if err.code == "challenge_gone" else 409
        raise HTTPException(status_code=status_code, detail=err.message)

    payload = await _build_active_game_payload(db, int(game_id), current_user.id)
    if not payload:
        raise HTTPException(status_code=404, detail="Не удалось инициализировать состояние партии")

    return payload


@router.get("/games/live")
async def get_live_games(db: AsyncSession = Depends(get_db)):
    """Возвращает список активных партий, идущих прямо сейчас (для вкладки Live и режима зрителя)."""
    from src.services.game_state import state_manager
    from src.services.connection_manager import manager

    game_ids = await state_manager.redis.smembers("active_games")
    if not game_ids:
        return []

    active_states: list[tuple[int, dict]] = []
    user_ids: set[int] = set()

    for gid_raw in game_ids:
        try:
            gid = int(gid_raw)
        except ValueError:
            continue
        state = await state_manager.get_game_state(gid)
        if not state or state.get("status") != "in_progress":
            continue

        white_id = int(state["white_id"])
        black_id = int(state["black_id"])
        user_ids.add(white_id)
        user_ids.add(black_id)
        active_states.append((gid, state))

    if not active_states:
        return []

    users_res = await db.execute(select(User).where(User.id.in_(user_ids)))
    users_map = {u.id: u for u in users_res.scalars().all()}

    live_list = []
    for gid, state in active_states:
        white_id = int(state["white_id"])
        black_id = int(state["black_id"])
        white_user = users_map.get(white_id)
        black_user = users_map.get(black_id)
        spectators_set = await manager.get_spectator_ids(gid)

        live_list.append({
            "game_id": gid,
            "white_id": white_id,
            "black_id": black_id,
            "white_username": white_user.username if white_user else "Белые",
            "white_rating": white_user.rating if white_user else 1200,
            "black_username": black_user.username if black_user else "Черные",
            "black_rating": black_user.rating if black_user else 1200,
            "base_time_ms": int(state.get("base_time_ms", 180000)),
            "increment_ms": int(state.get("increment_ms", 0)),
            "state_version": int(state.get("state_version", 0)),
            "spectators_count": len(spectators_set),
        })

    live_list.sort(key=lambda x: x["game_id"], reverse=True)
    return live_list

@router.get("/{username}/games")
async def get_user_games(
        username: str,
        limit: int = Query(25, ge=1, le=2000),
        offset: int = Query(0, ge=0),
        sort: str = Query("desc"),
        db: AsyncSession = Depends(get_db),
):
    user = await _resolve_user(db, username)
    if not user:
        raise HTTPException(status_code=404, detail=f"User {username} not found")

    WhiteUser = aliased(User)
    BlackUser = aliased(User)

    conditions = [
        or_(Game.white_player_id == user.id, Game.black_player_id == user.id),
        Game.status == GameStatus.COMPLETED,
        Game.pgn.is_not(None),
        Game.pgn != "",
    ]

    count_query = select(func.count(Game.id)).where(*conditions)
    total_count = (await db.execute(count_query)).scalar() or 0

    null_date_flag = case(
        (or_(Game.pgn_date.is_(None), Game.pgn_date == "", Game.pgn_date.like("?%")), 1),
        else_=0,
    )

    if sort == "asc":
        order_clauses = [
            asc(null_date_flag),
            asc(Game.pgn_date),
            asc(Game.finished_at),
            asc(Game.id),
        ]
    else:
        order_clauses = [
            asc(null_date_flag),
            desc(Game.pgn_date),
            desc(Game.finished_at),
            desc(Game.id),
        ]

    games_query = (
        select(
            Game.id,
            Game.status,
            Game.is_rated,
            Game.result,
            Game.pgn_date,
            Game.site,
            Game.eco,
            Game.finished_at,
            Game.pgn,
            WhiteUser.username,
            WhiteUser.rating,
            BlackUser.username,
            BlackUser.rating,
        )
        .join(WhiteUser, Game.white_player_id == WhiteUser.id)
        .join(BlackUser, Game.black_player_id == BlackUser.id)
        .where(*conditions)
        .order_by(*order_clauses)
        .limit(limit)
        .offset(offset)
    )

    rows = (await db.execute(games_query)).all()

    items = []
    for (
            game_id,
            status_val,
            is_rated,
            result_val,
            pgn_date,
            site,
            eco,
            finished_at,
            pgn_text,
            w_name,
            w_rating,
            b_name,
            b_rating,
    ) in rows:
        if (not pgn_date or not site or not eco or not result_val or result_val == "*") and pgn_text:
            fallback_meta = extract_pgn_metadata(pgn_text, fallback_result=result_val or "*")
            result_val = result_val if (result_val and result_val != "*") else fallback_meta["result"]
            pgn_date = pgn_date or fallback_meta["pgn_date"]
            site = site or fallback_meta["site"]
            eco = eco or fallback_meta["eco"]

        items.append({
            "game_id": game_id,
            "white_username": w_name,
            "white_rating": w_rating,
            "black_username": b_name,
            "black_rating": b_rating,
            "status": status_val.value if hasattr(status_val, "value") else str(status_val),
            "is_rated": is_rated,
            "result": result_val or "*",
            "pgn_date": pgn_date,
            "site": site,
            "eco": eco,
            "finished_at": finished_at.isoformat() if finished_at else None,
        })

    return {
        "total": total_count,
        "items": items,
    }

"""
Ограничение частоты запросов к эндпоинтам, нагружающим Stockfish.

Фиксированное окно в Redis (INCR + EXPIRE). Квота считается на пользователя, если запрос
пришёл с валидным Bearer-токеном, иначе — на IP-адрес (гостевые партии с ботами и просмотр
анализа доступны без входа). Токен только декодируется, без обращения к БД.
"""

import logging
import time

import jwt
from fastapi import HTTPException, Request, status

from src.core.config import settings
from src.db import redis as redis_module

logger = logging.getLogger(__name__)


def _client_identity(request: Request) -> str:
    auth_header = request.headers.get("authorization", "")
    scheme, _, token = auth_header.partition(" ")
    if scheme.lower() == "bearer" and token:
        try:
            payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
            subject = payload.get("sub")
            if subject:
                return f"user:{subject}"
        except jwt.InvalidTokenError:
            pass
    host = request.client.host if request.client else "unknown"
    return f"ip:{host}"


def rate_limit(scope: str, limit_setting: str):
    """
    Фабрика зависимости FastAPI. limit_setting — имя поля Settings с лимитом на окно,
    читается при каждом запросе (удобно переопределять в тестах и через .env).
    """

    async def _dependency(request: Request) -> None:
        limit = int(getattr(settings, limit_setting))
        window_s = int(settings.RATE_LIMIT_WINDOW_S)
        if limit <= 0:
            return

        window_id = int(time.time()) // window_s
        key = f"ratelimit:{scope}:{_client_identity(request)}:{window_id}"
        redis_client = redis_module.redis_client
        try:
            count = await redis_client.incr(key)
            if count == 1:
                await redis_client.expire(key, window_s + 1)
        except Exception as exc:
            # Redis недоступен — не блокируем пользователей (fail-open), но фиксируем проблему.
            logger.warning("Rate limiter unavailable for %s: %s", scope, exc)
            return

        if count > limit:
            retry_after = window_s - int(time.time()) % window_s
            raise HTTPException(
                status_code=status.HTTP_429_TOO_MANY_REQUESTS,
                detail="Слишком много запросов к движку, попробуйте чуть позже",
                headers={"Retry-After": str(max(1, retry_after))},
            )

    return _dependency

class GameRef:
    game_id: int
    state: dict
    white_id: int
    black_id: int

    def opponent_of(self, user_id: int) -> int:
        return self.black_id if user_id == self.white_id else self.white_id

    async def broadcast(self, payload: dict) -> None:
        await broadcast_to_game_participants_and_spectators(self.game_id, self.white_id, self.black_id, payload)


def _required_game_id(data: dict) -> int:
    game_id = data.get("game_id")
    if not game_id:
        raise _Ignore
    return int(game_id)


async def _load_game(data: dict) -> GameRef:
    gid = _required_game_id(data)
    state = await state_manager.get_game_state(gid)
    if not state:
        raise _Ignore
    return GameRef(gid, state, int(state["white_id"]), int(state["black_id"]))


async def _participant_game(ctx: ActionContext, data: dict, *, in_progress: bool | None) -> GameRef:
    """
    Партия из data["game_id"], в которой текущий пользователь — игрок.
    in_progress=True — только идущая, False — только завершённая, None — любая.
    """
    game = await _load_game(data)
    is_active = game.state.get("status") == "in_progress"
    if in_progress is not None and is_active != in_progress:
        raise _Ignore
    if ctx.user.id not in (game.white_id, game.black_id):
        raise _Ignore
    return game


ActionHandler = Callable[[ActionContext, dict], Awaitable[None]]
ACTION_HANDLERS: dict[str, ActionHandler] = {}


def action(name: str):
    def register(handler: ActionHandler) -> ActionHandler:
        ACTION_HANDLERS[name] = handler
        return handler
    return register

@action("spectate_game")
async def _spectate_game(ctx: ActionContext, data: dict) -> None:
    game = await _load_game(data)
    if game.state.get("status") != "in_progress":
        await ctx.error("Партия уже завершена или не найдена")
        return

    await manager.add_spectator(game.game_id, ctx.user.id)
    white_user = await ctx.db.get(User, game.white_id)
    black_user = await ctx.db.get(User, game.black_id)
    white_time, black_time = _compute_live_clocks(game.state)

    await ctx.send({
        "event": "spectate_started",
        "game_id": game.game_id,
        "white_username": white_user.username if white_user else "Белые",
        "white_rating": white_user.rating if white_user else 1200,
        "black_username": black_user.username if black_user else "Черные",
        "black_rating": black_user.rating if black_user else 1200,
        "base_time_ms": int(game.state.get("base_time_ms", 180000)),
        "increment_ms": int(game.state.get("increment_ms", 0)),
        "white_time_ms": white_time,
        "black_time_ms": black_time,
        "fen": game.state.get("board_fen"),
        "pgn": game.state.get("pgn", ""),
        "state_version": int(game.state.get("state_version", 0)),
    })


@action("leave_spectate")

        await manager.remove_spectator(int(game_id), ctx.user.id)
    else:

@action("chat_message")
async def _chat_message(ctx: ActionContext, data: dict) -> None:
    raw_text = str(data.get("text", "")).strip()
    if not raw_text:
        raise _Ignore
    game = await _load_game(data)
    if ctx.user.id not in (game.white_id, game.black_id):
        # Писать в чат партии могут только игроки: зрители (и кто угодно, знающий id партии)
        # не должны спамить или подсказывать ходы.
        await ctx.error("Писать в чат партии могут только её участники", code="chat_forbidden")
        return

    await game.broadcast({
        "event": "chat_message",
        "game_id": game.game_id,
        "sender_id": ctx.user.id,
        "sender": ctx.user.username,
        "text": raw_text[:140],
        "timestamp": int(time.time() * 1000),
    })

@action("abort_game")
async def _abort_game(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    abort_res = await state_manager.abort_game(ctx.db, game.game_id)
    if abort_res.get("status") == "error":
        await ctx.error(abort_res.get("reason", "Cannot abort game"))
        return

    await game.broadcast({
        "event": "game_over",
        "game_id": game.game_id,
        "reason": "aborted",
        "result": "*",
        "white_delta": 0,
        "black_delta": 0,
        "white_new_rating": abort_res["white_new_rating"],
        "black_new_rating": abort_res["black_new_rating"],
    })


@action("resign")
async def _resign(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    game_result = "0-1" if ctx.user.id == game.white_id else "1-0"
    if not await state_manager.claim_game_end(game.game_id, game_result, "resignation"):
        return  # партию уже завершил другой обработчик (мат, флаг, ничья) — итог разослан им

    rating_updates = await state_manager.finish_game(ctx.db, game.game_id, game_result, termination="resignation")
    await game.broadcast({
        "event": "game_over",
        "game_id": game.game_id,
        "reason": "resignation",
        "winner_id": game.opponent_of(ctx.user.id),
        "result": game_result,
        **rating_updates,
    })


@action("offer_draw")
async def _offer_draw(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    await state_manager.update_game_state(game.game_id, {"draw_offer": str(ctx.user.id)})
    await manager.send_personal_message(
        {"event": "draw_offered", "game_id": game.game_id, "player_id": ctx.user.id},
        game.opponent_of(ctx.user.id),
    )


@action("accept_draw")
async def _accept_draw(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    draw_offer = str(game.state.get("draw_offer") or "")
    if not draw_offer.isdigit() or int(draw_offer) == ctx.user.id:
        await ctx.error("No active draw offer to accept")
        return

    def _draw_offer_still_valid(fresh: dict, acceptor_id: int = ctx.user.id) -> bool:
        offer = str(fresh.get("draw_offer") or "")
        return offer.isdigit() and int(offer) != acceptor_id

    game_result = "1/2-1/2"
    if not await state_manager.claim_game_end(game.game_id, game_result, "agreement", guard=_draw_offer_still_valid):
        return

    rating_updates = await state_manager.finish_game(ctx.db, game.game_id, game_result, termination="agreement")
    await game.broadcast({
        "event": "game_over",
        "game_id": game.game_id,
        "reason": "draw_agreement",
        "result": game_result,
        **rating_updates,
    })


@action("decline_draw")
async def _decline_draw(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    await state_manager.update_game_state(game.game_id, {"draw_offer": ""})
    await manager.send_personal_message(
        {"event": "draw_declined", "game_id": game.game_id, "player_id": ctx.user.id},
        game.opponent_of(ctx.user.id),
    )

def _move_payload(ctx: ActionContext, gid: int, command_id: str, res: dict, move_data: dict, game_over: bool) -> dict:
    return {
        "event": "move_made",
        "game_id": gid,
        "player_id": ctx.user.id,
        "command_id": command_id,
        "state_version": res.get("state_version", 0),
        "move": move_data["uci"],
        "san": move_data["san"],
        "fen": move_data["new_fen"],
        "pgn": move_data["new_pgn"],
        "white_time_ms": res["white_time_ms"],
        "black_time_ms": res["black_time_ms"],
        "game_over": game_over,
    }


@action("make_move")
async def _make_move(ctx: ActionContext, data: dict) -> None:
    move_str = data.get("move")
    if not move_str:
        raise _Ignore
    gid = _required_game_id(data)
    command_id = data.get("command_id") or uuid.uuid4().hex

    res = await state_manager.process_atomic_move(
        game_id=gid,
        user_id=ctx.user.id,
        command_id=command_id,
        move_str=move_str,
        expected_version=data.get("state_version"),
    )
    if res["status"] == "ignored":
        return
    if res["status"] == "error":
        await ctx.error(
            res["reason"], fen=res.get("fen"), pgn=res.get("pgn"), state_version=res.get("state_version"),
        )
        return

    game = GameRef(gid, {}, res["white_id"], res["black_id"])
    move_data = res.get("move_data")

    if move_data:
        server_elapsed_ms = res.get("elapsed_ms", 0)
        client_time_ms = data.get("time_taken_ms", server_elapsed_ms)
        await state_manager.save_move_telemetry(gid, {
            "player_id": ctx.user.id,
            "move_number": move_data.get("fullmove_number", 1),
            "notation": move_data.get("uci", move_str),
            "san": move_data.get("san", move_str),
            "fen_after": move_data.get("new_fen", ""),
            "time_taken_ms": server_elapsed_ms,
            "window_blurred_before_move": bool(data.get("window_blurred", False)),
            "is_premove": trusted_premove(data.get("is_premove", False), server_elapsed_ms),
            "network_lag_ms": max(0, server_elapsed_ms - int(client_time_ms)),
        })
        await state_manager.sync_authoritative_state_to_db(ctx.db, gid, res)

    if res["status"] == "success":
        await game.broadcast(_move_payload(ctx, gid, command_id, res, move_data, game_over=False))
        return

    # game_over: мат/пат/ничья по правилам — или флаг, обнаруженный при попытке хода.
    game_result, termination = res["result"], res["termination"]
    rating_updates = await state_manager.finish_game(
        ctx.db, gid, game_result, termination=termination,
        final_pgn=move_data["new_pgn"] if move_data else None,
    )
    if move_data:
        await game.broadcast({
            **_move_payload(ctx, gid, command_id, res, move_data, game_over=True),
            "result": game_result,
            **rating_updates,
        })
    await game.broadcast({
        "event": "game_over",
        "game_id": gid,
        "reason": termination,
        "result": game_result,
        "white_time_ms": res["white_time_ms"],
        "black_time_ms": res["black_time_ms"],
        **rating_updates,
    })

@router.websocket("/matchmaking")
async def matchmaking_endpoint(
        websocket: WebSocket,
        current_user: User = Depends(get_current_user_ws),
        db: AsyncSession = Depends(get_db)
):
    await manager.connect(websocket, current_user.id)
    ctx = ActionContext(user=current_user, db=db)
    try:
        await ctx.send({"event": "connected", "message": f"Welcome, {current_user.username}!"})
        await _send_active_game_snapshot_if_exists(db, current_user)

        while True:
            data = await websocket.receive_json()
            if not isinstance(data, dict):
                continue
            action_name = data.get("action")
            handler = ACTION_HANDLERS.get(action_name)
            if handler is None:
                continue

            try:
                await handler(ctx, data)
            except _Ignore:
                pass
            except MatchmakingError as mm_err:
                await ctx.error(mm_err.message, code=mm_err.code)
            except Exception:
                logger.exception("Error processing WS action '%s' for user %s", action_name, current_user.id)
                await ctx.error("Internal server error processing action")

    except WebSocketDisconnect:
        # Закрываем только это соединение: другие вкладки пользователя остаются онлайн.
        await manager.disconnect(current_user.id, websocket)
    except Exception:
        logger.exception("WebSocket unexpected error for user %s", current_user.id)
        await manager.disconnect(current_user.id, websocket)

from datetime import datetime, timedelta, timezone
import jwt
from passlib.context import CryptContext
from src.core.config import settings

# Password hashing context using bcrypt algorithm
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def verify_password(plain_password: str, hashed_password: str) -> bool:
    """Verifies a plain password against a hashed counterpart.

    A hash passlib cannot identify (e.g. the marker stored for bot accounts) never matches
    instead of raising and turning a login attempt into a 500.
    """
    try:
        return pwd_context.verify(plain_password, hashed_password)
    except (ValueError, TypeError):
        return False


def get_password_hash(password: str) -> str:
    """Generates a secure bcrypt hash for a plain password."""
    return pwd_context.hash(password)


def create_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
    """Generates a JSON Web Token (JWT) for user authentication."""
    to_encode = data.copy()
    if expires_delta:
        expire = datetime.now(timezone.utc) + expires_delta
    else:
        expire = datetime.now(timezone.utc) + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)

    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    return encoded_jwt

    "carlsen": {"id": "Carlsen", "canonical": "Carlsen, Magnus", "rating": 2882, "exclude": ("henrik",)},
    "ding": {"id": "Ding", "canonical": "Ding, Liren", "rating": 2816, "exclude": ("yilun",)},
}

WHITE_TAG_RE = re.compile(r'\[White\s+"([^"]+)"\]')
BLACK_TAG_RE = re.compile(r'\[Black\s+"([^"]+)"\]')
RESULT_TAG_RE = re.compile(r'\[Result\s+"([^"]+)"\]')
DATE_TAG_RE = re.compile(r'\[(?:Event)?Date\s+"([^"]+)"\]')
SITE_TAG_RE = re.compile(r'\[Site\s+"([^"]+)"\]')
ECO_TAG_RE = re.compile(r'\[ECO\s+"([^"]+)"\]')

_SEED_LOCKS: dict[str, asyncio.Lock] = {}


def extract_pgn_metadata(raw_pgn: str, fallback_result: str = "*") -> dict[str, str | None]:
    result_str = fallback_result or "*"
    pgn_date: str | None = None
    site: str | None = None
    eco: str | None = None

    if not raw_pgn:
        return {"result": result_str, "pgn_date": None, "site": None, "eco": None}

    res_match = RESULT_TAG_RE.search(raw_pgn)
    if res_match:
        result_str = res_match.group(1).strip()[:10]
    elif result_str == "*":
        tail_match = re.search(r"(1-0|0-1|1/2-1/2)\s*$", raw_pgn.strip())
        if tail_match:
            result_str = tail_match.group(1)

    for d_match in DATE_TAG_RE.finditer(raw_pgn):
        val = d_match.group(1).strip()
        if val and not val.startswith("?"):
            pgn_date = val[:20]
            break

    s_match = SITE_TAG_RE.search(raw_pgn)
    if s_match and s_match.group(1).strip() != "?":
        site = s_match.group(1).strip()[:100]

    e_match = ECO_TAG_RE.search(raw_pgn)
    if e_match and e_match.group(1).strip() != "?":
        eco = e_match.group(1).strip()[:10]

    return {
        "result": result_str,
        "pgn_date": pgn_date,
        "site": site,
        "eco": eco,
    }


def get_champion_meta(username_or_id: str) -> dict | None:
    key = (username_or_id or "").strip().lower()
    if key in CHAMPIONS_META:
        return CHAMPIONS_META[key]
    for meta in CHAMPIONS_META.values():
        if meta["canonical"].lower() == key:
            return meta
    return None


def find_champion_pgn_file(champ_id: str) -> Path | None:
    target_lower = f"{champ_id.lower()}.pgn"
    alt_lower = f"{champ_id.lower()}_all_games.pgn"

    for folder in PGNS_DIRS:
        if not folder.exists():
            continue
        for pgn_path in folder.glob("*.pgn"):
            name_low = pgn_path.name.lower()
            if name_low in (target_lower, alt_lower) and pgn_path.stat().st_size > 100:
                return pgn_path
    return None


def _download_champion_pgn_sync(champ_id: str) -> Path | None:
    if "pytest" in sys.modules or os.environ.get("PYTEST_CURRENT_TEST"):
        return None

    PRIMARY_PGNS_DIR.mkdir(parents=True, exist_ok=True)
    target_path = PRIMARY_PGNS_DIR / f"{champ_id}.pgn"
    if target_path.exists() and target_path.stat().st_size > 100:
        return target_path

    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/131.0.0.0 Safari/537.36",
        "Referer": "https://www.pgnmentor.com/files.html",
    }

    zip_url = f"{PGN_MENTOR_BASE_URL}/{champ_id}.zip"
    pgn_url = f"{PGN_MENTOR_BASE_URL}/{champ_id}.pgn"

    try:
        resp = requests.get(zip_url, headers=headers, timeout=30)
        if resp.status_code == 200 and len(resp.content) > 200:
            with zipfile.ZipFile(io.BytesIO(resp.content)) as zf:
                pgn_members = [m for m in zf.namelist() if m.lower().endswith(".pgn")]
                if pgn_members:
                    raw_bytes = zf.read(pgn_members[0])
                    text = raw_bytes.decode("utf-8", errors="ignore")
                    if "[Event " in text:
                        target_path.write_text(text, encoding="utf-8")
                        return target_path
    except Exception as exc:
        logger.warning("Не удалось скачать ZIP для %s (%s), пробуем прямой .pgn...", champ_id, exc)

    try:
        resp = requests.get(pgn_url, headers=headers, timeout=30)
        if resp.status_code == 200 and "[Event " in resp.text:
            target_path.write_text(resp.text, encoding="utf-8")
            return target_path
    except Exception as exc:
        logger.error("Ошибка загрузки PGN для %s: %s", champ_id, exc)

    return None


def _is_target_champion_name(raw_name: str, champ_key: str, meta: dict) -> bool:
    low = raw_name.lower()
    if any(ex in low for ex in meta.get("exclude", ())):
        return False
    # Надёжное сопоставление по фамилии (даже при сокращении инициалов в PGN Mentor)
    surname = meta["canonical"].split(",")[0].strip().lower()
    if surname in low or champ_key in low:
        return True
    return False


def _split_pgn_games(content: str) -> list[str]:
    cleaned = content.lstrip("\ufeff").replace("\r\n", "\n").replace("\r", "\n").strip()
    if not cleaned:
        return []
    chunks = re.split(r'(?m)^(?=\[Event\s+")', cleaned)
    return [c.strip() for c in chunks if c.strip() and "[White " in c]


async def ensure_champion_seeded(db: AsyncSession, username_or_id: str) -> User | None:
    meta = get_champion_meta(username_or_id)
    if not meta:
        return None

    champ_key = meta["id"].lower()
    lock = _SEED_LOCKS.setdefault(champ_key, asyncio.Lock())

    async with lock:
        canonical_name = meta["canonical"]
        res = await db.execute(
            select(User).where(
                or_(User.username == canonical_name, User.username == meta["id"])
            )
        )
        champ_user = res.scalars().first()

        if champ_user is not None:
            count_q = select(func.count(Game.id)).where(
                or_(Game.white_player_id == champ_user.id, Game.black_player_id == champ_user.id),
                Game.status == GameStatus.COMPLETED,
            )
            games_count = (await db.execute(count_q)).scalar() or 0
            if games_count > 0:
                return champ_user

        all_users_res = await db.execute(select(User))
        users_by_name: dict[str, User] = {u.username: u for u in all_users_res.scalars().all()}

        if champ_user is None:
            champ_user = User(
                username=canonical_name,
                hashed_password="!historical_archive_no_login!",
                rating=meta["rating"],
                is_cheater=False,
                trust_factor=100,
            )
            db.add(champ_user)
            await db.flush()
            users_by_name[canonical_name] = champ_user

        pgn_file = find_champion_pgn_file(meta["id"])
        if not pgn_file or not pgn_file.exists():
            pgn_file = await asyncio.to_thread(_download_champion_pgn_sync, meta["id"])

        if not pgn_file or not pgn_file.exists():
            await db.commit()
            return champ_user

        content = pgn_file.read_text(encoding="utf-8", errors="ignore")
        raw_games = _split_pgn_games(content)
        if not raw_games:
            await db.commit()
            return champ_user

        new_opponents: dict[str, User] = {}
        parsed_entries: list[tuple[str, str, dict, str]] = []

        for raw_pgn in raw_games:
            w_match = WHITE_TAG_RE.search(raw_pgn)
            b_match = BLACK_TAG_RE.search(raw_pgn)
            if not w_match or not b_match:
                continue

            w_raw = w_match.group(1).strip()[:50]
            b_raw = b_match.group(1).strip()[:50]
            meta_fields = extract_pgn_metadata(raw_pgn)

            w_name = canonical_name if _is_target_champion_name(w_raw, champ_key, meta) else w_raw
            b_name = canonical_name if _is_target_champion_name(b_raw, champ_key, meta) else b_raw

            if w_name != canonical_name and b_name != canonical_name:
                continue

            for opp_name in (w_name, b_name):
                if opp_name not in users_by_name and opp_name not in new_opponents:
                    opp_meta = get_champion_meta(opp_name)
                    opp_rating = opp_meta["rating"] if opp_meta else 2450
                    new_opponents[opp_name] = User(
                        username=opp_name,
                        hashed_password="!historical_archive_no_login!",
                        rating=opp_rating,
                        is_cheater=False,
                        trust_factor=100,
                    )

            parsed_entries.append((w_name, b_name, meta_fields, raw_pgn))

        if new_opponents:
            db.add_all(new_opponents.values())
            await db.flush()
            users_by_name.update(new_opponents)

        existing_pgns_res = await db.execute(
            select(Game.pgn).where(
                or_(Game.white_player_id == champ_user.id, Game.black_player_id == champ_user.id)
            )
        )
        existing_pgns = {row[0] for row in existing_pgns_res.all() if row[0]}

        games_to_insert: list[Game] = []
        for w_name, b_name, meta_fields, raw_pgn in parsed_entries:
            if raw_pgn in existing_pgns:
                continue
            existing_pgns.add(raw_pgn)

            w_user = users_by_name[w_name]
            b_user = users_by_name[b_name]
            games_to_insert.append(
                Game(
                    white_player_id=w_user.id,
                    black_player_id=b_user.id,
                    pgn=raw_pgn,
                    status=GameStatus.COMPLETED,
                    result=meta_fields["result"],
                    pgn_date=meta_fields["pgn_date"],
                    site=meta_fields["site"],
                    eco=meta_fields["eco"],
                    is_rated=True,
                )
            )

        if games_to_insert:
            db.add_all(games_to_insert)

        await db.commit()
        logger.info(
            "Импортировано %d партий для чемпиона %s из %s",
            len(games_to_insert),
            canonical_name,
            pgn_file.name,
        )
        return champ_user


async def seed_all_champions() -> None:
    print(f"=== Полный импорт архивов 21 чемпиона мира (Корень: {PROJECT_ROOT}) ===")
    async with AsyncSessionLocal() as db:
        for _, meta in CHAMPIONS_META.items():
            user = await ensure_champion_seeded(db, meta["id"])
            if user:
                count_q = select(func.count(Game.id)).where(
                    or_(Game.white_player_id == user.id, Game.black_player_id == user.id),
                    Game.status == GameStatus.COMPLETED,

                total = (await db.execute(count_q)).scalar() or 0
                print(f"[+] {meta['id']:12s} ({meta['canonical']:24s}): в базе {total:4d} партий")
    print("=== Импорт всех чемпионов успешно завершён! ===")


if __name__ == "__main__":

@asynccontextmanager
async def lifespan(app: FastAPI):
    """Lifespan context manager for handling startup and shutdown events gracefully."""
    logger.info("Starting up %s...", settings.PROJECT_NAME)
    timeout_task = asyncio.create_task(state_manager.check_timeouts_loop(manager))

    yield

    timeout_task.cancel()
    try:
        await asyncio.gather(timeout_task, return_exceptions=True)
    except asyncio.CancelledError:
        pass

    logger.info("Shutting down %s...", settings.PROJECT_NAME)
    await asyncio.to_thread(close_all_engine_pools)
    await redis_client.aclose()
