def calculate_authentic_thinking_time_ms(
        bot_id: str,
        board: chess.Board,
        uci_move: str,
        source: str,
) -> int:
    """
    Вычисляет реалистичное человеческое время на ход (в мс) для исторического мастера
    с учётом его темпа, дебютной книги, единственного ответа или тактической сложности позиции.
    """
    legal_moves = list(board.legal_moves)
    num_legal = len(legal_moves)
    tempo = BOT_TEMPO_MULTIPLIER.get(bot_id, 1.0)

    if num_legal <= 1:
        return int(random.randint(240, 380) * tempo)

    if source in ("book", "opening_book"):
        base_ms = random.randint(360, 680) + board.fullmove_number * 22
        return int(base_ms * tempo)

    move_obj = chess.Move.from_uci(uci_move)
    is_capture = board.is_capture(move_obj)
    gives_check = board.gives_check(move_obj)

    captures_available = sum(1 for m in legal_moves if board.is_capture(m))
    complexity_ms = min(num_legal * 26, 950) + min(captures_available * 95, 550)

    if board.is_check():
        complexity_ms += 250
    if is_capture or gives_check:
        complexity_ms += 280

    piece_count = len(board.piece_map())
    if piece_count <= 10:
        complexity_ms = int(complexity_ms * 0.65)

    raw_ms = (520 + complexity_ms + random.randint(-110, 180)) * tempo
    return max(320, min(int(raw_ms), 2900))


def _detect_eco_code(moves_uci: list[str]) -> str:
    """Определяет базовый код дебюта ECO по первым ходам партии."""
    prefix = " ".join(moves_uci[:6])
    if prefix.startswith("e2e4 e7e5 g1f3 d7d6"):
        return "C41"
    if prefix.startswith("e2e4 e7e5 f2f4"):
        return "C30"
    if prefix.startswith("e2e4 e7e5 f1c4"):
        return "C23"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1c4 f8c5 b2b4"):
        return "C51"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1c4"):
        return "C50"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1b5"):
        return "C60"
    if prefix.startswith("e2e4 c7c5"):
        return "B20"
    if prefix.startswith("e2e4 e7e6"):
        return "C00"
    if prefix.startswith("c2c4"):
        return "A10"
    if prefix.startswith("g1f3"):
        return "A04"
    if prefix.startswith("d2d4 d7d5 c2c4"):
        return "D06"
    if prefix.startswith("d2d4"):
        return "D00"
    if prefix.startswith("e2e4"):
        return "B00"
    return "A00"


def _replay_moves(moves_uci: list[str]) -> chess.Board:
    """Переигрывает ходы с историей (нужна для правил повторения); ValueError при нелегальном ходе."""
    board = chess.Board()
    for uci_str in moves_uci:
        try:
            move = chess.Move.from_uci(uci_str)
        except ValueError:
            raise ValueError(f"Некорректный ход в истории: {uci_str}") from None
        if move not in board.legal_moves:
            raise ValueError(f"Нелегальный ход в истории: {uci_str}")
        board.push(move)
    return board


def resolve_bot_game_outcome(board: chess.Board, payload: "BotGameSaveRequest") -> tuple[str, str]:
    """
    Итог партии с ботом определяет сервер, а не клиент:
    - позиция на доске финальная (мат, пат, повторение, ...) — результат с доски;
    - игрок сдался — только поражение игрока;
    - упал флаг — победа стороны с неупавшим флагом, но ничья, если у неё нет материала
      для мата (правило ФИДЕ 6.9). Часы партии с ботом живут на клиенте, поэтому сторону
      с упавшим флагом сервер берёт из присланного результата.
    Иначе партия не завершена, и «победу» сохранить нельзя.
    """
    outcome = terminal_outcome(board)
    if outcome is not None:
        return outcome

    if payload.termination == "resignation":
        return ("0-1" if payload.player_color == "white" else "1-0"), "resignation"

    if payload.termination == "timeout":
        if payload.result == "1-0":
            return flag_outcome(board.fen(), "black")
        if payload.result == "0-1":
            return flag_outcome(board.fen(), "white")
        raise ValueError("Для флага укажите, чья сторона победила по времени")

    raise ValueError("Партия не завершена на доске: укажите termination ('resignation' или 'timeout')")


def _build_pgn_string(
        moves_uci: list[str],
        white_name: str,
        white_elo: int,
        black_name: str,
        black_elo: int,
        result: str,
) -> tuple[str, str]:
    """Строит валидный PGN-документ с заголовками и возвращает (pgn_text, final_fen)."""
    board = chess.Board()
    pgn_game = chess.pgn.Game()
    pgn_game.headers["Event"] = "Historical Bot Match"
    pgn_game.headers["Site"] = "Michess Bot Arena"
    pgn_game.headers["Date"] = datetime.now(UTC).strftime("%Y.%m.%d")
    pgn_game.headers["Round"] = "-"
    pgn_game.headers["White"] = white_name
    pgn_game.headers["Black"] = black_name
    pgn_game.headers["Result"] = result
    pgn_game.headers["WhiteElo"] = str(white_elo)
    pgn_game.headers["BlackElo"] = str(black_elo)
    pgn_game.headers["ECO"] = _detect_eco_code(moves_uci)

    node: chess.pgn.GameNode = pgn_game
    for uci_str in moves_uci:
        move = chess.Move.from_uci(uci_str)
        if move not in board.legal_moves:
            raise ValueError(f"Нелегальный ход в истории: {uci_str}")
        board.push(move)
        node = node.add_variation(move)

    exporter = chess.pgn.StringExporter(headers=True, variations=False, comments=False)
    return pgn_game.accept(exporter).strip(), board.fen()


async def _get_or_create_bot_user(db: AsyncSession, bot_id: str) -> User:
    """Находит или создаёт запись исторического бота в таблице пользователей."""
    meta = BOT_METADATA[bot_id]
    bot_username = meta["name"]

    stmt = select(User).where(User.username == bot_username)
    res = await db.execute(stmt)
    bot_user = res.scalar_one_or_none()
    if bot_user is not None:
        if bot_user.hashed_password == BOT_PASSWORD_MARKER:
            return bot_user
        # Имя занято человеком (до запрета регистрации имён ботов) — не приписываем ему
        # партии бота, а заводим бота под служебным именем.
        bot_username = f"{bot_username} [бот]"
        res = await db.execute(select(User).where(User.username == bot_username))
        bot_user = res.scalar_one_or_none()
        if bot_user is not None:
            return bot_user

    user_cols = {c.name for c in User.__table__.columns}
    user_kwargs: dict = {
        "username": bot_username,
        "rating": meta["rating"],
    }
    if "hashed_password" in user_cols:
        user_kwargs["hashed_password"] = BOT_PASSWORD_MARKER
    elif "password_hash" in user_cols:
        user_kwargs["password_hash"] = BOT_PASSWORD_MARKER
    if "email" in user_cols:
        user_kwargs["email"] = f"{bot_id}@michess.local"
    if "is_cheater" in user_cols:
        user_kwargs["is_cheater"] = False
    if "trust_factor" in user_cols:
        user_kwargs["trust_factor"] = 100

    bot_user = User(**user_kwargs)
    db.add(bot_user)
    await db.flush()
    return bot_user


def _prepare_game_kwargs(
        white_id: int,
        black_id: int,
        pgn_text: str,
        final_fen: str,
        payload: BotGameSaveRequest,
) -> dict:
    """Формирует словарь полей модели Game в соответствии со схемой БД."""
    game_cols = {c.name: c for c in Game.__table__.columns}
    kwargs: dict = {}

    if "pgn_date" in game_cols:
        kwargs["pgn_date"] = datetime.now(UTC).strftime("%Y.%m.%d")
    if "site" in game_cols:
        kwargs["site"] = "Michess Bot Arena"
    if "eco" in game_cols:
        kwargs["eco"] = _detect_eco_code(payload.moves_uci)

    if "white_player_id" in game_cols:
        kwargs["white_player_id"] = white_id
    elif "white_id" in game_cols:
        kwargs["white_id"] = white_id

    if "black_player_id" in game_cols:
        kwargs["black_player_id"] = black_id
    elif "black_id" in game_cols:
        kwargs["black_id"] = black_id

    if "pgn" in game_cols:
        kwargs["pgn"] = pgn_text
    if "board_fen" in game_cols:
        kwargs["board_fen"] = final_fen
    elif "fen" in game_cols:
        kwargs["fen"] = final_fen
    elif "current_fen" in game_cols:
        kwargs["current_fen"] = final_fen

    if "result" in game_cols:
        kwargs["result"] = payload.result
    if "termination" in game_cols:
        kwargs["termination"] = payload.termination or "normal"
    if "is_rated" in game_cols:
        kwargs["is_rated"] = False
    if "base_time_ms" in game_cols:
        kwargs["base_time_ms"] = payload.base_time_ms
    if "increment_ms" in game_cols:
        kwargs["increment_ms"] = payload.increment_ms

    if "status" in game_cols:
        col_type = game_cols["status"].type
        enums = getattr(col_type, "enums", None)
        if enums and "COMPLETED" in enums:
            kwargs["status"] = "COMPLETED"
        elif enums and "FINISHED" in enums:
            kwargs["status"] = "FINISHED"
        else:
            kwargs["status"] = "COMPLETED"

    if "finished_at" in game_cols:
        is_tz = getattr(game_cols["finished_at"].type, "timezone", False)
        now_utc = datetime.now(UTC)
        kwargs["finished_at"] = now_utc if is_tz else now_utc.replace(tzinfo=None)

    return kwargs

@router.post(
    "/move",
    response_model=BotMoveResponse,
    dependencies=[Depends(rate_limit("bot_move", "RATE_LIMIT_BOT_MOVE"))],
)
async def get_bot_move(payload: BotMoveRequest):
    """Принимает FEN позиции и возвращает ответный ход выбранного исторического бота."""
    bot = BOT_INSTANCES.get(payload.bot_id)
    if bot is None:
        raise HTTPException(status_code=404, detail=f"Бот '{payload.bot_id}' не найден")

    try:
        board = chess.Board(payload.fen)
    except ValueError:
        raise HTTPException(status_code=400, detail="Некорректный формат FEN") from None

    if board.is_game_over():
        raise HTTPException(status_code=400, detail="Партия уже завершена")

    result = await asyncio.to_thread(bot.choose_move, payload.fen)
    thinking_time_ms = calculate_authentic_thinking_time_ms(
        bot_id=payload.bot_id,
        board=board,
        uci_move=result["uci"],
        source=result["source"],
    )

    return BotMoveResponse(
        bot_id=payload.bot_id,
        uci=result["uci"],
        san=result["san"],
        fen_after=result["fen_after"],
        source=result["source"],
        comment=result["comment"],
        thinking_time_ms=thinking_time_ms,
    )


@router.post("/save-game", response_model=BotGameSaveResponse)
async def save_bot_game(
        payload: BotGameSaveRequest,
        db: AsyncSession = Depends(get_db),
        current_user: User = Depends(get_current_user),
):
    """Сохраняет завершённую партию против исторического бота в базу данных."""
    if payload.bot_id not in BOT_METADATA:
        raise HTTPException(status_code=404, detail=f"Бот '{payload.bot_id}' не найден")

    if payload.player_color not in ("white", "black"):
        raise HTTPException(status_code=400, detail="Некорректный цвет игрока")

    if payload.result not in ("1-0", "0-1", "1/2-1/2"):
        raise HTTPException(status_code=400, detail="Некорректный результат партии")

    if not payload.moves_uci:
        raise HTTPException(status_code=400, detail="Список ходов пуст")

    try:
        final_board = _replay_moves(payload.moves_uci)
        result, termination = resolve_bot_game_outcome(final_board, payload)
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc
    payload = payload.model_copy(update={"result": result, "termination": termination})

    bot_user = await _get_or_create_bot_user(db, payload.bot_id)

    if payload.player_color == "white":
        white_user, black_user = current_user, bot_user
    else:
        white_user, black_user = bot_user, current_user

    try:
        pgn_text, final_fen = _build_pgn_string(
            moves_uci=payload.moves_uci,
            white_name=white_user.username,
            white_elo=white_user.rating,
            black_name=black_user.username,
            black_elo=black_user.rating,
            result=payload.result,
        )
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc

    game_kwargs = _prepare_game_kwargs(
        white_id=white_user.id,
        black_id=black_user.id,
        pgn_text=pgn_text,
        final_fen=final_fen,
        payload=payload,
    )

    new_game = Game(**game_kwargs)
    db.add(new_game)
    await db.commit()
    await db.refresh(new_game)

    return BotGameSaveResponse(
        game_id=new_game.id,
        status="finished",
        result=result,
    )

async def user_from_token(db: AsyncSession, token: str) -> User | None:
    """
    Пользователь по JWT или None. Токен действителен, только пока его версия совпадает с
    users.token_version: смена пароля и «выйти на всех устройствах» её увеличивают.
    """
    try:
        user_id, version = decode_user_token(token)
    except InvalidTokenError:
        return None
    user = await db.get(User, user_id)
    if user is None or user.token_version != version:
        return None
    return user


async def get_current_user(
        token: str = Depends(oauth2_scheme),
        db: AsyncSession = Depends(get_db)
) -> User:
    """
    Dependency for HTTP endpoints to authenticate users via Bearer token in headers.
    """
    user = await user_from_token(db, token)
    if user is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return user

        ticket: str = Query(..., max_length=128, description="One-time ticket from POST /auth/ws-ticket"),
        session_factory: async_sessionmaker[AsyncSession] = Depends(get_session_factory),
) -> User:
    """
    Dependency for WebSocket endpoints: authenticates by a one-time ticket.

    JWT в адресе WebSocket попадал бы в access-логи uvicorn и прокси и жил там сутки. Вместо него
    клиент получает по HTTP (токен в заголовке) одноразовый тикет на WS_TICKET_TTL_S секунд;
    GETDEL гарантирует, что тикет сработает ровно один раз.

    Пользователь читается в короткой сессии, которая закрывается сразу: зависимость с get_db
    держала бы соединение пула всё время, пока открыт WebSocket. Возвращаемый объект
    отсоединён от сессии, его поля уже загружены.
    """
    rejected = WebSocketException(code=status.WS_1008_POLICY_VIOLATION, reason="Invalid or expired ticket")
    try:
        raw = await redis_module.redis_client.getdel(f"{WS_TICKET_PREFIX}{ticket}")
    except Exception as exc:
        raise WebSocketException(code=status.WS_1011_INTERNAL_ERROR, reason="Ticket store unavailable") from exc
    if not raw:
        raise rejected

    user_id_str, _, version_str = str(raw).partition(":")
    async with session_factory() as db:
        user = await db.get(User, int(user_id_str))

    # Токены отозвали между выдачей тикета и подключением — тикет тоже недействителен.
    if user is None or str(user.token_version) != version_str:
        raise rejected
    return user

"""
Ограничение частоты запросов: эндпоинты, нагружающие Stockfish, вход и регистрация.

Фиксированное окно в Redis (INCR + EXPIRE). Квота считается на пользователя, если запрос
пришёл с валидным Bearer-токеном, иначе — на IP-адрес (гостевые партии с ботами и просмотр
анализа доступны без входа). Токен только декодируется, без обращения к БД.
"""

import logging
import time

import jwt
from fastapi import HTTPException, Request, status

from src.core.config import settings
from src.db import redis as redis_module

logger = logging.getLogger(__name__)


def _client_identity(request: Request) -> str:
    auth_header = request.headers.get("authorization", "")
    scheme, _, token = auth_header.partition(" ")
    if scheme.lower() == "bearer" and token:
        try:
            payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
            subject = payload.get("sub")
            if subject:
                return f"user:{subject}"
        except jwt.InvalidTokenError:
            pass
    return f"ip:{client_ip(request)}"


def client_ip(request: Request) -> str:
    return request.client.host if request.client else "unknown"


def _window(window_s: int) -> tuple[int, int]:
    """(номер текущего окна, секунд до его конца)."""
    now = int(time.time())
    return now // window_s, max(1, window_s - now % window_s)


def _too_many(detail: str, retry_after: int) -> HTTPException:
    return HTTPException(
        status_code=status.HTTP_429_TOO_MANY_REQUESTS,
        detail=detail,
        headers={"Retry-After": str(retry_after)},
    )


async def try_consume(scope: str, identity: str, limit: int, window_s: int) -> bool:
    """
    Засчитывает попытку; False, если в текущем окне их уже больше limit.
    limit <= 0 — без ограничения. При недоступном Redis пропускает (fail-open).
    """
    if limit <= 0:
        return True
    window_id, _ = _window(window_s)
    key = f"ratelimit:{scope}:{identity}:{window_id}"
    redis_client = redis_module.redis_client
    try:
        count = await redis_client.incr(key)
        if count == 1:
            await redis_client.expire(key, window_s + 1)
    except Exception as exc:
        # Redis недоступен — не блокируем пользователей (fail-open), но фиксируем проблему.
        logger.warning("Rate limiter unavailable for %s: %s", scope, exc)
        return True
    return count <= limit


async def enforce_rate_limit(scope: str, identity: str, limit: int, window_s: int, detail: str) -> None:
    """Как try_consume, но для HTTP: бросает 429 с Retry-After."""
    if not await try_consume(scope, identity, limit, window_s):
        raise _too_many(detail, _window(window_s)[1])


async def ensure_below_limit(scope: str, identity: str, limit: int, window_s: int, detail: str) -> None:
    """Проверяет счётчик, не увеличивая его (например, неудачных входов): 429, если лимит исчерпан."""
    if limit <= 0:
        return
    window_id, retry_after = _window(window_s)
    try:
        count = int(await redis_module.redis_client.get(f"ratelimit:{scope}:{identity}:{window_id}") or 0)
    except Exception as exc:
        logger.warning("Rate limiter unavailable for %s: %s", scope, exc)
        return
    if count >= limit:
        raise _too_many(detail, retry_after)


async def record_attempt(scope: str, identity: str, window_s: int) -> None:
    """Увеличивает счётчик без проверки (парный к ensure_below_limit)."""
    window_id, _ = _window(window_s)
    key = f"ratelimit:{scope}:{identity}:{window_id}"
    try:
        if await redis_module.redis_client.incr(key) == 1:
            await redis_module.redis_client.expire(key, window_s + 1)
    except Exception as exc:
        logger.warning("Rate limiter unavailable for %s: %s", scope, exc)


def rate_limit(scope: str, limit_setting: str):
    """
    Фабрика зависимости FastAPI для эндпоинтов движка. limit_setting — имя поля Settings
    с лимитом на окно, читается при каждом запросе (удобно переопределять в тестах и через .env).
    """

    async def _dependency(request: Request) -> None:
        await enforce_rate_limit(
            scope,
            _client_identity(request),
            int(getattr(settings, limit_setting)),
            int(settings.RATE_LIMIT_WINDOW_S),
            "Слишком много запросов к движку, попробуйте чуть позже",
        )

    return _dependency

    white_id = int(state["white_id"])
    black_id = int(state["black_id"])
    white_user = await db.get(User, white_id)
    black_user = await db.get(User, black_id)
    white_time, black_time = _compute_live_clocks(state)

    payload = {
        "event": "match_found",
        "game_id": game_id,
        "color": "white" if user.id == white_id else "black",
        "opponent_id": black_id if user.id == white_id else white_id,
        "white_username": white_user.username if white_user else "Белые",
        "white_rating": white_user.rating if white_user else 1200,
        "black_username": black_user.username if black_user else "Черные",
        "black_rating": black_user.rating if black_user else 1200,
        "base_time_ms": int(state.get("base_time_ms", 180000)),
        "increment_ms": int(state.get("increment_ms", 0)),
        "white_time_ms": white_time,
        "black_time_ms": black_time,
        "fen": state.get("board_fen"),
        "pgn": state.get("pgn", ""),
        "state_version": int(state.get("state_version", 0)),
    }
    await manager.send_personal_message(payload, user.id)


async def create_rematch_game(db: AsyncSession, prev_state: dict, connection_manager=None) -> dict:
    """
    Создаёт партию-реванш со сменой цветов (white_id <-> black_id) и тем же контролем времени.
    Проверки занятости игроков и старт партии — в matchmaker.start_pvp_game.
    """
    return await matchmaker.start_pvp_game(
        white_id=int(prev_state["black_id"]),
        black_id=int(prev_state["white_id"]),
        base_time_ms=int(prev_state.get("base_time_ms", 180000)),
        increment_ms=int(prev_state.get("increment_ms", 0)),
        connection_manager=connection_manager,
        db=db,
    )


async def _fresh_current_user(db: AsyncSession, user: User) -> User:
    """
    Объект пользователя из аутентификации загружен при подключении и не видит изменения
    рейтинга после сыгранных партий — перечитываем его в сессии текущего действия.
    """
    return await db.get(User, user.id, populate_existing=True) or user


async def broadcast_to_game_participants_and_spectators(
        game_id: int,
        white_id: int,
        black_id: int,
        payload: dict,
) -> None:
    """Отправляет событие обоим игрокам и всем зрителям, наблюдающим за партией."""
    await manager.send_personal_message(payload, white_id)
    await manager.send_personal_message(payload, black_id)
    await manager.broadcast_to_spectators(game_id, payload)


# ---------------------------------------------------------------------------------------
# Обработчики действий WebSocket. Каждое действие клиента ("action") — отдельная функция,
# зарегистрированная в ACTION_HANDLERS; общие проверки (id партии, её статус, участие
# игрока) собраны в _participant_game.
# ---------------------------------------------------------------------------------------


class _Ignore(Exception):
    """Действие молча игнорируется: некорректный ввод, чужая или неподходящая по статусу партия."""


@dataclass
class ActionContext:
    user: User
    db: AsyncSession

    async def send(self, payload: dict) -> None:
        await manager.send_personal_message(payload, self.user.id)

    async def error(self, message: str, **extra) -> None:
        await self.send({"event": "error", "message": message, **extra})


@dataclass
class GameRef:
    game_id: int
    state: dict
    white_id: int
    black_id: int

    def opponent_of(self, user_id: int) -> int:
        return self.black_id if user_id == self.white_id else self.white_id

    async def broadcast(self, payload: dict) -> None:
        await broadcast_to_game_participants_and_spectators(self.game_id, self.white_id, self.black_id, payload)


def _required_game_id(data: dict) -> int:
    game_id = data.get("game_id")
    if not game_id:
        raise _Ignore
    return int(game_id)


async def _load_game(data: dict) -> GameRef:
    gid = _required_game_id(data)
    state = await state_manager.get_game_state(gid)
    if not state:
        raise _Ignore
    return GameRef(gid, state, int(state["white_id"]), int(state["black_id"]))


async def _participant_game(ctx: ActionContext, data: dict, *, in_progress: bool | None) -> GameRef:
    """
    Партия из data["game_id"], в которой текущий пользователь — игрок.
    in_progress=True — только идущая, False — только завершённая, None — любая.
    """
    game = await _load_game(data)
    is_active = game.state.get("status") == "in_progress"
    if in_progress is not None and is_active != in_progress:
        raise _Ignore
    if ctx.user.id not in (game.white_id, game.black_id):
        raise _Ignore
    return game


ActionHandler = Callable[[ActionContext, dict], Awaitable[None]]
ACTION_HANDLERS: dict[str, ActionHandler] = {}


def action(name: str):
    def register(handler: ActionHandler) -> ActionHandler:
        ACTION_HANDLERS[name] = handler
        return handler
    return register

        "white_rating": white_user.rating if white_user else 1200,
        "black_username": black_user.username if black_user else "Черные",
        "black_rating": black_user.rating if black_user else 1200,
        "base_time_ms": int(game.state.get("base_time_ms", 180000)),
        "increment_ms": int(game.state.get("increment_ms", 0)),
        "white_time_ms": white_time,
        "black_time_ms": black_time,
        "fen": game.state.get("board_fen"),
        "pgn": game.state.get("pgn", ""),

    })

    if game_id:
        await manager.remove_spectator(int(game_id), ctx.user.id)
    else:
        await manager.remove_user_from_all_spectating(ctx.user.id)
    await ctx.send({"event": "spectate_left"})


@action("chat_message")
async def _chat_message(ctx: ActionContext, data: dict) -> None:
    raw_text = str(data.get("text", "")).strip()
    if not raw_text:
        raise _Ignore
    game = await _load_game(data)
    if ctx.user.id not in (game.white_id, game.black_id):
        # Писать в чат партии могут только игроки: зрители (и кто угодно, знающий id партии)
        # не должны спамить или подсказывать ходы.
        await ctx.error("Писать в чат партии могут только её участники", code="chat_forbidden")
        return

    await game.broadcast({
        "event": "chat_message",
        "game_id": game.game_id,
        "sender_id": ctx.user.id,
        "sender": ctx.user.username,
        "text": raw_text[:140],
        "timestamp": int(time.time() * 1000),
    })


# --- Завершение партии: отмена, сдача, ничья --------------------------------------------

@action("abort_game")
async def _abort_game(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    abort_res = await state_manager.abort_game(ctx.db, game.game_id)
    if abort_res.get("status") == "error":
        await ctx.error(abort_res.get("reason", "Cannot abort game"))
        return

    await game.broadcast({
        "event": "game_over",
        "game_id": game.game_id,
        "reason": "aborted",
        "result": "*",
        "white_delta": 0,
        "black_delta": 0,
        "white_new_rating": abort_res["white_new_rating"],
        "black_new_rating": abort_res["black_new_rating"],
    })


@action("resign")
async def _resign(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    game_result = "0-1" if ctx.user.id == game.white_id else "1-0"
    if not await state_manager.claim_game_end(game.game_id, game_result, "resignation"):
        return  # партию уже завершил другой обработчик (мат, флаг, ничья) — итог разослан им

    rating_updates = await state_manager.finish_game(ctx.db, game.game_id, game_result, termination="resignation")
    await game.broadcast({
        "event": "game_over",
        "game_id": game.game_id,
        "reason": "resignation",
        "winner_id": game.opponent_of(ctx.user.id),
        "result": game_result,
        **rating_updates,
    })


@action("offer_draw")
async def _offer_draw(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    # Одно предложение на позицию: повторное (или после отказа) — только после следующего хода.
    # Иначе предложения ничьей превращаются в способ дёргать соперника в цейтноте.
    version = str(game.state.get("state_version", 0))
    offered_key = f"draw_offered_at_{ctx.user.id}"
    if str(game.state.get(offered_key, "")) == version:
        await ctx.error("Ничью можно предложить один раз за ход", code="draw_offer_too_soon")
        return
    await state_manager.update_game_state(game.game_id, {"draw_offer": str(ctx.user.id), offered_key: version})
    await manager.send_personal_message(
        {"event": "draw_offered", "game_id": game.game_id, "player_id": ctx.user.id},
        game.opponent_of(ctx.user.id),
    )


@action("accept_draw")
async def _accept_draw(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    draw_offer = str(game.state.get("draw_offer") or "")
    if not draw_offer.isdigit() or int(draw_offer) == ctx.user.id:
        await ctx.error("No active draw offer to accept")
        return

    def _draw_offer_still_valid(fresh: dict, acceptor_id: int = ctx.user.id) -> bool:
        offer = str(fresh.get("draw_offer") or "")
        return offer.isdigit() and int(offer) != acceptor_id

    game_result = "1/2-1/2"
    if not await state_manager.claim_game_end(game.game_id, game_result, "agreement", guard=_draw_offer_still_valid):
        return

    rating_updates = await state_manager.finish_game(ctx.db, game.game_id, game_result, termination="agreement")
    await game.broadcast({
        "event": "game_over",
        "game_id": game.game_id,
        "reason": "draw_agreement",
        "result": game_result,
        **rating_updates,
    })


@action("decline_draw")
async def _decline_draw(ctx: ActionContext, data: dict) -> None:
    game = await _participant_game(ctx, data, in_progress=True)
    await state_manager.update_game_state(game.game_id, {"draw_offer": ""})
    await manager.send_personal_message(
        {"event": "draw_declined", "game_id": game.game_id, "player_id": ctx.user.id},
        game.opponent_of(ctx.user.id),
    )

def _move_payload(ctx: ActionContext, gid: int, command_id: str, res: dict, move_data: dict, game_over: bool) -> dict:
    return {
        "event": "move_made",
        "game_id": gid,
        "player_id": ctx.user.id,
        "command_id": command_id,
        "state_version": res.get("state_version", 0),
        "move": move_data["uci"],
        "san": move_data["san"],
        "fen": move_data["new_fen"],
        "pgn": move_data["new_pgn"],
        "white_time_ms": res["white_time_ms"],
        "black_time_ms": res["black_time_ms"],
        "game_over": game_over,
    }


@action("make_move")
async def _make_move(ctx: ActionContext, data: dict) -> None:
    move_str = data.get("move")
    if not move_str:
        raise _Ignore
    gid = _required_game_id(data)
    command_id = data.get("command_id") or uuid.uuid4().hex

    res = await state_manager.process_atomic_move(
        game_id=gid,
        user_id=ctx.user.id,
        command_id=command_id,
        move_str=move_str,
        expected_version=data.get("state_version"),
    )
    if res["status"] == "ignored":
        return
    if res["status"] == "error":
        await ctx.error(
            res["reason"], fen=res.get("fen"), pgn=res.get("pgn"), state_version=res.get("state_version"),
        )
        return

    game = GameRef(gid, {}, res["white_id"], res["black_id"])
    move_data: dict = res.get("move_data") or {}

    if move_data:
        server_elapsed_ms = res.get("elapsed_ms", 0)
        client_time_ms = data.get("time_taken_ms", server_elapsed_ms)
        await state_manager.save_move_telemetry(gid, {
            "player_id": ctx.user.id,
            "move_number": move_data.get("fullmove_number", 1),
            "notation": move_data.get("uci", move_str),
            "san": move_data.get("san", move_str),
            "fen_after": move_data.get("new_fen", ""),
            "time_taken_ms": server_elapsed_ms,
            "window_blurred_before_move": bool(data.get("window_blurred", False)),
            "is_premove": trusted_premove(data.get("is_premove", False), server_elapsed_ms),
            "network_lag_ms": max(0, server_elapsed_ms - int(client_time_ms)),
        })
        await state_manager.sync_authoritative_state_to_db(ctx.db, gid, res)

    if res["status"] == "success":
        await game.broadcast(_move_payload(ctx, gid, command_id, res, move_data, game_over=False))
        return

    # game_over: мат/пат/ничья по правилам — или флаг, обнаруженный при попытке хода.
    game_result, termination = res["result"], res["termination"]
    rating_updates = await state_manager.finish_game(
        ctx.db, gid, game_result, termination=termination,
        final_pgn=move_data["new_pgn"] if move_data else None,
    )
    if move_data:
        await game.broadcast({
            **_move_payload(ctx, gid, command_id, res, move_data, game_over=True),
            "result": game_result,
            **rating_updates,
        })
    await game.broadcast({
        "event": "game_over",
        "game_id": gid,
        "reason": termination,
        "result": game_result,
        "white_time_ms": res["white_time_ms"],
        "black_time_ms": res["black_time_ms"],
        **rating_updates,
    })

async def _dispatch_action(
        session_factory: async_sessionmaker[AsyncSession],
        user: User,
        action_name: str,
        handler: ActionHandler,
        data: dict,
) -> None:
    """
    Выполняет одно действие в собственной короткой сессии БД.

    Сессия закрывается сразу после действия, и соединение возвращается в пул. Одна сессия на
    всё WebSocket-соединение держала бы транзакцию открытой, пока игрок подключён: при паре
    десятков игроков в лобби пул бы кончился и все запросы к БД вставали бы по таймауту.
    """
    async with session_factory() as db:
        ctx = ActionContext(user=user, db=db)
        try:
            await handler(ctx, data)
        except _Ignore:
            pass
        except MatchmakingError as mm_err:
            await ctx.error(mm_err.message, code=mm_err.code)
        except Exception:
            logger.exception("Error processing WS action '%s' for user %s", action_name, user.id)
            await ctx.error("Internal server error processing action")


# --- Ограничение частоты сообщений ------------------------------------------------------
#
# Лимиты считаются на пользователя (все его вкладки и воркеры вместе) в Redis, по группам
# действий: (сообщений, окно в секундах). Ходы ограничены щедро — пуля с премувами не должна
# упираться в лимит, — а чат и предложения строго, чтобы их нельзя было использовать для спама.

WS_MAX_MESSAGE_CHARS = 4096
WS_RATE_LIMITS: dict[str, tuple[int, int]] = {
    "moves": (60, 10),
    "chat": (5, 10),
    "offers": (6, 60),
    "lobby": (20, 10),
    "other": (30, 10),
}
_ACTION_GROUPS = {
    "make_move": "moves",
    "chat_message": "chat",
    "offer_draw": "offers", "offer_rematch": "offers",
    "join_queue": "lobby", "leave_queue": "lobby", "accept_challenge": "lobby",
    "create_invite": "lobby", "accept_invite": "lobby",
}
# Соединение, которое продолжает слать сообщения сверх лимита, закрывается: столько
# отклонённых сообщений за WS_STRIKE_WINDOW_S честный клиент не набирает.
WS_MAX_STRIKES = 30
WS_STRIKE_WINDOW_S = 10.0
WS_CLOSE_POLICY_VIOLATION = 1008


class _StrikeCounter:
    def __init__(self):
        self._times: list[float] = []

    def add(self) -> int:
        now = time.monotonic()
        self._times = [t for t in self._times if now - t < WS_STRIKE_WINDOW_S]
        self._times.append(now)
        return len(self._times)


async def _within_rate_limit(user_id: int, action_name: str) -> bool:
    group = _ACTION_GROUPS.get(action_name, "other")
    limit, window_s = WS_RATE_LIMITS[group]
    return await try_consume(f"ws:{group}", f"user:{user_id}", limit, window_s)


async def _reject_rate_limited(user: User, action_name: str, data: dict) -> None:
    payload: dict[str, Any] = {
        "event": "error",
        "code": "rate_limited",
        "action": action_name,
        "message": "Слишком много действий подряд, подождите немного",
    }
    if action_name == "make_move" and data.get("game_id"):
        # Клиент уже показал ход на доске — возвращаем ему актуальную позицию.
        try:
            state = await state_manager.get_game_state(int(data["game_id"]))
        except (TypeError, ValueError):
            state = None
        if state:
            payload.update(
                fen=state.get("board_fen"), pgn=state.get("pgn", ""),
                state_version=int(state.get("state_version", 0)),
            )
    await manager.send_personal_message(payload, user.id)


async def _handle_message(
        session_factory: async_sessionmaker[AsyncSession], user: User, raw: str,
) -> bool:
    """Разбирает и выполняет одно сообщение. False — сообщение отклонено (засчитывается в флуд)."""
    if len(raw) > WS_MAX_MESSAGE_CHARS:
        await manager.send_personal_message(
            {"event": "error", "code": "message_too_large", "message": "Слишком длинное сообщение"}, user.id,
        )
        return False
    try:
        data = json.loads(raw)
    except ValueError:
        return False  # мусор: ответа не шлём, но засчитываем
    if not isinstance(data, dict):
        return False

    action_name = str(data.get("action") or "")
    handler = ACTION_HANDLERS.get(action_name)
    if handler is None:
        return True  # неизвестное действие (например, от более новой версии клиента) — пропускаем

    if not await _within_rate_limit(user.id, action_name):
        await _reject_rate_limited(user, action_name, data)
        return False

    await _dispatch_action(session_factory, user, action_name, handler, data)
    return True


@router.websocket("/matchmaking")
async def matchmaking_endpoint(
        websocket: WebSocket,
        current_user: User = Depends(get_current_user_ws),
        session_factory: async_sessionmaker[AsyncSession] = Depends(get_session_factory),
):
    await manager.connect(websocket, current_user.id)
    strikes = _StrikeCounter()
    try:
        await manager.send_personal_message(
            {"event": "connected", "message": f"Welcome, {current_user.username}!"}, current_user.id,
        )
        async with session_factory() as db:
            await _send_active_game_snapshot_if_exists(db, current_user)

        while True:
            raw = await websocket.receive_text()
            accepted = await _handle_message(session_factory, current_user, raw)
            if not accepted and strikes.add() >= WS_MAX_STRIKES:
                logger.warning("Closing WebSocket of user %s: message flood", current_user.id)
                await websocket.close(code=WS_CLOSE_POLICY_VIOLATION)
                break

    except WebSocketDisconnect:
        pass
    except Exception:
        logger.exception("WebSocket unexpected error for user %s", current_user.id)
    finally:
        # Закрываем только это соединение: другие вкладки пользователя остаются онлайн.
        await manager.disconnect(current_user.id, websocket)
