async def _resolve_user(db: AsyncSession, username: str) -> User | None:
    """
    Находит пользователя по точному имени или ID чемпиона.
    Если запрашивается чемпион из раздела «Легенды шахмат», автоматически засеивает его PGN-базу при первом обращении.
    """
    champ_meta = get_champion_meta(username)
    if champ_meta is not None:
        champ_user = await ensure_champion_seeded(db, champ_meta["id"])
        if champ_user is not None:
            return champ_user

    result = await db.execute(select(User).where(User.username == username))
    user = result.scalars().first()
    if user:
        return user

    result = await db.execute(select(User).where(User.username.ilike(f"%{username}%")))
    return result.scalars().first()

        db: AsyncSession = Depends(get_db),
):
    query = select(User).where(User.is_cheater == False).order_by(desc(User.rating)).limit(limit)  # noqa: E712
    result = await db.execute(query)
    top_users = result.scalars().all()

        {"rank": index + 1, "username": user.username, "rating": user.rating}
        for index, user in enumerate(top_users)
    ]


@router.get("/games/live")
async def get_live_games(db: AsyncSession = Depends(get_db)):
    """Возвращает список активных партий, идущих прямо сейчас (для вкладки Live и режима зрителя)."""
    from src.services.game_state import state_manager
    from src.services.connection_manager import manager

    game_ids = await state_manager.redis.smembers("active_games")
    live_list = []

    for gid_raw in game_ids:
        try:
            gid = int(gid_raw)
        except ValueError:
            continue
        state = await state_manager.get_game_state(gid)
        if not state or state.get("status") != "in_progress":
            continue

        white_id = int(state["white_id"])
        black_id = int(state["black_id"])
        white_user = await db.get(User, white_id)
        black_user = await db.get(User, black_id)

        spectators_set = getattr(manager, "game_spectators", {}).get(gid, set())

        live_list.append({
            "game_id": gid,
            "white_id": white_id,
            "black_id": black_id,
            "white_username": white_user.username if white_user else "Белые",
            "white_rating": white_user.rating if white_user else 1200,
            "black_username": black_user.username if black_user else "Черные",
            "black_rating": black_user.rating if black_user else 1200,
            "base_time_ms": int(state.get("base_time_ms", 180000)),
            "increment_ms": int(state.get("increment_ms", 0)),
            "state_version": int(state.get("state_version", 0)),
            "spectators_count": len(spectators_set),
        })

    live_list.sort(key=lambda x: x["game_id"], reverse=True)
    return live_list

@router.get("/{username}/games")
async def get_user_games(
        username: str,
        limit: int = Query(25, ge=1, le=2000),
        offset: int = Query(0, ge=0),
        sort: str = Query("desc"),
        db: AsyncSession = Depends(get_db),
):
    user = await _resolve_user(db, username)
    if not user:
        raise HTTPException(status_code=404, detail=f"User {username} not found")

    WhiteUser = aliased(User)
    BlackUser = aliased(User)

    conditions = [
        or_(Game.white_player_id == user.id, Game.black_player_id == user.id),
        Game.status == GameStatus.COMPLETED,
        Game.pgn.is_not(None),
        Game.pgn != "",
    ]

    count_query = select(func.count(Game.id)).where(*conditions)
    total_count = (await db.execute(count_query)).scalar() or 0

    null_date_flag = case(
        (or_(Game.pgn_date.is_(None), Game.pgn_date == "", Game.pgn_date.like("?%")), 1),
        else_=0,
    )

    if sort == "asc":
        order_clauses = [
            asc(null_date_flag),
            asc(Game.pgn_date),
            asc(Game.finished_at),
            asc(Game.id),
        ]
    else:
        order_clauses = [
            asc(null_date_flag),
            desc(Game.pgn_date),
            desc(Game.finished_at),
            desc(Game.id),
        ]

    games_query = (
        select(
            Game.id,
            Game.status,
            Game.is_rated,
            Game.result,
            Game.pgn_date,
            Game.site,
            Game.eco,
            Game.finished_at,
            Game.pgn,
            WhiteUser.username,
            WhiteUser.rating,
            BlackUser.username,
            BlackUser.rating,
        )
        .join(WhiteUser, Game.white_player_id == WhiteUser.id)
        .join(BlackUser, Game.black_player_id == BlackUser.id)
        .where(*conditions)
        .order_by(*order_clauses)
        .limit(limit)
        .offset(offset)
    )

    rows = (await db.execute(games_query)).all()

    items = []
    for (
            game_id,
            status_val,
            is_rated,
            result_val,
            pgn_date,
            site,
            eco,
            finished_at,
            pgn_text,
            w_name,
            w_rating,
            b_name,
            b_rating,
    ) in rows:
        if (not pgn_date or not site or not eco or not result_val or result_val == "*") and pgn_text:
            fallback_meta = extract_pgn_metadata(pgn_text, fallback_result=result_val or "*")
            result_val = result_val if (result_val and result_val != "*") else fallback_meta["result"]
            pgn_date = pgn_date or fallback_meta["pgn_date"]
            site = site or fallback_meta["site"]
            eco = eco or fallback_meta["eco"]

        items.append({
            "game_id": game_id,
            "white_username": w_name,
            "white_rating": w_rating,
            "black_username": b_name,
            "black_rating": b_rating,
            "status": status_val.value if hasattr(status_val, "value") else str(status_val),
            "is_rated": is_rated,
            "result": result_val or "*",
            "pgn_date": pgn_date,
            "site": site,
            "eco": eco,
            "finished_at": finished_at.isoformat() if finished_at else None,
        })

    return {
        "total": total_count,
        "items": items,
    }

def _compute_live_clocks(state: dict) -> tuple[int, int]:
    now_ms = int(time.time() * 1000)
    elapsed = max(0, now_ms - int(state.get("last_move_at", now_ms)))
    turn = state.get("turn", "white")
    w_ms = int(state.get("white_time_ms", 180000))
    b_ms = int(state.get("black_time_ms", 180000))

    white_time = max(0, w_ms - elapsed) if turn in ("white", "w") else w_ms
    black_time = max(0, b_ms - elapsed) if turn in ("black", "b") else b_ms
    return white_time, black_time


async def _send_active_game_snapshot_if_exists(db: AsyncSession, user: User):
    """Если игрок переподключился во время идущей партии, восстанавливаем ему доску и часы."""
    game_id_str = await state_manager.redis.get(f"user:{user.id}:current_game")
    if not game_id_str:
        return

    game_id = int(game_id_str)
    state = await state_manager.get_game_state(game_id)
    if not state or state.get("status") != "in_progress":
        return

    white_id = int(state["white_id"])
    black_id = int(state["black_id"])
    white_user = await db.get(User, white_id)
    black_user = await db.get(User, black_id)
    white_time, black_time = _compute_live_clocks(state)

    payload = {
        "event": "match_found",
        "game_id": game_id,
        "color": "white" if user.id == white_id else "black",
        "opponent_id": black_id if user.id == white_id else white_id,
        "white_username": white_user.username if white_user else "Белые",
        "white_rating": white_user.rating if white_user else 1200,
        "black_username": black_user.username if black_user else "Черные",
        "black_rating": black_user.rating if black_user else 1200,
        "base_time_ms": int(state.get("base_time_ms", 180000)),
        "increment_ms": int(state.get("increment_ms", 0)),
        "white_time_ms": white_time,
        "black_time_ms": black_time,
        "fen": state.get("board_fen"),
        "pgn": state.get("pgn", ""),
        "state_version": int(state.get("state_version", 0)),
    }
    await manager.send_personal_message(payload, user.id)


async def _init_new_pvp_game_in_db_and_redis(
        db: AsyncSession,
        white_id: int,
        black_id: int,
        base_time_ms: int,
        increment_ms: int,
) -> dict:
    """Создаёт новую рейтинговую PvP-партию в PostgreSQL и инициализирует её стейт в Redis."""
    white_user = await db.get(User, white_id)
    black_user = await db.get(User, black_id)

    white_rating = white_user.rating if white_user else 1200
    black_rating = black_user.rating if black_user else 1200
    white_username = white_user.username if white_user else "Белые"
    black_username = black_user.username if black_user else "Черные"

    now_ms = int(time.time() * 1000)
    initial_fen = chess.STARTING_FEN

    new_game = Game(
        white_player_id=white_id,
        black_player_id=black_id,
        status=GameStatus.IN_PROGRESS,
        base_time_ms=base_time_ms,
        increment_ms=increment_ms,
        is_rated=True,
        state_version=0,
        turn="w",
        board_fen=initial_fen,
        white_time_ms=base_time_ms,
        black_time_ms=base_time_ms,
        last_move_at=now_ms,
        white_initial_rating=white_rating,
        black_initial_rating=black_rating,
    )
    db.add(new_game)
    await db.commit()
    await db.refresh(new_game)

    new_game_id = new_game.id

    if hasattr(state_manager, "init_game"):
        await state_manager.init_game(
            game_id=new_game_id,
            white_id=white_id,
            black_id=black_id,
            base_time_ms=base_time_ms,
            increment_ms=increment_ms,
        )
    else:
        await state_manager.redis.hset(f"game:{new_game_id}:state", mapping={
            "white_id": str(white_id),
            "black_id": str(black_id),
            "board_fen": initial_fen,
            "pgn": "",
            "turn": "white",
            "status": "in_progress",
            "state_version": "0",
            "base_time_ms": str(base_time_ms),
            "increment_ms": str(increment_ms),
            "white_time_ms": str(base_time_ms),
            "black_time_ms": str(base_time_ms),
            "last_move_at": str(now_ms),
            "draw_offer": "",
            "rematch_offer": "",
        })
        await state_manager.redis.sadd("active_games", str(new_game_id))
        await state_manager.redis.set(f"user:{white_id}:current_game", str(new_game_id))
        await state_manager.redis.set(f"user:{black_id}:current_game", str(new_game_id))

    return {
        "game_id": new_game_id,
        "white_id": white_id,
        "black_id": black_id,
        "white_username": white_username,
        "white_rating": white_rating,
        "black_username": black_username,
        "black_rating": black_rating,
        "base_time_ms": base_time_ms,
        "increment_ms": increment_ms,
        "fen": initial_fen,
    }


async def create_rematch_game(db: AsyncSession, prev_state: dict) -> dict:
    """
    Создаёт новую партию-реванш в БД и Redis с переменой цвета фигур (white_id <-> black_id)

    return await _init_new_pvp_game_in_db_and_redis(

        black_id=int(prev_state["white_id"]),
        base_time_ms=int(prev_state.get("base_time_ms", 180000)),
        increment_ms=int(prev_state.get("increment_ms", 0)),
    )

@router.websocket("/matchmaking")
async def matchmaking_endpoint(
        websocket: WebSocket,
        current_user: User = Depends(get_current_user_ws),
        db: AsyncSession = Depends(get_db)
):
    await manager.connect(websocket, current_user.id)
    try:
        await manager.send_personal_message(
            {"event": "connected", "message": f"Welcome, {current_user.username}!"},
            current_user.id
        )

        await _send_active_game_snapshot_if_exists(db, current_user)

        while True:
            data = await websocket.receive_json()
            action = data.get("action")

            try:
                if action == "join_queue":
                    base_time = data.get("base_time_ms", 180000)
                    inc_time = data.get("increment_ms", 0)

                    default_min = max(0, current_user.rating - 150)
                    default_max = current_user.rating + 150

                    min_rating = data.get("min_rating", default_min)
                    max_rating = data.get("max_rating", default_max)

                    from src.services.matchmaker import matchmaker
                    await matchmaker.join_queue(
                        current_user.id, current_user.username, current_user.rating,
                        base_time, inc_time, min_rating, max_rating
                    )
                    await manager.send_personal_message({"event": "queue_joined"}, current_user.id)

                elif action == "leave_queue":
                    from src.services.matchmaker import matchmaker
                    await matchmaker.leave_queue(current_user.id)
                    await manager.send_personal_message({"event": "queue_left"}, current_user.id)

                elif action == "accept_challenge":
                    target_user_id = data.get("target_user_id")
                    if target_user_id and target_user_id != current_user.id:
                        from src.services.matchmaker import matchmaker
                        await matchmaker.accept_challenge(target_user_id, current_user.id, manager, state_manager)

                elif action == "create_invite":
                    base_time = int(data.get("base_time_ms", 180000))
                    inc_time = int(data.get("increment_ms", 2000))
                    color_pref = str(data.get("color_pref", "random"))

                    from src.services.matchmaker import matchmaker
                    invite_data = await matchmaker.create_private_invite(
                        user_id=current_user.id,
                        username=current_user.username,
                        rating=current_user.rating,
                        base_time_ms=base_time,
                        increment_ms=inc_time,
                        color_pref=color_pref,
                    )
                    await manager.send_personal_message(
                        {"event": "invite_created", **invite_data},
                        current_user.id,
                    )

                elif action == "accept_invite":
                    code = str(data.get("code", "")).strip()
                    if not code:
                        continue

                    from src.services.matchmaker import matchmaker
                    accepted = await matchmaker.accept_private_invite(code, current_user.id)
                    if not accepted:
                        await manager.send_personal_message(
                            {"event": "error", "message": "Приглашение недействительно или истекло"},
                            current_user.id,
                        )
                        continue

                    game_info = await _init_new_pvp_game_in_db_and_redis(
                        db=db,
                        white_id=accepted["white_id"],
                        black_id=accepted["black_id"],
                        base_time_ms=accepted["base_time_ms"],
                        increment_ms=accepted["increment_ms"],
                    )

                    common_payload = {
                        "event": "match_found",
                        "game_id": game_info["game_id"],
                        "white_username": game_info["white_username"],
                        "white_rating": game_info["white_rating"],
                        "black_username": game_info["black_username"],
                        "black_rating": game_info["black_rating"],
                        "base_time_ms": game_info["base_time_ms"],
                        "increment_ms": game_info["increment_ms"],
                        "white_time_ms": game_info["base_time_ms"],
                        "black_time_ms": game_info["base_time_ms"],
                        "fen": game_info["fen"],
                        "pgn": "",
                        "state_version": 0,
                    }

                    await manager.send_personal_message(
                        {**common_payload, "color": "white", "opponent_id": game_info["black_id"]},
                        game_info["white_id"],
                    )
                    await manager.send_personal_message(
                        {**common_payload, "color": "black", "opponent_id": game_info["white_id"]},
                        game_info["black_id"],
                    )

                elif action == "spectate_game":
                    game_id = data.get("game_id")
                    if not game_id:
                        continue

                    gid = int(game_id)
                    state = await state_manager.get_game_state(gid)
                    if not state or state.get("status") != "in_progress":
                        await manager.send_personal_message(
                            {"event": "error", "message": "Партия уже завершена или не найдена"},
                            current_user.id,
                        )
                        continue

                    if hasattr(manager, "remove_user_from_all_spectating"):
                        manager.remove_user_from_all_spectating(current_user.id)
                    if hasattr(manager, "add_spectator"):
                        manager.add_spectator(gid, current_user.id)

                    white_id = int(state["white_id"])
                    black_id = int(state["black_id"])
                    white_user = await db.get(User, white_id)
                    black_user = await db.get(User, black_id)
                    white_time, black_time = _compute_live_clocks(state)

                    await manager.send_personal_message({
                        "event": "spectate_started",
                        "game_id": gid,
                        "white_username": white_user.username if white_user else "Белые",
                        "white_rating": white_user.rating if white_user else 1200,
                        "black_username": black_user.username if black_user else "Черные",
                        "black_rating": black_user.rating if black_user else 1200,
                        "base_time_ms": int(state.get("base_time_ms", 180000)),
                        "increment_ms": int(state.get("increment_ms", 0)),
                        "white_time_ms": white_time,
                        "black_time_ms": black_time,
                        "fen": state.get("board_fen"),
                        "pgn": state.get("pgn", ""),
                        "state_version": int(state.get("state_version", 0)),
                    }, current_user.id)

                elif action == "leave_spectate":
                    game_id = data.get("game_id")
                    if game_id and hasattr(manager, "remove_spectator"):
                        manager.remove_spectator(int(game_id), current_user.id)
                    elif hasattr(manager, "remove_user_from_all_spectating"):
                        manager.remove_user_from_all_spectating(current_user.id)
                    await manager.send_personal_message({"event": "spectate_left"}, current_user.id)

                elif action == "chat_message":
                    game_id = data.get("game_id")
                    raw_text = str(data.get("text", "")).strip()
                    if not game_id or not raw_text:
                        continue

                    clean_text = raw_text[:140]
                    gid = int(game_id)
                    game_state = await state_manager.get_game_state(gid)
                    if not game_state:
                        continue

                    white_id = int(game_state["white_id"])
                    black_id = int(game_state["black_id"])

                    chat_payload = {
                        "event": "chat_message",
                        "game_id": gid,
                        "sender_id": current_user.id,
                        "sender": current_user.username,
                        "text": clean_text,
                        "timestamp": int(time.time() * 1000),
                    }
                    await broadcast_to_game_participants_and_spectators(gid, white_id, black_id, chat_payload)

                elif action == "abort_game":
                    game_id = data.get("game_id")
                    if not game_id:
                        continue

                    gid = int(game_id)
                    game_state = await state_manager.get_game_state(gid)
                    if not game_state or game_state.get("status") != "in_progress":
                        continue

                    white_id, black_id = int(game_state["white_id"]), int(game_state["black_id"])
                    if current_user.id not in (white_id, black_id):
                        continue

                    abort_res = await state_manager.abort_game(db, gid)
                    if abort_res.get("status") == "error":
                        await manager.send_personal_message(
                            {"event": "error", "message": abort_res.get("reason", "Cannot abort game")},
                            current_user.id
                        )
                        continue

                    game_over_payload = {
                        "event": "game_over",
                        "game_id": gid,
                        "reason": "aborted",
                        "result": "*",
                        "white_delta": 0,
                        "black_delta": 0,
                        "white_new_rating": abort_res["white_new_rating"],

import jwt

# Password hashing context using bcrypt algorithm
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def verify_password(plain_password: str, hashed_password: str) -> bool:
    """Verifies a plain password against a hashed counterpart."""
    return pwd_context.verify(plain_password, hashed_password)


def get_password_hash(password: str) -> str:
    """Generates a secure bcrypt hash for a plain password."""
    return pwd_context.hash(password)


def create_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
    """Generates a JSON Web Token (JWT) for user authentication."""
    to_encode = data.copy()
    if expires_delta:
        expire = datetime.now(timezone.utc) + expires_delta
    else:
        expire = datetime.now(timezone.utc) + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)

    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    return encoded_jwt

def extract_pgn_metadata(raw_pgn: str, fallback_result: str = "*") -> dict[str, str | None]:
    """Извлекает result, pgn_date, site и eco из PGN-текста один раз перед сохранением в БД."""
    result_str = fallback_result or "*"
    pgn_date: str | None = None
    site: str | None = None
    eco: str | None = None

    if not raw_pgn:
        return {"result": result_str, "pgn_date": None, "site": None, "eco": None}

    res_match = RESULT_TAG_RE.search(raw_pgn)
    if res_match:
        result_str = res_match.group(1).strip()[:10]
    elif result_str == "*":
        tail_match = re.search(r"(1-0|0-1|1/2-1/2)\s*$", raw_pgn.strip())
        if tail_match:
            result_str = tail_match.group(1)

    for d_match in DATE_TAG_RE.finditer(raw_pgn):
        val = d_match.group(1).strip()
        if val and not val.startswith("?"):
            pgn_date = val[:20]
            break

    s_match = SITE_TAG_RE.search(raw_pgn)
    if s_match and s_match.group(1).strip() != "?":
        site = s_match.group(1).strip()[:100]

    e_match = ECO_TAG_RE.search(raw_pgn)
    if e_match and e_match.group(1).strip() != "?":
        eco = e_match.group(1).strip()[:10]

    return {
        "result": result_str,
        "pgn_date": pgn_date,
        "site": site,
        "eco": eco,
    }

            w_name = canonical_name if _is_target_champion_name(w_raw, champ_key, meta) else w_raw
            b_name = canonical_name if _is_target_champion_name(b_raw, champ_key, meta) else b_raw

            if w_name != canonical_name and b_name != canonical_name:
                continue

            for opp_name in (w_name, b_name):
                if opp_name not in users_by_name and opp_name not in new_opponents:
                    opp_meta = get_champion_meta(opp_name)
                    opp_rating = opp_meta["rating"] if opp_meta else 2450
                    new_opponents[opp_name] = User(
                        username=opp_name,
                        hashed_password="!historical_archive_no_login!",
                        rating=opp_rating,
                        is_cheater=False,
                        trust_factor=100,
                    )

            parsed_entries.append((w_name, b_name, meta_fields, raw_pgn))

        if new_opponents:
            db.add_all(new_opponents.values())
            await db.flush()
            users_by_name.update(new_opponents)

        existing_pgns_res = await db.execute(
            select(Game.pgn).where(
                or_(Game.white_player_id == champ_user.id, Game.black_player_id == champ_user.id)
            )
        )
        existing_pgns = {row[0] for row in existing_pgns_res.all() if row[0]}

        games_to_insert: list[Game] = []
        for w_name, b_name, meta_fields, raw_pgn in parsed_entries:
            if raw_pgn in existing_pgns:
                continue
            existing_pgns.add(raw_pgn)

            w_user = users_by_name[w_name]
            b_user = users_by_name[b_name]
            games_to_insert.append(
                Game(
                    white_player_id=w_user.id,
                    black_player_id=b_user.id,
                    pgn=raw_pgn,
                    status=GameStatus.COMPLETED,
                    result=meta_fields["result"],
                    pgn_date=meta_fields["pgn_date"],
                    site=meta_fields["site"],
                    eco=meta_fields["eco"],
                    is_rated=True,
                )
            )

        if games_to_insert:
            db.add_all(games_to_insert)

        await db.commit()
        logger.info(
            "Импортировано %d партий для чемпиона %s из %s",
            len(games_to_insert),
            canonical_name,
            pgn_file.name,
        )
        return champ_user


async def seed_all_champions() -> None:
    print(f"=== Импорт архивов чемпионов мира в БД (Корень: {PROJECT_ROOT}) ===")

@asynccontextmanager
async def lifespan(app: FastAPI):
    """
    Lifespan context manager for handling startup and shutdown events gracefully.
    This is the recommended place to initialize DB connection pools, Redis instances, and background tasks.
    """
    logger.info("Starting up %s...", settings.PROJECT_NAME)

    # Запускаем фоновый мониторинг таймаутов партий
    timeout_task = asyncio.create_task(state_manager.check_timeouts_loop(manager))

    yield

    # Корректно завершаем фоновые таски и пул соединений Redis
    timeout_task.cancel()
    try:
        await asyncio.gather(timeout_task, return_exceptions=True)
    except asyncio.CancelledError:
        pass

    logger.info("Shutting down %s...", settings.PROJECT_NAME)
    await redis_client.aclose()
