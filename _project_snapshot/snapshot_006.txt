"""
Разрыв соединения: часы при переподключении, флаг и брошенная партия у отключённого игрока,
несколько вкладок одного пользователя и гонка «перезагрузка страницы».
"""

import time

from src.services.connection_manager import ConnectionManager
from src.services.game_state import ABANDON_TIMEOUT_MS, evaluate_clock_expiry, state_manager


class DummyManager:
    def __init__(self):
        self.sent: list[tuple[int, dict]] = []

    async def send_personal_message(self, message: dict, user_id: int):
        self.sent.append((user_id, message))


class DummySocket:
    async def accept(self):
        pass

    async def send_text(self, text: str):
        pass


def _state(now: int, **overrides) -> dict:
    state = {
        "white_id": 1, "black_id": 2, "turn": "white",
        "white_time_ms": 60000, "black_time_ms": 60000, "last_move_at": now,
        "white_disconnected": "false", "black_disconnected": "false",
        "white_disconnect_started_at": 0, "black_disconnect_started_at": 0,
    }
    state.update(overrides)
    return state


async def test_reconnect_during_opponent_turn_does_not_give_opponent_time():
    """Раньше last_move_at сдвигался на время офлайна, и часы соперника стояли всё это время."""
    await state_manager.initialize_game(game_id=31, white_id=10, black_id=20, base_time_ms=60000, increment_ms=0)
    now = int(time.time() * 1000)
    # Ход белых; чёрные отключились 20 секунд назад.
    await state_manager.update_game_state(31, {
        "last_move_at": now - 30000,
        "black_disconnected": "true",
        "black_disconnect_started_at": now - 20000,
    })

    await state_manager.handle_player_reconnect(20, DummyManager())

    state = await state_manager.get_game_state(31)
    assert state["black_disconnected"] == "false"
    assert state["last_move_at"] == now - 30000
    assert state["white_time_ms"] == 60000


def test_disconnected_player_flags_counting_time_before_disconnect():
    """Время между последним ходом и разрывом тоже идёт в зачёт (раньше терялось)."""
    now = int(time.time() * 1000)
    state = _state(
        now,
        white_time_ms=8000, last_move_at=now - 10000,
        white_disconnected="true", white_disconnect_started_at=now - 3000,
    )
    assert evaluate_clock_expiry(state, now) == ("0-1", 2, "timeout")


def test_abandon_timer_ignores_stopped_clock_on_opponent_turn():
    """
    Если сейчас ход соперника, часы отключённого стоят: при 5 секундах на часах он не должен
    проигрывать через 5 секунд офлайна — только по общему таймеру брошенной партии.
    """
    now = int(time.time() * 1000)
    state = _state(now, black_time_ms=5000, black_disconnected="true", black_disconnect_started_at=now - 10000)
    assert evaluate_clock_expiry(state, now) is None

    abandoned = {**state, "black_disconnect_started_at": now - ABANDON_TIMEOUT_MS}
    assert evaluate_clock_expiry(abandoned, now) == ("1-0", 1, "abandoned")

async def test_reload_race_does_not_leave_player_marked_disconnected(monkeypatch):
    """
    Перезагрузка страницы: новая вкладка подключается (и выполняет handle_player_reconnect,
    который ничего не делает), пока старая ещё обрабатывает разрыв и только потом ставит флаг.
    """
    await state_manager.initialize_game(game_id=35, white_id=10, black_id=20, base_time_ms=60000, increment_ms=0)
    cm = ConnectionManager()
    old_tab, new_tab = DummySocket(), DummySocket()
    await cm.connect(old_tab, user_id=10)

    original_disconnect_handler = state_manager.handle_player_disconnect

    async def reconnect_then_mark(user_id, manager):
        await cm.connect(new_tab, user_id=10)  # новая вкладка успела подключиться
        await original_disconnect_handler(user_id, manager)

    monkeypatch.setattr(state_manager, "handle_player_disconnect", reconnect_then_mark)
    await cm.disconnect(10, old_tab)

    state = await state_manager.get_game_state(35)
    assert state["white_disconnected"] == "false"
    assert cm._has_open_sockets(10)

@needs_stockfish
def test_pool_reuses_process_and_limits_concurrency():
    pool = EnginePool(SF_PATH, size=1, acquire_timeout_s=0.2)
    try:
        with pool.acquire() as first:
            pass
        with pool.acquire() as second:
            assert second is first  # процесс переиспользован, а не запущен заново

            errors = []

            def _try_acquire():
                try:
                    with pool.acquire():
                        pass
                except EngineBusyError as exc:
                    errors.append(exc)

            t = threading.Thread(target=_try_acquire)
            t.start()
            t.join()
            assert len(errors) == 1
    finally:
        pool.close()


@needs_stockfish
def test_pool_replaces_dead_engine():
    pool = EnginePool(SF_PATH, size=1, acquire_timeout_s=1.0)
    try:
        with pool.acquire() as engine:
            engine.quit()  # процесс завершился, а вызывающий код ошибку «проглотил»
        with pool.acquire() as fresh:
            assert fresh is not engine
            info = fresh.analyse(chess.Board(), chess.engine.Limit(depth=1))
            assert "score" in info
    finally:
        pool.close()

        state_manager.claim_game_end(game.id, "0-1", "resignation"),

async def test_finish_game_with_stale_session_does_not_double_rate(real_redis, db_session):
    """
    Сессия WS-обработчика долгоживущая (expire_on_commit=False) и может держать устаревший
    объект Game со статусом IN_PROGRESS. finish_game обязан перечитать строку из БД.
    """
    game, white, black = await _make_game(db_session)
    # Фабрика на том же тестовом движке (импортировать conftest напрямую нельзя: это создаст его копию).
    TestSessionLocal = async_sessionmaker(db_session.bind, expire_on_commit=False)

    async with TestSessionLocal() as stale_session:
        stale_game = await stale_session.get(Game, game.id)
        assert stale_game.status == GameStatus.IN_PROGRESS  # объект закэширован в identity map

        async with TestSessionLocal() as other_session:
            first = await state_manager.finish_game(other_session, game.id, "1-0", termination="resignation")
        assert first["white_delta"] > 0

        second = await state_manager.finish_game(stale_session, game.id, "1-0", termination="resignation")
        assert second["white_delta"] == first["white_delta"]

    async with TestSessionLocal() as check:
        w = await check.get(User, white.id)
        b = await check.get(User, black.id)
        assert w.rating == 1500 + first["white_delta"]
        assert b.rating == 1500 + first["black_delta"]


def test_evaluate_clock_expiry():

from src.services.game_engine import engine

STARTING_FEN = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"

def test_underpromotion_to_knight_and_rook():
    """Проверка превращения пешки в коня (a7a8n) и ладью (a7a8r) с корректной записью SAN."""
    promotion_fen = "4k3/P7/8/8/8/8/8/4K3 w - - 0 1"

    assert is_valid_n is True

    assert res_knight["san"].startswith("a8=N")

    is_valid_r, res_rook = engine.process_move(promotion_fen, "a7a8r")
    assert is_valid_r is True
    assert res_rook["uci"] == "a7a8r"
    assert res_rook["san"].startswith("a8=R")


def _play(moves: list[str]) -> list[dict]:
    """Проигрывает ходы через engine.process_move, передавая FEN и PGN, как это делает сервер."""
    fen, pgn, results = STARTING_FEN, "", []
    for mv in moves:
        is_valid, data = engine.process_move(fen, mv, pgn)
        assert is_valid, f"move {mv} rejected"
        fen, pgn = data["new_fen"], data["new_pgn"]
        results.append(data)
    return results


def test_threefold_repetition_ends_game():
    """Третье повторение позиции завершает партию ничьей (раньше не детектировалось: доска строилась из FEN)."""
    dance = ["g1f3", "g8f6", "f3g1", "f6g8"]
    results = _play(dance * 2)

    assert all(not r["is_game_over"] for r in results[:-1])
    last = results[-1]
    assert last["is_game_over"] is True
    assert last["termination"] == "repetition"
    assert last["result"] == "1/2-1/2"

def _worker(client, worker_id: str) -> ConnectionManager:
    cm = ConnectionManager()
    cm.worker_id = worker_id
    cm.redis = client
    cm.pubsub = client.pubsub()
    return cm


async def _stop(*managers: ConnectionManager):
    for cm in managers:
        for task in (cm._listener_task, cm._heartbeat_task):
            if task is not None:
                task.cancel()
        await asyncio.gather(
            *(t for t in (cm._listener_task, cm._heartbeat_task) if t is not None), return_exceptions=True,
        )
        await cm.pubsub.aclose()


async def _wait_for(predicate, timeout_s: float = 3.0):
    deadline = time.monotonic() + timeout_s
    while time.monotonic() < deadline:
        if predicate():
            return True
        await asyncio.sleep(0.02)
    return False

"""
Celery-задача анализа: каждая задача идёт в собственном asyncio.run, поэтому глобальный
пул соединений использовать нельзя — со второй задачи он ломался («different loop»).
"""

import asyncio
import json

import pytest
from sqlalchemy.ext.asyncio import async_sessionmaker, create_async_engine

from src.models.base import Base
from src.models.game import Game, GameStatus
from src.models.user import User
from src.services.bots.base import resolve_stockfish_path
from src.worker import tasks

PGN = '[Event "t"]\n[Result "1-0"]\n\n1. e4 e5 2. Qh5 Nc6 3. Bc4 Nf6 4. Qxf7# 1-0'


async def _prepare_db(url: str) -> list[int]:
    engine = create_async_engine(url)
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
    session_factory = async_sessionmaker(engine, expire_on_commit=False)
    async with session_factory() as db:
        white = User(username="worker_w", hashed_password="pw", rating=1500)
        black = User(username="worker_b", hashed_password="pw", rating=1500)
        db.add_all([white, black])
        await db.commit()
        games = [
            Game(white_player_id=white.id, black_player_id=black.id, status=GameStatus.COMPLETED,
                 base_time_ms=180000, increment_ms=0, is_rated=True, pgn=PGN)
            for _ in range(2)
        ]
        db.add_all(games)
        await db.commit()
        ids = [g.id for g in games]
    await engine.dispose()
    return ids


async def _cached_analysis(url: str, game_id: int) -> str | None:
    engine = create_async_engine(url)
    async with async_sessionmaker(engine)() as db:
        game = await db.get(Game, game_id)
        value = game.analysis_json
    await engine.dispose()
    return value


@pytest.mark.skipif(resolve_stockfish_path() is None, reason="Stockfish binary not available")
def test_consecutive_tasks_each_in_own_event_loop(tmp_path, monkeypatch):
    url = f"sqlite+aiosqlite:///{(tmp_path / 'worker.db').as_posix()}"
    game_ids = asyncio.run(_prepare_db(url))
    monkeypatch.setattr(tasks, "_task_database_url", lambda: url)

    for game_id in game_ids:  # две задачи подряд — как в живом воркере
        report = tasks.analyze_game_task(game_id)
        assert report.get("status") != "error"
        assert len(report["moves"]) == 7

    for game_id in game_ids:
        assert json.loads(asyncio.run(_cached_analysis(url, game_id)))["moves"]

def test_zukertort_style_rewards_development_and_castling():
    """Проверяем, что стилевой фильтр Цукерторта поощряет развитие лёгких фигур и рокировку."""
    board = chess.Board("r1bqk2r/pppp1ppp/2n2n2/2b1p3/2B1P3/5N2/PPPP1PPP/RNBQK2R w KQkq - 4 4")
