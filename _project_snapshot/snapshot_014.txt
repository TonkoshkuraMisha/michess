import {useState, useEffect} from 'react';
import {useParams, Link, useNavigate} from 'react-router-dom';
import {format, parseISO} from 'date-fns';
import {ru} from 'date-fns/locale';
import {
    LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer
} from 'recharts';
import {
    Trophy,
    Swords,
    Calendar,
    Download,
    Crosshair,
    Target
} from 'lucide-react';
import {API_BASE_URL} from '../config/api';

interface UserProfile {
    id: number;
    username: string;
    rating: number;
    created_at: string;
}

interface GameData {
    game_id: number;
    white_username: string;
    white_rating: number;
    black_username: string;
    black_rating: number;
    status: string;
    is_rated: boolean;
    result: string;
    pgn_date: string | null;
    site: string | null;
    eco: string | null;
    finished_at: string | null;
}

const displayDate = (game: GameData) => {
    if (game.pgn_date && !game.pgn_date.includes('????')) {
        return game.pgn_date.replace(/\??/g, '').replace(/\.$/, '');
    }
    if (game.finished_at) {
        return format(parseISO(game.finished_at), 'd MMM yyyy', {locale: ru});
    }
    return '—';
};

export default function Profile() {
    const {targetUsername} = useParams<{ targetUsername: string }>();
    const navigate = useNavigate();
    const [profile, setProfile] = useState<UserProfile | null>(null);
    const [games, setGames] = useState<GameData[]>([]);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState<string | null>(null);

    useEffect(() => {
        const fetchProfileData = async () => {
            setLoading(true);
            setError(null);
            try {
                const profileRes = await fetch(`${API_BASE_URL}/api/v1/profile/${targetUsername}`);
                if (!profileRes.ok) throw new Error('Пользователь не найден');
                const profileData = await profileRes.json();
                setProfile(profileData);

                const gamesRes = await fetch(`${API_BASE_URL}/api/v1/profile/${targetUsername}/games?limit=30`);
                if (gamesRes.ok) {
                    const gamesData = await gamesRes.json();
                    setGames(gamesData.items || []);
                }
            } catch (err: any) {
                setError(err.message);
            } finally {
                setLoading(false);
            }
        };

        if (targetUsername) {
            fetchProfileData();
        }
    }, [targetUsername]);

    if (loading) {
        return (
            <div className="flex flex-col items-center justify-center w-full h-full text-boxwood-light font-serif">
                <div className="animate-spin text-amber-500 mb-4"><Swords size={40}/></div>
                <div>Загрузка профиля...</div>
            </div>
        );
    }

    if (error || !profile) {
        return (
            <div className="bg-red-950/80 border border-red-800 p-6 rounded text-red-200 text-center font-serif">
                <h2 className="text-xl font-bold mb-2">Ошибка</h2>
                <p>{error || 'Не удалось загрузить профиль'}</p>
                <Link to="/" className="mt-4 inline-block text-amber-400 hover:underline">На главную</Link>
            </div>
        );
    }

    let wins = 0, losses = 0, draws = 0;
    let tempRating = profile.rating;

    const chartDataReversed = games.map((game) => {
        const isWhite = game.white_username === profile.username;
        let delta = 0;

        if (game.result === '1-0') {
            if (isWhite) {
                wins++;
                delta = 15;
            } else {
                losses++;
                delta = -15;
            }
        } else if (game.result === '0-1') {
            if (isWhite) {
                losses++;
                delta = -15;
            } else {
                wins++;
                delta = 15;
            }
        } else if (game.result === '1/2-1/2') {
            draws++;
        }

        const pointRating = tempRating;
        tempRating -= delta;

        const chartDate = game.pgn_date ? game.pgn_date.split('.')[0] : (game.finished_at ? format(parseISO(game.finished_at), 'd MMM', {locale: ru}) : '');

        return {
            name: 'Партия',
            rating: pointRating,
            date: chartDate
        };
    });

    const chartData = chartDataReversed.reverse();
    const winRate = games.length > 0 ? Math.round((wins / games.length) * 100) : 0;

    return (
        <div className="w-full max-w-6xl flex flex-col gap-6 font-serif">
            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
                <div
                    className="bg-acacia-dark border border-amber-900/40 p-6 rounded shadow-heavy flex flex-col items-center justify-center text-center relative overflow-hidden">
                    <img
                        src={`https://api.dicebear.com/9.x/bottts-neutral/svg?seed=${profile.username}&backgroundColor=D0A972`}
                        alt="Avatar Background"
                        className="absolute top-0 right-0 w-48 h-48 opacity-10 pointer-events-none translate-x-10 -translate-y-10"
                    />
                    <img
                        src={`https://api.dicebear.com/9.x/bottts-neutral/svg?seed=${profile.username}&backgroundColor=E6C998`}
                        alt="Avatar"
                        className="w-24 h-24 rounded-lg bg-black/40 shadow-inner border-2 border-amber-600/40 mb-3 z-10"
                    />
                    <h1 className="text-3xl font-bold text-amber-200 z-10">{profile.username}</h1>

                    <div className="flex items-center gap-2 mt-3 text-boxwood-light/80 text-sm z-10">
                        <Calendar size={16} className="text-amber-600"/>
                        На сайте с {format(parseISO(profile.created_at), 'd MMMM yyyy', {locale: ru})}
                    </div>
                </div>

                <div className="md:col-span-2 grid grid-cols-2 lg:grid-cols-4 gap-4">
                    <div
                        className="bg-[#1A120B] border border-amber-900/60 p-5 rounded shadow-inner flex flex-col justify-center">
                        <div
                            className="text-boxwood/60 text-xs uppercase tracking-widest mb-1 flex items-center gap-1.5">
                            <Trophy size={14}/> Рейтинг
                        </div>
                        <div className="text-4xl font-mono font-bold text-amber-400">{profile.rating}</div>
                    </div>
                    <div
                        className="bg-[#1A120B] border border-amber-900/60 p-5 rounded shadow-inner flex flex-col justify-center">
                        <div
                            className="text-boxwood/60 text-xs uppercase tracking-widest mb-1 flex items-center gap-1.5">
                            <Swords size={14}/> Игр сыграно
                        </div>
                        <div className="text-4xl font-mono font-bold text-blue-400">{games.length}</div>
                    </div>
                    <div
                        className="bg-[#1A120B] border border-amber-900/60 p-5 rounded shadow-inner flex flex-col justify-center">
                        <div
                            className="text-boxwood/60 text-xs uppercase tracking-widest mb-1 flex items-center gap-1.5">
                            <Target size={14}/> Винрейт
                        </div>
                        <div className="text-4xl font-mono font-bold text-emerald-400">{winRate}%</div>
                    </div>
                    <div
                        className="bg-[#1A120B] border border-amber-900/60 p-5 rounded shadow-inner flex flex-col justify-center">
                        <div

                            <span className="text-emerald-400">{wins}</span>
                            <span className="text-boxwood/40">/</span>
                            <span className="text-gray-400">{draws}</span>
                            <span className="text-boxwood/40">/</span>
                            <span className="text-red-400">{losses}</span>
                        </div>

                </div>
            </div>

            <div className="bg-acacia-dark border border-amber-900/40 p-5 rounded shadow-heavy">
                <h2 className="text-sm uppercase tracking-widest text-amber-500/80 mb-6 font-bold">Динамика рейтинга
                    (последние партии)</h2>
                <div className="h-64 w-full font-sans">
                    {chartData.length > 1 ? (
                        <ResponsiveContainer width="100%" height="100%">
                            <LineChart data={chartData}>
                                <CartesianGrid strokeDasharray="3 3" stroke="#5C3A21" vertical={false}/>
                                <XAxis dataKey="date" stroke="#8B5A2B" tick={{fill: '#D0A972', fontSize: 12}} dy={10}/>
                                <YAxis domain={['dataMin - 30', 'dataMax + 30']} stroke="#8B5A2B"
                                       tick={{fill: '#D0A972', fontSize: 12}} dx={-10}/>
                                <Tooltip contentStyle={{
                                    backgroundColor: '#1A120B',
                                    borderColor: '#7A4B31',

                                <Line type="monotone" dataKey="rating" stroke="#fbbf24" strokeWidth={3}

                            Недостаточно данных для графика

                    )}

            </div>

            <div className="bg-acacia-dark border border-amber-900/40 rounded shadow-heavy overflow-hidden">
                <div className="p-5 border-b border-amber-900/40 bg-black/20">
                    <h2 className="text-sm uppercase tracking-widest text-amber-500/80 font-bold">История партий</h2>
                </div>

                <div className="overflow-x-auto">
                    <table className="w-full text-left text-sm font-sans whitespace-nowrap">
                        <thead className="bg-black/40 text-boxwood/60 uppercase text-xs tracking-wider">
                        <tr>
                            <th className="px-4 py-4 font-medium">Белые</th>
                            <th className="px-4 py-4 font-medium">Черные</th>
                            <th className="px-4 py-4 font-medium text-center">Результат</th>
                            <th className="px-4 py-4 font-medium text-center">Дебют</th>

/**
 * Адреса API и WebSocket.
 *
 * Разработка: Vite (порт 3000) и uvicorn (порт 8000) — разные адреса, по умолчанию 127.0.0.1:8000.
 * Продакшен (сборка за nginx): фронтенд и API на одном домене, поэтому по умолчанию API —
 * относительный путь, а WebSocket — тот же хост с ws/wss по протоколу страницы.
 * VITE_API_URL / VITE_WS_URL переопределяют оба варианта.
 */

const DEV_API_URL = 'http://127.0.0.1:8000';
const DEV_WS_URL = 'ws://127.0.0.1:8000';

function sameOriginWsUrl(): string {
    if (typeof window === 'undefined' || !window.location) return DEV_WS_URL;
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    return `${protocol}//${window.location.host}`;
}

export const API_BASE_URL: string =
    import.meta.env.VITE_API_URL ?? (import.meta.env.PROD ? '' : DEV_API_URL);

export const WS_BASE_URL: string =
    import.meta.env.VITE_WS_URL || (import.meta.env.PROD ? sameOriginWsUrl() : DEV_WS_URL);

    if (typeof window !== 'undefined') {

export function getPieceSvgUrl(pieceSetKey: string, color: string, type: string): string {
    const setConfig = PIECE_SETS[pieceSetKey] || PIECE_SETS.merida;
    return `/assets/pieces/${setConfig.folder}/${color}${type.toUpperCase()}.svg`;
}

export function getHarmoniousPieceStyle(
    pieceSetKey: string,
    boardThemeKey: string,
    pieceColor: string
): { width: string; height: string; filter: string } {
    const setConfig = PIECE_SETS[pieceSetKey] || PIECE_SETS.merida;
    const theme = BOARD_THEMES[boardThemeKey] || BOARD_THEMES.walnut;
    const isWhite = pieceColor === 'w';
    const size = `${setConfig.scalePercent}%`;

    if (setConfig.id === 'cardinal') {
        return {
            width: size,
            height: size,
            filter: 'drop-shadow(0 2px 3px rgba(0, 0, 0, 0.45))'
        };
    }

    if (theme.id === 'walnut' || theme.id === 'maple') {
        return {
            width: size,
            height: size,
            filter: isWhite
                ? 'sepia(0.08) brightness(1.01) drop-shadow(0 2px 2.5px rgba(0, 0, 0, 0.55))'
                : 'drop-shadow(0 0 1px rgba(255, 248, 225, 0.55)) drop-shadow(0 2.5px 3px rgba(0, 0, 0, 0.6))'
        };
    }

    return {
        width: size,
        height: size,
        filter: isWhite
            ? 'drop-shadow(0 2px 2.5px rgba(0, 0, 0, 0.5))'
            : 'drop-shadow(0 0 1px rgba(255, 255, 255, 0.5)) drop-shadow(0 2.5px 3px rgba(0, 0, 0, 0.55))'
    };
}

// --- FILE: frontend/src/services/apiErrors.ts ---

/**
 * Текст ошибки из ответа API. detail бывает строкой (HTTPException) или списком ошибок
 * валидации Pydantic — его без разбора React показал бы как «[object Object]».
 */
export function formatErrorDetail(detail: unknown, fallback: string): string {
    if (typeof detail === 'string' && detail) return detail;
    if (Array.isArray(detail)) {
        const messages = detail
            .map((item) => (item && typeof item === 'object' ? String((item as {msg?: unknown}).msg ?? '') : ''))
            .map((msg) => msg.replace(/^Value error, /, ''))
            .filter(Boolean);
        if (messages.length) return messages.join('; ');
    }
    return fallback;
}

export async function readErrorDetail(res: Response, fallback: string): Promise<string> {
    try {
        const data = await res.json();
        return formatErrorDetail(data?.detail, fallback);
    } catch {
        return fallback;
    }
}

const RATE_LIMITED_RECONNECT_DELAY_MS = 15000;

class AuthExpiredError extends Error {}

class RateLimitedError extends Error {}

/**
 * Одноразовый тикет для WebSocket. JWT в адрес WebSocket не кладём: адреса попадают в логи
 * сервера и прокси. Тикет получаем по HTTP с токеном в заголовке, он действует ~30 с и один раз.
 */
async function fetchWsTicket(token: string): Promise<string> {
    const res = await fetch(`${API_BASE_URL}/api/v1/auth/ws-ticket`, {
        method: 'POST',
        headers: {Authorization: `Bearer ${token}`},
    });
    if (res.status === 401) throw new AuthExpiredError('Сессия истекла');
    if (res.status === 429) throw new RateLimitedError('Слишком частые переподключения');
    if (!res.ok) throw new Error(`ws-ticket: HTTP ${res.status}`);
    const data = await res.json();
    return data.ticket;
}

class GameWebSocketService {
    private ws: WebSocket | null = null;
    private currentToken: string | null = null;
    private listeners: { [event: string]: EventCallback[] } = {};
    private pendingMessages: string[] = [];
    private reconnectTimer: ReturnType<typeof setTimeout> | null = null;
    // Идёт запрос тикета; номер попытки отсекает устаревшие (после disconnect или смены токена).
    private connecting = false;
    private attempt = 0;

    connect(token: string) {
        if (!token) return;

        // Если токен сменился (выход и вход под другим игроком), переоткрываем соединение
        if (this.currentToken && this.currentToken !== token) {
            this.closeSocket();
            this.attempt += 1;
            this.connecting = false;
        }

        this.currentToken = token;

        if (this.ws && (this.ws.readyState === WebSocket.OPEN || this.ws.readyState === WebSocket.CONNECTING)) {
            return;
        }
        if (this.connecting) return;

        if (this.reconnectTimer) {
            clearTimeout(this.reconnectTimer);
            this.reconnectTimer = null;
        }

        void this.openSocket(token);
    }

    private async openSocket(token: string) {
        this.connecting = true;
        const attempt = ++this.attempt;

        let ticket: string;
        try {
            ticket = await fetchWsTicket(token);
        } catch (err) {
            if (attempt !== this.attempt) return;
            this.connecting = false;
            if (err instanceof AuthExpiredError) {
                this.handleAuthExpired();
            } else {
                console.warn('Не удалось получить тикет WebSocket:', err);
                this.scheduleReconnect(err instanceof RateLimitedError ? RATE_LIMITED_RECONNECT_DELAY_MS : RECONNECT_DELAY_MS);
            }
            return;
        }

        if (attempt !== this.attempt || this.currentToken !== token) return;  // отключились, пока ждали
        this.connecting = false;

        const wsUrl = `${WS_BASE_URL}/api/v1/ws/matchmaking?ticket=${encodeURIComponent(ticket)}`;
        this.ws = new WebSocket(wsUrl);

        this.ws.onopen = () => {
            while (this.pendingMessages.length > 0 && this.ws && this.ws.readyState === WebSocket.OPEN) {
                const msg = this.pendingMessages.shift();
                if (msg) {
                    this.ws.send(msg);
                }
            }
        };

        this.ws.onmessage = (event) => {
            try {
                const data = JSON.parse(event.data);
                const eventType = data.event;

                if (this.listeners[eventType]) {
                    this.listeners[eventType].forEach(cb => cb(data));
                }
                if (this.listeners['*']) {
                    this.listeners['*'].forEach(cb => cb(data));
                }
            } catch (e) {
                console.error('Ошибка парсинга JSON от WebSocket:', e);
            }
        };

        this.ws.onclose = (event) => {
            this.ws = null;
            if (event.code === 1008) {
                // Тикет недействителен или токены отозваны (смена пароля, выход на всех устройствах)
                this.handleAuthExpired();
                return;
            }
            if (this.currentToken && event.code !== 1000) {
                this.scheduleReconnect(RECONNECT_DELAY_MS);
            }
        };

        this.ws.onerror = (error) => {
            console.error('WebSocket ошибка:', error);
        };
    }

    private scheduleReconnect(delayMs: number) {
        if (this.reconnectTimer) clearTimeout(this.reconnectTimer);
        this.reconnectTimer = setTimeout(() => {
            this.reconnectTimer = null;
            if (this.currentToken) {
                this.connect(this.currentToken);
            }
        }, delayMs);
    }

    private handleAuthExpired() {
        // Токен истёк или отозван — очищаем его и уведомляем приложение
        this.currentToken = null;
        this.pendingMessages = [];
        localStorage.removeItem('token');
        window.dispatchEvent(new Event('michess-auth-expired'));
    }

    private closeSocket() {
        if (this.ws) {
            try {
                this.ws.close(1000);
            } catch {
                // ignore
            }
            this.ws = null;
        }
    }

    on(event: string, callback: EventCallback) {
        if (!this.listeners[event]) {
            this.listeners[event] = [];
        }
        if (!this.listeners[event].includes(callback)) {
            this.listeners[event].push(callback);
        }
    }

    off(event: string, callback: EventCallback) {
        if (!this.listeners[event]) return;
        this.listeners[event] = this.listeners[event].filter(cb => cb !== callback);
    }

    send(action: string, payload: any = {}) {
        const serialized = JSON.stringify({action, ...payload});

        if (this.ws && this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(serialized);
            return;
        }

        const fallbackToken = this.currentToken || localStorage.getItem('token');
        if (fallbackToken) {
            this.pendingMessages.push(serialized);
            this.connect(fallbackToken);
        } else {
            console.error('Невозможно отправить сообщение: пользователь не авторизован');
        }
    }

    disconnect() {
        this.currentToken = null;
        this.pendingMessages = [];
        this.attempt += 1;  // отменяет запрос тикета, если он ещё идёт
        this.connecting = false;
        if (this.reconnectTimer) {
            clearTimeout(this.reconnectTimer);
            this.reconnectTimer = null;
        }
        this.closeSocket();
    }
}

export const gameSocket = new GameWebSocketService();

const MUTE_STORAGE_KEY = 'michess_sound_muted';

class SoundEngine {
    private ctx: AudioContext | null = null;
    private muted: boolean = false;

    constructor() {
        if (typeof window !== 'undefined') {
            this.muted = localStorage.getItem(MUTE_STORAGE_KEY) === 'true';
        }
    }

    public isMuted(): boolean {
        return this.muted;
    }

    public setMuted(mute: boolean): void {
        this.muted = mute;
        if (typeof window !== 'undefined') {
            localStorage.setItem(MUTE_STORAGE_KEY, String(mute));
        }
    }

    public toggleMute(): boolean {
        this.setMuted(!this.muted);
        return this.muted;
    }

    private getContext(): AudioContext | null {
        if (typeof window === 'undefined') return null;
        if (!this.ctx) {
            const AudioCtx = window.AudioContext || (window as unknown as {
                webkitAudioContext?: typeof AudioContext
            }).webkitAudioContext;
            if (AudioCtx) {
                this.ctx = new AudioCtx();
            }
        }
        if (this.ctx && this.ctx.state === 'suspended') {
            this.ctx.resume().catch(() => {
            });
        }
        return this.ctx;
    }

    private playTone(
        freq: number,
        durationSec: number,
        delaySec: number = 0,
        type: OscillatorType = 'sine',
        peakGain: number = 0.14,
        endFreq?: number
    ): void {
        const ctx = this.getContext();
        if (!ctx) return;

        const startTime = ctx.currentTime + delaySec;
        const stopTime = startTime + durationSec;

        const osc = ctx.createOscillator();
        const gain = ctx.createGain();

        osc.type = type;
        osc.frequency.setValueAtTime(freq, startTime);
        if (endFreq !== undefined) {
            osc.frequency.exponentialRampToValueAtTime(Math.max(20, endFreq), stopTime);
        }

        gain.gain.setValueAtTime(0.0001, startTime);
        gain.gain.linearRampToValueAtTime(peakGain, startTime + Math.min(0.015, durationSec * 0.2));
        gain.gain.exponentialRampToValueAtTime(0.0001, stopTime);

        osc.connect(gain);
        gain.connect(ctx.destination);

        osc.start(startTime);
        osc.stop(stopTime + 0.01);
    }

    private playSample(file: string, volume: number = 0.75, fallbackFn?: () => void): void {
        if (typeof window === 'undefined' || typeof Audio === 'undefined') {
            fallbackFn?.();
            return;
        }
        try {
            const audio = new Audio(`/assets/sounds/${file}`);
            audio.volume = volume;
            audio.currentTime = 0;
            audio.play().catch(() => {
                fallbackFn?.();
            });
        } catch {
            fallbackFn?.();
        }
    }

    public play(event: SoundEventType): void {
        if (this.muted) return;

        switch (event) {
            case 'move':

/**

 */
export function resolveMoveSound(san?: string | null): SoundEventType {
    if (!san) return 'move';
    if (san.includes('#') || san.includes('+')) return 'check';
    if (san.includes('=')) return 'promote';
    if (san.startsWith('O-O')) return 'castle';
    if (san.includes('x')) return 'capture';
    return 'move';
}

# Развёртывание michess (продакшен)

Стек поднимается одной командой через `docker-compose.prod.yml`:

| Сервис    | Что делает                                                        | Наружу       |
|-----------|-------------------------------------------------------------------|--------------|
| `web`     | nginx: статика фронтенда, прокси `/api/` и WebSocket, CSP          | порт 80      |
| `api`     | FastAPI (uvicorn, `WEB_CONCURRENCY` процессов)                     | —            |
| `worker`  | Celery: фоновый анализ завершённых партий (Stockfish)              | —            |
| `migrate` | одноразово применяет миграции Alembic перед стартом api и worker  | —            |
| `db`      | PostgreSQL 16                                                      | —            |
| `redis`   | Redis 7.4 с AOF (состояние партий, очереди, лимиты, брокер Celery) | —            |

## 1. Настройка

    cp .env.example .env

Обязательно поменять в `.env`:

- `SECRET_KEY` — случайная строка не короче 32 символов. Со слабым ключом API в продакшене
  **не стартует** (по нему подписываются токены всех пользователей):

      python -c "import secrets; print(secrets.token_urlsafe(48))"

- `POSTGRES_PASSWORD` — свой пароль БД.

`ENVIRONMENT=production`, адреса БД и Redis задаёт сам `docker-compose.prod.yml`.

## 2. Запуск

    docker compose -f docker-compose.prod.yml up -d --build
    docker compose -f docker-compose.prod.yml ps
    curl http://localhost/health      # {"status": "healthy", "checks": {...}}

Загрузка архива партий чемпионов мира (однократно, по желанию):

    docker compose -f docker-compose.prod.yml exec api python -m src.db.seed

## 3. Обновление

    git pull
    docker compose -f docker-compose.prod.yml up -d --build

Миграции применяются автоматически сервисом `migrate` до перезапуска `api` и `worker`.

> Релиз со спринта 7 меняет формат токенов: все пользователи один раз войдут заново.

## 4. HTTPS

nginx в образе `web` слушает только HTTP. TLS завершается снаружи — балансировщиком облака
или отдельным прокси (Caddy, Traefik) перед портом `HTTP_PORT`. Прокси должен передавать
`Upgrade`/`Connection` для WebSocket (`/api/v1/ws/`). Если TLS завершается прямо в nginx,
добавьте `listen 443 ssl` в `frontend/nginx.conf` и включите `Strict-Transport-Security`
в `frontend/security_headers.inc`.

Внешний прокси должен подменять, а не дописывать `X-Forwarded-For`: лимиты попыток входа
и запросов к движку считаются по IP клиента.

## 5. Что важно знать

- **Наружу открыт только `web`.** API доверяет `X-Forwarded-For` от любого адреса внутренней
  сети compose (`FORWARDED_ALLOW_IPS=*`) — это безопасно, пока порт `api` не опубликован.
- **Несколько процессов API** поддерживаются: присутствие, зрители, рассылка событий и цикл
  таймаутов (с выбором лидера) работают через Redis.
- **Мониторинг:** `GET /health` возвращает 503, если недоступна БД или Redis.
- **Документация API** (`/docs`) в продакшене отключена.
- **Celery-воркер** обязателен для фонового анализа партий; без него анализ всё равно
  считается по кнопке на странице партии.
