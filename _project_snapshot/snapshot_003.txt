def classify_move_by_cp_loss(cp_loss: int, is_best_move: bool = False) -> tuple[str, str]:
    """
    Классифицирует ход по потере сантипешек (cp_loss):
    - Best: 0..10 cp (или точное совпадение с первой линией движка)
    - Good: 11..45 cp
    - Inaccuracy (?!): 46..95 cp
    - Mistake (?): 96..199 cp
    - Blunder (??): >= 200 cp
    """
    loss = max(0, int(cp_loss))
    if is_best_move or loss <= 10:
        return "Best", "!"
    if loss <= 45:
        return "Good", ""
    if loss <= 95:
        return "Inaccuracy", "?!"
    if loss < 200:
        return "Mistake", "?"
    return "Blunder", "??"


def calculate_move_accuracy(cp_loss: int) -> float:
    """Формула точности одного хода (0..100%) на основе потери сантипешек."""
    loss = max(0.0, float(cp_loss))
    raw = 103.1668 * math.exp(-0.004 * loss) - 3.1669
    return round(max(0.0, min(100.0, raw)), 1)


def calculate_accuracy(cp_losses: list[int]) -> float:
    """Рассчитывает общую точность игрока в партии (0..100%) по списку потерь сантипешек."""
    if not cp_losses:
        return 100.0
    per_move = [calculate_move_accuracy(loss) for loss in cp_losses]
    return round(sum(per_move) / len(per_move), 1)


def format_eval_display(eval_white_cp: int, mate_white: int | None = None) -> str:
    """Форматирует оценку с точки зрения Белых для шкалы EvalBar (+1.4, -0.8, M3, -M2)."""
    if mate_white is not None and mate_white != 0:
        return f"M{mate_white}" if mate_white > 0 else f"-M{abs(mate_white)}"
    pawns = eval_white_cp / 100.0
    sign = "+" if pawns > 0 else ""
    return f"{sign}{pawns:.1f}"


def _evaluate_board_sequence_sync(pgn_text: str, depth: int = 10) -> dict:
    """Синхронный проход по всем позициям партии со Stockfish или статическим fallback-оценщиком."""
    pgn_game = chess.pgn.read_game(io.StringIO(pgn_text))
    if pgn_game is None:
        raise ValueError("Не удалось распарсить PGN партии")

    board = pgn_game.board()
    mainline_moves = list(pgn_game.mainline_moves())

    sf_path = resolve_stockfish_path()
    engine = None
    engine_label = "Static Fallback Evaluator"

    if sf_path and os.path.exists(sf_path):
        try:
            engine = chess.engine.SimpleEngine.popen_uci(sf_path)
            engine_label = engine.id.get("name", "Stockfish")
        except Exception as exc:
            logger.warning("Не удалось запустить Stockfish (%s), используем fallback: %s", sf_path, exc)
            engine = None

    # Оценки каждой позиции с точки зрения Белых (индекс 0 = стартовая позиция)
    evals_white_cp: list[int] = []
    mates_white: list[int | None] = []
    best_moves_uci: list[str | None] = []

    def eval_single_board(b: chess.Board) -> tuple[int, int | None, str | None]:
        if b.is_checkmate():
            mate_val = -1 if b.turn == chess.WHITE else 1
            cp_val = -10000 if b.turn == chess.WHITE else 10000
            return cp_val, mate_val, None
        if b.is_game_over():
            return 0, None, None

        if engine is not None:
            try:
                info = engine.analyse(b, chess.engine.Limit(depth=depth, time=0.06))
                if isinstance(info, list):
                    info = info[0]
                pov_white = info["score"].white()
                mate = pov_white.mate()
                cp = pov_white.score(mate_score=10000) or 0
                cp = max(-2000, min(2000, int(cp)))
                pv = info.get("pv")
                best_uci = pv[0].uci() if pv else None
                return cp, mate, best_uci
            except Exception:
                pass

        # Fallback без бинарника Stockfish
        best_uci = None
        best_side_score = -100000.0
        for candidate in b.legal_moves:
            b_copy = b.copy()
            b_copy.push(candidate)
            sc = static_board_eval_cp(b_copy, b.turn)
            if sc > best_side_score:
                best_side_score = sc
                best_uci = candidate.uci()

        white_cp = int(static_board_eval_cp(b, chess.WHITE))
        white_cp = max(-2000, min(2000, white_cp))
        return white_cp, None, best_uci

    try:
        cp0, mate0, best0 = eval_single_board(board)
        evals_white_cp.append(cp0)
        mates_white.append(mate0)
        best_moves_uci.append(best0)

        san_list: list[str] = []
        uci_list: list[str] = []
        colors: list[str] = []

        for mv in mainline_moves:
            colors.append("white" if board.turn == chess.WHITE else "black")
            san_list.append(board.san(mv))
            uci_list.append(mv.uci())
            board.push(mv)
            cp_i, mate_i, best_i = eval_single_board(board)
            evals_white_cp.append(cp_i)
            mates_white.append(mate_i)
            best_moves_uci.append(best_i)
    finally:
        if engine is not None:
            try:
                engine.quit()
            except Exception:
                pass

    moves_report: list[dict] = []
    white_losses: list[int] = []
    black_losses: list[int] = []

    counts = {
        "white": {"Best": 0, "Good": 0, "Inaccuracy": 0, "Mistake": 0, "Blunder": 0},
        "black": {"Best": 0, "Good": 0, "Inaccuracy": 0, "Mistake": 0, "Blunder": 0},
    }

    for idx, uci_played in enumerate(uci_list):
        mover_color = colors[idx]
        before_white_cp = evals_white_cp[idx]
        after_white_cp = evals_white_cp[idx + 1]
        after_mate = mates_white[idx + 1]
        recommended_uci = best_moves_uci[idx]

        if mover_color == "white":
            cp_loss = max(0, before_white_cp - after_white_cp)
            white_losses.append(cp_loss)
        else:
            cp_loss = max(0, after_white_cp - before_white_cp)
            black_losses.append(cp_loss)

        is_exact_best = recommended_uci is not None and recommended_uci == uci_played
        if is_exact_best:
            cp_loss = 0

        label, symbol = classify_move_by_cp_loss(cp_loss, is_best_move=is_exact_best)
        counts[mover_color][label] += 1

        moves_report.append({
            "ply": idx + 1,
            "move_number": (idx // 2) + 1,
            "color": mover_color,
            "san": san_list[idx],
            "uci": uci_played,
            "eval_cp": after_white_cp,
            "eval_display": format_eval_display(after_white_cp, after_mate),
            "mate": after_mate,
            "best_move_uci": recommended_uci,
            "cp_loss": cp_loss,
            "classification": label,
            "symbol": symbol,
        })

    def build_side_summary(losses: list[int], side_counts: dict[str, int]) -> dict:
        avg_loss = round(sum(losses) / len(losses), 1) if losses else 0.0
        return {
            "accuracy": calculate_accuracy(losses),
            "best": side_counts["Best"],
            "good": side_counts["Good"],
            "inaccuracies": side_counts["Inaccuracy"],
            "mistakes": side_counts["Mistake"],
            "blunders": side_counts["Blunder"],
            "avg_cp_loss": avg_loss,
        }

    return {
        "engine": engine_label,
        "white": build_side_summary(white_losses, counts["white"]),
        "black": build_side_summary(black_losses, counts["black"]),
        "moves": moves_report,
    }


class AnalysisService:
    async def analyze_position(self, db: AsyncSession, moves: list[str]) -> dict:
        board = chess.Board()
        for move_str in moves:
            try:
                board.push_san(move_str)
            except ValueError:
                try:
                    board.push_uci(move_str)
                except ValueError:
                    break

        sf_path = resolve_stockfish_path()
        eval_cp = int(static_board_eval_cp(board, chess.WHITE))
        best_move = None

        if sf_path and os.path.exists(sf_path) and not board.is_game_over():
            def _run_pos():
                with chess.engine.SimpleEngine.popen_uci(sf_path) as eng:
                    info = eng.analyse(board, chess.engine.Limit(depth=10, time=0.1))
                    if isinstance(info, list):
                        info = info[0]
                    cp = info["score"].white().score(mate_score=10000) or 0
                    pv = info.get("pv")
                    return int(cp), (pv[0].uci() if pv else None)

            try:
                eval_cp, best_move = await asyncio.to_thread(_run_pos)
            except Exception:
                pass

        return {
            "fen": board.fen(),
            "evaluation_cp": eval_cp,
            "evaluation_pawns": round(eval_cp / 100.0, 2),
            "best_move": best_move,
        }

    async def analyze_full_game(self, db: AsyncSession, game_id: int, force: bool = False) -> dict:
        result = await db.execute(select(Game).where(Game.id == game_id))
        game = result.scalars().first()
        if not game or not game.pgn:
            raise LookupError(f"Партия {game_id} или её PGN не найдены")

        # Если анализ уже сохранён в БД и не запрошен пересчёт — отдаём мгновенно из кэша
        cached_raw = getattr(game, "analysis_json", None)
        if cached_raw and not force:
            try:
                cached_data = json.loads(cached_raw)
                if isinstance(cached_data, dict) and "moves" in cached_data:
                    cached_data["game_id"] = game_id
                    return cached_data
            except Exception:
                pass

        report = await asyncio.to_thread(_evaluate_board_sequence_sync, game.pgn, 10)
        report["game_id"] = game_id

        if hasattr(game, "analysis_json"):
            try:
                game.analysis_json = json.dumps(report, ensure_ascii=False)
                await db.commit()
            except Exception as exc:
                await db.rollback()
                logger.warning("Не удалось сохранить кэш анализа в БД: %s", exc)

        return report


analysis_service = AnalysisService()

import logging
import math
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from src.models.game import Game
from src.models.move import Move
from src.models.user import User

logger = logging.getLogger(__name__)

MIN_MOVES_FOR_ANALYSIS = 5
OPENING_MOVE_CUTOFF = 8
CHEATER_TRUST_THRESHOLD = 30


def calculate_blur_ratio(moves: list[Move]) -> float:
    """
    Считает долю ходов с потерей фокуса окна (window_blurred_before_move == True)
    после 8-го хода. Если после 8-го хода менее 3 ходов, оценивает все ходы игрока.
    """
    if not moves:
        return 0.0

    post_opening = [m for m in moves if (m.move_number or 0) > OPENING_MOVE_CUTOFF]
    target_moves = post_opening if len(post_opening) >= 3 else moves

    blurred_count = sum(1 for m in target_moves if bool(m.window_blurred_before_move))
    return round(blurred_count / len(target_moves), 4)


def calculate_timing_cv(moves: list[Move]) -> tuple[float, float]:
    """
    Считает коэффициент вариации времени ходов (CV = std_dev / mean),
    исключая предходы (is_premove == True). Возвращает (cv, mean_time_ms).
    """
    normal_times = [
        float(m.time_taken_ms)
        for m in moves
        if not bool(m.is_premove) and (m.time_taken_ms or 0) > 0
    ]

    if len(normal_times) < 3:
        return 1.0, 0.0

    mean_time = sum(normal_times) / len(normal_times)
    if mean_time <= 0:
        return 1.0, 0.0

    variance = sum((t - mean_time) ** 2 for t in normal_times) / len(normal_times)
    std_dev = math.sqrt(variance)
    return round(std_dev / mean_time, 4), round(mean_time, 1)


def compute_telemetry_penalty(moves: list[Move]) -> dict:
    """Вычисляет штраф к фактору доверия по паттернам Tab-Switch и Timing Variance."""
    if len(moves) < MIN_MOVES_FOR_ANALYSIS:
        return {
            "penalty": 0,
            "blur_ratio": 0.0,
            "timing_cv": 1.0,
            "mean_time_ms": 0.0,
        }

    blur_ratio = calculate_blur_ratio(moves)
    timing_cv, mean_time_ms = calculate_timing_cv(moves)

    penalty = 0

    if blur_ratio >= 0.75:
        penalty += 45
    elif blur_ratio >= 0.50:
        penalty += 25
    elif blur_ratio >= 0.35:
        penalty += 12

    is_engine_pacing = 1500.0 <= mean_time_ms <= 15000.0
    if is_engine_pacing:
        if timing_cv <= 0.12:
            penalty += 40
        elif timing_cv <= 0.22:
            penalty += 22

    if blur_ratio >= 0.65 and is_engine_pacing and timing_cv <= 0.20:
        penalty += 15

    return {
        "penalty": penalty,
        "blur_ratio": blur_ratio,
        "timing_cv": timing_cv,
        "mean_time_ms": mean_time_ms,
    }


async def evaluate_game_telemetry(db: AsyncSession, game_id: int) -> dict[int, dict]:
    """
    Анализирует телеметрию завершённой партии для обоих игроков,
    снижает user.trust_factor при аномалиях и выставляет is_cheater = True при trust_factor < 30.
    """
    game = await db.get(Game, game_id)
    if not game:
        return {}

    result = await db.execute(
        select(Move).where(Move.game_id == game_id).order_by(Move.id.asc())
    )
    all_moves = list(result.scalars().all())
    if not all_moves:
        return {}

    moves_by_player: dict[int, list[Move]] = {
        game.white_player_id: [],
        game.black_player_id: [],
    }
    for move in all_moves:
        moves_by_player.setdefault(move.player_id, []).append(move)

    report: dict[int, dict] = {}

    for player_id, player_moves in moves_by_player.items():
        metrics = compute_telemetry_penalty(player_moves)
        penalty = metrics["penalty"]

        user = await db.get(User, player_id)
        if not user:
            continue

        if penalty > 0:
            user.trust_factor = max(0, int(user.trust_factor) - penalty)
            if user.trust_factor < CHEATER_TRUST_THRESHOLD:
                user.is_cheater = True
                logger.warning(
                    "Fair Play Alert: user %s (id=%s) flagged in game %s (trust_factor=%s, blur=%.2f, cv=%.2f)",
                    user.username,
                    user.id,
                    game_id,
                    user.trust_factor,
                    metrics["blur_ratio"],
                    metrics["timing_cv"],
                )

        report[player_id] = {
            **metrics,
            "trust_factor": user.trust_factor,
            "is_cheater": user.is_cheater,
        }

    await db.commit()
    return report

def evaluate_anderssen_style_bonus(board_before: chess.Board, move: chess.Move) -> tuple[float, str | None]:
    piece = board_before.piece_at(move.from_square)
    if not piece:
        return 0.0, None

    bonus = 0.0
    note = None

    if board_before.gives_check(move):
        bonus += 36.0
        note = "Шах! В духе «Бессмертной партии» — ни одного темпа передышки королю соперника!"

    if board_before.is_capture(move):
        bonus += 24.0
        if not note:
            note = "Вскрываем линии и устраняем защитные редуты неприятеля!"

    if board_before.is_castling(move):
        bonus += 38.0
        note = "Рокировка! Подключаем ладью к решающему штурму."

    if piece.piece_type in {chess.KNIGHT, chess.BISHOP} and board_before.fullmove_number <= 14:
        back_rank = 0 if board_before.turn == chess.WHITE else 7
        if chess.square_rank(move.from_square) == back_rank:
            bonus += 28.0
            if not note:
                note = "Мобилизуем лёгкие фигуры для комбинационного удара!"
        if move.to_square in EXTENDED_CENTER:
            bonus += 18.0

    if move.to_square in CENTER_SQUARES:
        bonus += 22.0
        if not note:
            note = "Борьба за инициативу и центральные диагонали!"

    return bonus, note

def normalize_fen_key(board: chess.Board) -> str:
    """Нормализует FEN до первых 3 полей (позиция, очередь хода, права рокировки)."""
    parts = board.fen().split(" ")
    return f"{parts[0]} {parts[1]} {parts[2]} -"


def resolve_stockfish_path(custom_path: str | None = None) -> str | None:
    """Ищет бинарник Stockfish в локальной папке bin/ проекта, системном PATH или стандартных путях Linux."""
    if custom_path and os.path.exists(custom_path):
        return custom_path

    for binary_name in ("stockfish.exe", "stockfish"):
        candidate = PROJECT_ROOT / "bin" / binary_name
        if candidate.exists() and candidate.is_file():
            return str(candidate)

    which_path = shutil.which("stockfish")
    if which_path:
        return which_path

    for linux_path in ("/usr/games/stockfish", "/usr/bin/stockfish", "/usr/local/bin/stockfish"):
        if os.path.exists(linux_path) and os.path.isfile(linux_path):
            return linux_path

    return None


def load_bot_opening_book(
        bot_id: str,
        fallback_book: dict[str, list[tuple[str, int, str]]] | None = None,
) -> dict[str, list[tuple[str, int, str]]]:
    """
    Загружает дебютную книгу бота из src/services/bots/books/{bot_id}.json
    и аккуратно объединяет её с фирменными дебютными линиями fallback_book.
    """
    book: dict[str, list[tuple[str, int, str]]] = {}
    if fallback_book:
        for k, v in fallback_book.items():
            book[k] = list(v)

    candidates = [
        BOOKS_DIR / f"{bot_id}.json",
        PROJECT_ROOT / f"{bot_id}_generated_book.json",
    ]

    for json_path in candidates:
        if json_path.exists():
            try:
                with open(json_path, "r", encoding="utf-8") as f:
                    data = json.load(f)
                if isinstance(data, dict) and data:
                    for fen_key, moves in data.items():
                        parsed_moves = [
                            (str(m[0]), int(m[1]), str(m[2]))
                            for m in moves
                            if len(m) >= 3 and str(m[0]) != "0000"
                        ]
                        if not parsed_moves:
                            continue

                        if fen_key in book:
                            existing_ucis = {m[0] for m in parsed_moves}
                            for fb_move in book[fen_key]:
                                if fb_move[0] not in existing_ucis:
                                    parsed_moves.append(fb_move)
                        book[fen_key] = parsed_moves
                    break
            except Exception as exc:
                logger.warning("Не удалось загрузить книгу %s: %s", json_path, exc)

    return book


def _simulate_human_thinking_time(
        board: chess.Board,
        is_book_move: bool,
        bot_time_left_ms: int = 180000,
        base_time_ms: int = 180000,
) -> None:
    """
    Имитирует человеческое время раздумий шахматиста.
    При запуске тестов (pytest) задержка автоматически отключается.
    """
    if "pytest" in sys.modules or os.environ.get("PYTEST_CURRENT_TEST"):
        return

    piece_count = len(board.piece_map())
    move_number = board.fullmove_number

    is_opening = move_number <= 10 and piece_count >= 28
    is_endgame = piece_count <= 12

    if is_opening:
        base_delay = random.uniform(0.8, 1.8) if is_book_move else random.uniform(3.5, 8.0)
    elif is_endgame:
        base_delay = random.uniform(0.8, 2.0) if piece_count <= 6 else random.uniform(4.0, 12.0)
    else:
        legal_moves_count = board.legal_moves.count()
        complexity_mult = 0.7 + (min(legal_moves_count, 45) / 45.0) * 0.8
        base_delay = random.uniform(4.0, 14.0) * complexity_mult
        if random.random() < 0.08 and base_time_ms > 60000:
            base_delay = random.uniform(18.0, 35.0)

    control_scale_factor = max(0.3, min(3.0, base_time_ms / 300000.0))
    if base_time_ms > 600000:
        control_scale_factor = 1.5 + (base_time_ms / 1800000.0) * 1.5

    adjusted_delay = base_delay * control_scale_factor

    if bot_time_left_ms < 30000:
        time_ratio = bot_time_left_ms / 30000.0
        adjusted_delay = max(0.5, adjusted_delay * time_ratio)

    noise = random.gauss(0, 0.3) * adjusted_delay
    final_thinking_time = max(0.6, adjusted_delay + noise)
    max_allowed = max(0.5, (bot_time_left_ms - 2000) / 1000.0)
    time.sleep(min(final_thinking_time, max_allowed))


def static_board_eval_cp(board: chess.Board, perspective: chess.Color) -> float:
    """Базовая статическая оценка позиции в сантипешках (fallback при отсутствии Stockfish)."""
    if board.is_checkmate():
        return -100000.0 if board.turn == perspective else 100000.0
    if board.is_stalemate() or board.is_insufficient_material():
        return 0.0

    score = 0.0
    for sq in chess.SQUARES:
        p = board.piece_at(sq)
        if not p:
            continue
        val = float(PIECE_VALUES[p.piece_type])
        if sq in CENTER_SQUARES:
            val += 25.0 if p.piece_type == chess.PAWN else 15.0
        elif sq in EXTENDED_CENTER:
            val += 15.0 if p.piece_type == chess.PAWN else 8.0

        if p.color == perspective:
            score += val
        else:
            score -= val
    return score


def analyse_with_stockfish(
        board: chess.Board,
        engine_path: str | None,
        skill_level: int = 7,
        uci_elo: int | None = 1850,
        depth: int = 6,
        time_limit: float = 0.25,
        multipv: int = 6,
) -> dict[chess.Move, float]:
    """Запрашивает оценку топ-ходов у движка Stockfish."""
    if not engine_path or not os.path.exists(engine_path):
        return {}

    scores: dict[chess.Move, float] = {}
    try:
        with chess.engine.SimpleEngine.popen_uci(engine_path) as engine:
            options: dict[str, int | bool] = {"Skill Level": skill_level}
            if uci_elo is not None:
                if "UCI_LimitStrength" in engine.options:
                    options["UCI_LimitStrength"] = True
                if "UCI_Elo" in engine.options:
                    min_elo = getattr(engine.options["UCI_Elo"], "min", 1320) or 1320
                    max_elo = getattr(engine.options["UCI_Elo"], "max", 3190) or 3190
                    options["UCI_Elo"] = max(min_elo, min(max_elo, uci_elo))
            engine.configure(options)

            infos = engine.analyse(
                board,
                chess.engine.Limit(depth=depth, time=time_limit),
                multipv=min(multipv, board.legal_moves.count()),
            )
            if isinstance(infos, dict):
                infos = [infos]
            for info in infos:
                pv = info.get("pv")
                if pv:
                    mv = pv[0]
                    cp = info["score"].pov(board.turn).score(mate_score=100000)
                    if cp is not None:
                        scores[mv] = float(cp)
    except Exception as exc:
        logger.warning("Ошибка запуска Stockfish (%s): %s", engine_path, exc)
        return {}

    return scores


class HistoricalBot:
    """
    Базовый класс исторической шахматной персоны.
    Наследники задают только параметры силы, дебютную книгу и функцию стилевого бонуса.
    """

    bot_id: str = "base"
    engine_source_name: str = "historical_style_engine"
    default_comment: str = "Классический позиционный ход."
    blunder_guard_cp: float = 130.0
    jitter_range: float = 8.0
    multipv: int = 6
    opening_book: dict[str, list[tuple[str, int, str]]] = {}

    def __init__(
            self,
            stockfish_path: str | None = None,
            skill_level: int = 7,
            uci_elo: int = 1850,
            search_depth: int = 6,
    ):
        self._explicit_stockfish_path = stockfish_path
        self.stockfish_path = resolve_stockfish_path(stockfish_path)
        self.skill_level = skill_level
        self.uci_elo = uci_elo
        self.search_depth = search_depth

    def _get_active_engine_path(self) -> str | None:
        if self._explicit_stockfish_path is not None:
            return self._explicit_stockfish_path if os.path.exists(self._explicit_stockfish_path) else None
        if self.stockfish_path and os.path.exists(self.stockfish_path):
            return self.stockfish_path
        self.stockfish_path = resolve_stockfish_path()
        return self.stockfish_path

    def evaluate_style_bonus(self, board_before: chess.Board, move: chess.Move) -> tuple[float, str | None]:
        """Переопределяется в подклассах для придания уникального шахматного стиля."""
        return 0.0, None

    def choose_move(self, fen: str) -> dict:
        board = chess.Board(fen)
        if board.is_game_over():
            raise ValueError("Game is already over")

        # 1. Поиск хода в дебютной книге чемпиона
        fen_key = normalize_fen_key(board)
        if fen_key in self.opening_book:
            candidates = self.opening_book[fen_key]
            valid_book_moves = []
            for item in candidates:
                uci_str, weight, comment = item[0], int(item[1]), item[2]
                if uci_str == "0000":
                    continue
                try:
                    move = chess.Move.from_uci(uci_str)
                except ValueError:
                    continue
                if move in board.legal_moves:
                    valid_book_moves.append((move, max(weight, 1), comment))

            if valid_book_moves:
                _simulate_human_thinking_time(board, is_book_move=True)
                moves, weights, comments = zip(*valid_book_moves)
                idx = random.choices(range(len(moves)), weights=weights, k=1)[0]
                chosen_move = moves[idx]
                san = board.san(chosen_move)
                board.push(chosen_move)
                return {
                    "uci": chosen_move.uci(),
                    "san": san,
                    "fen_after": board.fen(),
                    "source": "opening_book",
                    "comment": comments[idx],
                }

        _simulate_human_thinking_time(board, is_book_move=False)

        # 2. Тактический расчёт Stockfish
        engine_path = self._get_active_engine_path() if self.stockfish_path is not None else None
        base_scores = analyse_with_stockfish(
            board=board,
            engine_path=engine_path,
            skill_level=self.skill_level,
            uci_elo=self.uci_elo,
            depth=self.search_depth,
            time_limit=0.25,
            multipv=self.multipv,
        )

        used_stockfish = bool(base_scores)
        # Fallback-расчёт при отсутствии бинарника Stockfish
        if not base_scores:
            us = board.turn
            for mv in board.legal_moves:
                b1 = board.copy()
                b1.push(mv)
                if b1.is_checkmate():
                    base_scores[mv] = 100000.0
                    continue
                worst_reply_score = 100000.0
                for reply in b1.legal_moves:
                    b2 = b1.copy()
                    b2.push(reply)
                    s = static_board_eval_cp(b2, us)
                    if s < worst_reply_score:
                        worst_reply_score = s
                base_scores[mv] = worst_reply_score if b1.legal_moves.count() > 0 else 0.0

        # 3. Коридор допустимой уступки + индивидуальный стилевой бонус чемпиона
        max_tactical_cp = max(base_scores.values())
        candidate_scores = {
                               mv: cp for mv, cp in base_scores.items()
                               if cp >= max_tactical_cp - self.blunder_guard_cp
                           } or base_scores

        best_move = None
        best_total_score = -float("inf")
        best_comment = None

        for mv, tactical_cp in candidate_scores.items():
            style_bonus, note = self.evaluate_style_bonus(board, mv)
            jitter = random.uniform(-self.jitter_range, self.jitter_range)
            total_score = tactical_cp + style_bonus + jitter

            if total_score > best_total_score:
                best_total_score = total_score
                best_move = mv
                best_comment = note

        if best_move is None:
            best_move = next(iter(board.legal_moves))

        san = board.san(best_move)
        board.push(best_move)

        engine_backend = "stockfish" if used_stockfish else "fallback_minimax"
        logger.info("Bot %s move %s via %s [%s] (engine=%s)", self.bot_id, san, self.engine_source_name, engine_backend, engine_path)
        return {
            "uci": best_move.uci(),
            "san": san,
            "fen_after": board.fen(),
            "source": self.engine_source_name,
            "engine_backend": engine_backend,
            "comment": best_comment or self.default_comment,
        }

def _evaluate_pawn_structure(board: chess.Board, us: chess.Color) -> float:
    our_pawns = board.pieces(chess.PAWN, us)
    start_rank = 1 if us == chess.WHITE else 6
    pawn_files = [chess.square_file(sq) for sq in our_pawns]

    score = 0.0
    for sq in our_pawns:
        f = chess.square_file(sq)
        r = chess.square_rank(sq)
        if r != start_rank:
            score += float(r if us == chess.WHITE else (7 - r)) * 3.0
        if f < 7 and chess.square(f + 1, r) in our_pawns:
            score += 15.0
        support_rank = (r - 1) if us == chess.WHITE else (r + 1)
        if 0 <= support_rank <= 7:
            if (f > 0 and chess.square(f - 1, support_rank) in our_pawns) or (
                    f < 7 and chess.square(f + 1, support_rank) in our_pawns
            ):
                score += 10.0

    doubled_pawns = len(pawn_files) - len(set(pawn_files))
    score -= float(doubled_pawns) * 20.0
    return score

    bonus = 0.0
    note = None

    if piece.piece_type == chess.PAWN:
        bonus += 14.0
        if move.to_square in CENTER_SQUARES:
            bonus += 35.0
            note = "Пешки — душа шахмат! Мощный удар в центр доски."
        elif move.to_square in EXTENDED_CENTER:
            bonus += 22.0
            note = "Занимаем пространство пешечной фалангой."
        elif move.to_square in {chess.C3, chess.C6, chess.D3, chess.D6}:
            bonus += 16.0
            note = "Подготавливаем продвижение центральной пешки по методу Филидора."

    if piece.piece_type == chess.KNIGHT and board_before.fullmove_number <= 10:
        if us == chess.WHITE:
            if move.to_square == chess.C3 and board_before.piece_at(chess.C2) == chess.Piece(chess.PAWN, chess.WHITE):
                bonus -= 55.0
            if move.to_square == chess.F3 and board_before.piece_at(chess.F2) == chess.Piece(chess.PAWN, chess.WHITE):
                bonus -= 45.0
        else:
            if move.to_square == chess.C6 and board_before.piece_at(chess.C7) == chess.Piece(chess.PAWN, chess.BLACK):
                bonus -= 55.0
            if move.to_square == chess.F6 and board_before.piece_at(chess.F7) == chess.Piece(chess.PAWN, chess.BLACK):
                bonus -= 45.0

    structure_before = _evaluate_pawn_structure(board_before, us)
    board_after = board_before.copy()
    board_after.push(move)
    structure_after = _evaluate_pawn_structure(board_after, us)

def evaluate_staunton_style_bonus(board_before: chess.Board, move: chess.Move) -> tuple[float, str | None]:
    piece = board_before.piece_at(move.from_square)
    if not piece:
        return 0.0, None

    bonus = 0.0
    note = None

    if board_before.is_castling(move):
        bonus += 40.0
        note = "Рокировка в духе английской классической школы: король в безопасности, ладья входит в игру."

    if piece.piece_type == chess.PAWN and board_before.fullmove_number <= 12:
        if move.to_square in {chess.C4, chess.C5}:
            bonus += 30.0
            note = "Берём под фланговый контроль центральный пункт по системе Стаунтона!"
        elif move.to_square in CENTER_SQUARES:
            bonus += 24.0
            note = "Укрепляем позиции в центре доски."

    if piece.piece_type in {chess.KNIGHT, chess.BISHOP} and board_before.fullmove_number <= 12:
        back_rank = 0 if board_before.turn == chess.WHITE else 7
        if chess.square_rank(move.from_square) == back_rank:
            bonus += 26.0
            if not note:
                note = "Развиваем фигуры на стратегически важные поля."
        if move.to_square in EXTENDED_CENTER:
            bonus += 16.0

    if board_before.gives_check(move):
        bonus += 18.0
        if not note:
            note = "Тактический шах для перехвата инициативы."
    elif board_before.is_capture(move):
        bonus += 14.0

    return bonus, note
