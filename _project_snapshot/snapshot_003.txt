    @classmethod
    def _split_origins(cls, value):
        if isinstance(value, str):
            text = value.strip()
            if text.startswith("["):
                return json.loads(text)
            return [origin.strip() for origin in text.split(",") if origin.strip()]
        return value

    @model_validator(mode="after")
    def _check_production_secrets(self):
        """
        В продакшене слабый SECRET_KEY = возможность подделать токен любого пользователя.
        Лучше не стартовать вовсе, чем работать с ключом из примера или коротким ключом.
        """
        if self.is_production:
            key = self.SECRET_KEY
            if len(key) < MIN_SECRET_KEY_LENGTH or key.lower() in _WEAK_SECRET_KEYS or "change-me" in key.lower():
                raise ValueError(
                    f"SECRET_KEY must be a random string of at least {MIN_SECRET_KEY_LENGTH} characters "
                    "in production (e.g. `python -c \"import secrets; print(secrets.token_urlsafe(48))\"`)"
                )
        return self

    def is_production(self) -> bool:
        return self.ENVIRONMENT.strip().lower() == "production"

    @property

            f"postgresql+asyncpg://{self.POSTGRES_USER}:{self.POSTGRES_PASSWORD}"
            f"@{self.POSTGRES_SERVER}:{self.POSTGRES_PORT}/{self.POSTGRES_DB}"
        )

    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        case_sensitive=True,
        extra="ignore",
    )


settings = Settings()

from datetime import UTC, datetime, timedelta

import jwt
from passlib.context import CryptContext

from src.core.config import settings

# Password hashing context using bcrypt algorithm
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def verify_password(plain_password: str, hashed_password: str) -> bool:
    """Verifies a plain password against a hashed counterpart.

    A hash passlib cannot identify (e.g. the marker stored for bot accounts) never matches
    instead of raising and turning a login attempt into a 500.
    """
    try:
        return pwd_context.verify(plain_password, hashed_password)
    except (ValueError, TypeError):
        return False


def get_password_hash(password: str) -> str:
    """Generates a secure bcrypt hash for a plain password."""
    return pwd_context.hash(password)


class InvalidTokenError(Exception):
    """Токен не подписан нами, истёк или имеет старый формат."""


def create_user_token(user, expires_delta: timedelta | None = None) -> str:
    """
    JWT пользователя: sub — id (имя можно было бы сменить, id — нет), ver — версия токенов,
    по которой отзываются все выданные ранее токены.
    """
    return create_access_token({"sub": str(user.id), "ver": int(user.token_version or 0)}, expires_delta)


def decode_user_token(token: str) -> tuple[int, int]:
    """(user_id, token_version) из токена; InvalidTokenError для чужих, истёкших и старых токенов."""
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
    except jwt.InvalidTokenError as exc:
        raise InvalidTokenError(str(exc)) from exc
    subject, version = payload.get("sub"), payload.get("ver")
    # Токены старого формата (sub = имя пользователя, без ver) не принимаются: войти заново.
    if not isinstance(subject, str) or not subject.isdigit() or not isinstance(version, int):
        raise InvalidTokenError("unsupported token format")
    return int(subject), version


def create_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
    """Generates a JSON Web Token (JWT) for user authentication."""
    to_encode = data.copy()
    if expires_delta:
        expire = datetime.now(UTC) + expires_delta
    else:
        expire = datetime.now(UTC) + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)

    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    return encoded_jwt

WHITE_TAG_RE = re.compile(r'\[White\s+"([^"]+)"\]')

def extract_pgn_metadata(raw_pgn: str, fallback_result: str = "*") -> dict[str, str | None]:
    result_str = fallback_result or "*"
    pgn_date: str | None = None
    site: str | None = None
    eco: str | None = None

    if not raw_pgn:
        return {"result": result_str, "pgn_date": None, "site": None, "eco": None}

    res_match = RESULT_TAG_RE.search(raw_pgn)
    if res_match:
        result_str = res_match.group(1).strip()[:10]
    elif result_str == "*":
        tail_match = re.search(r"(1-0|0-1|1/2-1/2)\s*$", raw_pgn.strip())
        if tail_match:
            result_str = tail_match.group(1)

    for d_match in DATE_TAG_RE.finditer(raw_pgn):
        val = d_match.group(1).strip()
        if val and not val.startswith("?"):
            pgn_date = val[:20]
            break

    s_match = SITE_TAG_RE.search(raw_pgn)
    if s_match and s_match.group(1).strip() != "?":
        site = s_match.group(1).strip()[:100]

    e_match = ECO_TAG_RE.search(raw_pgn)
    if e_match and e_match.group(1).strip() != "?":
        eco = e_match.group(1).strip()[:10]

    return {
        "result": result_str,
        "pgn_date": pgn_date,
        "site": site,
        "eco": eco,
    }


def get_champion_meta(username_or_id: str) -> dict | None:
    key = (username_or_id or "").strip().lower()
    if key in CHAMPIONS_META:
        return CHAMPIONS_META[key]
    for meta in CHAMPIONS_META.values():
        if meta["canonical"].lower() == key:
            return meta
    return None


def find_champion_pgn_file(champ_id: str) -> Path | None:
    target_lower = f"{champ_id.lower()}.pgn"
    alt_lower = f"{champ_id.lower()}_all_games.pgn"

    for folder in PGNS_DIRS:
        if not folder.exists():
            continue
        for pgn_path in folder.glob("*.pgn"):
            name_low = pgn_path.name.lower()
            if name_low in (target_lower, alt_lower) and pgn_path.stat().st_size > 100:
                return pgn_path
    return None


def _download_champion_pgn_sync(champ_id: str) -> Path | None:
    if "pytest" in sys.modules or os.environ.get("PYTEST_CURRENT_TEST"):
        return None

    PRIMARY_PGNS_DIR.mkdir(parents=True, exist_ok=True)
    target_path = PRIMARY_PGNS_DIR / f"{champ_id}.pgn"
    if target_path.exists() and target_path.stat().st_size > 100:
        return target_path

    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/131.0.0.0 Safari/537.36",
        "Referer": "https://www.pgnmentor.com/files.html",
    }

    zip_url = f"{PGN_MENTOR_BASE_URL}/{champ_id}.zip"
    pgn_url = f"{PGN_MENTOR_BASE_URL}/{champ_id}.pgn"

    try:
        resp = requests.get(zip_url, headers=headers, timeout=30)
        if resp.status_code == 200 and len(resp.content) > 200:
            with zipfile.ZipFile(io.BytesIO(resp.content)) as zf:
                pgn_members = [m for m in zf.namelist() if m.lower().endswith(".pgn")]
                if pgn_members:
                    raw_bytes = zf.read(pgn_members[0])
                    text = raw_bytes.decode("utf-8", errors="ignore")
                    if "[Event " in text:
                        target_path.write_text(text, encoding="utf-8")
                        return target_path
    except Exception as exc:
        logger.warning("Не удалось скачать ZIP для %s (%s), пробуем прямой .pgn...", champ_id, exc)

    try:
        resp = requests.get(pgn_url, headers=headers, timeout=30)
        if resp.status_code == 200 and "[Event " in resp.text:
            target_path.write_text(resp.text, encoding="utf-8")
            return target_path
    except Exception as exc:
        logger.error("Ошибка загрузки PGN для %s: %s", champ_id, exc)

    return None


def _is_target_champion_name(raw_name: str, champ_key: str, meta: dict) -> bool:
    low = raw_name.lower()
    if any(ex in low for ex in meta.get("exclude", ())):
        return False
    # Надёжное сопоставление по фамилии (даже при сокращении инициалов в PGN Mentor)
    surname = meta["canonical"].split(",")[0].strip().lower()
    return surname in low or champ_key in low


def _split_pgn_games(content: str) -> list[str]:
    cleaned = content.lstrip("\ufeff").replace("\r\n", "\n").replace("\r", "\n").strip()
    if not cleaned:
        return []
    chunks = re.split(r'(?m)^(?=\[Event\s+")', cleaned)
    return [c.strip() for c in chunks if c.strip() and "[White " in c]


async def ensure_champion_seeded(db: AsyncSession, username_or_id: str) -> User | None:
    meta = get_champion_meta(username_or_id)
    if not meta:
        return None

    champ_key = meta["id"].lower()
    lock = _SEED_LOCKS.setdefault(champ_key, asyncio.Lock())

    async with lock:
        canonical_name = meta["canonical"]
        res = await db.execute(
            select(User).where(
                or_(User.username == canonical_name, User.username == meta["id"])
            )
        )
        champ_user = res.scalars().first()

        if champ_user is not None:
            count_q = select(func.count(Game.id)).where(
                or_(Game.white_player_id == champ_user.id, Game.black_player_id == champ_user.id),
                Game.status == GameStatus.COMPLETED,
            )
            games_count = (await db.execute(count_q)).scalar() or 0
            if games_count > 0:
                return champ_user

        all_users_res = await db.execute(select(User))
        users_by_name: dict[str, User] = {u.username: u for u in all_users_res.scalars().all()}

        if champ_user is None:
            champ_user = User(
                username=canonical_name,
                hashed_password="!historical_archive_no_login!",
                rating=meta["rating"],
                is_cheater=False,
                trust_factor=100,
            )
            db.add(champ_user)
            await db.flush()
            users_by_name[canonical_name] = champ_user

        pgn_file = find_champion_pgn_file(meta["id"])
        if not pgn_file or not pgn_file.exists():
            pgn_file = await asyncio.to_thread(_download_champion_pgn_sync, meta["id"])

        if not pgn_file or not pgn_file.exists():
            await db.commit()
            return champ_user

        content = pgn_file.read_text(encoding="utf-8", errors="ignore")
        raw_games = _split_pgn_games(content)
        if not raw_games:
            await db.commit()
            return champ_user

        new_opponents: dict[str, User] = {}
        parsed_entries: list[tuple[str, str, dict, str]] = []

        for raw_pgn in raw_games:
            w_match = WHITE_TAG_RE.search(raw_pgn)
            b_match = BLACK_TAG_RE.search(raw_pgn)
            if not w_match or not b_match:
                continue

            w_raw = w_match.group(1).strip()[:50]
            b_raw = b_match.group(1).strip()[:50]
            meta_fields = extract_pgn_metadata(raw_pgn)

            w_name = canonical_name if _is_target_champion_name(w_raw, champ_key, meta) else w_raw
            b_name = canonical_name if _is_target_champion_name(b_raw, champ_key, meta) else b_raw

            if w_name != canonical_name and b_name != canonical_name:
                continue

            for opp_name in (w_name, b_name):
                if opp_name not in users_by_name and opp_name not in new_opponents:
                    opp_meta = get_champion_meta(opp_name)
                    opp_rating = opp_meta["rating"] if opp_meta else 2450
                    new_opponents[opp_name] = User(
                        username=opp_name,
                        hashed_password="!historical_archive_no_login!",
                        rating=opp_rating,
                        is_cheater=False,
                        trust_factor=100,
                    )

            parsed_entries.append((w_name, b_name, meta_fields, raw_pgn))

        if new_opponents:
            db.add_all(new_opponents.values())
            await db.flush()
            users_by_name.update(new_opponents)

        existing_pgns_res = await db.execute(
            select(Game.pgn).where(
                or_(Game.white_player_id == champ_user.id, Game.black_player_id == champ_user.id)
            )
        )
        existing_pgns = {row[0] for row in existing_pgns_res.all() if row[0]}

        games_to_insert: list[Game] = []
        for w_name, b_name, meta_fields, raw_pgn in parsed_entries:
            if raw_pgn in existing_pgns:
                continue
            existing_pgns.add(raw_pgn)

            w_user = users_by_name[w_name]
            b_user = users_by_name[b_name]
            games_to_insert.append(
                Game(
                    white_player_id=w_user.id,
                    black_player_id=b_user.id,
                    pgn=raw_pgn,
                    status=GameStatus.COMPLETED,
                    result=meta_fields["result"],
                    pgn_date=meta_fields["pgn_date"],
                    site=meta_fields["site"],
                    eco=meta_fields["eco"],
                    is_rated=True,
                )
            )

        if games_to_insert:
            db.add_all(games_to_insert)

        await db.commit()
        logger.info(
            "Импортировано %d партий для чемпиона %s из %s",
            len(games_to_insert),
            canonical_name,
            pgn_file.name,
        )
        return champ_user


async def seed_all_champions() -> None:
    print(f"=== Полный импорт архивов 21 чемпиона мира (Корень: {PROJECT_ROOT}) ===")
    async with AsyncSessionLocal() as db:
        for _, meta in CHAMPIONS_META.items():
            user = await ensure_champion_seeded(db, meta["id"])
            if user:
                count_q = select(func.count(Game.id)).where(
                    or_(Game.white_player_id == user.id, Game.black_player_id == user.id),

async def get_db():
    """
    Dependency generator that yields an asynchronous database session.
    Ensures proper cleanup after request completion.

    Только для HTTP-запросов: сессия живёт до конца запроса. Долгоживущим обработчикам
    (WebSocket) нужна фабрика get_session_factory и короткая сессия на каждую операцию.
    """
    async with AsyncSessionLocal() as session:
        yield session


def get_session_factory() -> async_sessionmaker[AsyncSession]:
    """
    Dependency: фабрика сессий для долгоживущих соединений.

    Сессия из get_db, взятая WebSocket-обработчиком, жила бы всё соединение: первый же запрос
    открывает транзакцию, и соединение пула остаётся занятым («idle in transaction»), пока
    игрок подключён. Поэтому WebSocket открывает по короткой сессии на каждое действие.
    """
    return AsyncSessionLocal

    return JSONResponse(
        status_code=200 if healthy else 503,
        content={"status": "healthy" if healthy else "degraded", "version": settings.VERSION, "checks": checks},

import re
from datetime import datetime

from pydantic import BaseModel, ConfigDict, Field, field_validator

USERNAME_MIN_LENGTH = 3
USERNAME_MAX_LENGTH = 30
_USERNAME_RE = re.compile(r"[A-Za-z0-9А-Яа-яЁё_-]+")
_LATIN_RE = re.compile(r"[A-Za-z]")
_CYRILLIC_RE = re.compile(r"[А-Яа-яЁё]")


def username_key(username: str) -> str:
    """Ключ для сравнения имён без учёта регистра."""
    return username.strip().casefold()


def validate_username(value: str) -> str:
    """
    Правила имени при регистрации. Белый список символов отсекает пробелы, невидимые и
    управляющие символы, а запрет смешивать алфавиты — подделки вида «Мorphy» (кириллическая М).
    """
    if not _USERNAME_RE.fullmatch(value):
        raise ValueError("Имя может содержать только буквы (латиница или кириллица), цифры, «_» и «-»")
    if _LATIN_RE.search(value) and _CYRILLIC_RE.search(value):
        raise ValueError("Имя не может смешивать латиницу и кириллицу")
    if not any(ch.isalpha() for ch in value):
        raise ValueError("Имя должно содержать хотя бы одну букву")
    return value


class UserCreate(BaseModel):
    """Schema for incoming user registration data."""
    username: str = Field(
        ..., min_length=USERNAME_MIN_LENGTH, max_length=USERNAME_MAX_LENGTH, description="Unique username",
    )
    # Верхняя граница — чтобы не хэшировать мегабайтные «пароли»; bcrypt всё равно учитывает 72 байта.
    password: str = Field(..., min_length=8, max_length=128, description="Strong password")

    @field_validator("username")
    @classmethod
    def _check_username(cls, value: str) -> str:

def classify_move_by_cp_loss(cp_loss: int, is_best_move: bool = False) -> tuple[str, str]:
    """
    Классифицирует ход по потере сантипешек (cp_loss):
    - Best: 0..10 cp (или точное совпадение с первой линией движка)
    - Good: 11..45 cp
    - Inaccuracy (?!): 46..95 cp
    - Mistake (?): 96..199 cp
    - Blunder (??): >= 200 cp
    """
    loss = max(0, int(cp_loss))
    if is_best_move or loss <= 10:
        return "Best", "!"
    if loss <= 45:
        return "Good", ""
    if loss <= 95:
        return "Inaccuracy", "?!"
    if loss < 200:
        return "Mistake", "?"
    return "Blunder", "??"


def calculate_move_accuracy(cp_loss: int) -> float:
    """Формула точности одного хода (0..100%) на основе потери сантипешек."""
    loss = max(0.0, float(cp_loss))
    raw = 103.1668 * math.exp(-0.004 * loss) - 3.1669
    return round(max(0.0, min(100.0, raw)), 1)


def calculate_accuracy(cp_losses: list[int]) -> float:
    """Рассчитывает общую точность игрока в партии (0..100%) по списку потерь сантипешек."""
    if not cp_losses:
        return 100.0
    per_move = [calculate_move_accuracy(loss) for loss in cp_losses]
    return round(sum(per_move) / len(per_move), 1)


def format_eval_display(eval_white_cp: int, mate_white: int | None = None) -> str:
    """Форматирует оценку с точки зрения Белых для шкалы EvalBar (+1.4, -0.8, M3, -M2)."""
    if mate_white is not None and mate_white != 0:
        return f"M{mate_white}" if mate_white > 0 else f"-M{abs(mate_white)}"
    pawns = eval_white_cp / 100.0
    sign = "+" if pawns > 0 else ""
    return f"{sign}{pawns:.1f}"


def _evaluate_board_sequence_sync(pgn_text: str, depth: int = 10) -> dict:
    """Синхронный проход по всем позициям партии со Stockfish или статическим fallback-оценщиком."""
    pgn_game = chess.pgn.read_game(io.StringIO(pgn_text))
    if pgn_game is None:
        raise ValueError("Не удалось распарсить PGN партии")

    board = pgn_game.board()
    mainline_moves = list(pgn_game.mainline_moves())

    sf_path = resolve_stockfish_path()
    engine = None
    engine_label = "Static Fallback Evaluator"
    engine_stack = contextlib.ExitStack()

    if sf_path and os.path.exists(sf_path):
        # EngineBusyError намеренно не перехватывается: под нагрузкой лучше вернуть 503, чем
        # закэшировать в БД анализ статическим оценщиком вместо Stockfish.
        try:
            engine = engine_stack.enter_context(get_engine_pool(sf_path).acquire())
            engine_label = engine.id.get("name", "Stockfish")
        except EngineUnavailableError as exc:
            logger.warning("Не удалось запустить Stockfish (%s), используем fallback: %s", sf_path, exc)
            engine = None

    # Оценки каждой позиции с точки зрения Белых (индекс 0 = стартовая позиция)
    evals_white_cp: list[int] = []
    mates_white: list[int | None] = []
    best_moves_uci: list[str | None] = []

    def eval_single_board(b: chess.Board) -> tuple[int, int | None, str | None]:
        if b.is_checkmate():
            mate_val = -1 if b.turn == chess.WHITE else 1
            cp_val = -10000 if b.turn == chess.WHITE else 10000
            return cp_val, mate_val, None
        if b.is_game_over():
            return 0, None, None

        if engine is not None:
            try:
                info = engine.analyse(b, chess.engine.Limit(depth=depth, time=0.06))
                if isinstance(info, list):
                    info = info[0]
                pov_white = info["score"].white()
                mate = pov_white.mate()
                cp = pov_white.score(mate_score=10000) or 0
                cp = max(-2000, min(2000, int(cp)))
                pv = info.get("pv")
                best_uci = pv[0].uci() if pv else None
                return cp, mate, best_uci
            except Exception:
                pass

        # Fallback без бинарника Stockfish
        best_uci = None
        best_side_score = -100000.0
        for candidate in b.legal_moves:
            b_copy = b.copy()
            b_copy.push(candidate)
            sc = static_board_eval_cp(b_copy, b.turn)
            if sc > best_side_score:
                best_side_score = sc
                best_uci = candidate.uci()

        white_cp = int(static_board_eval_cp(b, chess.WHITE))
        white_cp = max(-2000, min(2000, white_cp))
        return white_cp, None, best_uci

    try:
        cp0, mate0, best0 = eval_single_board(board)
        evals_white_cp.append(cp0)
        mates_white.append(mate0)
        best_moves_uci.append(best0)

        san_list: list[str] = []
        uci_list: list[str] = []
        colors: list[str] = []

        for mv in mainline_moves:
            colors.append("white" if board.turn == chess.WHITE else "black")
            san_list.append(board.san(mv))
            uci_list.append(mv.uci())
            board.push(mv)
            cp_i, mate_i, best_i = eval_single_board(board)
            evals_white_cp.append(cp_i)
            mates_white.append(mate_i)
            best_moves_uci.append(best_i)
    finally:
        engine_stack.close()  # возвращает движок в пул

    moves_report: list[dict] = []
    white_losses: list[int] = []
    black_losses: list[int] = []

    counts = {
        "white": {"Best": 0, "Good": 0, "Inaccuracy": 0, "Mistake": 0, "Blunder": 0},
        "black": {"Best": 0, "Good": 0, "Inaccuracy": 0, "Mistake": 0, "Blunder": 0},
    }

    for idx, uci_played in enumerate(uci_list):
        mover_color = colors[idx]
        before_white_cp = evals_white_cp[idx]
        after_white_cp = evals_white_cp[idx + 1]
        after_mate = mates_white[idx + 1]
        recommended_uci = best_moves_uci[idx]

        if mover_color == "white":
            cp_loss = max(0, before_white_cp - after_white_cp)
            white_losses.append(cp_loss)
        else:
            cp_loss = max(0, after_white_cp - before_white_cp)
            black_losses.append(cp_loss)

        is_exact_best = recommended_uci is not None and recommended_uci == uci_played
        if is_exact_best:
            cp_loss = 0

        label, symbol = classify_move_by_cp_loss(cp_loss, is_best_move=is_exact_best)
        counts[mover_color][label] += 1

        moves_report.append({
            "ply": idx + 1,
            "move_number": (idx // 2) + 1,
            "color": mover_color,
            "san": san_list[idx],
            "uci": uci_played,
            "eval_cp": after_white_cp,
            "eval_display": format_eval_display(after_white_cp, after_mate),
            "mate": after_mate,
            "best_move_uci": recommended_uci,
            "cp_loss": cp_loss,
            "classification": label,
            "symbol": symbol,
        })

    def build_side_summary(losses: list[int], side_counts: dict[str, int]) -> dict:
        avg_loss = round(sum(losses) / len(losses), 1) if losses else 0.0
        return {
            "accuracy": calculate_accuracy(losses),
            "best": side_counts["Best"],
            "good": side_counts["Good"],
            "inaccuracies": side_counts["Inaccuracy"],
            "mistakes": side_counts["Mistake"],
            "blunders": side_counts["Blunder"],
            "avg_cp_loss": avg_loss,
        }

    return {
        "engine": engine_label,
        "white": build_side_summary(white_losses, counts["white"]),
        "black": build_side_summary(black_losses, counts["black"]),
        "moves": moves_report,
    }


class GameNotFinishedError(Exception):
    """Анализ идущей партии запрещён: он работал бы как подсказка движка игрокам."""


_FINISHED_STATUSES = (GameStatus.COMPLETED, GameStatus.ABORTED)


def ensure_game_finished(game: Game) -> None:
    if game.status not in _FINISHED_STATUSES:
        raise GameNotFinishedError(f"Партия {game.id} ещё не завершена")


def _count_mainline_plies(pgn_text: str) -> int:
    pgn_game = chess.pgn.read_game(io.StringIO(pgn_text))
    if pgn_game is None:
        return -1
    return sum(1 for _ in pgn_game.mainline_moves())


class AnalysisService:
    async def analyze_position(self, db: AsyncSession, moves: list[str]) -> dict:
        board = chess.Board()
        for move_str in moves:
            try:
                board.push_san(move_str)
            except ValueError:
                try:
                    board.push_uci(move_str)
                except ValueError:
                    break

        sf_path = resolve_stockfish_path()
        eval_cp = int(static_board_eval_cp(board, chess.WHITE))
        best_move = None

        if sf_path and not board.is_game_over():  # resolve_stockfish_path уже проверил файл
            def _run_pos():
                with get_engine_pool(sf_path).acquire() as eng:
                    info = eng.analyse(board, chess.engine.Limit(depth=10, time=0.1))
                    if isinstance(info, list):
                        info = info[0]
                    cp = info["score"].white().score(mate_score=10000) or 0
                    pv = info.get("pv")
                    return int(cp), (pv[0].uci() if pv else None)

            try:
                eval_cp, best_move = await asyncio.to_thread(_run_pos)
            except EngineUnavailableError as exc:
                logger.warning("Stockfish недоступен, отдаём статическую оценку: %s", exc)

        return {
            "fen": board.fen(),
            "evaluation_cp": eval_cp,
            "evaluation_pawns": round(eval_cp / 100.0, 2),
            "best_move": best_move,
        }

    async def analyze_full_game(self, db: AsyncSession, game_id: int, force: bool = False) -> dict:
        result = await db.execute(select(Game).where(Game.id == game_id))
        game = result.scalars().first()
        if not game or not game.pgn:
            raise LookupError(f"Партия {game_id} или её PGN не найдены")
        ensure_game_finished(game)

        # Если анализ уже сохранён в БД и не запрошен пересчёт — отдаём мгновенно из кэша
        cached_raw = getattr(game, "analysis_json", None)
        if cached_raw and not force:
            try:
                cached_data = json.loads(cached_raw)
                if (
                        isinstance(cached_data, dict)
                        and "moves" in cached_data
                        and len(cached_data["moves"]) == _count_mainline_plies(game.pgn)
                ):
                    cached_data["game_id"] = game_id
                    return cached_data
                # Кэш не соответствует итоговому PGN (например, сохранён по незавершённой партии
                # до запрета анализа идущих игр) — пересчитываем.
            except Exception:
                pass

        report = await asyncio.to_thread(_evaluate_board_sequence_sync, game.pgn, 10)
        report["game_id"] = game_id

        if hasattr(game, "analysis_json"):
            try:
                game.analysis_json = json.dumps(report, ensure_ascii=False)
                await db.commit()
            except Exception as exc:
                await db.rollback()
                logger.warning("Не удалось сохранить кэш анализа в БД: %s", exc)

        return report

import logging

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from src.models.game import Game

from src.models.user import User

logger = logging.getLogger(__name__)

MIN_MOVES_FOR_ANALYSIS = 5
OPENING_MOVE_CUTOFF = 8
CHEATER_TRUST_THRESHOLD = 30
EVALUATED_MARKER_TTL_S = 90 * 24 * 60 * 60


def trusted_premove(client_flag, server_elapsed_ms: int) -> bool:
    """
    Премув исключает ход из анализа ритма, поэтому флаг клиента принимаем, только если сервер
    сам видит почти мгновенный ход. Иначе читер мог бы пометить премувом все ходы.
    """
    from src.core.config import settings

    return bool(client_flag) and int(server_elapsed_ms) <= settings.AC_PREMOVE_MAX_ELAPSED_MS


async def _claim_first_evaluation(game_id: int) -> bool:
    """
    True, если партия оценивается впервые. Отметка хранится в Redis (SET NX), а не в памяти
    процесса: после рестарта или на другом воркере штраф за ту же партию не начислится снова.
    """
    from src.db import redis as redis_module

    try:
        claimed = await redis_module.redis_client.set(
            f"anticheat:evaluated:{int(game_id)}", "1", nx=True, ex=EVALUATED_MARKER_TTL_S,
        )
        return bool(claimed)
    except Exception as exc:
        logger.warning("Anti-cheat idempotency marker unavailable for game %s: %s", game_id, exc)
        return False  # без отметки не штрафуем: повторный штраф хуже пропущенного


def calculate_blur_ratio(moves: list[Move]) -> float:
    """
    Считает долю ходов с потерей фокуса окна (window_blurred_before_move == True)
    после 8-го хода. Если после 8-го хода менее 3 ходов, оценивает все ходы игрока.
    """
    if not moves:
        return 0.0

    post_opening = [m for m in moves if (m.move_number or 0) > OPENING_MOVE_CUTOFF]
    target_moves = post_opening if len(post_opening) >= 3 else moves

    blurred_count = sum(1 for m in target_moves if bool(m.window_blurred_before_move))
    return round(blurred_count / len(target_moves), 4)


def calculate_timing_cv(moves: list[Move]) -> tuple[float, float]:
    """
    Считает коэффициент вариации времени ходов (CV = std_dev / mean),
    исключая предходы (is_premove == True). Возвращает (cv, mean_time_ms).
    """
    normal_times = [
        float(m.time_taken_ms)
        for m in moves
        if not bool(m.is_premove) and (m.time_taken_ms or 0) > 0
    ]

    if len(normal_times) < 3:
        return 1.0, 0.0

    mean_time = sum(normal_times) / len(normal_times)
    if mean_time <= 0:
        return 1.0, 0.0

    variance = sum((t - mean_time) ** 2 for t in normal_times) / len(normal_times)
    std_dev = math.sqrt(variance)
    return round(std_dev / mean_time, 4), round(mean_time, 1)


def compute_telemetry_penalty(moves: list[Move]) -> dict:
    """Вычисляет штраф к фактору доверия по паттернам Tab-Switch и Timing Variance."""
    if len(moves) < MIN_MOVES_FOR_ANALYSIS:
        return {
            "penalty": 0,
            "blur_ratio": 0.0,
            "timing_cv": 1.0,
            "mean_time_ms": 0.0,
        }

    blur_ratio = calculate_blur_ratio(moves)
    timing_cv, mean_time_ms = calculate_timing_cv(moves)

    # Основной сигнал — ритм ходов, который измеряет сервер (time_taken_ms — серверное время
    # между ходами). Потеря фокуса окна сообщается клиентом и подделывается в обе стороны,
    # поэтому сама по себе не штрафует (иначе страдают честные игроки со вторым монитором)
    # и лишь усиливает штраф, когда серверный ритм уже похож на движок.
    timing_penalty = 0
    is_engine_pacing = 1500.0 <= mean_time_ms <= 15000.0
    if is_engine_pacing:
        if timing_cv <= 0.12:
            timing_penalty = 40
        elif timing_cv <= 0.22:
            timing_penalty = 22

    blur_penalty = 0
    if timing_penalty > 0:
        if blur_ratio >= 0.75:
            blur_penalty = 45
        elif blur_ratio >= 0.50:
            blur_penalty = 25
        elif blur_ratio >= 0.35:
            blur_penalty = 12
        if blur_ratio >= 0.65 and timing_cv <= 0.20:
            blur_penalty += 15

    penalty = timing_penalty + blur_penalty

    return {
        "penalty": penalty,
        "blur_ratio": blur_ratio,
        "timing_cv": timing_cv,
        "mean_time_ms": mean_time_ms,
    }


async def evaluate_game_telemetry(db: AsyncSession, game_id: int, force: bool = False) -> dict[int, dict]:
    """
    Анализирует телеметрию завершённой партии для обоих игроков,
    снижает user.trust_factor при аномалиях и выставляет is_cheater = True при trust_factor < 30.
    Идемпотентна по game_id (повторный вызов для той же партии не штрафует игрока дважды).
    """
    game = await db.get(Game, game_id)
    if not game:
        return {}

    result = await db.execute(
        select(Move).where(Move.game_id == game_id).order_by(Move.id.asc())
    )
    all_moves = list(result.scalars().all())
    if not all_moves:
        return {}

    already_evaluated = not force and not await _claim_first_evaluation(game_id)

    moves_by_player: dict[int, list[Move]] = {
        game.white_player_id: [],
        game.black_player_id: [],
    }
    for move in all_moves:
        moves_by_player.setdefault(move.player_id, []).append(move)

    report: dict[int, dict] = {}

    for player_id, player_moves in moves_by_player.items():
        metrics = compute_telemetry_penalty(player_moves)
        penalty = 0 if already_evaluated else metrics["penalty"]

        user = await db.get(User, player_id)
        if not user:
            continue

        if penalty > 0:
            user.trust_factor = max(0, int(user.trust_factor) - penalty)
            if user.trust_factor < CHEATER_TRUST_THRESHOLD:
                user.is_cheater = True
                logger.warning(
                    "Fair Play Alert: user %s (id=%s) flagged in game %s (trust_factor=%s, blur=%.2f, cv=%.2f)",
                    user.username,
                    user.id,
                    game_id,
                    user.trust_factor,
                    metrics["blur_ratio"],
                    metrics["timing_cv"],
                )

        report[player_id] = {
            **metrics,
            "trust_factor": user.trust_factor,
            "is_cheater": user.is_cheater,
        }

    await db.commit()
    return report

def evaluate_anderssen_style_bonus(board_before: chess.Board, move: chess.Move) -> tuple[float, str | None]:
    piece = board_before.piece_at(move.from_square)
    if not piece:
        return 0.0, None

    bonus = 0.0
    note = None

    if board_before.gives_check(move):
        bonus += 36.0
        note = "Шах! В духе «Бессмертной партии» — ни одного темпа передышки королю соперника!"

    if board_before.is_capture(move):
        bonus += 24.0
        if not note:
            note = "Вскрываем линии и устраняем защитные редуты неприятеля!"

    if board_before.is_castling(move):
        bonus += 38.0
        note = "Рокировка! Подключаем ладью к решающему штурму."

    if piece.piece_type in {chess.KNIGHT, chess.BISHOP} and board_before.fullmove_number <= 14:
        back_rank = 0 if board_before.turn == chess.WHITE else 7
        if chess.square_rank(move.from_square) == back_rank:
            bonus += 28.0
            if not note:
                note = "Мобилизуем лёгкие фигуры для комбинационного удара!"
        if move.to_square in EXTENDED_CENTER:
            bonus += 18.0

    if move.to_square in CENTER_SQUARES:
        bonus += 22.0
        if not note:
            note = "Борьба за инициативу и центральные диагонали!"

    return bonus, note
