"""
Правила, доделанные после аудита: итог партии с ботом определяет сервер, имена ботов

FOOLS_MATE = ["f2f3", "e7e5", "g2g4", "d8h4"]


def _payload(**kwargs) -> BotGameSaveRequest:
    base = {"bot_id": "morphy", "player_color": "white", "result": "1-0", "moves_uci": ["e2e4"]}
    return BotGameSaveRequest(**{**base, **kwargs})


def _board(moves: list[str]) -> chess.Board:
    board = chess.Board()
    for mv in moves:
        board.push_uci(mv)
    return board

def _receive_event(ws, event: str, attempts: int = 6) -> dict:
    for _ in range(attempts):
        message = ws.receive_json()
        if message.get("event") == event:
            return message
    raise AssertionError(f"event {event} not received")

"""
Разрыв соединения: часы при переподключении, флаг и брошенная партия у отключённого игрока,
несколько вкладок одного пользователя и гонка «перезагрузка страницы».
"""

import time

from src.services.connection_manager import ConnectionManager
from src.services.game_state import ABANDON_TIMEOUT_MS, evaluate_clock_expiry, state_manager


class DummyManager:
    def __init__(self):
        self.sent: list[tuple[int, dict]] = []

    async def send_personal_message(self, message: dict, user_id: int):
        self.sent.append((user_id, message))


class DummySocket:
    async def accept(self):
        pass

    async def send_text(self, text: str):
        pass


def _state(now: int, **overrides) -> dict:
    state = {
        "white_id": 1, "black_id": 2, "turn": "white",
        "white_time_ms": 60000, "black_time_ms": 60000, "last_move_at": now,
        "white_disconnected": "false", "black_disconnected": "false",
        "white_disconnect_started_at": 0, "black_disconnect_started_at": 0,
    }
    state.update(overrides)
    return state


async def test_reconnect_during_opponent_turn_does_not_give_opponent_time():
    """Раньше last_move_at сдвигался на время офлайна, и часы соперника стояли всё это время."""
    await state_manager.initialize_game(game_id=31, white_id=10, black_id=20, base_time_ms=60000, increment_ms=0)
    now = int(time.time() * 1000)
    # Ход белых; чёрные отключились 20 секунд назад.
    await state_manager.update_game_state(31, {
        "last_move_at": now - 30000,
        "black_disconnected": "true",
        "black_disconnect_started_at": now - 20000,
    })

    await state_manager.handle_player_reconnect(20, DummyManager())

    state = await state_manager.get_game_state(31)
    assert state["black_disconnected"] == "false"
    assert state["last_move_at"] == now - 30000
    assert state["white_time_ms"] == 60000


def test_disconnected_player_flags_counting_time_before_disconnect():
    """Время между последним ходом и разрывом тоже идёт в зачёт (раньше терялось)."""
    now = int(time.time() * 1000)
    state = _state(
        now,
        white_time_ms=8000, last_move_at=now - 10000,
        white_disconnected="true", white_disconnect_started_at=now - 3000,
    )
    assert evaluate_clock_expiry(state, now) == ("0-1", 2, "timeout")


def test_abandon_timer_ignores_stopped_clock_on_opponent_turn():
    """
    Если сейчас ход соперника, часы отключённого стоят: при 5 секундах на часах он не должен
    проигрывать через 5 секунд офлайна — только по общему таймеру брошенной партии.
    """
    now = int(time.time() * 1000)
    state = _state(now, black_time_ms=5000, black_disconnected="true", black_disconnect_started_at=now - 10000)
    assert evaluate_clock_expiry(state, now) is None

    abandoned = {**state, "black_disconnect_started_at": now - ABANDON_TIMEOUT_MS}
    assert evaluate_clock_expiry(abandoned, now) == ("1-0", 1, "abandoned")

@needs_stockfish
def test_pool_reuses_process_and_limits_concurrency():
    pool = EnginePool(SF_PATH, size=1, acquire_timeout_s=0.2)
    try:
        with pool.acquire() as first:
            pass
        with pool.acquire() as second:
            assert second is first  # процесс переиспользован, а не запущен заново

            errors = []

            def _try_acquire():
                try:
                    with pool.acquire():
                        pass
                except EngineBusyError as exc:
                    errors.append(exc)

            t = threading.Thread(target=_try_acquire)
            t.start()
            t.join()
            assert len(errors) == 1
    finally:
        pool.close()


@needs_stockfish
def test_pool_replaces_dead_engine():
    pool = EnginePool(SF_PATH, size=1, acquire_timeout_s=1.0)
    try:
        with pool.acquire() as engine:
            engine.quit()  # процесс завершился, а вызывающий код ошибку «проглотил»
        with pool.acquire() as fresh:
            assert fresh is not engine
            info = fresh.analyse(chess.Board(), chess.engine.Limit(depth=1))
            assert "score" in info
    finally:
        pool.close()

def test_checkmate_detection():

    fen = "r1bqkbnr/pppp1ppp/2n5/4p3/2B1P3/5Q2/PPPP1PPP/RNB1K1NR w KQkq - 4 3"
    is_valid, data = engine.process_move(fen, "f3f7")

    assert is_valid is True
    assert data["is_game_over"] is True
    assert data["result"] == "1-0"
    assert data["termination"] == "checkmate"


def test_underpromotion_to_knight_and_rook():
    """Проверка превращения пешки в коня (a7a8n) и ладью (a7a8r) с корректной записью SAN."""
    promotion_fen = "4k3/P7/8/8/8/8/8/4K3 w - - 0 1"

    is_valid_n, res_knight = engine.process_move(promotion_fen, "a7a8n")
    assert is_valid_n is True
    assert res_knight["uci"] == "a7a8n"
    assert res_knight["san"].startswith("a8=N")

    is_valid_r, res_rook = engine.process_move(promotion_fen, "a7a8r")
    assert is_valid_r is True
    assert res_rook["uci"] == "a7a8r"
    assert res_rook["san"].startswith("a8=R")


def _play(moves: list[str]) -> list[dict]:
    """Проигрывает ходы через engine.process_move, передавая FEN и PGN, как это делает сервер."""
    fen, pgn, results = STARTING_FEN, "", []
    for mv in moves:
        is_valid, data = engine.process_move(fen, mv, pgn)
        assert is_valid, f"move {mv} rejected"
        fen, pgn = data["new_fen"], data["new_pgn"]
        results.append(data)
    return results


def test_threefold_repetition_ends_game():
    """Третье повторение позиции завершает партию ничьей (раньше не детектировалось: доска строилась из FEN)."""
    dance = ["g1f3", "g8f6", "f3g1", "f6g8"]
    results = _play(dance * 2)

    assert all(not r["is_game_over"] for r in results[:-1])
    last = results[-1]
    assert last["is_game_over"] is True
    assert last["termination"] == "repetition"
    assert last["result"] == "1/2-1/2"


def test_repetition_not_ended_when_only_reachable():
    """Позиция, из которой повторение лишь достижимо следующим ходом, партию не завершает."""
    results = _play(["g1f3", "g8f6", "f3g1", "f6g8", "g1f3", "g8f6", "f3g1"])
    assert results[-1]["is_game_over"] is False


def test_inconsistent_pgn_falls_back_to_fen():
    """Если PGN не соответствует FEN, ход валидируется по FEN, как раньше."""
    is_valid, data = engine.process_move(STARTING_FEN, "e2e4", "1. d4 d5")
    assert is_valid is True
    assert data["san"] == "e4"
