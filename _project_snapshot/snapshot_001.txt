## СПРИНТ 5. Прагматичный Античит на базе уже собираемой телеметрии (ОЧЕРЕДЬ РАЗРАБОТКИ ⏳)

В таблице `moves` уже сохраняются `time_taken_ms`, `window_blurred_before_move`, `is_premove`, `network_lag_ms`, а у
пользователя есть `trust_factor` и `is_cheater`. Заставляем эти данные работать без нагрузки на CPU.

### - [ ] Таск 5.1. Детектор аномалий поведения (Timing Variance + Tab-Switch)

* **Техническая причина:** Читеры, использующие подсказки с соседней вкладки или второй программы, выдают себя двумя
  детерминированными паттернами:
    1. Регулярный уход из вкладки (`window_blurred_before_move == True`) перед сложными ходами в миттельшпиле.
    2. «Роботизированный тайминг» — почти одинаковое время на каждый ход (низкое среднеквадратичное отклонение времени
       хода, например, каждые 4–6 секунд вне зависимости от сложности позиции).
* **Ветка Git:** `feature/telemetry-anticheat-service`
* **Команда старта:**

  git checkout dev && git pull origin dev && git checkout -b feature/telemetry-anticheat-service

* **Что сделать:**
    1. Создать сервис `src/services/anticheat.py` с функцией `evaluate_game_telemetry(db, game_id)`.
    2. После завершения рейтинговой партии (в `state_manager.finish_game`) вызывать этот анализатор:
        * Считать долю ходов с потерей фокуса окна (`blur_ratio`) после 8-го хода.
        * Считать коэффициент вариации времени ходов (исключая `is_premove == True`).
        * При подозрительных показателях снижать `user.trust_factor` (со 100 вниз). Если `trust_factor < 30`, выставлять
          `is_cheater = True`.
    3. В профиле игрока (`Profile.tsx`) отображать бейдж репутации («Высокий фактор доверия» / «Под наблюдением Fair
       Play»).
* **Тестирование:**
    * Создать `tests/test_anticheat.py`: протестировать расчёт `trust_factor` на серии «человеческих» ходов (разное
      время, без ухода из вкладки) и «читерских» ходов (постоянный `window_blurred_before_move=True` и монотонное время
      4500 мс).

      python -m pytest tests/test_anticheat.py -v

* **Коммит:** `feat: implement deterministic anti-cheat telemetry analyzer and trust factor updates`

import asyncio
from logging.config import fileConfig

from sqlalchemy import pool
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import async_engine_from_config

from alembic import context

# Импорт конфигурации и базовой модели
from src.core.config import settings
from src.models import Base

config = context.config

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


def run_migrations_offline() -> None:
    url = settings.async_database_url
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata)

    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    configuration = config.get_section(config.config_ini_section, {})
    configuration["sqlalchemy.url"] = settings.async_database_url

    connectable = async_engine_from_config(
        configuration,
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)

    await connectable.dispose()


def run_migrations_online() -> None:
    asyncio.run(run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

        pgn_date = None
        site = None
        eco = None
        res_val = cur_result

        for m in DATE_RE.finditer(pgn_text):
            val = m.group(1).strip()
            if val and not val.startswith("?"):
                pgn_date = val[:20]
                break

        s_match = SITE_RE.search(pgn_text)
        if s_match and s_match.group(1).strip() != "?":
            site = s_match.group(1).strip()[:100]

        e_match = ECO_RE.search(pgn_text)
        if e_match and e_match.group(1).strip() != "?":
            eco = e_match.group(1).strip()[:10]

        if not res_val or res_val == "*":
            r_match = RESULT_RE.search(pgn_text)
            if r_match:
                res_val = r_match.group(1).strip()[:10]

        bind.execute(
            sa.text(
                "UPDATE games SET pgn_date = :d, site = :s, eco = :e, result = :r WHERE id = :id"
            ),
            {"d": pgn_date, "s": site, "e": eco, "r": res_val, "id": game_id},
        )

@router.post("/register", response_model=UserResponse, status_code=status.HTTP_201_CREATED)
async def register_user(user_in: UserCreate, db: AsyncSession = Depends(get_db)):
    """
    Register a new user.
    """
    result = await db.execute(select(User).where(User.username == user_in.username))
    existing_user = result.scalars().first()

    if existing_user:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Username already registered."
        )

    hashed_pwd = get_password_hash(user_in.password)

    new_user = User(
        username=user_in.username,
        hashed_password=hashed_pwd,
    )

    db.add(new_user)
    await db.commit()
    await db.refresh(new_user)

    return new_user


@router.post("/login")
async def login(
        form_data: OAuth2PasswordRequestForm = Depends(),
        db: AsyncSession = Depends(get_db)
):
    """
    OAuth2 compatible token login, get an access token for future requests.
    """
    # 1. Fetch user from database
    result = await db.execute(select(User).where(User.username == form_data.username))
    user = result.scalars().first()

    # 2. Verify existence and password
    if not user or not verify_password(form_data.password, user.hashed_password):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Incorrect username or password",
            headers={"WWW-Authenticate": "Bearer"},
        )

    # 3. Generate JWT
    access_token = create_access_token(data={"sub": user.username})

    return {
        "access_token": access_token,
        "token_type": "bearer"
    }

def calculate_authentic_thinking_time_ms(
        bot_id: str,
        board: chess.Board,
        uci_move: str,
        source: str,
) -> int:
    """
    Вычисляет реалистичное человеческое время на ход (в мс) для исторического мастера
    с учётом его темпа, дебютной книги, единственного ответа или тактической сложности позиции.
    """
    legal_moves = list(board.legal_moves)
    num_legal = len(legal_moves)
    tempo = BOT_TEMPO_MULTIPLIER.get(bot_id, 1.0)

    # Единственный легальный ход (например, выход из шаха) — почти мгновенно
    if num_legal <= 1:
        return int(random.randint(240, 380) * tempo)

    # Ход из дебютной книги — быстрое воспроизведение по памяти
    if source == "book":
        base_ms = random.randint(360, 680) + board.fullmove_number * 22
        return int(base_ms * tempo)

    move_obj = chess.Move.from_uci(uci_move)
    is_capture = board.is_capture(move_obj)
    gives_check = board.gives_check(move_obj)

    # Оцениваем напряжённость позиции (количество возможных взятий и разветвлений)
    captures_available = sum(1 for m in legal_moves if board.is_capture(m))
    complexity_ms = min(num_legal * 26, 950) + min(captures_available * 95, 550)

    if board.is_check():
        complexity_ms += 250
    if is_capture or gives_check:
        complexity_ms += 280

    # В глубоком эндшпиле (мало фигур) техническая реализация идёт быстрее
    piece_count = len(board.piece_map())
    if piece_count <= 10:
        complexity_ms = int(complexity_ms * 0.65)

    raw_ms = (520 + complexity_ms + random.randint(-110, 180)) * tempo
    return max(320, min(int(raw_ms), 2900))


def _detect_eco_code(moves_uci: list[str]) -> str:
    """Определяет базовый код дебюта ECO по первым ходам партии."""
    prefix = " ".join(moves_uci[:6])
    if prefix.startswith("e2e4 e7e5 g1f3 d7d6"):
        return "C41"
    if prefix.startswith("e2e4 e7e5 f2f4"):
        return "C30"
    if prefix.startswith("e2e4 e7e5 f1c4"):
        return "C23"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1c4 f8c5 b2b4"):
        return "C51"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1c4"):
        return "C50"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1b5"):
        return "C60"
    if prefix.startswith("e2e4 c7c5"):
        return "B20"
    if prefix.startswith("e2e4 e7e6"):
        return "C00"
    if prefix.startswith("c2c4"):
        return "A10"
    if prefix.startswith("g1f3"):
        return "A04"
    if prefix.startswith("d2d4 d7d5 c2c4"):
        return "D06"
    if prefix.startswith("d2d4"):
        return "D00"
    if prefix.startswith("e2e4"):
        return "B00"
    return "A00"


def _build_pgn_string(
        moves_uci: list[str],
        white_name: str,
        white_elo: int,
        black_name: str,
        black_elo: int,
        result: str,
) -> tuple[str, str]:
    """Строит валидный PGN-документ с заголовками и возвращает (pgn_text, final_fen)."""
    board = chess.Board()
    pgn_game = chess.pgn.Game()
    pgn_game.headers["Event"] = "Historical Bot Match"
    pgn_game.headers["Site"] = "Michess Bot Arena"
    pgn_game.headers["Date"] = datetime.now(timezone.utc).strftime("%Y.%m.%d")
    pgn_game.headers["Round"] = "-"
    pgn_game.headers["White"] = white_name
    pgn_game.headers["Black"] = black_name
    pgn_game.headers["Result"] = result
    pgn_game.headers["WhiteElo"] = str(white_elo)
    pgn_game.headers["BlackElo"] = str(black_elo)
    pgn_game.headers["ECO"] = _detect_eco_code(moves_uci)

    node = pgn_game
    for uci_str in moves_uci:
        move = chess.Move.from_uci(uci_str)
        if move not in board.legal_moves:
            raise ValueError(f"Нелегальный ход в истории: {uci_str}")
        board.push(move)
        node = node.add_variation(move)

    exporter = chess.pgn.StringExporter(headers=True, variations=False, comments=False)
    return pgn_game.accept(exporter).strip(), board.fen()


async def _get_or_create_bot_user(db: AsyncSession, bot_id: str) -> User:

    res = await db.execute(stmt)

    if bot_user is not None:
        return bot_user

    user_cols = {c.name for c in User.__table__.columns}
    user_kwargs: dict = {
        "username": bot_username,

    if "hashed_password" in user_cols:
        user_kwargs["hashed_password"] = "!historical_bot_no_login!"
    elif "password_hash" in user_cols:
        user_kwargs["password_hash"] = "!historical_bot_no_login!"

    if "is_cheater" in user_cols:
        user_kwargs["is_cheater"] = False
    if "trust_factor" in user_cols:
        user_kwargs["trust_factor"] = 100

    bot_user = User(**user_kwargs)
    db.add(bot_user)
    await db.flush()
    return bot_user


def _prepare_game_kwargs(
        white_id: int,
        black_id: int,
        pgn_text: str,
        final_fen: str,
        payload: BotGameSaveRequest,
) -> dict:
    """Формирует словарь полей модели Game в соответствии со схемой БД."""
    game_cols = {c.name: c for c in Game.__table__.columns}
    kwargs: dict = {}

    if "pgn_date" in game_cols:
        kwargs["pgn_date"] = datetime.now(timezone.utc).strftime("%Y.%m.%d")
    if "site" in game_cols:
        kwargs["site"] = "Michess Bot Arena"
    if "eco" in game_cols:
        kwargs["eco"] = _detect_eco_code(payload.moves_uci)
        
    if "white_player_id" in game_cols:
        kwargs["white_player_id"] = white_id
    elif "white_id" in game_cols:
        kwargs["white_id"] = white_id

    if "black_player_id" in game_cols:
        kwargs["black_player_id"] = black_id
    elif "black_id" in game_cols:
        kwargs["black_id"] = black_id

    if "pgn" in game_cols:
        kwargs["pgn"] = pgn_text
    if "board_fen" in game_cols:
        kwargs["board_fen"] = final_fen
    elif "fen" in game_cols:
        kwargs["fen"] = final_fen
    elif "current_fen" in game_cols:
        kwargs["current_fen"] = final_fen

    if "result" in game_cols:
        kwargs["result"] = payload.result
    if "termination" in game_cols:
        kwargs["termination"] = "normal"
    if "is_rated" in game_cols:
        kwargs["is_rated"] = False
    if "base_time_ms" in game_cols:
        kwargs["base_time_ms"] = payload.base_time_ms
    if "increment_ms" in game_cols:
        kwargs["increment_ms"] = payload.increment_ms

    if "status" in game_cols:
        col_type = game_cols["status"].type
        enums = getattr(col_type, "enums", None)
        if enums and "COMPLETED" in enums:
            kwargs["status"] = "COMPLETED"
        elif enums and "FINISHED" in enums:
            kwargs["status"] = "FINISHED"
        else:
            kwargs["status"] = "COMPLETED"

    if "finished_at" in game_cols:
        is_tz = getattr(game_cols["finished_at"].type, "timezone", False)
        now_utc = datetime.now(timezone.utc)
        kwargs["finished_at"] = now_utc if is_tz else now_utc.replace(tzinfo=None)

    return kwargs

@router.post("/move", response_model=BotMoveResponse)
async def get_bot_move(payload: BotMoveRequest):
    """Принимает FEN позиции и возвращает ответный ход выбранного исторического бота."""
    bot = BOT_INSTANCES.get(payload.bot_id)
    if bot is None:
        raise HTTPException(status_code=404, detail=f"Бот '{payload.bot_id}' не найден")

    try:
        board = chess.Board(payload.fen)
    except ValueError:
        raise HTTPException(status_code=400, detail="Некорректный формат FEN")

    if board.is_game_over():
        raise HTTPException(status_code=400, detail="Партия уже завершена")

    result = await asyncio.to_thread(bot.choose_move, payload.fen)
    thinking_time_ms = calculate_authentic_thinking_time_ms(
        bot_id=payload.bot_id,
        board=board,
        uci_move=result["uci"],
        source=result["source"],
    )

    return BotMoveResponse(
        bot_id=payload.bot_id,
        uci=result["uci"],
        san=result["san"],
        fen_after=result["fen_after"],
        source=result["source"],
        comment=result["comment"],
        thinking_time_ms=thinking_time_ms,
    )


@router.post("/save-game", response_model=BotGameSaveResponse)
async def save_bot_game(
        payload: BotGameSaveRequest,
        db: AsyncSession = Depends(get_db),
        current_user: User = Depends(get_current_user),
):
    """Сохраняет завершённую партию против исторического бота в базу данных."""
    if payload.bot_id not in BOT_METADATA:
        raise HTTPException(status_code=404, detail=f"Бот '{payload.bot_id}' не найден")

    if payload.player_color not in ("white", "black"):
        raise HTTPException(status_code=400, detail="Некорректный цвет игрока")

    if payload.result not in ("1-0", "0-1", "1/2-1/2"):
        raise HTTPException(status_code=400, detail="Некорректный результат партии")

    if not payload.moves_uci:
        raise HTTPException(status_code=400, detail="Список ходов пуст")

    bot_user = await _get_or_create_bot_user(db, payload.bot_id)

    if payload.player_color == "white":
        white_user, black_user = current_user, bot_user
    else:
        white_user, black_user = bot_user, current_user

    try:
        pgn_text, final_fen = _build_pgn_string(
            moves_uci=payload.moves_uci,
            white_name=white_user.username,
            white_elo=white_user.rating,
            black_name=black_user.username,
            black_elo=black_user.rating,
            result=payload.result,
        )
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc))

    game_kwargs = _prepare_game_kwargs(
        white_id=white_user.id,
        black_id=black_user.id,
        pgn_text=pgn_text,
        final_fen=final_fen,
        payload=payload,
    )

    new_game = Game(**game_kwargs)
    db.add(new_game)
    await db.commit()
    await db.refresh(new_game)

    return BotGameSaveResponse(
        game_id=new_game.id,
        status="finished",
        result=payload.result,
    )

from fastapi import Depends, Query, WebSocketException, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
import jwt

from src.core.config import settings
from src.db.session import get_db
from src.models.user import User

# Схема для извлечения токена из заголовка Authorization: Bearer
oauth2_scheme = OAuth2PasswordBearer(tokenUrl=f"{settings.API_V1_STR}/auth/login")


async def get_current_user(
        token: str = Depends(oauth2_scheme),
        db: AsyncSession = Depends(get_db)
) -> User:
    """
    Dependency for HTTP endpoints to authenticate users via Bearer token in headers.
    """
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
        username: str | None = payload.get("sub")
        if username is None:
            raise credentials_exception
    except jwt.InvalidTokenError:
        raise credentials_exception

    result = await db.execute(select(User).where(User.username == username))
    user = result.scalars().first()

    if user is None:
        raise credentials_exception

    return user


async def get_current_user_ws(
        token: str = Query(..., description="JWT access token passed as a query parameter"),
        db: AsyncSession = Depends(get_db)
) -> User:
    """
    Dependency for WebSocket endpoints to authenticate users via query parameter token.
    """
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
        username: str | None = payload.get("sub")
        if username is None:
            raise WebSocketException(code=status.WS_1008_POLICY_VIOLATION, reason="Invalid token payload")
    except jwt.InvalidTokenError:
        raise WebSocketException(code=status.WS_1008_POLICY_VIOLATION, reason="Invalid or expired token")

    result = await db.execute(select(User).where(User.username == username))
    user = result.scalars().first()

    if user is None:
        raise WebSocketException(code=status.WS_1008_POLICY_VIOLATION, reason="User not found")

    return user
