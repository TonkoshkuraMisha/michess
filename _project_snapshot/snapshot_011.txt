// --- FILE: frontend/src/components/Profile.tsx ---

import {useState, useEffect} from 'react';
import {useParams, Link, useNavigate} from 'react-router-dom';
import {format, parseISO} from 'date-fns';
import {ru} from 'date-fns/locale';
import {
    LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer
} from 'recharts';
import {
    Trophy,
    Swords,
    Calendar,
    Download,
    Crosshair,
    Target,
    ShieldCheck,
    ShieldAlert
} from 'lucide-react';
import {API_BASE_URL} from '../config/api';

interface UserProfile {
    id: number;
    username: string;
    rating: number;
    is_cheater?: boolean;
    trust_factor?: number;
    created_at: string;
}

interface GameData {
    game_id: number;
    white_username: string;
    white_rating: number;
    black_username: string;
    black_rating: number;
    status: string;
    is_rated: boolean;
    result: string;
    pgn_date: string | null;
    site: string | null;
    eco: string | null;
    finished_at: string | null;
}

const displayDate = (game: GameData) => {
    if (game.pgn_date && !game.pgn_date.includes('????')) {
        return game.pgn_date.replace(/\??/g, '').replace(/\.$/, '');
    }
    if (game.finished_at) {
        return format(parseISO(game.finished_at), 'd MMM yyyy', {locale: ru});
    }
    return '—';
};

export default function Profile() {
    const {targetUsername} = useParams<{ targetUsername: string }>();
    const navigate = useNavigate();
    const [profile, setProfile] = useState<UserProfile | null>(null);
    const [games, setGames] = useState<GameData[]>([]);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState<string | null>(null);

    useEffect(() => {
        const fetchProfileData = async () => {
            setLoading(true);
            setError(null);
            try {
                const profileRes = await fetch(`${API_BASE_URL}/api/v1/profile/${targetUsername}`);
                if (!profileRes.ok) throw new Error('Пользователь не найден');
                const profileData = await profileRes.json();
                setProfile(profileData);

                const gamesRes = await fetch(`${API_BASE_URL}/api/v1/profile/${targetUsername}/games?limit=30`);
                if (gamesRes.ok) {
                    const gamesData = await gamesRes.json();
                    setGames(gamesData.items || []);
                }
            } catch (err: any) {
                setError(err.message);
            } finally {
                setLoading(false);
            }
        };

        if (targetUsername) {
            fetchProfileData();
        }
    }, [targetUsername]);

    if (loading) {
        return (
            <div className="flex flex-col items-center justify-center w-full h-full text-boxwood-light font-serif">
                <div className="animate-spin text-amber-500 mb-4"><Swords size={40}/></div>
                <div>Загрузка профиля...</div>
            </div>
        );
    }

    if (error || !profile) {
        return (
            <div className="bg-red-950/80 border border-red-800 p-6 rounded text-red-200 text-center font-serif">
                <h2 className="text-xl font-bold mb-2">Ошибка</h2>
                <p>{error || 'Не удалось загрузить профиль'}</p>
                <Link to="/" className="mt-4 inline-block text-amber-400 hover:underline">На главную</Link>
            </div>
        );
    }

    let wins = 0, losses = 0, draws = 0;
    let tempRating = profile.rating;

    const chartDataReversed = games.map((game) => {
        const isWhite = game.white_username === profile.username;
        let delta = 0;

        if (game.result === '1-0') {
            if (isWhite) {
                wins++;
                delta = 15;
            } else {
                losses++;
                delta = -15;
            }
        } else if (game.result === '0-1') {
            if (isWhite) {
                losses++;
                delta = -15;
            } else {
                wins++;
                delta = 15;
            }
        } else if (game.result === '1/2-1/2') {
            draws++;
        }

        const pointRating = tempRating;
        tempRating -= delta;

        const chartDate = game.pgn_date ? game.pgn_date.split('.')[0] : (game.finished_at ? format(parseISO(game.finished_at), 'd MMM', {locale: ru}) : '');

        return {
            name: 'Партия',
            rating: pointRating,
            date: chartDate
        };
    });

    const chartData = chartDataReversed.reverse();
    const winRate = games.length > 0 ? Math.round((wins / games.length) * 100) : 0;
    const trustFactor = profile.trust_factor ?? 100;
    const isUnderFairPlayWatch = Boolean(profile.is_cheater) || trustFactor < 60;

    return (
        <div className="w-full max-w-6xl flex flex-col gap-6 font-serif">
            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
                <div
                    className="bg-acacia-dark border border-amber-900/40 p-6 rounded shadow-heavy flex flex-col items-center justify-center text-center relative overflow-hidden">
                    <img
                        src={`https://api.dicebear.com/9.x/bottts-neutral/svg?seed=${profile.username}&backgroundColor=D0A972`}
                        alt="Avatar Background"
                        className="absolute top-0 right-0 w-48 h-48 opacity-10 pointer-events-none translate-x-10 -translate-y-10"
                    />
                    <img
                        src={`https://api.dicebear.com/9.x/bottts-neutral/svg?seed=${profile.username}&backgroundColor=E6C998`}
                        alt="Avatar"
                        className="w-24 h-24 rounded-lg bg-black/40 shadow-inner border-2 border-amber-600/40 mb-3 z-10"
                    />
                    <h1 className="text-3xl font-bold text-amber-200 z-10">{profile.username}</h1>

                    {/* Бейдж репутации Fair Play (Таск 5.1) */}
                    <div className="mt-2.5 z-10">
                        {isUnderFairPlayWatch ? (
                            <span
                                data-testid="reputation-badge"
                                className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-sans font-bold bg-rose-950/85 border border-rose-600/60 text-rose-200 shadow"
                                title={`Фактор доверия: ${trustFactor}%`}
                            >
                                <ShieldAlert size={14} className="text-rose-400"/>
                                Под наблюдением Fair Play
                            </span>
                        ) : (
                            <span
                                data-testid="reputation-badge"
                                className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-sans font-bold bg-emerald-950/85 border border-emerald-600/50 text-emerald-200 shadow"
                                title={`Фактор доверия: ${trustFactor}%`}
                            >
                                <ShieldCheck size={14} className="text-emerald-400"/>
                                Высокий фактор доверия
                            </span>
                        )}
                    </div>

                    <div className="flex items-center gap-2 mt-3 text-boxwood-light/80 text-sm z-10">
                        <Calendar size={16} className="text-amber-600"/>
                        На сайте с {format(parseISO(profile.created_at), 'd MMMM yyyy', {locale: ru})}
                    </div>
                </div>

                <div className="md:col-span-2 grid grid-cols-2 lg:grid-cols-4 gap-4">
                    <div
                        className="bg-[#1A120B] border border-amber-900/60 p-5 rounded shadow-inner flex flex-col justify-center">
                        <div
                            className="text-boxwood/60 text-xs uppercase tracking-widest mb-1 flex items-center gap-1.5">
                            <Trophy size={14}/> Рейтинг
                        </div>
                        <div className="text-4xl font-mono font-bold text-amber-400">{profile.rating}</div>
                    </div>
                    <div
                        className="bg-[#1A120B] border border-amber-900/60 p-5 rounded shadow-inner flex flex-col justify-center">
                        <div
                            className="text-boxwood/60 text-xs uppercase tracking-widest mb-1 flex items-center gap-1.5">
                            <Swords size={14}/> Игр сыграно
                        </div>
                        <div className="text-4xl font-mono font-bold text-blue-400">{games.length}</div>
                    </div>
                    <div
                        className="bg-[#1A120B] border border-amber-900/60 p-5 rounded shadow-inner flex flex-col justify-center">
                        <div
                            className="text-boxwood/60 text-xs uppercase tracking-widest mb-1 flex items-center gap-1.5">
                            <Target size={14}/> Винрейт
                        </div>
                        <div className="text-4xl font-mono font-bold text-emerald-400">{winRate}%</div>
                    </div>
                    <div
                        className="bg-[#1A120B] border border-amber-900/60 p-5 rounded shadow-inner flex flex-col justify-center">
                        <div
                            className="text-boxwood/60 text-[10px] xl:text-xs uppercase tracking-wider mb-1 flex items-center gap-1.5 whitespace-nowrap">
                            <Crosshair size={14}/> WIN / DRAW / LOSE
                        </div>
                        <div className="text-2xl font-mono font-bold flex gap-2">
                            <span className="text-emerald-400">{wins}</span>
                            <span className="text-boxwood/40">/</span>
                            <span className="text-gray-400">{draws}</span>
                            <span className="text-boxwood/40">/</span>
                            <span className="text-red-400">{losses}</span>
                        </div>
                    </div>
                </div>
            </div>

            <div className="bg-acacia-dark border border-amber-900/40 p-5 rounded shadow-heavy">
                <h2 className="text-sm uppercase tracking-widest text-amber-500/80 mb-6 font-bold">Динамика рейтинга
                    (последние партии)</h2>
                <div className="h-64 w-full font-sans">
                    {chartData.length > 1 ? (
                        <ResponsiveContainer width="100%" height="100%">
                            <LineChart data={chartData}>
                                <CartesianGrid strokeDasharray="3 3" stroke="#5C3A21" vertical={false}/>
                                <XAxis dataKey="date" stroke="#8B5A2B" tick={{fill: '#D0A972', fontSize: 12}} dy={10}/>
                                <YAxis domain={['dataMin - 30', 'dataMax + 30']} stroke="#8B5A2B"
                                       tick={{fill: '#D0A972', fontSize: 12}} dx={-10}/>
                                <Tooltip contentStyle={{
                                    backgroundColor: '#1A120B',
                                    borderColor: '#7A4B31',
                                    color: '#F5DEB3',
                                    borderRadius: '8px'
                                }} itemStyle={{color: '#fbbf24', fontWeight: 'bold'}}/>
                                <Line type="monotone" dataKey="rating" stroke="#fbbf24" strokeWidth={3}
                                      dot={{r: 4, fill: '#1A120B', stroke: '#fbbf24', strokeWidth: 2}}
                                      activeDot={{r: 6, fill: '#fbbf24'}}/>
                            </LineChart>
                        </ResponsiveContainer>
                    ) : (
                        <div className="h-full flex items-center justify-center text-boxwood/50">
                            Недостаточно данных для графика
                        </div>
                    )}
                </div>
            </div>

            <div className="bg-acacia-dark border border-amber-900/40 rounded shadow-heavy overflow-hidden">
                <div className="p-5 border-b border-amber-900/40 bg-black/20">
                    <h2 className="text-sm uppercase tracking-widest text-amber-500/80 font-bold">История партий</h2>
                </div>

                <div className="overflow-x-auto">
                    <table className="w-full text-left text-sm font-sans whitespace-nowrap">
                        <thead className="bg-black/40 text-boxwood/60 uppercase text-xs tracking-wider">
                        <tr>
                            <th className="px-4 py-4 font-medium">Белые</th>
                            <th className="px-4 py-4 font-medium">Черные</th>
                            <th className="px-4 py-4 font-medium text-center">Результат</th>
                            <th className="px-4 py-4 font-medium text-center">Дебют</th>
                            <th className="px-4 py-4 font-medium text-center">Место</th>
                            <th className="px-4 py-4 font-medium text-center">Дата</th>
                            <th className="px-4 py-4 font-medium text-right">PGN</th>
                        </tr>
                        </thead>
                        <tbody className="divide-y divide-amber-900/20 text-boxwood-light">
                        {games.length === 0 ? (
                            <tr>
                                <td colSpan={7} className="px-6 py-8 text-center text-boxwood/50 font-serif">
                                    Партий пока нет.
                                </td>
                            </tr>
                        ) : (
                            games.map((game) => {
                                const isWhite = game.white_username === profile.username;
                                let resultColor = 'text-gray-400';
                                let resultText = 'Ничья';

                                if (game.result === '1-0') {
                                    resultColor = isWhite ? 'text-emerald-400' : 'text-red-400';
                                    resultText = isWhite ? 'Победа' : 'Поражение';
                                } else if (game.result === '0-1') {
                                    resultColor = isWhite ? 'text-red-400' : 'text-emerald-400';
                                    resultText = isWhite ? 'Поражение' : 'Победа';
                                }

                                return (
                                    <tr
                                        key={game.game_id}
                                        onClick={() => navigate(`/game/${game.game_id}`, {state: {game}})}
                                        className="hover:bg-black/30 transition group cursor-pointer"
                                        title="Нажмите, чтобы просмотреть партию"
                                    >
                                        <td className="px-4 py-4">
                                            <div className="flex items-center gap-2">
                                                <span
                                                    className="w-3 h-3 rounded-sm border border-gray-400 bg-white inline-block"></span>
                                                <Link
                                                    to={`/profile/${game.white_username}`}
                                                    onClick={(e) => e.stopPropagation()}
                                                    className={`font-bold hover:underline ${isWhite ? 'text-amber-300' : ''}`}
                                                >
                                                    {game.white_username}
                                                </Link>
                                                <span
                                                    className="text-xs font-mono text-boxwood/50">({game.white_rating})</span>
                                            </div>
                                        </td>
                                        <td className="px-4 py-4">
                                            <div className="flex items-center gap-2">
                                                <span
                                                    className="w-3 h-3 rounded-sm border border-gray-600 bg-black inline-block"></span>
                                                <Link
                                                    to={`/profile/${game.black_username}`}
                                                    onClick={(e) => e.stopPropagation()}
                                                    className={`font-bold hover:underline ${!isWhite ? 'text-amber-300' : ''}`}
                                                >
                                                    {game.black_username}
                                                </Link>
                                                <span
                                                    className="text-xs font-mono text-boxwood/50">({game.black_rating})</span>
                                            </div>
                                        </td>
                                        <td className="px-4 py-4 text-center">
                                            <div className={`font-bold font-serif ${resultColor}`}>
                                                {resultText} <span
                                                className="text-xs opacity-70 ml-1">({game.result})</span>
                                            </div>
                                        </td>
                                        <td className="px-4 py-4 text-center font-mono text-amber-500/80">
                                            {game.eco || '—'}
                                        </td>
                                        <td className="px-4 py-4 text-center text-boxwood/70">
                                            {game.site || '—'}
                                        </td>
                                        <td className="px-4 py-4 text-center text-boxwood/70 font-mono">
                                            {displayDate(game)}
                                        </td>
                                        <td className="px-4 py-4 text-right">
                                            <a
                                                href={`${API_BASE_URL}/api/v1/profile/games/${game.game_id}/pgn`}
                                                download
                                                onClick={(e) => e.stopPropagation()}
                                                className="inline-flex items-center gap-1.5 px-3 py-1.5 bg-black/30 hover:bg-amber-900/60 text-amber-400 rounded transition border border-amber-900/40 text-xs font-bold"
                                                title="Скачать PGN"
                                            >
                                                <Download size={14}/>
                                            </a>
                                        </td>
                                    </tr>
                                );
                            })
                        )}
                        </tbody>
                    </table>
                </div>
            </div>
        </div>
    );
}

export function getSavedBoardThemeKey(): string {
    if (typeof window !== 'undefined') {
        const saved = localStorage.getItem('michess_board_theme');
        if (saved && BOARD_THEMES[saved]) return saved;
    }
    return 'walnut';
}

export function getSavedPieceSetKey(): string {
    if (typeof window !== 'undefined') {
        const saved = localStorage.getItem('michess_piece_set');
        if (saved && PIECE_SETS[saved]) return saved;
    }
    return 'merida';
}

export function getPieceSvgUrl(pieceSetKey: string, color: string, type: string): string {
    const setConfig = PIECE_SETS[pieceSetKey] || PIECE_SETS.merida;
    return `/assets/pieces/${setConfig.folder}/${color}${type.toUpperCase()}.svg`;
}

export function getHarmoniousPieceStyle(
    pieceSetKey: string,
    boardThemeKey: string,
    pieceColor: string
): { width: string; height: string; filter: string } {
    const setConfig = PIECE_SETS[pieceSetKey] || PIECE_SETS.merida;
    const theme = BOARD_THEMES[boardThemeKey] || BOARD_THEMES.walnut;
    const isWhite = pieceColor === 'w';
    const size = `${setConfig.scalePercent}%`;

    if (setConfig.id === 'cardinal') {
        return {
            width: size,
            height: size,
            filter: 'drop-shadow(0 2px 3px rgba(0, 0, 0, 0.45))'
        };
    }

    if (theme.id === 'walnut' || theme.id === 'maple') {
        return {
            width: size,
            height: size,
            filter: isWhite
                ? 'sepia(0.08) brightness(1.01) drop-shadow(0 2px 2.5px rgba(0, 0, 0, 0.55))'
                : 'drop-shadow(0 0 1px rgba(255, 248, 225, 0.55)) drop-shadow(0 2.5px 3px rgba(0, 0, 0, 0.6))'
        };
    }

    return {
        width: size,
        height: size,
        filter: isWhite
            ? 'drop-shadow(0 2px 2.5px rgba(0, 0, 0, 0.5))'
            : 'drop-shadow(0 0 1px rgba(255, 255, 255, 0.5)) drop-shadow(0 2.5px 3px rgba(0, 0, 0, 0.55))'
    };
}

// --- FILE: frontend/src/services/gameSocket.ts ---

import {WS_BASE_URL} from '../config/api';

type EventCallback = (data: any) => void;

class GameWebSocketService {
    private ws: WebSocket | null = null;
    private listeners: { [event: string]: EventCallback[] } = {};

    connect(token: string) {
        if (this.ws && this.ws.readyState === WebSocket.OPEN) return;

        const wsUrl = `${WS_BASE_URL}/api/v1/ws/matchmaking?token=${token}`;
        this.ws = new WebSocket(wsUrl);

        this.ws.onopen = () => {
            console.log('🔗 WebSocket соединение установлено');
        };

        this.ws.onmessage = (event) => {
            try {
                const data = JSON.parse(event.data);
                console.log('📥 WS получено сообщение:', data);
                const eventType = data.event;

                if (this.listeners[eventType]) {
                    this.listeners[eventType].forEach(cb => cb(data));
                }
                if (this.listeners['*']) {
                    this.listeners['*'].forEach(cb => cb(data));
                }
            } catch (e) {
                console.error('Ошибка парсинга JSON от WebSocket:', e);
            }
        };

        this.ws.onclose = () => {
            console.log('🔌 WebSocket соединение закрыто');
        };

        this.ws.onerror = (error) => {
            console.error('❌ WebSocket ошибка:', error);
        };
    }

    on(event: string, callback: EventCallback) {
        if (!this.listeners[event]) {
            this.listeners[event] = [];
        }
        this.listeners[event].push(callback);
    }

    off(event: string, callback: EventCallback) {
        if (!this.listeners[event]) return;
        this.listeners[event] = this.listeners[event].filter(cb => cb !== callback);
    }

    send(action: string, payload: any = {}) {
        if (this.ws && this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(JSON.stringify({action, ...payload}));
        } else {
            console.error('Невозможно отправить сообщение: WebSocket не подключен');
        }
    }

    disconnect() {
        if (this.ws) {
            this.ws.close();
            this.ws = null;
        }
    }
}

export const gameSocket = new GameWebSocketService();

"""

def is_text_candidate(path: Path, relative: str) -> bool:
    name = path.name
    name_lower = name.lower()
    suffix = path.suffix.lower()
    rel_parts = {part.lower() for part in Path(relative).parts}

    if name in EXCLUDED_FILES:
        return False
    # Исключаем любые старые текстовые снэпшоты, чтобы они не попадали внутрь новых частей
    if "snapshot" in name_lower and suffix == ".txt":
        return False
    if name_lower.endswith("_generated_book.json"):
        return False
    if suffix in EXCLUDED_SUFFIXES:
        return False
    if rel_parts & SECRET_PATH_PARTS:
        return False
    if re.search(r"(^|/)\.env(?:\.|$)", relative.replace("\\", "/"), re.I):
        return False

    if suffix in TEXT_SUFFIXES or name in TEXT_FILENAMES:
        return True

    if not suffix:
        try:
            sample = path.read_bytes()[:4096]
        except OSError:
            return False
        return b"\x00" not in sample

    return False


def get_group_priority(relative_posix: str) -> tuple[int, str]:
    """
    Упорядочивает файлы по смысловым блокам:
    0..N — приоритетные манифесты (ToDo.md, requirements.txt и т.д.)
    100  — миграции БД (migrations/)
    200  — исходный код бэкенда (src/)
    300  — тесты бэкенда (tests/)
    400  — скрипты (scripts/)
    500  — исходный код фронтенда (frontend/)
    600  — остальные файлы
    """
    lower = relative_posix.lower()
    for idx, prio_name in enumerate(PRIORITY_FILES):
        if lower == prio_name.lower():
            return (idx, lower)

    if lower.startswith("migrations/"):
        return (100, lower)
    if lower.startswith("src/"):
        return (200, lower)
    if lower.startswith("tests/"):
        return (300, lower)
    if lower.startswith("scripts/"):
        return (400, lower)
    if lower.startswith("frontend/"):
        return (500, lower)
    return (600, lower)


def collect_files(root: Path) -> list[Path]:
    files: list[Path] = []

    for current, dirs, filenames in os.walk(root, topdown=True, followlinks=False):
        current_path = Path(current)
        dirs[:] = sorted(
            d for d in dirs
            if d not in EXCLUDED_DIRS
            and not (current_path / d).is_symlink()
        )

        for filename in filenames:
            path = current_path / filename
            if path.is_symlink() or not path.is_file():
                continue

            relative = path.relative_to(root).as_posix()
            if is_text_candidate(path, relative):
                files.append(path)

    return sorted(files, key=lambda p: get_group_priority(p.relative_to(root).as_posix()))


def decode_file(path: Path) -> str:
    raw = path.read_bytes()
    for encoding in ("utf-8-sig", "utf-8", "cp1251"):
        try:
            return raw.decode(encoding)
        except UnicodeDecodeError:
            pass
    return raw.decode("utf-8", errors="replace")


def maybe_summarize_large_json(path: Path, relative: str, content: str, full_json: bool) -> str:
    """
    Если это большая JSON-книга дебютов (> 15 КБ), сворачивает её до первых 3 позиций
    и общей сводки, чтобы не забивать контекст тысячами строк FEN.
    """
    if full_json or path.suffix.lower() != ".json":
        return content

    raw_size = len(content.encode("utf-8"))
    if raw_size <= MAX_DATA_JSON_BYTES:
        return content

    try:
        parsed = json.loads(content)
        if isinstance(parsed, dict):
            total_keys = len(parsed)
            sample_items = list(parsed.items())[:3]
            sample_dict = dict(sample_items)
            sample_str = json.dumps(sample_dict, ensure_ascii=False, indent=2)
            return (
                f"// [AUTO-SUMMARY: Большой JSON-файл ({raw_size // 1024} KiB, {total_keys} ключей). "
                f"Показаны первые 3 записи. Для полного вывода запустите с --full-json]\n"
                f"{sample_str}\n"
                f"// ... ещё {max(0, total_keys - 3)} записей скрыто ..."
            )
    except Exception:
        pass

    return content


def make_file_blocks(
        root: Path,
        files: list[Path],
        full_json: bool = False,
) -> tuple[list[tuple[str, str]], list[str]]:
    blocks: list[tuple[str, str]] = []
    unreadable: list[str] = []

    for path in files:
        relative = path.relative_to(root).as_posix()
        try:
            content = decode_file(path)
        except OSError as exc:
            unreadable.append(f"{relative}: {exc}")
            continue

        content = content.replace("\r\n", "\n").replace("\r", "\n")
        content = maybe_summarize_large_json(path, relative, content, full_json)

        sep = "=" * 88
        block = (
            f"\n{sep}\n"
            f"FILE: {relative}\n"
            f"{sep}\n"
            f"{content.rstrip()}\n"
        )
        blocks.append((relative, block))

    return blocks, unreadable


def split_oversized_block(relative: str, block: str, max_bytes: int) -> list[tuple[str, str]]:
    """Делит один очень крупный файл по границам строк, если он сам превышает лимит части."""
    if len(block.encode("utf-8")) <= max_bytes:
        return [(relative, block)]

    pieces: list[tuple[str, str]] = []
    remaining = block
    part_idx = 1

    while remaining:
        low, high = 1, len(remaining)
        best = 1

        while low <= high:
            middle = (low + high) // 2
            size = len(remaining[:middle].encode("utf-8"))
            if size <= max_bytes:
                best = middle
                low = middle + 1
            else:
                high = middle - 1

        if best < len(remaining):
            newline = remaining.rfind("\n", 0, best)
            if newline > best // 2:
                best = newline + 1

        chunk_text = remaining[:best]
        label = f"{relative} (chunk {part_idx})"
        pieces.append((label, chunk_text))
        remaining = remaining[best:]
        part_idx += 1

    return pieces


def pack_parts(
        blocks: list[tuple[str, str]],
        max_bytes: int,
) -> list[tuple[list[str], str]]:
    """Упаковывает блоки файлов в части так, чтобы каждая часть была строго <= max_bytes."""
    chunks: list[tuple[str, str]] = []
    for relative, block in blocks:
        chunks.extend(split_oversized_block(relative, block, max_bytes))

    parts: list[tuple[list[str], str]] = []
    current_files: list[str] = []
    current_texts: list[str] = []
    current_size = 0

    for label, chunk in chunks:
        chunk_size = len(chunk.encode("utf-8"))
        if current_texts and current_size + chunk_size > max_bytes:
            parts.append((current_files, "".join(current_texts)))
            current_files = []
            current_texts = []
            current_size = 0

        current_files.append(label)
        current_texts.append(chunk)
        current_size += chunk_size

    if current_texts:
        parts.append((current_files, "".join(current_texts)))

    return parts or [([], "[Подходящих текстовых файлов не найдено]\n")]


def main() -> int:
    parser = argparse.ArgumentParser(
        description="Создать многочастный Snapshot проекта без обрезки в чате."
    )
    parser.add_argument(
        "--root",
        type=Path,
        default=Path.cwd(),
        help="Корень проекта (по умолчанию текущая папка).",
    )
    parser.add_argument(
        "--max-kb",
        type=float,
        default=DEFAULT_MAX_KB,
        help=f"Максимальный размер одной части в KiB (по умолчанию {DEFAULT_MAX_KB} KiB).",
    )
    parser.add_argument(
        "--max-mb",
        type=float,
        default=None,
        help="Альтернативный лимит в MiB (переопределяет --max-kb, если указан).",
    )
    parser.add_argument(
        "--full-json",
        action="store_true",
        help="Не сворачивать большие JSON-книги дебютов.",
    )
    parser.add_argument(
        "--output",
        type=Path,
        default=None,
        help=f"Папка результата (по умолчанию ./{OUTPUT_DIR_NAME}).",
    )
    args = parser.parse_args()

    root = args.root.expanduser().resolve()
    if not root.is_dir():
        print(f"Ошибка: папка не найдена: {root}", file=sys.stderr)
        return 2

    if args.max_mb is not None:
        max_kb = args.max_mb * 1024.0
    else:
        max_kb = args.max_kb

    if max_kb <= 10:
        print("Ошибка: размер части должен быть больше 10 KiB.", file=sys.stderr)
        return 2

    output = (
        args.output.expanduser().resolve()
        if args.output
        else root / OUTPUT_DIR_NAME
    )
    # Оставляем запас 3 KiB под шапку и футер части
    max_bytes = max(8192, int(max_kb * 1024) - 3072)

    files = collect_files(root)
    blocks, unreadable = make_file_blocks(root, files, full_json=args.full_json)
    parts = pack_parts(blocks, max_bytes)

    output.mkdir(parents=True, exist_ok=True)

    for old in output.glob("snapshot_[0-9][0-9][0-9].txt"):
        old.unlink()

    total_parts = len(parts)
    index_lines = [
        f"PROJECT: {root.name}",
        f"ROOT: {root}",
        f"FILES INCLUDED: {len(blocks)}",
        f"PARTS: {total_parts} (target <= {max_kb:.0f} KiB per part)",
        "",
        "UPLOAD ALL PARTS IN NUMERICAL ORDER (snapshot_001.txt .. snapshot_00N.txt).",
        "",
        "TABLE OF CONTENTS BY PART:",
    ]

    print(f"Проект: {root}")
    print(f"Текстовых файлов включено: {len(blocks)}")
    print(f"Количество частей: {total_parts} (лимит ~{max_kb:.0f} KiB на файл)\n")

    total_size = 0
    dash_line = "-" * 88
    for number, (part_files, content) in enumerate(parts, start=1):
        filename = f"snapshot_{number:03d}.txt"
        header = (
            f"PROJECT SNAPSHOT — PART {number}/{total_parts}\n"
            f"Project: {root.name}\n"
            f"Files in this part ({len(part_files)}): {', '.join(part_files)}\n"
            f"{dash_line}\n"
        )
        footer = (
            f"\n{dash_line}\n"
            f"END OF PART {number}/{total_parts}\n"
        )
        destination = output / filename
        destination.write_text(header + content + footer, encoding="utf-8", newline="\n")
        size = destination.stat().st_size
        total_size += size

        print(f"  [{number:02d}/{total_parts:02d}] {filename}: {size / 1024:.1f} KiB ({len(part_files)} файлов)")
        index_lines.append(f"\n--- {filename} ({size / 1024:.1f} KiB) ---")
        for pf in part_files:
            index_lines.append(f"  - {pf}")

    (output / "README.txt").write_text("\n".join(index_lines) + "\n", encoding="utf-8")

    if unreadable:
        report = output / "unreadable_files.txt"
        report.write_text("\n".join(unreadable), encoding="utf-8")
        print(f"\nНе удалось прочитать файлов: {len(unreadable)} (см. {report})")

    print(f"\nГотово! Папка со снэпшотом: {output}")
    print(f"Суммарный объём всех частей: {total_size / 1024:.1f} KiB")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

# Michess ♟️ — Full-Stack Шахматная Платформа

Веб-платформа для игры в шахматы в реальном времени (PvP), матчей против исторических шахматных мастеров (с аутентичными
дебютными книгами и стилем игры на базе Stockfish 19), изучения архивов чемпионов мира и компьютерного разбора партий.

---

## 🛠 Технологический стек

### Бэкенд (Python 3.11+)

* **Framework:** FastAPI (REST API + WebSockets)
* **База данных:** PostgreSQL 16 (асинхронный `asyncpg` + SQLAlchemy 2.0)
* **Кэш, очереди и Pub/Sub:** Redis 8 (менеджер WebSocket-сессий, очереди матчмейкинга `Sorted Sets`, приватные вызовы)
* **Шахматное ядро и движок:** `python-chess` + Stockfish 19 UCI
* **Миграции:** Alembic
* **Тестирование и линтеры:** Pytest (`pytest-asyncio` + `fakeredis` + `aiosqlite`), Flake8

### Фронтенд (React 19 + TypeScript)

* **Сборка и стили:** Vite 6 + Tailwind CSS v4
* **Шахматная логика и графики:** `chess.js`, `recharts`, `lucide-react`
* **Тестирование и линтеры:** Vitest, Oxlint

---

## 🚀 Ключевые возможности

* **Асинхронный PvP-матчмейкинг:** Поиск соперника через Redis `Sorted Sets`, 36 программ шахматных часов FIDE DGT2500,
  предходы (`Premoves`), превращение пешки, авто-отмена на 1-м ходу, мгновенные реванши со сменой цвета, приватные
  вызовы по ссылке (`/challenge/:code`), режим зрителя (`Live`) и игровой чат.
* **Исторические боты (7 персон):** Филидор, Лабурдонне, Стаунтон, Андерсен, Морфи, Паульсен, Цукерторт. Единый базовый
  класс `HistoricalBot`, JSON-книги дебютов из реальных PGN-партий, тактический расчёт через Stockfish 19 с
  индивидуальными стилевыми бонусами и имитацией человеческого времени раздумий.
* **Студия разбора партий (`GameViewer`):** Экспресс-анализ всей партии движком Stockfish по кнопке (`Accuracy %`, поиск
  зевков `??`, ошибок `?`, неточностей `?!`), вертикальная шкала оценки (`EvalBar`), интерактивный график перевеса
  `Recharts` и рисование тактических стрелок правой кнопкой мыши.
* **Архив Чемпионов мира:** Хронологическая база партий 21 чемпиона мира и претендентов с просмотром на доске и
  скачиванием PGN.
* **Античит-телеметрия:** Сбор детерминированных метрик каждого хода (`time_taken_ms`, `window_blurred_before_move`,

$root = $PSScriptRoot

$stockfishBin = Join-Path $root "bin\stockfish.exe"

{
    $conns = Get-NetTCPConnection -LocalPort $port -State Listen -ErrorAction SilentlyContinue
    if ($conns)
    {
        $pids = $conns | Select-Object -ExpandProperty OwningProcess -Unique
        foreach ($procId in $pids)
        {
            if ($procId -gt 0)
            {
                Write-Host "[*] Освобождаем порт $port (PID: $procId)..." -ForegroundColor DarkYellow

docker compose up -d db redis

{
    Write-Host "[!] Внимание: Docker Desktop не запущен или недоступен!" -ForegroundColor Red
}

Write-Host "[*] Применяем миграции базы данных (Alembic)..." -ForegroundColor Cyan

} -ArgumentList $pythonExe, $root | Out-Null

    }

}

Start-Process "http://127.0.0.1:3000"
}
else
{
    Write-Host "[!] Порт 3000 не ответил за 15 секунд. Проверьте лог: Receive-Job -Name MichessFrontend -Keep" -ForegroundColor Red
}

Write-Host "========================================================" -ForegroundColor Green
Write-Host " Все сервисы запущены! Локальный сервер работает." -ForegroundColor Green
Write-Host " Фронтенд: [http://127.0.0.1:3000](http://127.0.0.1:3000)" -ForegroundColor Cyan
Write-Host " API Docs: [http://127.0.0.1:8000/docs](http://127.0.0.1:8000/docs)" -ForegroundColor Cyan

# Регламент разработки и Git-процессов (Michess Workflow)

Пошаговая шпаргалка для работы над каждой новой задачей в проекте.

---

## 1. Схема веток проекта

1. `master` — **Релизная ветка (Production).**
    * Здесь лежит только проверенный и полностью рабочий код.
    * Напрямую в `master` код никогда не пишем. Обновляем его только слиянием из `dev`.

2. `dev` — **Основная ветка разработки (Integration).**
    * База, в которую сливаются все готовые фичи и исправления.
    * Все новые ветки создаются от `dev` и после завершения вливаются обратно в `dev`.

3. Рабочие ветки (`feature/...`, `fix/...`, `chore/...`, `perf/...`) — **Ветки под конкретную задачу.**
    * `feature/название` — новая функциональность или доработка UI (например, `feature/board-layout`);
    * `fix/название` — исправление бага (например, `fix/timer-sync`);
    * `chore/название` — настройка инфраструктуры, тестов, документации, CI/CD;
    * `perf/название` — оптимизация производительности и SQL-запросов.

---

## 2. Пошаговый алгоритм работы над новой фичей (От и До)

### Шаг 1. Создание ветки перед началом работы

Всегда начинай новую задачу в корневой папке проекта `michess` с переключения на `dev` и создания новой рабочей ветки:

    git checkout dev
    git pull origin dev
    git checkout -b feature/название-задачи

*(Если ты забыл создать ветку и уже начал менять файлы прямо в `dev` — просто выполни
`git checkout -b feature/название-задачи`, и все незакоммиченные изменения автоматически переедут в новую ветку).*

---

### Шаг 2. Написание кода и правило миграций БД (Alembic)

ВАЖНОЕ ПРАВИЛО ПО БАЗЕ ДАННЫХ:

* Если ты менял **только фронтенд** (`frontend/...`), стили, WebSocket-логику или сервисы Python без изменения таблиц
  БД — **миграции Alembic делать НЕ НУЖНО**.
* Миграция создается **ТОЛЬКО** тогда, когда ты изменил или добавил колонки/таблицы в папке `src/models/`. В этом случае
  выполни:

  alembic revision --autogenerate -m "описание_изменений_моделей"
  alembic upgrade head

---

### Шаг 3. Локальная проверка перед коммитом

1. Если менялся Python-код (из корня проекта):

   python -m flake8 src tests scripts --count --select=E9,F63,F7,F82 --show-source --statistics
   python -m pytest -v

2. Если менялся React/TypeScript-код (фронтенд):

   cd frontend
   npm run lint
   npm run test
   npm run build
   cd ..

### Шаг 4. Коммит изменений в рабочую ветку

Убедись по `git status`, что в список не попали временные файлы, и зафиксируй изменения:

    git status
    git add .
    git commit -m "feat: краткое описание сделанной задачи"

Префиксы коммитов:

* `feat:` — новая фича или улучшение интерфейса
* `fix:` — исправление ошибки
* `refactor:` — рефакторинг кода
* `perf:` — ускорение работы и оптимизация запросов
* `chore:` — инфраструктура, конфиги, зависимости
* `docs:` — документация (`README.md`, `QUICKSTART.md`, `WORKFLOW.md`, `ToDo.md`)

### Шаг 5. Слияние готовой задачи в `dev` и отправка на GitHub

Когда задача проверена в браузере и тесты проходят:

    git checkout dev
    git merge --no-ff feature/название-задачи -m "Merge feature/название-задачи into dev"
    git push origin dev
    git branch -d feature/название-задачи

После `git push origin dev` на GitHub во вкладке **Actions** автоматически запустится пайплайн `Michess CI`, который
проверит бэкенд и фронтенд в облаке.

---

### Шаг 6. Выпуск стабильного релиза в `master`

Выполняется, когда в `dev` завершен логический этап (или спринт) и всё стабильно работает:

    git checkout master
    git pull origin master
    git merge --no-ff dev -m "Release: merge stable dev into master"
    git push origin master
    git checkout dev

*(Последняя команда `git checkout dev` обязательна, чтобы никогда не оставаться в ветке `master`).*
