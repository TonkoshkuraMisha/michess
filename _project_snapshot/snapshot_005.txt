import pytest
import time
from src.services.game_state import state_manager


class DummyManager:
    """Заглушка WebSocket-менеджера для изоляции тестов логики состояний."""

@pytest.mark.asyncio
async def test_game_state_initialization():

@pytest.mark.asyncio
async def test_disconnect_and_reconnect_time_deduction():
    await state_manager.initialize_game(
        game_id=2, white_id=10, black_id=20, base_time_ms=180000, increment_ms=0
    )

    dummy_manager = DummyManager()

    # Эмулируем дисконект белых (ход белых)
    await state_manager.handle_player_disconnect(10, dummy_manager)

    state_after_dc = await state_manager.get_game_state(2)
    assert state_after_dc["white_disconnected"] == "true"

    # Искусственно состарим время начала дисконекта на 5 секунд (5000 мс)
    old_dc_time = int(state_after_dc["white_disconnect_started_at"]) - 5000
    await state_manager.update_game_state(2, {"white_disconnect_started_at": old_dc_time})

    # Эмулируем переподключение
    await state_manager.handle_player_reconnect(10, dummy_manager)

    state_after_reconnect = await state_manager.get_game_state(2)
    assert state_after_reconnect["white_disconnected"] == "false"
    # Время должно уменьшиться примерно на 5 секунд (180000 - 5000 = 175000)
    assert state_after_reconnect["white_time_ms"] <= 175000


@pytest.mark.asyncio
async def test_abort_game_without_rating_change(db_session):
    """Проверяет, что отмена партии на 1-м ходу (до 2 полуходов) не меняет рейтинг и ставит статус ABORTED."""
    from src.models.user import User
    from src.models.game import Game, GameStatus
    from src.services.game_state import state_manager

    u1 = User(username="abort_white_player", hashed_password="pw", rating=1500)
    u2 = User(username="abort_black_player", hashed_password="pw", rating=1500)
    db_session.add_all([u1, u2])
    await db_session.commit()
    await db_session.refresh(u1)
    await db_session.refresh(u2)

    game = Game(
        white_player_id=u1.id,
        black_player_id=u2.id,
        status=GameStatus.IN_PROGRESS,
        base_time_ms=180000,
        increment_ms=2000,
        is_rated=True,
    )
    db_session.add(game)
    await db_session.commit()
    await db_session.refresh(game)

    await state_manager.redis.hset(f"game:{game.id}:state", mapping={
        "white_id": str(u1.id),
        "black_id": str(u2.id),
        "status": "in_progress",
        "state_version": "1",
        "last_move_at": "1000",
    })

    res = await state_manager.abort_game(db_session, game.id)
    assert res["status"] == "aborted"
    assert res["result"] == "*"
    assert res["white_delta"] == 0
    assert res["black_delta"] == 0

    await db_session.refresh(u1)
    await db_session.refresh(u2)
    await db_session.refresh(game)
    assert u1.rating == 1500
    assert u2.rating == 1500
    assert game.status == GameStatus.ABORTED


@pytest.mark.asyncio
async def test_private_invite_create_and_redeem():
    """Проверяет создание приватного вызова по ссылке в Redis и его одноразовое погашение."""
    from src.services.matchmaker import matchmaker

    invite = await matchmaker.create_private_invite(
        user_id=101,
        username="teacher_misha",
        rating=2050,
        base_time_ms=600000,
        increment_ms=5000,
        color_pref="white",
    )
    code = invite["code"]
    assert len(code) >= 6

    # Создатель не может принять собственный вызов
    self_accept = await matchmaker.accept_private_invite(code, acceptor_id=101)
    assert self_accept is None

    # Второй игрок успешно принимает вызов
    redeemed = await matchmaker.accept_private_invite(code, acceptor_id=202)
    assert redeemed is not None
    assert redeemed["white_id"] == 101
    assert redeemed["black_id"] == 202
    assert redeemed["base_time_ms"] == 600000
    assert redeemed["increment_ms"] == 5000

    # Повторное погашение того же кода невозможно
    second_try = await matchmaker.accept_private_invite(code, acceptor_id=303)
    assert second_try is None

def matches_player(header_name: str, surname: str, exclude: tuple[str, ...]) -> bool:
    name_lower = (header_name or "").lower()
    if any(ex in name_lower for ex in exclude):
        return False
    return surname in name_lower


def process_pgn_stream(pgn_stream, surname: str, exclude: tuple[str, ...], book_additions: dict) -> int:
    games_processed = 0
    while True:
        try:
            game = chess.pgn.read_game(pgn_stream)
            if game is None:
                break

            is_w = matches_player(game.headers.get("White", ""), surname, exclude)
            is_b = matches_player(game.headers.get("Black", ""), surname, exclude)
            track_both = not (is_w or is_b)

            board = game.board()
            ply = 0
            for move in game.mainline_moves():
                if ply >= MAX_BOOK_PLY:
                    break
                uci = move.uci()
                if uci == "0000" or not bool(move):
                    break

                white_turn = board.turn == chess.WHITE
                if track_both or (white_turn and is_w) or (not white_turn and is_b):
                    parts = board.fen().split(" ")
                    fen_key = f"{parts[0]} {parts[1]} {parts[2]} -"
                    book_additions.setdefault(fen_key, {})
                    book_additions[fen_key][uci] = book_additions[fen_key].get(uci, 0) + 1

                try:
                    board.push(move)
                except ValueError:
                    break
                ply += 1

            games_processed += 1
        except Exception:
            continue
    return games_processed


def find_pgn_files_for_bot(identifiers: tuple[str, ...]) -> list[Path]:
    """Ищет PGN-файлы чемпиона во всех возможных директориях проекта."""
    search_dirs = [
        PROJECT_ROOT / "data",
        PROJECT_ROOT / "downloaded_pgns",
        PROJECT_ROOT / "src" / "db" / "data",
        PROJECT_ROOT,
    ]
    found: list[Path] = []
    seen: set[Path] = set()

    for directory in search_dirs:
        if not directory.exists():
            continue
        iterator = directory.glob("*.pgn") if directory == PROJECT_ROOT else directory.rglob("*.pgn")
        for pgn_path in iterator:
            resolved = pgn_path.resolve()
            if resolved in seen or not resolved.is_file():
                continue
            path_lower = str(resolved).lower()
            if any(ident in path_lower for ident in identifiers):
                seen.add(resolved)
                found.append(resolved)

    # Если есть как объединённый файл (*_all_*.pgn), так и мелкие файлы из папки —
    # отдаём приоритет объединённому файлу, чтобы не считать одни и те же партии дважды
    combined = [p for p in found if "all" in p.name.lower() or "все партии" in p.name.lower()]
    if combined:
        return [max(combined, key=lambda p: p.stat().st_size)]
    return found


def build_book_for_bot(bot_id: str) -> None:
    cfg = BOT_BOOK_CONFIG[bot_id]
    target_json = BOOKS_DIR / f"{bot_id}.json"

    existing_book: dict = {}
    if target_json.exists():
        try:
            with open(target_json, "r", encoding="utf-8") as f:
                loaded = json.load(f)
                if isinstance(loaded, dict):
                    existing_book = loaded
        except Exception:
            existing_book = {}

    matched_files = find_pgn_files_for_bot(cfg["identifiers"])
    book_additions: dict[str, dict[str, int]] = {}
    total_games = 0

    for filepath in matched_files:
        with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
            total_games += process_pgn_stream(f, cfg["surname_filter"], cfg["exclude"], book_additions)

    if total_games > 0:
        formatted_book: dict[str, list] = dict(existing_book)
        for fen, moves_dict in book_additions.items():
            sorted_moves = sorted(moves_dict.items(), key=lambda x: x[1], reverse=True)
            total_count = sum(moves_dict.values())
            formatted_book[fen] = [
                [
                    uci,
                    max(int((count / total_count) * 100), 1),
                    cfg["comment_tpl"].format(count=count),
                ]
                for uci, count in sorted_moves
            ]

        with open(target_json, "w", encoding="utf-8") as out_f:
            json.dump(formatted_book, out_f, ensure_ascii=False, indent=2)
        print(
            f"[+] {bot_id:13s}: спарсено партий из PGN: {total_games:4d} | "
            f"Позиций в книге: {len(formatted_book):4d} -> {target_json.relative_to(PROJECT_ROOT)}"
        )
        return

    # Если сырых PGN-файлов нет, но JSON-книга уже перенесена и лежит в src/services/bots/books/
    if existing_book:
        print(
            f"[+] {bot_id:13s}: книга активна и проверена | "
            f"Позиций в книге: {len(existing_book):4d} -> {target_json.relative_to(PROJECT_ROOT)}"
        )
        return

    # Если нет ни PGN, ни JSON-книги — инициализируем JSON-файл базовым репертуаром из модуля бота
    try:
        module = __import__(f"src.services.bots.{bot_id}", fromlist=["*"])
        default_attr = f"DEFAULT_{bot_id.upper()}_BOOK"
        fallback_data = getattr(module, default_attr, {})
        if fallback_data:
            serializable = {
                fen: [[m[0], int(m[1]), str(m[2])] for m in moves]
                for fen, moves in fallback_data.items()
            }
            with open(target_json, "w", encoding="utf-8") as out_f:
                json.dump(serializable, out_f, ensure_ascii=False, indent=2)
            print(
                f"[~] {bot_id:13s}: создана базовая книга | "
                f"Позиций в книге: {len(serializable):4d} -> {target_json.relative_to(PROJECT_ROOT)}"
            )
            return
    except Exception as exc:
        print(f"[!] Ошибка инициализации базовой книги для {bot_id}: {exc}")


def main() -> None:

if __name__ == "__main__":

TABLE_GAME_REGEX = re.compile(
    r'\d+\.(?:&nbsp;|\xa0|\s)*<a[^>]*href="/perl/chessgame\?gid=(\d+)"',
    re.IGNORECASE,
)
PAGINATION_INFO_REGEX = re.compile(
    r"page\s+\d+\s+of\s+(\d+)\s*;\s*games\s+\d+-\d+\s+of\s+(\d+)",
    re.IGNORECASE,
)


async def fetch_gids_from_page(
        client: httpx.AsyncClient, pid: int, page: int
) -> tuple[list[str], int | None, int | None]:
    url = (
        f"{BASE_URL}/perl/chessplayer?pid={pid}"
        if page == 1
        else f"{BASE_URL}/perl/chess.pl?page={page}&pid={pid}"
    )

    for attempt in range(1, 7):
        try:
            response = await client.get(url)
            if response.status_code in (429, 403, 503):
                wait_sec = 10 * attempt
                print(f"[!] Лимит ({response.status_code}) на стр. {page}, пауза {wait_sec} сек...")
                await asyncio.sleep(wait_sec)
                continue

            response.raise_for_status()
            gids = list(dict.fromkeys(TABLE_GAME_REGEX.findall(response.text)))

            detected_pages, detected_total = None, None
            match = PAGINATION_INFO_REGEX.search(response.text)
            if match:
                detected_pages, detected_total = int(match.group(1)), int(match.group(2))

            if not gids:
                wait_sec = 8 * attempt
                print(f"[!] Стр. {page} вернула 0 партий (попытка {attempt}/6), пауза {wait_sec} сек...")
                await asyncio.sleep(wait_sec)
                continue

            return gids, detected_pages, detected_total
        except httpx.HTTPError as exc:
            wait_sec = 4 * attempt
            print(f"[!] Ошибка сети на стр. {page} (попытка {attempt}): {exc}")
            await asyncio.sleep(wait_sec)

    return [], None, None


async def download_pgn(
        client: httpx.AsyncClient,
        player_key: str,
        player_dir: Path,
        gid: str,
        semaphore: asyncio.Semaphore,
        index: int,
        total: int,
) -> tuple[int, str, str]:
    file_path = player_dir / f"{index:03d}_{player_key}_{gid}.pgn"
    if file_path.exists():
        cached = file_path.read_text(encoding="utf-8", errors="ignore").strip()
        if "[Event " in cached:
            return index, gid, cached

    url = f"{BASE_URL}/njs/api/game/downloadPGN/{gid}"
    async with semaphore:
        for attempt in range(1, 7):
            try:
                resp = await client.get(url)
                text = resp.text.strip()
                if resp.status_code == 200 and "[Event " in text:
                    file_path.write_text(text + "\n", encoding="utf-8")
                    if index % 25 == 0 or index == total:
                        print(f"[{index:03d}/{total}] Скачано (gid={gid})...")
                    await asyncio.sleep(0.35)
                    return index, gid, text
                await asyncio.sleep(15 * attempt)
            except httpx.RequestError:
                await asyncio.sleep(3 * attempt)

    return index, gid, ""

    player_dir = DATA_PGNS_DIR / f"{player_key}_pgns"

    combined_file = DATA_PGNS_DIR / f"{player_key}_all_games.pgn"

    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/130.0.0.0 Safari/537.36",
        "Referer": f"{BASE_URL}/perl/chessplayer?pid={pid}",
    }
    semaphore = asyncio.Semaphore(3)

    async with httpx.AsyncClient(headers=headers, timeout=25.0, follow_redirects=True) as client:
        print(f"=== Скачивание базы: {cfg['name']} (pid={pid}) ===")
        all_gids: list[str] = []
        page = 1
        while page <= total_pages:
            gids, det_pages, det_total = await fetch_gids_from_page(client, pid, page)
            if det_pages:
                total_pages = det_pages
            for gid in gids:
                if gid not in all_gids:
                    all_gids.append(gid)
            total_info = f"/{det_total}" if det_total else ""
            print(
                f"[Страница {page:02d}/{total_pages:02d}] Найдено: {len(gids)} | Всего ID: {len(all_gids)}{total_info}")
            page += 1
            await asyncio.sleep(0.35)

        if not all_gids:
            print(f"[-] Партии для {player_key} не найдены.")
            return

        tasks = [
            download_pgn(client, player_key, player_dir, gid, semaphore, idx, len(all_gids))
            for idx, gid in enumerate(all_gids, start=1)
        ]
        results = await asyncio.gather(*tasks)
        results.sort(key=lambda x: x[0])
        valid_pgns = [pgn for _, _, pgn in results if pgn]

        combined_file.write_text("\n\n".join(valid_pgns) + "\n", encoding="utf-8")
        print(f"[+] Сохранено {len(valid_pgns)} партий в {combined_file}")

    build_book_for_bot(player_key)


async def main() -> None:
    parser = argparse.ArgumentParser(description="Единый загрузчик PGN исторических чемпионов с Chessgames.")

    headers = {"User-Agent": "MiChess/1.0 (chess application; python-requests)"}
    session = requests.Session()
    session.headers.update(headers)

    print("Проверка и докачка портретов чемпионов...\n")

    for champ_id, wiki_title in champions.items():
        save_path = PUBLIC_CHAMPIONS_DIR / f"{champ_id}.jpg"
        if save_path.exists() and save_path.stat().st_size > 0:
            print(f"[~] {champ_id}.jpg уже существует, пропускаем.")
            continue

        params = {
            "action": "query",
            "titles": wiki_title,
            "prop": "pageimages",
            "piprop": "thumbnail",
            "pithumbsize": "500",
            "format": "json",
            "formatversion": "2",
        }

        for attempt in range(3):
            try:
                response = session.get(api_url, params=params, timeout=15)
                if response.status_code == 429:
                    time.sleep(15)
                    continue
                if response.status_code != 200:
                    break

                pages = response.json().get("query", {}).get("pages", [])
                if not pages or not pages[0].get("thumbnail"):
                    break

                img_url = pages[0]["thumbnail"]["source"].split("?")[0]
                cmd = [
                    "curl", "-sS", "-L",
                    "-A", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/131.0.0.0 Safari/537.36",
                    "-e", "https://ru.wikipedia.org/",
                    "--fail", img_url, "-o", str(save_path),
                ]
                result = subprocess.run(cmd, capture_output=True)
                if result.returncode == 0 and save_path.exists() and save_path.stat().st_size > 0:
                    print(f"[+] {champ_id}.jpg успешно скачан!")
                    break
            except Exception as exc:
                print(f"[-] Ошибка для {champ_id}: {exc}")
                break
        time.sleep(2)

    sync_historical_portraits()
    print("\nГотово! Все портреты синхронизированы.")


if __name__ == "__main__":
    main()

import shutil
import urllib.request
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent.parent
PIECES_ROOT = PROJECT_ROOT / "frontend" / "public" / "assets" / "pieces"

PIECE_CODES = ["wP", "wN", "wB", "wR", "wQ", "wK", "bP", "bN", "bB", "bR", "bQ", "bK"]

REMOTE_SETS = {
    "merida": "https://raw.githubusercontent.com/lichess-org/lila/master/public/piece/merida/{code}.svg",
    "alpha": "https://raw.githubusercontent.com/lichess-org/lila/master/public/piece/alpha/{code}.svg",
    "cburnett": "https://raw.githubusercontent.com/lichess-org/lila/master/public/piece/cburnett/{code}.svg",
    "maestro": "https://raw.githubusercontent.com/lichess-org/lila/master/public/piece/maestro/{code}.svg",
}


def make_cardinal_outlined_svg(base_svg: str, code: str) -> str:
    is_black = code.startswith("b")
    halo_color = "#FFFFFF" if is_black else "#111827"
    filter_id = f"cardinal_halo_{code}"

    halo_def = (
        f'<defs><filter id="{filter_id}" x="-12%" y="-12%" width="124%" height="124%">'
        f'<feMorphology in="SourceAlpha" operator="dilate" radius="1.15" result="dilated"/>'
        f'<feFlood flood-color="{halo_color}" flood-opacity="0.98" result="flood"/>'
        f'<feComposite in="flood" in2="dilated" operator="in" result="outline"/>'
        f'<feMerge><feMergeNode in="outline"/><feMergeNode in="SourceGraphic"/></feMerge>'
        f'</filter></defs>'
    )

    idx = base_svg.find(">")
    if idx == -1 or "</svg>" not in base_svg:
        return base_svg

    header = base_svg[: idx + 1]
    body = base_svg[idx + 1:].replace("</svg>", "")
    return f'{header}{halo_def}<g filter="url(#{filter_id})">{body}</g></svg>\n'


def backup_existing_classic_pieces() -> None:
    classic_dir = PIECES_ROOT / "classic"
    classic_dir.mkdir(parents=True, exist_ok=True)
    for code in PIECE_CODES:
        root_svg = PIECES_ROOT / f"{code}.svg"
        target_svg = classic_dir / f"{code}.svg"
        if root_svg.exists() and not target_svg.exists():
            shutil.copy2(root_svg, target_svg)


def download_svg(url: str) -> str | None:
    req = urllib.request.Request(
        url,
        headers={"User-Agent": "Michess-PieceSetup/1.0"},
    )
    try:
        with urllib.request.urlopen(req, timeout=15) as resp:
            if resp.status == 200:
                text = resp.read().decode("utf-8", errors="ignore")
                if "<svg" in text:
                    return text
    except Exception as exc:
        print(f"[!] Предупреждение при загрузке {url}: {exc}")
    return None


def setup_all_piece_sets() -> None:
    PIECES_ROOT.mkdir(parents=True, exist_ok=True)
    backup_existing_classic_pieces()
    classic_dir = PIECES_ROOT / "classic"

    for set_name, url_tpl in REMOTE_SETS.items():
        set_dir = PIECES_ROOT / set_name
        set_dir.mkdir(parents=True, exist_ok=True)
        downloaded_count = 0

        for code in PIECE_CODES:
            target_file = set_dir / f"{code}.svg"
            if target_file.exists() and target_file.stat().st_size > 200:
                downloaded_count += 1
                continue

            svg_content = download_svg(url_tpl.format(code=code))
            if svg_content:
                target_file.write_text(svg_content, encoding="utf-8")
                downloaded_count += 1
            elif (classic_dir / f"{code}.svg").exists():
                shutil.copy2(classic_dir / f"{code}.svg", target_file)

        print(f"[+] Набор '{set_name:8s}' готов ({downloaded_count}/12 фигур) -> {set_dir.relative_to(PROJECT_ROOT)}")

    cardinal_dir = PIECES_ROOT / "cardinal"
    cardinal_dir.mkdir(parents=True, exist_ok=True)
    merida_dir = PIECES_ROOT / "merida"

    for code in PIECE_CODES:
        src_file = merida_dir / f"{code}.svg"
        if not src_file.exists():
            src_file = classic_dir / f"{code}.svg"
        if src_file.exists():
            raw_svg = src_file.read_text(encoding="utf-8")
            outlined_svg = make_cardinal_outlined_svg(raw_svg, code)
            (cardinal_dir / f"{code}.svg").write_text(outlined_svg, encoding="utf-8")

    print(f"[+] Набор 'cardinal' готов (12/12 фигур) -> {cardinal_dir.relative_to(PROJECT_ROOT)}")

    for code in PIECE_CODES:
        merida_svg = merida_dir / f"{code}.svg"
        if merida_svg.exists():
            shutil.copy2(merida_svg, PIECES_ROOT / f"{code}.svg")

    print("[+] Все 5 комплектов шахматных фигур (60 SVG-файлов) успешно установлены!")


if __name__ == "__main__":
    setup_all_piece_sets()

# --- FILE: setup_stockfish.py ---

import io
import os
import platform
import shutil
import stat
import tarfile
import zipfile
import requests
import chess.engine

GITHUB_RELEASE_API = "https://api.github.com/repos/official-stockfish/Stockfish/releases/latest"
PROJECT_ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), ".."))
BIN_DIR = os.path.join(PROJECT_ROOT, "bin")


def get_target_asset_url(assets: list[dict]) -> tuple[str, str]:
    """Подбирает оптимальный бинарник Stockfish под текущую ОС (включая universal-сборки SF 19+)."""
    sys_name = platform.system().lower()

    if sys_name == "windows":
        preferred = [
            "windows-x86-64-universal.zip",
            "windows-x86-64-avx2.zip",
            "windows-x86-64-sse41-popcnt.zip",
            "windows-x86-64.zip",
        ]
    elif sys_name == "linux":
        preferred = [
            "linux-x86-64-universal.tar.gz",
            "linux-x86-64-universal.tar",
            "ubuntu-x86-64-avx2.tar",
            "ubuntu-x86-64-sse41-popcnt.tar",
            "ubuntu-x86-64.tar",
        ]
    else:
        preferred = [
            "macos-universal.tar.gz",
            "macos-m1-apple-silicon.tar",
            "macos-x86-64-avx2.tar",
        ]

    for suffix in preferred:
        for asset in assets:
            name = asset.get("name", "")
            if suffix in name:
                return name, asset["browser_download_url"]

    # Fallback: любой x86-64 архив под нужную ОС
    for asset in assets:
        name = asset.get("name", "").lower()
        if sys_name == "windows" and "windows" in name and "x86-64" in name and name.endswith(".zip"):
            return asset["name"], asset["browser_download_url"]
        if sys_name == "linux" and ("linux" in name or "ubuntu" in name) and "x86-64" in name:
            return asset["name"], asset["browser_download_url"]

    available = [a.get("name", "") for a in assets]
    raise RuntimeError(f"Не найден подходящий архив Stockfish для ОС '{sys_name}'. Доступные: {available}")


def download_and_install_stockfish() -> str:
    os.makedirs(BIN_DIR, exist_ok=True)
    is_windows = platform.system().lower() == "windows"
    target_filename = "stockfish.exe" if is_windows else "stockfish"
    target_path = os.path.join(BIN_DIR, target_filename)

    if os.path.exists(target_path) and os.path.getsize(target_path) > 1_000_000:
        print(f"[~] Бинарник Stockfish уже установлен: {target_path}")
        return target_path

    print("[1/3] Запрашиваем последний релиз Stockfish с GitHub...")
    headers = {"User-Agent": "Michess-Setup/1.0"}
    resp = requests.get(GITHUB_RELEASE_API, headers=headers, timeout=20)
    resp.raise_for_status()
    release_data = resp.json()
    tag_name = release_data.get("tag_name", "latest")

    asset_name, download_url = get_target_asset_url(release_data.get("assets", []))
    print(f"[2/3] Скачиваем {asset_name} (версия {tag_name})...")

    archive_resp = requests.get(download_url, headers=headers, stream=True, timeout=180)
    archive_resp.raise_for_status()
    archive_bytes = io.BytesIO(archive_resp.content)

    print(f"[3/3] Распаковываем движок в {target_path}...")
    extracted = False

    if asset_name.endswith(".zip"):
        with zipfile.ZipFile(archive_bytes) as zf:
            # Берём самый крупный .exe файл со словом stockfish (илиavx2/universal), игнорируя мелкие обёртки
            exe_members = [
                m for m in zf.infolist()
                if not m.is_dir() and m.filename.lower().endswith(".exe") and "stockfish" in m.filename.lower()
            ]
            exe_members.sort(key=lambda m: m.file_size, reverse=True)
            if exe_members:
                best_member = exe_members[0]
                with zf.open(best_member) as src, open(target_path, "wb") as dst:
                    shutil.copyfileobj(src, dst)
                extracted = True
    elif asset_name.endswith((".tar", ".tar.gz", ".tgz")):
        with tarfile.open(fileobj=archive_bytes, mode="r:*") as tf:
            bin_members = [
                m for m in tf.getmembers()
                if m.isfile()
                   and "stockfish" in os.path.basename(m.name).lower()
                   and not m.name.lower().endswith((".txt", ".md", ".1", ".cpp", ".h", ".sh"))
            ]
            bin_members.sort(key=lambda m: m.size, reverse=True)
            if bin_members:
                src = tf.extractfile(bin_members[0])
                if src:
                    with src, open(target_path, "wb") as dst:
                        shutil.copyfileobj(src, dst)
                    os.chmod(target_path, os.stat(target_path).st_mode | stat.S_IEXEC)
                    extracted = True

    if not extracted:
        raise RuntimeError("Не удалось найти исполняемый файл Stockfish внутри архива.")

    with chess.engine.SimpleEngine.popen_uci(target_path) as engine:
        engine_name = engine.id.get("name", "Stockfish")
        print(f"[+] Успешно! Движок готов к работе: {engine_name} -> {target_path}")

    return target_path


if __name__ == "__main__":
    download_and_install_stockfish()
