import json
import logging
import os
import random
import shutil
from pathlib import Path

import chess
import chess.engine

from src.services.engine_pool import get_engine_pool

logger = logging.getLogger(__name__)

BOTS_DIR = Path(__file__).resolve().parent
BOOKS_DIR = BOTS_DIR / "books"
PROJECT_ROOT = BOTS_DIR.parent.parent.parent

CENTER_SQUARES = {chess.D4, chess.E4, chess.D5, chess.E5}
EXTENDED_CENTER = {
    chess.C4, chess.D4, chess.E4, chess.F4,
    chess.C5, chess.D5, chess.E5, chess.F5,
}

PIECE_VALUES = {
    chess.PAWN: 100,
    chess.KNIGHT: 320,
    chess.BISHOP: 330,
    chess.ROOK: 500,
    chess.QUEEN: 900,
    chess.KING: 20000,
}


def normalize_fen_key(board: chess.Board) -> str:
    """Нормализует FEN до первых 3 полей (позиция, очередь хода, права рокировки)."""
    parts = board.fen().split(" ")
    return f"{parts[0]} {parts[1]} {parts[2]} -"


def resolve_stockfish_path(custom_path: str | None = None) -> str | None:
    """Ищет бинарник Stockfish в локальной папке bin/ проекта, системном PATH или стандартных путях Linux."""
    if custom_path and os.path.exists(custom_path):
        return custom_path

    for binary_name in ("stockfish.exe", "stockfish"):
        candidate = PROJECT_ROOT / "bin" / binary_name
        if candidate.exists() and candidate.is_file():
            return str(candidate)

    which_path = shutil.which("stockfish")
    if which_path:
        return which_path

    for linux_path in ("/usr/games/stockfish", "/usr/bin/stockfish", "/usr/local/bin/stockfish"):
        if os.path.exists(linux_path) and os.path.isfile(linux_path):
            return linux_path

    return None


def load_bot_opening_book(
        bot_id: str,
        fallback_book: dict[str, list[tuple[str, int, str]]] | None = None,
) -> dict[str, list[tuple[str, int, str]]]:
    """
    Загружает дебютную книгу бота из src/services/bots/books/{bot_id}.json
    и аккуратно объединяет её с фирменными дебютными линиями fallback_book.
    """
    book: dict[str, list[tuple[str, int, str]]] = {}
    if fallback_book:
        for k, v in fallback_book.items():
            book[k] = list(v)

    candidates = [
        BOOKS_DIR / f"{bot_id}.json",
        PROJECT_ROOT / f"{bot_id}_generated_book.json",
    ]

    for json_path in candidates:
        if json_path.exists():
            try:
                with open(json_path, encoding="utf-8") as f:
                    data = json.load(f)
                if isinstance(data, dict) and data:
                    for fen_key, moves in data.items():
                        parsed_moves = [
                            (str(m[0]), int(m[1]), str(m[2]))
                            for m in moves
                            if len(m) >= 3 and str(m[0]) != "0000"
                        ]
                        if not parsed_moves:
                            continue

                        if fen_key in book:
                            existing_ucis = {m[0] for m in parsed_moves}
                            for fb_move in book[fen_key]:
                                if fb_move[0] not in existing_ucis:
                                    parsed_moves.append(fb_move)
                        book[fen_key] = parsed_moves
                    break
            except Exception as exc:
                logger.warning("Не удалось загрузить книгу %s: %s", json_path, exc)

    return book


def _simulate_human_thinking_time(
        board: chess.Board,
        is_book_move: bool,
        bot_time_left_ms: int = 180000,
        base_time_ms: int = 180000,
) -> float:
    """
    Вычисляет рекомендуемое человеческое время раздумий шахматиста (в секундах)
    без блокировки рабочих потоков сервера через time.sleep().
    """
    piece_count = len(board.piece_map())
    move_number = board.fullmove_number

    is_opening = move_number <= 10 and piece_count >= 28
    is_endgame = piece_count <= 12

    if is_opening:
        base_delay = random.uniform(0.4, 0.9) if is_book_move else random.uniform(1.2, 2.8)
    elif is_endgame:
        base_delay = random.uniform(0.4, 1.0) if piece_count <= 6 else random.uniform(1.2, 2.6)
    else:
        legal_moves_count = board.legal_moves.count()
        complexity_mult = 0.7 + (min(legal_moves_count, 45) / 45.0) * 0.8
        base_delay = random.uniform(1.2, 3.0) * complexity_mult

    control_scale_factor = max(0.3, min(1.5, base_time_ms / 300000.0))
    adjusted_delay = base_delay * control_scale_factor

    if bot_time_left_ms < 30000:
        time_ratio = max(0.15, bot_time_left_ms / 30000.0)
        adjusted_delay = max(0.25, adjusted_delay * time_ratio)

    max_allowed = max(0.25, (bot_time_left_ms - 2000) / 1000.0)
    return round(min(adjusted_delay, max_allowed), 3)


def static_board_eval_cp(board: chess.Board, perspective: chess.Color) -> float:
    """Базовая статическая оценка позиции в сантипешках (fallback при отсутствии Stockfish)."""
    if board.is_checkmate():
        return -100000.0 if board.turn == perspective else 100000.0
    if board.is_stalemate() or board.is_insufficient_material():
        return 0.0

    score = 0.0
    for sq in chess.SQUARES:
        p = board.piece_at(sq)
        if not p:
            continue
        val = float(PIECE_VALUES[p.piece_type])
        if sq in CENTER_SQUARES:
            val += 25.0 if p.piece_type == chess.PAWN else 15.0
        elif sq in EXTENDED_CENTER:
            val += 15.0 if p.piece_type == chess.PAWN else 8.0

        if p.color == perspective:
            score += val
        else:
            score -= val
    return score


def analyse_with_stockfish(
        board: chess.Board,
        engine_path: str | None,
        skill_level: int = 7,
        uci_elo: int | None = 1850,
        depth: int = 6,
        time_limit: float = 0.25,
        multipv: int = 6,
) -> dict[chess.Move, float]:
    """Запрашивает оценку топ-ходов у движка Stockfish."""
    if not engine_path or not os.path.exists(engine_path):
        return {}

    scores: dict[chess.Move, float] = {}
    options: dict[str, int | bool] = {"Skill Level": skill_level}
    if uci_elo is not None:
        options["UCI_LimitStrength"] = True
        options["UCI_Elo"] = uci_elo

    try:
        # Пул сам приводит UCI_Elo / Skill Level к допустимому диапазону движка.
        with get_engine_pool(engine_path).acquire(options) as engine:
            infos = engine.analyse(
                board,
                chess.engine.Limit(depth=depth, time=time_limit),
                multipv=min(multipv, board.legal_moves.count()),
            )
            if isinstance(infos, dict):
                infos = [infos]
            for info in infos:
                pv = info.get("pv")
                if pv:
                    mv = pv[0]
                    cp = info["score"].pov(board.turn).score(mate_score=100000)
                    if cp is not None:
                        scores[mv] = float(cp)
    except Exception as exc:
        logger.warning("Ошибка запуска Stockfish (%s): %s", engine_path, exc)
        return {}

    return scores


class HistoricalBot:
    """
    Базовый класс исторической шахматной персоны.
    Наследники задают только параметры силы, дебютную книгу и функцию стилевого бонуса.
    """

    bot_id: str = "base"
    engine_source_name: str = "historical_style_engine"
    default_comment: str = "Классический позиционный ход."
    blunder_guard_cp: float = 130.0
    jitter_range: float = 8.0
    multipv: int = 6
    opening_book: dict[str, list[tuple[str, int, str]]] = {}

    def __init__(
            self,
            stockfish_path: str | None = None,
            skill_level: int = 7,
            uci_elo: int = 1850,
            search_depth: int = 6,
    ):
        self._explicit_stockfish_path = stockfish_path
        self.stockfish_path = resolve_stockfish_path(stockfish_path)
        self.skill_level = skill_level
        self.uci_elo = uci_elo
        self.search_depth = search_depth

    def _get_active_engine_path(self) -> str | None:
        if self._explicit_stockfish_path is not None:
            return self._explicit_stockfish_path if os.path.exists(self._explicit_stockfish_path) else None
        if self.stockfish_path and os.path.exists(self.stockfish_path):
            return self.stockfish_path
        self.stockfish_path = resolve_stockfish_path()
        return self.stockfish_path

    def evaluate_style_bonus(self, board_before: chess.Board, move: chess.Move) -> tuple[float, str | None]:
        """Переопределяется в подклассах для придания уникального шахматного стиля."""
        return 0.0, None

    def choose_move(self, fen: str) -> dict:
        board = chess.Board(fen)
        if board.is_game_over():
            raise ValueError("Game is already over")

        # 1. Поиск хода в дебютной книге чемпиона
        fen_key = normalize_fen_key(board)
        if fen_key in self.opening_book:
            candidates = self.opening_book[fen_key]
            valid_book_moves = []
            for item in candidates:
                uci_str, weight, comment = item[0], int(item[1]), item[2]
                if uci_str == "0000":
                    continue
                try:
                    move = chess.Move.from_uci(uci_str)
                except ValueError:
                    continue
                if move in board.legal_moves:
                    valid_book_moves.append((move, max(weight, 1), comment))

            if valid_book_moves:
                _simulate_human_thinking_time(board, is_book_move=True)
                moves, weights, comments = zip(*valid_book_moves, strict=True)
                idx = random.choices(range(len(moves)), weights=weights, k=1)[0]
                chosen_move = moves[idx]
                san = board.san(chosen_move)
                board.push(chosen_move)
                return {
                    "uci": chosen_move.uci(),
                    "san": san,
                    "fen_after": board.fen(),
                    "source": "opening_book",
                    "comment": comments[idx],
                }

        _simulate_human_thinking_time(board, is_book_move=False)

        # 2. Тактический расчёт Stockfish
        engine_path = self._get_active_engine_path() if self.stockfish_path is not None else None
        base_scores = analyse_with_stockfish(
            board=board,
            engine_path=engine_path,
            skill_level=self.skill_level,
            uci_elo=self.uci_elo,
            depth=self.search_depth,
            time_limit=0.25,
            multipv=self.multipv,
        )

        used_stockfish = bool(base_scores)
        # Fallback-расчёт при отсутствии бинарника Stockfish
        if not base_scores:
            us = board.turn
            for mv in board.legal_moves:
                b1 = board.copy()
                b1.push(mv)
                if b1.is_checkmate():
                    base_scores[mv] = 100000.0
                    continue
                worst_reply_score = 100000.0
                for reply in b1.legal_moves:
                    b2 = b1.copy()
                    b2.push(reply)
                    s = static_board_eval_cp(b2, us)
                    if s < worst_reply_score:
                        worst_reply_score = s
                base_scores[mv] = worst_reply_score if b1.legal_moves.count() > 0 else 0.0

        # 3. Коридор допустимой уступки + индивидуальный стилевой бонус чемпиона
        max_tactical_cp = max(base_scores.values())
        candidate_scores = {
                               mv: cp for mv, cp in base_scores.items()
                               if cp >= max_tactical_cp - self.blunder_guard_cp
                           } or base_scores

        best_move = None
        best_total_score = -float("inf")
        best_comment = None

        for mv, tactical_cp in candidate_scores.items():
            style_bonus, note = self.evaluate_style_bonus(board, mv)
            jitter = random.uniform(-self.jitter_range, self.jitter_range)
            total_score = tactical_cp + style_bonus + jitter

            if total_score > best_total_score:
                best_total_score = total_score
                best_move = mv
                best_comment = note

        if best_move is None:
            best_move = next(iter(board.legal_moves))

        san = board.san(best_move)
        board.push(best_move)

        engine_backend = "stockfish" if used_stockfish else "fallback_minimax"
        logger.info(
            "Bot %s move %s via %s [%s] (engine=%s)",
            self.bot_id,
            san,
            self.engine_source_name,
            engine_backend,
            engine_path,
        )
        return {
            "uci": best_move.uci(),
            "san": san,
            "fen_after": board.fen(),
            "source": self.engine_source_name,
            "engine_backend": engine_backend,
            "comment": best_comment or self.default_comment,
        }

def _evaluate_pawn_structure(board: chess.Board, us: chess.Color) -> float:
    our_pawns = board.pieces(chess.PAWN, us)
    start_rank = 1 if us == chess.WHITE else 6
    pawn_files = [chess.square_file(sq) for sq in our_pawns]

    score = 0.0
    for sq in our_pawns:
        f = chess.square_file(sq)
        r = chess.square_rank(sq)
        if r != start_rank:
            score += float(r if us == chess.WHITE else (7 - r)) * 3.0
        if f < 7 and chess.square(f + 1, r) in our_pawns:
            score += 15.0
        support_rank = (r - 1) if us == chess.WHITE else (r + 1)
        # Диапазон проверяется первым: для крайних горизонталей support_rank выходит за доску.
        if 0 <= support_rank <= 7 and (
                (f > 0 and chess.square(f - 1, support_rank) in our_pawns)
                or (f < 7 and chess.square(f + 1, support_rank) in our_pawns)
        ):
            score += 10.0

    doubled_pawns = len(pawn_files) - len(set(pawn_files))
    score -= float(doubled_pawns) * 20.0
    return score

    if piece.piece_type == chess.KNIGHT and board_before.fullmove_number <= 10:
        if us == chess.WHITE:
            if move.to_square == chess.C3 and board_before.piece_at(chess.C2) == chess.Piece(chess.PAWN, chess.WHITE):
                bonus -= 55.0
            if move.to_square == chess.F3 and board_before.piece_at(chess.F2) == chess.Piece(chess.PAWN, chess.WHITE):
                bonus -= 45.0
        else:
            if move.to_square == chess.C6 and board_before.piece_at(chess.C7) == chess.Piece(chess.PAWN, chess.BLACK):
                bonus -= 55.0
            if move.to_square == chess.F6 and board_before.piece_at(chess.F7) == chess.Piece(chess.PAWN, chess.BLACK):
                bonus -= 45.0

    structure_before = _evaluate_pawn_structure(board_before, us)
    board_after = board_before.copy()
    board_after.push(move)
    structure_after = _evaluate_pawn_structure(board_after, us)
    bonus += structure_after - structure_before

    return bonus, note

# --- FILE: src/services/connection_manager.py ---
"""
Менеджер WebSocket-соединений, корректный при нескольких воркерах (процессах) и вкладках.

Состояние, которое должно быть общим для всех воркеров, хранится в Redis:
- присутствие: каждое соединение — элемент sorted set `presence:user:{uid}` с отметкой
  последнего пульса. Игрок считается офлайн, только когда живых соединений нет ни на одном
  воркере; записи упавшего воркера устаревают через PRESENCE_STALE_MS;
- зрители партий: `game:{gid}:spectators` и обратный индекс `user:{uid}:spectating`.

Доставка: сообщение сразу отправляется в локальные сокеты пользователя и всегда публикуется
в канал `user:{uid}` с id воркера-отправителя. Остальные воркеры доставляют его в свои
вкладки этого пользователя, отправитель своё сообщение пропускает (без дублей).
"""

import asyncio
import json
import logging
import time
import uuid

from fastapi import WebSocket

from src.core.instance import WORKER_ID
from src.db.redis import redis_client

logger = logging.getLogger(__name__)

PRESENCE_HEARTBEAT_S = 10
PRESENCE_STALE_MS = 35_000

SESSION_REVOKED_EVENT = "session_revoked"
WS_CLOSE_SESSION_REVOKED = 1008


def _is_session_revoked(raw: str) -> bool:
    if SESSION_REVOKED_EVENT not in raw:  # дешёвый фильтр до разбора JSON
        return False
    try:
        return json.loads(raw).get("event") == SESSION_REVOKED_EVENT
    except (ValueError, AttributeError):
        return False


SPECTATING_TTL_S = 6 * 60 * 60


def _presence_key(uid: int) -> str:
    return f"presence:user:{uid}"


def _spectators_key(gid: int) -> str:
    return f"game:{gid}:spectators"


def _spectating_key(uid: int) -> str:
    return f"user:{uid}:spectating"


def _now_ms() -> int:
    return int(time.time() * 1000)


class ConnectionManager:
    def __init__(self):
        self.worker_id = WORKER_ID
        self.active_connections: dict[int, WebSocket] = {}
        self._user_sockets: dict[int, set[WebSocket]] = {}
        # websocket -> (user_id, connection_id в Redis-присутствии)
        self._connections: dict[WebSocket, tuple[int, str]] = {}
        self.redis = redis_client
        self.pubsub = self.redis.pubsub()
        self._listener_task: asyncio.Task | None = None
        self._heartbeat_task: asyncio.Task | None = None

    # ------------------------------------------------------------------ фоновые задачи

    def _ensure_background_tasks(self) -> None:
        try:
            if self._listener_task is None or self._listener_task.done():
                self._listener_task = asyncio.create_task(self._listen_pubsub())
            if self._heartbeat_task is None or self._heartbeat_task.done():
                self._heartbeat_task = asyncio.create_task(self._heartbeat_loop())
        except RuntimeError:
            pass

    async def _heartbeat_loop(self):
        """Обновляет отметки присутствия всех локальных соединений, пока они открыты."""
        while True:
            try:
                await asyncio.sleep(PRESENCE_HEARTBEAT_S)
                now = _now_ms()
                for uid, conn_id in list(self._connections.values()):
                    await self.redis.zadd(_presence_key(uid), {conn_id: now})
                    await self.redis.expire(_presence_key(uid), PRESENCE_STALE_MS // 1000 * 2)
            except asyncio.CancelledError:
                raise
            except Exception as exc:
                logger.warning("Presence heartbeat failed: %s", exc)

    async def _listen_pubsub(self):
        """Доставляет в локальные сокеты сообщения, опубликованные другими воркерами."""
        while True:
            try:
                if not getattr(self.pubsub, "subscribed", False):
                    await asyncio.sleep(0.25)
                    continue

                message = await self.pubsub.get_message(ignore_subscribe_messages=True, timeout=1.0)
                if not message or message.get("type") != "message":
                    continue
                channel = str(message.get("channel", ""))
                if not channel.startswith("user:"):
                    continue
                raw = self._unwrap_envelope(str(message.get("data", "")))
                if raw is not None:
                    await self._on_remote_message(int(channel.split(":")[1]), raw)
            except asyncio.CancelledError:
                raise
            except Exception as e:
                logger.debug("PubSub listener loop warning: %s", e)
                await asyncio.sleep(0.5)

    async def _on_remote_message(self, uid: int, raw: str) -> None:
        """Сообщение для пользователя, опубликованное другим воркером."""
        await self._deliver_local(uid, raw)
        if _is_session_revoked(raw):
            # Токены отозваны на другом воркере — закрываем и здешние вкладки.
            await self._close_local_sockets(uid)

    def _unwrap_envelope(self, data: str) -> str | None:
        """Возвращает полезную нагрузку или None, если сообщение опубликовал этот же воркер."""
        try:
            envelope = json.loads(data)
        except ValueError:
            return data
        if isinstance(envelope, dict) and "origin" in envelope and "data" in envelope:
            if envelope["origin"] == self.worker_id:
                return None
            return str(envelope["data"])
        return data

    # ------------------------------------------------------------------ присутствие

    def _has_open_sockets(self, uid: int) -> bool:
        return bool(self._user_sockets.get(uid))

    async def _presence_add(self, uid: int, conn_id: str) -> None:
        try:
            await self.redis.zadd(_presence_key(uid), {conn_id: _now_ms()})
            await self.redis.expire(_presence_key(uid), PRESENCE_STALE_MS // 1000 * 2)
        except Exception as exc:
            logger.warning("Presence add failed for user %s: %s", uid, exc)

    async def _presence_remove(self, uid: int, conn_id: str) -> None:
        try:
            await self.redis.zrem(_presence_key(uid), conn_id)
        except Exception as exc:
            logger.warning("Presence remove failed for user %s: %s", uid, exc)

    async def is_user_online(self, user_id: int) -> bool:
        """Есть ли у пользователя живое соединение на любом воркере."""
        uid = int(user_id)
        if self._has_open_sockets(uid):
            return True
        try:
            key = _presence_key(uid)
            await self.redis.zremrangebyscore(key, 0, _now_ms() - PRESENCE_STALE_MS)
            return int(await self.redis.zcard(key)) > 0
        except Exception as exc:
            logger.warning("Presence check failed for user %s: %s", uid, exc)
            return False

    # ------------------------------------------------------------------ соединения

    async def connect(self, websocket: WebSocket, user_id: int):
        uid = int(user_id)
        await websocket.accept()
        conn_id = uuid.uuid4().hex
        self._connections[websocket] = (uid, conn_id)
        self.active_connections[uid] = websocket
        self._user_sockets.setdefault(uid, set()).add(websocket)
        await self._presence_add(uid, conn_id)

        try:
            await self.pubsub.subscribe(f"user:{uid}")
        except Exception as exc:
            logger.warning("PubSub subscribe warning for user %s: %s", uid, exc)
        self._ensure_background_tasks()

        from src.services.game_state import state_manager
        await state_manager.handle_player_reconnect(uid, self)

        logger.info("User %s connected. Active users on worker: %s", uid, len(self.active_connections))

    async def disconnect(self, user_id: int, websocket: WebSocket | None = None):
        """
        Закрывает одно соединение пользователя. Пока у него открыта хотя бы одна вкладка на
        любом воркере, он считается онлайн: очередь, зрительство и партия не затрагиваются.
        websocket=None закрывает все локальные соединения пользователя.

        Между await-ами пользователь может переподключиться (перезагрузка страницы, в том числе
        на другой воркер), поэтому присутствие перепроверяется. Новое соединение сначала
        регистрируется в Redis и только потом вызывает handle_player_reconnect, поэтому при
        любом порядке событий флаг отключения либо не ставится, либо снимается.
        """
        uid = int(user_id)

        closing = list(self._user_sockets.pop(uid, set())) if websocket is None else [websocket]
        for ws in closing:
            sockets = self._user_sockets.get(uid)
            if sockets is not None:
                sockets.discard(ws)
            conn = self._connections.pop(ws, None)
            if conn is not None:
                await self._presence_remove(uid, conn[1])

        if self._has_open_sockets(uid):
            self.active_connections[uid] = next(iter(self._user_sockets[uid]))
            return

        self._user_sockets.pop(uid, None)
        self.active_connections.pop(uid, None)

        try:
            await self.pubsub.unsubscribe(f"user:{uid}")
        except Exception:
            pass

        if self._has_open_sockets(uid):
            # Переподключился на этом воркере во время отписки — восстанавливаем подписку.
            try:
                await self.pubsub.subscribe(f"user:{uid}")
            except Exception as exc:
                logger.warning("PubSub resubscribe warning for user %s: %s", uid, exc)
            return

        if await self.is_user_online(uid):
            return  # открыта вкладка на другом воркере

        await self.remove_user_from_all_spectating(uid)

        from src.services.matchmaker import matchmaker
        await matchmaker.leave_queue(uid)

        if await self.is_user_online(uid):
            return

        from src.services.game_state import state_manager
        await state_manager.handle_player_disconnect(uid, self)

        if await self.is_user_online(uid):
            await state_manager.handle_player_reconnect(uid, self)
            return

        logger.info("User %s disconnected.", uid)

    # ------------------------------------------------------------------ доставка

    async def _deliver_local(self, user_id: int, raw_text: str) -> bool:
        """Мгновенно отправляет JSON во все открытые на этом воркере вкладки пользователя."""
        uid = int(user_id)
        sockets = list(self._user_sockets.get(uid, set()))

        delivered = False
        dead_sockets: list[WebSocket] = []

        for ws in sockets:
            try:
                await ws.send_text(raw_text)
                delivered = True
            except Exception:
                dead_sockets.append(ws)

        for dead_ws in dead_sockets:
            if uid in self._user_sockets:
                self._user_sockets[uid].discard(dead_ws)
            if self.active_connections.get(uid) is dead_ws:
                self.active_connections.pop(uid, None)

        return delivered

    async def revoke_sessions(self, user_id: int) -> None:
        """
        Токены пользователя отозваны (смена пароля, «выйти на всех устройствах»): сообщаем всем
        его вкладкам и закрываем их WebSocket — на этом воркере сразу, на остальных при получении
        события через pub/sub. Иначе уже открытое соединение продолжало бы работать со старым
        токеном. Код 1008 клиент понимает как «сессия недействительна, войдите заново».
        """
        await self.send_personal_message({"event": SESSION_REVOKED_EVENT}, user_id)
        await self._close_local_sockets(int(user_id))

    async def _close_local_sockets(self, uid: int) -> None:
        for ws in list(self._user_sockets.get(uid, set())):
            try:
                await ws.close(code=WS_CLOSE_SESSION_REVOKED)
            except Exception:
                pass  # сокет уже закрыт; его уберёт disconnect из обработчика соединения

    async def send_personal_message(self, message: dict, user_id: int):
        """
        Доставляет сообщение во вкладки пользователя на этом воркере и публикует его для
        остальных воркеров (у пользователя могут быть вкладки на нескольких процессах).
        """
        uid = int(user_id)
        payload_str = json.dumps(message)
        await self._deliver_local(uid, payload_str)

        envelope = json.dumps({"origin": self.worker_id, "data": payload_str})
        try:
            await self.redis.publish(f"user:{uid}", envelope)
        except Exception as exc:
            logger.error("Failed to publish message for user %s: %s", uid, exc)

    # ------------------------------------------------------------------ зрители

    async def add_spectator(self, game_id: int, user_id: int) -> None:
        gid, uid = int(game_id), int(user_id)
        await self.remove_user_from_all_spectating(uid)
        await self.redis.sadd(_spectators_key(gid), str(uid))
        await self.redis.expire(_spectators_key(gid), SPECTATING_TTL_S)
        await self.redis.set(_spectating_key(uid), str(gid), ex=SPECTATING_TTL_S)

    async def remove_spectator(self, game_id: int, user_id: int) -> None:
        gid, uid = int(game_id), int(user_id)
        await self.redis.srem(_spectators_key(gid), str(uid))
        if await self.redis.get(_spectating_key(uid)) == str(gid):
            await self.redis.delete(_spectating_key(uid))

    async def remove_user_from_all_spectating(self, user_id: int) -> None:
        uid = int(user_id)
        gid = await self.redis.get(_spectating_key(uid))
        if gid:
            await self.redis.srem(_spectators_key(int(gid)), str(uid))
            await self.redis.delete(_spectating_key(uid))

    async def get_spectator_ids(self, game_id: int) -> set[int]:
        members = await self.redis.smembers(_spectators_key(int(game_id)))
        return {int(m) for m in members}

    async def broadcast_to_spectators(self, game_id: int, message: dict) -> None:
        for uid in await self.get_spectator_ids(game_id):
            await self.send_personal_message(message, uid)

"""
Пул долгоживущих процессов Stockfish.

Раньше каждый запрос к боту или анализу запускал новый процесс движка (popen_uci) — это
сотни миллисекунд на старт и неограниченное число параллельных процессов под нагрузкой.
Пул держит не более ENGINE_POOL_SIZE процессов на один бинарник, переиспользует их между
запросами и ограничивает число одновременных расчётов: при исчерпании пула ожидающий
вызов получает EngineBusyError по таймауту, а не порождает новый процесс.

Пул синхронный и потокобезопасный: им пользуются из asyncio.to_thread (API) и из Celery.
"""

import atexit
import logging
import os
import queue
import threading
from collections.abc import Iterator
from contextlib import contextmanager

import chess.engine

from src.core.config import settings

logger = logging.getLogger(__name__)

# Опции, которые боты меняют под свой стиль. Перед каждой выдачей движка они сбрасываются
# к значениям полной силы, чтобы настройки одного вызова не «протекали» в следующий.
_DEFAULT_OPTIONS: dict[str, int | bool] = {
    "Skill Level": 20,
    "UCI_LimitStrength": False,
}


class EngineUnavailableError(RuntimeError):
    """Бинарник Stockfish отсутствует или не запускается."""


class EngineBusyError(RuntimeError):
    """Все движки пула заняты дольше допустимого времени ожидания."""


class EnginePool:
    def __init__(self, engine_path: str, size: int, acquire_timeout_s: float):
        self.engine_path = engine_path
        self.size = max(1, int(size))
        self.acquire_timeout_s = acquire_timeout_s
        self._slots = threading.BoundedSemaphore(self.size)
        self._idle: queue.LifoQueue[chess.engine.SimpleEngine] = queue.LifoQueue()
        self._closed = False

    def _spawn(self) -> chess.engine.SimpleEngine:
        if not os.path.exists(self.engine_path):
            raise EngineUnavailableError(f"Stockfish not found: {self.engine_path}")
        try:
            return chess.engine.SimpleEngine.popen_uci(self.engine_path)
        except Exception as exc:
            raise EngineUnavailableError(f"Cannot start Stockfish ({self.engine_path}): {exc}") from exc

    @staticmethod
    def _quit(engine: chess.engine.SimpleEngine) -> None:
        try:
            engine.quit()
        except Exception:
            try:
                engine.close()
            except Exception:
                pass

    @staticmethod
    def _is_alive(engine: chess.engine.SimpleEngine) -> bool:
        """Процесс движка ещё работает (вызывающий код мог проглотить EngineTerminatedError)."""
        try:
            return not engine.protocol.returncode.done()
        except Exception:
            return False

    @staticmethod
    def _configure(engine: chess.engine.SimpleEngine, options: dict) -> None:
        merged = {**_DEFAULT_OPTIONS, **options}
        supported: dict = {}
        for name, value in merged.items():
            option = engine.options.get(name)
            if option is None:
                continue
            # Числовые (spin) опции приводим к диапазону конкретной сборки движка.
            if option.type == "spin" and isinstance(value, (int, float)) and not isinstance(value, bool):
                if option.min is not None:
                    value = max(option.min, value)
                if option.max is not None:
                    value = min(option.max, value)
                value = int(value)
            supported[name] = value
        if supported:
            engine.configure(supported)

    @contextmanager
    def acquire(self, options: dict | None = None, timeout_s: float | None = None) -> Iterator[chess.engine.SimpleEngine]:
        """
        Выдаёт движок в монопольное пользование на время блока with.

        options — UCI-опции поверх значений по умолчанию (полная сила).
        Движок, упавший с ошибкой протокола или по таймауту, не возвращается в пул, а
        завершается; вместо него при следующем запросе будет запущен новый процесс.
        """
        if self._closed:
            raise EngineUnavailableError("Engine pool is closed")

        wait = self.acquire_timeout_s if timeout_s is None else timeout_s
        if not self._slots.acquire(timeout=wait):
            raise EngineBusyError("All chess engines are busy")

        engine: chess.engine.SimpleEngine | None = None
        healthy = True
        try:
            try:
                engine = self._idle.get_nowait()
            except queue.Empty:
                engine = self._spawn()

            self._configure(engine, options or {})
            yield engine
        except (chess.engine.EngineError, TimeoutError):
            healthy = False
            raise
        finally:
            if engine is not None:
                if healthy and not self._closed and self._is_alive(engine):
                    self._idle.put(engine)
                else:
                    self._quit(engine)
            self._slots.release()

    def close(self) -> None:
        self._closed = True
        while True:
            try:
                engine = self._idle.get_nowait()
            except queue.Empty:
                break
            self._quit(engine)


_pools: dict[str, EnginePool] = {}
_pools_lock = threading.Lock()


def get_engine_pool(engine_path: str | None) -> EnginePool:
    """Возвращает пул для указанного бинарника (по одному пулу на путь, создаётся лениво)."""
    if not engine_path:
        raise EngineUnavailableError("Stockfish path is not configured")
    key = os.path.abspath(engine_path)
    with _pools_lock:
        pool = _pools.get(key)
        if pool is None or pool._closed:
            pool = EnginePool(key, settings.ENGINE_POOL_SIZE, settings.ENGINE_ACQUIRE_TIMEOUT_S)
            _pools[key] = pool
        return pool


def close_all_engine_pools() -> None:
    with _pools_lock:
        pools = list(_pools.values())
        _pools.clear()
    for pool in pools:
        pool.close()


# python-chess держит для каждого SimpleEngine фоновый НЕ-daemon поток. Интерпретатор ждёт
# такие потоки до вызова обычных atexit-обработчиков, поэтому процесс (Celery-воркер, pytest)
# с открытым пулом зависал бы на выходе. threading._register_atexit срабатывает раньше этого
# ожидания (так же завершается concurrent.futures); atexit — запасной вариант.
_register_early_exit_hook = getattr(threading, "_register_atexit", None)
if _register_early_exit_hook is not None:
    _register_early_exit_hook(close_all_engine_pools)
else:
    atexit.register(close_all_engine_pools)
