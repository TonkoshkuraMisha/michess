class ConnectionManager:
    """
    Scalable WebSocket manager powered by Redis Pub/Sub.
    Handles personal messages and global lobby broadcasts across multiple processes.
    """

    def __init__(self):
        self.game_spectators: dict[int, set[int]] = {}
        self.active_connections: dict[int, WebSocket] = {}
        self.redis = redis_client
        self.pubsub = self.redis.pubsub()
        self._listener_task: asyncio.Task | None = None

    async def connect(self, websocket: WebSocket, user_id: int):
        await websocket.accept()
        self.active_connections[user_id] = websocket

        await self.pubsub.subscribe(f"user:{user_id}")

        if self._listener_task is None:
            self._listener_task = asyncio.create_task(self._listen_pubsub())

        # Уведомляем менеджер состояния о переподключении игрока
        from src.services.game_state import state_manager
        await state_manager.handle_player_reconnect(user_id, self)

        logger.info("User %s connected. Active on this worker: %s", user_id, len(self.active_connections))

    async def disconnect(self, user_id: int):
        if user_id in self.active_connections:
            del self.active_connections[user_id]

        await self.pubsub.unsubscribe(f"user:{user_id}")

        # Снимаем игрока с поиска, если он там был
        from src.services.matchmaker import matchmaker
        await matchmaker.leave_queue(user_id)

        # Уведомляем менеджер состояния об обрыве связи (запуск Grace Period)
        from src.services.game_state import state_manager
        await state_manager.handle_player_disconnect(user_id, self)

        logger.info("User %s disconnected.", user_id)

    async def _listen_pubsub(self):
        """Фоновый таск, который слушает Redis Pub/Sub и прокидывает сообщения в нужные WebSocket-ы."""
        try:
            async for message in self.pubsub.listen():
                if message["type"] == "message":
                    channel = message["channel"]
                    data = message["data"]

                    if channel.startswith("user:"):
                        user_id = int(channel.split(":")[1])
                        if user_id in self.active_connections:
                            ws = self.active_connections[user_id]
                            await ws.send_text(data)
        except Exception as e:
            logger.error("PubSub listener error: %s", e)

    async def send_personal_message(self, message: dict, user_id: int):
        """Отправляет сообщение пользователю через Redis."""
        await self.redis.publish(f"user:{user_id}", json.dumps(message))



    def add_spectator(self, game_id: int, user_id: int) -> None:
        if not hasattr(self, "game_spectators"):
            self.game_spectators: dict[int, set[int]] = {}
        self.game_spectators.setdefault(int(game_id), set()).add(int(user_id))

    def remove_spectator(self, game_id: int, user_id: int) -> None:
        if not hasattr(self, "game_spectators"):
            return
        gid = int(game_id)
        if gid in self.game_spectators:
            self.game_spectators[gid].discard(int(user_id))
            if not self.game_spectators[gid]:
                del self.game_spectators[gid]

    def remove_user_from_all_spectating(self, user_id: int) -> None:
        if not hasattr(self, "game_spectators"):
            return
        uid = int(user_id)
        empty_games = []
        for gid, watchers in self.game_spectators.items():
            watchers.discard(uid)
            if not watchers:
                empty_games.append(gid)
        for gid in empty_games:
            del self.game_spectators[gid]

    async def broadcast_to_spectators(self, game_id: int, message: dict) -> None:
        if not hasattr(self, "game_spectators"):
            return
        watchers = list(self.game_spectators.get(int(game_id), set()))
        for uid in watchers:
            await self.send_personal_message(message, uid)


manager = ConnectionManager()

import chess

STARTING_FEN = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"


class GameEngine:
    @staticmethod
    def process_move(fen: str, move_str: str, current_pgn: str = "") -> tuple[bool, dict | None]:
        """
        Проверяет ход, применяет его к текущей позиции (FEN) и возвращает новые данные.

        Возвращает: (is_valid, move_data)
        Где move_data содержит:
        - new_fen: str (Новое состояние доски)
        - new_pgn: str (Обновленная строка ходов PGN)
        - san: str (Красивая нотация, например "Nf3")
        - uci: str (Универсальная нотация, например "g1f3")
        - fullmove_number: int (Номер хода)
        - is_obvious: bool (Флаг единственного легального хода для античита)
        - is_game_over: bool
        - result: str | None ("1-0", "0-1", "1/2-1/2")
        - termination: str | None ("checkmate", "stalemate", "repetition", etc.)
        """
        safe_fen = fen.strip() if fen and fen.strip() else STARTING_FEN
        try:
            board = chess.Board(safe_fen)
        except ValueError:
            return False, None

        # 1. Пытаемся распарсить ход (UCI или SAN)
        move = None
        try:
            candidate = chess.Move.from_uci(move_str)
            if candidate in board.legal_moves:
                move = candidate
        except ValueError:
            pass

        if move is None:
            try:
                candidate = board.parse_san(move_str)
                if candidate in board.legal_moves:
                    move = candidate
            except ValueError:
                return False, None

        if move is None or move not in board.legal_moves:
            return False, None

        # 2. Собираем метаданные ДО выполнения хода
        legal_moves_count = board.legal_moves.count()
        is_obvious = legal_moves_count == 1
        is_white_turn = board.turn == chess.WHITE
        fullmove_number = board.fullmove_number

        san = board.san(move)
        uci = move.uci()

        # Формируем PGN-строку за O(1) без перестроения всего дерева
        pgn_base = (current_pgn or "").strip()
        if is_white_turn:
            move_token = f"{fullmove_number}. {san}"
        else:
            move_token = san if pgn_base else f"{fullmove_number}... {san}"
        new_pgn = f"{pgn_base} {move_token}".strip()

        # 3. Применяем ход на доске
        board.push(move)
        new_fen = board.fen()

        # 4. Проверяем условия завершения партии
        is_game_over = board.is_game_over(claim_draw=True)
        result = None
        termination = None

        if is_game_over:
            res = board.result(claim_draw=True)
            result = res if res != "*" else None

            if board.is_checkmate():
                termination = "checkmate"
            elif board.is_stalemate():
                termination = "stalemate"
            elif board.is_insufficient_material():
                termination = "insufficient_material"
            elif board.is_fifty_moves():
                termination = "50_move_rule"
            elif board.is_repetition():
                termination = "repetition"
            else:
                termination = "draw"

        return True, {
            "new_fen": new_fen,
            "new_pgn": new_pgn,
            "san": san,
            "uci": uci,
            "fullmove_number": fullmove_number,
            "is_obvious": is_obvious,
            "is_game_over": is_game_over,
            "result": result,
            "termination": termination
        }


engine = GameEngine()

def _detect_eco_from_moves(moves_uci: list[str]) -> str:
    prefix = " ".join(moves_uci[:6])
    if prefix.startswith("e2e4 e7e5 g1f3 d7d6"):
        return "C41"
    if prefix.startswith("e2e4 e7e5 f2f4"):
        return "C30"
    if prefix.startswith("e2e4 e7e5 f1c4"):
        return "C23"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1c4 f8c5 b2b4"):
        return "C51"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1c4"):
        return "C50"
    if prefix.startswith("e2e4 e7e5 g1f3 b8c6 f1b5"):
        return "C60"
    if prefix.startswith("e2e4 c7c5"):
        return "B20"
    if prefix.startswith("e2e4 e7e6"):
        return "C00"
    if prefix.startswith("c2c4"):
        return "A10"
    if prefix.startswith("g1f3"):
        return "A04"
    if prefix.startswith("d2d4 d7d5 c2c4"):
        return "D06"
    if prefix.startswith("d2d4"):
        return "D00"
    if prefix.startswith("e2e4"):
        return "B00"
    return "A00"


class GameStateManager:
    def __init__(self):
        self.redis = redis_client

    def _get_key(self, game_id: int) -> str:
        return f"game:{game_id}:state"

    async def initialize_game(self, game_id: int, white_id: int, black_id: int, base_time_ms: int, increment_ms: int):
        now_ms = int(time.time() * 1000)
        state = {
            "white_id": str(white_id),
            "black_id": str(black_id),
            "board_fen": STARTING_FEN,
            "pgn": "",
            "state_version": "0",
            "white_time_ms": str(base_time_ms),
            "black_time_ms": str(base_time_ms),
            "base_time_ms": str(base_time_ms),
            "increment_ms": str(increment_ms),
            "last_move_at": str(now_ms),
            "turn": "white",
            "status": "in_progress",
            "white_disconnected": "false",
            "black_disconnected": "false",
            "white_disconnect_started_at": "0",
            "black_disconnect_started_at": "0",
            "draw_offer": "",
            "rematch_offer": "",
        }
        await self.redis.hset(self._get_key(game_id), mapping=state)
        await self.redis.sadd("active_games", str(game_id))
        await self.redis.set(f"user:{white_id}:current_game", str(game_id))
        await self.redis.set(f"user:{black_id}:current_game", str(game_id))

    async def get_game_state(self, game_id: int) -> dict | None:
        state = await self.redis.hgetall(self._get_key(game_id))
        if not state:
            return await self._recover_state_from_db(game_id)

        int_keys = [
            "white_id", "black_id", "white_time_ms", "black_time_ms",
            "base_time_ms", "increment_ms", "last_move_at",
            "white_disconnect_started_at", "black_disconnect_started_at",
            "state_version",
        ]
        for key in int_keys:
            if key in state and state[key] is not None and str(state[key]).lstrip("-").isdigit():
                state[key] = int(state[key])
        if "board_fen" not in state or not state["board_fen"]:
            state["board_fen"] = STARTING_FEN
        return state

    async def _recover_state_from_db(self, game_id: int) -> dict | None:
        """Восстанавливает состояние партии в Redis из PostgreSQL."""
        try:
            async with AsyncSessionLocal() as db:
                db_game = await db.get(Game, game_id)
                if not db_game:
                    return None

                now_ms = int(time.time() * 1000)
                turn_str = "white" if db_game.turn in ("w", "white") else "black"
                white_time = db_game.white_time_ms if db_game.white_time_ms is not None else db_game.base_time_ms
                black_time = db_game.black_time_ms if db_game.black_time_ms is not None else db_game.base_time_ms
                last_move = db_game.last_move_at if db_game.last_move_at is not None else now_ms

                status_map = {
                    GameStatus.IN_PROGRESS: "in_progress",
                    GameStatus.COMPLETED: "completed",
                    GameStatus.ABORTED: "aborted",
                    GameStatus.PENDING: "pending",
                }
                status_str = status_map.get(db_game.status, "completed")

                state = {
                    "white_id": str(db_game.white_player_id),
                    "black_id": str(db_game.black_player_id),
                    "board_fen": db_game.board_fen or STARTING_FEN,
                    "pgn": db_game.pgn or "",
                    "state_version": str(db_game.state_version or 0),
                    "white_time_ms": str(white_time),
                    "black_time_ms": str(black_time),
                    "base_time_ms": str(db_game.base_time_ms),
                    "increment_ms": str(db_game.increment_ms),
                    "last_move_at": str(last_move),
                    "turn": turn_str,
                    "status": status_str,
                    "white_disconnected": "false",
                    "black_disconnected": "false",
                    "white_disconnect_started_at": "0",
                    "black_disconnect_started_at": "0",
                }
                await self.redis.hset(self._get_key(game_id), mapping=state)
                if status_str == "in_progress":
                    await self.redis.sadd("active_games", str(game_id))
                    await self.redis.set(f"user:{db_game.white_player_id}:current_game", str(game_id))
                    await self.redis.set(f"user:{db_game.black_player_id}:current_game", str(game_id))
                return await self.get_game_state(game_id)
        except Exception as e:
            logger.error("Failed to recover game %s from DB: %s", game_id, e)
            return None

    async def update_game_state(self, game_id: int, updates: dict):
        str_updates = {k: str(v) for k, v in updates.items()}
        await self.redis.hset(self._get_key(game_id), mapping=str_updates)

    async def clear_game_state(self, game_id: int):
        state = await self.redis.hgetall(self._get_key(game_id))
        if state:
            await self.redis.delete(f"user:{state.get('white_id')}:current_game")
            await self.redis.delete(f"user:{state.get('black_id')}:current_game")
        await self.redis.delete(self._get_key(game_id))
        await self.redis.delete(f"game:{game_id}:commands")
        await self.redis.srem("active_games", str(game_id))

    async def save_move_telemetry(self, game_id: int, move_data: dict):
        key = f"game:{game_id}:moves"
        await self.redis.rpush(key, json.dumps(move_data))

    async def extract_all_moves(self, game_id: int) -> list[dict]:
        key = f"game:{game_id}:moves"
        moves_raw = await self.redis.lrange(key, 0, -1)
        await self.redis.delete(key)
        return [json.loads(m) for m in moves_raw]

    async def process_atomic_move(
            self,
            game_id: int,
            user_id: int,
            command_id: str,
            move_str: str,
            expected_version: int | None = None,
    ) -> dict:
        game_key = self._get_key(game_id)
        cmd_key = f"game:{game_id}:commands"

        async with self.redis.pipeline(transaction=True) as pipe:
            while True:
                try:
                    await pipe.watch(game_key, cmd_key)

                    is_processed = await pipe.sismember(cmd_key, command_id)
                    if is_processed:
                        await pipe.unwatch()
                        return {"status": "ignored", "reason": "already_processed"}

                    state = await pipe.hgetall(game_key)
                    if not state or state.get("status") != "in_progress":
                        await pipe.unwatch()
                        return {"status": "error", "reason": "game_inactive"}

                    current_version = int(state.get("state_version", 0))
                    if expected_version is not None and int(expected_version) != current_version:
                        await pipe.unwatch()
                        return {
                            "status": "error",
                            "reason": "version_mismatch",
                            "state_version": current_version,
                            "fen": state.get("board_fen", STARTING_FEN),
                            "pgn": state.get("pgn", ""),
                        }

                    turn = state.get("turn", "white")
                    white_id = int(state["white_id"])
                    black_id = int(state["black_id"])

                    is_white_turn = turn in ("white", "w")
                    expected_player_id = white_id if is_white_turn else black_id

                    if user_id != expected_player_id:
                        await pipe.unwatch()
                        return {"status": "error", "reason": "not_your_turn"}

                    now_ms = int(time.time() * 1000)
                    last_move_at = int(state["last_move_at"])
                    elapsed_ms = max(0, now_ms - last_move_at)

                    increment = int(state["increment_ms"])
                    white_time = int(state["white_time_ms"])
                    black_time = int(state["black_time_ms"])

                    if is_white_turn:
                        new_white_time = white_time - elapsed_ms
                        if new_white_time <= 0:
                            await pipe.unwatch()
                            return {
                                "status": "game_over",
                                "result": "0-1",
                                "termination": "timeout",
                                "white_id": white_id,
                                "black_id": black_id,
                                "white_time_ms": 0,
                                "black_time_ms": black_time,
                                "elapsed_ms": elapsed_ms,
                            }
                        new_white_time += increment
                        new_black_time = black_time
                    else:
                        new_black_time = black_time - elapsed_ms
                        if new_black_time <= 0:
                            await pipe.unwatch()
                            return {
                                "status": "game_over",
                                "result": "1-0",
                                "termination": "timeout",
                                "white_id": white_id,
                                "black_id": black_id,
                                "white_time_ms": white_time,
                                "black_time_ms": 0,
                                "elapsed_ms": elapsed_ms,
                            }
                        new_black_time += increment
                        new_white_time = white_time

                    current_fen = state.get("board_fen") or STARTING_FEN
                    current_pgn = state.get("pgn") or ""

                    is_valid, move_data = engine.process_move(current_fen, move_str, current_pgn)
                    if not is_valid or not move_data:
                        await pipe.unwatch()
                        return {"status": "error", "reason": "illegal_move"}

                    new_version = current_version + 1
                    new_turn = "black" if is_white_turn else "white"

                    pipe.multi()

                    if move_data["is_game_over"]:
                        updates = {
                            "status": "completed",
                            "state_version": str(new_version),
                            "board_fen": move_data["new_fen"],
                            "pgn": move_data["new_pgn"],
                            "last_move_at": str(now_ms),
                            "white_time_ms": str(new_white_time),
                            "black_time_ms": str(new_black_time),
                        }
                        pipe.hset(game_key, mapping=updates)
                        pipe.sadd(cmd_key, command_id)
                        await pipe.execute()
                        return {
                            "status": "game_over",
                            "result": move_data["result"],
                            "termination": move_data["termination"],
                            "move_data": move_data,
                            "state_version": new_version,
                            "white_time_ms": new_white_time,
                            "black_time_ms": new_black_time,
                            "white_id": white_id,
                            "black_id": black_id,
                            "elapsed_ms": elapsed_ms,
                            "now_ms": now_ms,
                        }

                    updates = {
                        "state_version": str(new_version),
                        "turn": new_turn,
                        "board_fen": move_data["new_fen"],
                        "pgn": move_data["new_pgn"],
                        "last_move_at": str(now_ms),
                        "white_time_ms": str(new_white_time),
                        "black_time_ms": str(new_black_time),
                    }

                    pipe.hset(game_key, mapping=updates)
                    pipe.sadd(cmd_key, command_id)
                    pipe.expire(cmd_key, 86400)

                    await pipe.execute()

                    return {
                        "status": "success",
                        "move_data": move_data,
                        "state_version": new_version,
                        "turn": new_turn,
                        "white_time_ms": new_white_time,
                        "black_time_ms": new_black_time,
                        "white_id": white_id,
                        "black_id": black_id,
                        "elapsed_ms": elapsed_ms,
                        "now_ms": now_ms,
                    }

                except WatchError:
                    continue

    async def sync_authoritative_state_to_db(self, db, game_id: int, res: dict):

    async def finish_game(self, db, game_id: int, result: str, termination: str = "unknown",
                          final_pgn: str = None) -> dict:
        db_game = await db.get(Game, game_id)
        if not db_game:
            return {}

        if db_game.status == GameStatus.COMPLETED:
            return {
                "white_delta": db_game.white_rating_delta or 0,
                "black_delta": db_game.black_rating_delta or 0
            }

        white_user = await db.get(User, db_game.white_player_id)
        black_user = await db.get(User, db_game.black_player_id)

        white_initial = white_user.rating
        black_initial = black_user.rating
        delta_w = delta_b = 0
        new_w = white_initial
        new_b = black_initial

        if db_game.is_rated:
            new_w, delta_w, new_b, delta_b = elo_calculator.calculate_new_ratings(
                white_initial, black_initial, result
            )
            white_user.rating = new_w
            black_user.rating = new_b

        db_game.result = result
        db_game.termination = termination

        if result == "1-0":
            db_game.winner_id = white_user.id
        elif result == "0-1":
            db_game.winner_id = black_user.id

        db_game.white_initial_rating = white_initial
        db_game.black_initial_rating = black_initial
        db_game.white_rating_delta = delta_w
        db_game.black_rating_delta = delta_b

        state = await self.redis.hgetall(self._get_key(game_id))
        if state:
            if state.get("state_version"):
                db_game.state_version = int(state["state_version"])
            if state.get("board_fen"):
                db_game.board_fen = state["board_fen"]
            if state.get("white_time_ms"):

    async def check_timeouts_loop(self, manager):
        while True:
            try:
                active_game_ids = list(await self.redis.smembers("active_games"))
                now_ms = int(time.time() * 1000)

                for game_id_str in active_game_ids:
                    game_id = int(game_id_str)
                    state = await self.get_game_state(game_id)
                    if not state or state["status"] != "in_progress":
                        await self.redis.srem("active_games", game_id_str)
                        continue

                    # 1. Авто-отмена партии при неявке на 1-м ходу (30 секунд)
                    if int(state.get("state_version", 0)) < 2 and now_ms - int(
                            state.get("last_move_at", now_ms)) >= 30000:
                        async with AsyncSessionLocal() as db:
                            await self.check_first_move_abort_timeout(db, game_id, manager=manager, timeout_ms=30000)
                        continue

                    white_id, black_id = int(state["white_id"]), int(state["black_id"])
                    turn = state.get("turn", "white")
                    game_result, winner_id, reason = None, None, "timeout"

                    for color in ["white", "black"]:
                        if state.get(f"{color}_disconnected") == "true":
                            offline_duration = now_ms - state.get(f"{color}_disconnect_started_at", now_ms)
                            remaining_time = state["white_time_ms"] if color == "white" else state["black_time_ms"]
                            current_player_time = remaining_time - offline_duration if turn == color else remaining_time

                            if current_player_time <= 0 or offline_duration >= min(remaining_time, 3 * 60 * 1000):
                                game_result = "0-1" if color == "white" else "1-0"
                                winner_id = black_id if color == "white" else white_id
                                reason = "abandoned"
                                break

                    if not game_result:
                        elapsed = now_ms - state["last_move_at"]
                        if turn in ("white", "w") and state.get("white_disconnected") != "true" and (
                                state["white_time_ms"] - elapsed <= 0
                        ):
                            game_result, winner_id = "0-1", black_id
                            reason = "timeout"
                        elif turn in ("black", "b") and state.get("black_disconnected") != "true" and (
                                state["black_time_ms"] - elapsed <= 0
                        ):
                            game_result, winner_id = "1-0", white_id
                            reason = "timeout"

                    if game_result:
                        logger.info("Game %s ended. Result: %s, Reason: %s", game_id, game_result, reason)

                        async with AsyncSessionLocal() as db:
                            rating_updates = await self.finish_game(db, game_id, game_result, termination=reason)

                        game_over_payload = {
                            "event": "game_over",
                            "game_id": game_id,
                            "reason": reason,
                            "winner_id": winner_id,
                            "result": game_result,
                        }
                        game_over_payload.update(rating_updates)

                        await manager.send_personal_message(game_over_payload, white_id)
                        await manager.send_personal_message(game_over_payload, black_id)
                        if hasattr(manager, "broadcast_to_spectators"):
                            await manager.broadcast_to_spectators(game_id, game_over_payload)

            except Exception as e:
                logger.error("Error in check_timeouts_loop: %s", e)

            await asyncio.sleep(0.5)

    async def abort_game(self, db, game_id: int) -> dict:
        """Отменяет партию на 1-м ходу (до 2 полуходов) без изменения рейтинга."""
        state = await self.get_game_state(game_id)
        if not state or state.get("status") != "in_progress":
            return {"status": "error", "reason": "Game is not in progress"}

        state_version = int(state.get("state_version", 0))
        if state_version >= 2:
            return {"status": "error", "reason": "Cannot abort game after move 1"}

        white_id = int(state["white_id"])
        black_id = int(state["black_id"])

        await self.redis.hset(f"game:{game_id}:state", mapping={
            "status": "aborted",
            "result": "*",
            "termination": "aborted",
        })
        await self.redis.srem("active_games", str(game_id))
        await self.redis.delete(f"user:{white_id}:current_game")
        await self.redis.delete(f"user:{black_id}:current_game")

        from datetime import datetime, timezone

        db_game = await db.get(Game, game_id)
        white_user = await db.get(User, white_id)
        black_user = await db.get(User, black_id)

        w_rating = white_user.rating if white_user else 1200
        b_rating = black_user.rating if black_user else 1200

        if db_game:
            db_game.status = GameStatus.ABORTED
            db_game.result = "*"
            db_game.termination = "aborted"
            db_game.white_rating_delta = 0
            db_game.black_rating_delta = 0
            is_tz = getattr(Game.__table__.c.finished_at.type, "timezone", False)
            now_utc = datetime.now(timezone.utc)
            db_game.finished_at = now_utc if is_tz else now_utc.replace(tzinfo=None)
            await db.commit()

        return {
            "status": "aborted",
            "result": "*",
            "termination": "aborted",
            "white_id": white_id,
            "black_id": black_id,
            "white_delta": 0,
            "black_delta": 0,
            "white_new_rating": w_rating,
            "black_new_rating": b_rating,
        }

    async def check_first_move_abort_timeout(self, db, game_id: int, manager=None, timeout_ms: int = 30000) -> bool:
        """Проверяет, не истекли ли 30 секунд на первые 2 полухода партии. Если истекли — отменяет игру."""
        state = await self.get_game_state(game_id)
        if not state or state.get("status") != "in_progress":
            return False

        if int(state.get("state_version", 0)) >= 2:
            return False

        now_ms = int(time.time() * 1000)
        last_move_at = int(state.get("last_move_at", now_ms))
        if now_ms - last_move_at < timeout_ms:
            return False

        res = await self.abort_game(db, game_id)
        if res.get("status") == "aborted" and manager is not None:
            payload = {
                "event": "game_over",
                "game_id": game_id,
                "reason": "aborted",
                "result": "*",
                "white_delta": 0,
                "black_delta": 0,
                "white_new_rating": res["white_new_rating"],
                "black_new_rating": res["black_new_rating"],
            }
            await manager.send_personal_message(payload, res["white_id"])
            await manager.send_personal_message(payload, res["black_id"])
            if hasattr(manager, "broadcast_to_spectators"):
                await manager.broadcast_to_spectators(game_id, payload)
        return res.get("status") == "aborted"

class FideEloSystem:
    """
    Система подсчета рейтинга по стандартам ФИДЕ.
    Включает динамический K-фактор в зависимости от текущего рейтинга и опыта.
    """

    @staticmethod
    def get_k_factor(rating: int, games_played: int = 30) -> int:
        if games_played < 30:
            return 40
        if rating >= 2400:
            return 10
        return 20

    @staticmethod
    def expected_score(rating_a: int, rating_b: int) -> float:
        return 1 / (1 + 10 ** ((rating_b - rating_a) / 400))

    @classmethod
    def calculate_new_ratings(
        cls,
        white_rating: int,
        black_rating: int,
        result: str,
        white_games: int = 30,
        black_games: int = 30
    ) -> tuple[int, int, int, int]:
        """
        Возвращает: (новый_рейтинг_белых, дельта_белых, новый_рейтинг_черных, дельта_черных)
        """
        if result in ("1-0", "white_won_on_time"):
            score_white, score_black = 1.0, 0.0
        elif result in ("0-1", "black_won_on_time"):
            score_white, score_black = 0.0, 1.0
        elif result in ("1/2-1/2", "draw_agreement"):
            score_white, score_black = 0.5, 0.5
        else:
            raise ValueError(f"Unknown game result: {result}")

        expected_white = cls.expected_score(white_rating, black_rating)
        expected_black = cls.expected_score(black_rating, white_rating)

        k_white = cls.get_k_factor(white_rating, white_games)
        k_black = cls.get_k_factor(black_rating, black_games)

        delta_white = round(k_white * (score_white - expected_white))
        delta_black = round(k_black * (score_black - expected_black))

        # Защита от падения рейтинга ниже 100
        new_white = max(100, white_rating + delta_white)
        new_black = max(100, black_rating + delta_black)

        return new_white, delta_white, new_black, delta_black

def analyze_game_task(game_id: int):
    """Синхронная обёртка Celery для запуска асинхронного анализатора телеметрии."""

        return asyncio.run(_process_anti_cheat_async(game_id))

class MockPipeline:
    def __init__(self, redis):
        self.redis = redis
        self.commands = []

    async def __aenter__(self):
        return self

    async def __aexit__(self, exc_type, exc, tb):
        pass

    async def watch(self, *names):
        pass

    async def unwatch(self):
        pass

    def multi(self):
        self.commands = []

    def hset(self, name, key=None, value=None, mapping=None):
        self.commands.append(("hset", name, key, value, mapping))

    def sadd(self, name, *values):
        self.commands.append(("sadd", name, values))

    def expire(self, name, time):
        self.commands.append(("expire", name, time))

    async def execute(self):
        for cmd in self.commands:
            if cmd[0] == "hset":
                await self.redis.hset(cmd[1], cmd[2], cmd[3], mapping=cmd[4])
            elif cmd[0] == "sadd":
                await self.redis.sadd(cmd[1], *cmd[2])
        self.commands = []

    async def sismember(self, name, value):
        return str(value) in self.redis.sets.get(name, set())

    async def hgetall(self, name):
        return await self.redis.hgetall(name)


class MockAsyncRedis:
    """Высокопроизводительный асинхронный мок Redis в оперативной памяти для тестов."""

    def __init__(self):
        self.store = {}
        self.sets = {}
        self.lists = {}
        self.kv = {}
        self.zsets = {}

    def pubsub(self):
        return MockPubSub()

    def pipeline(self, transaction=True):
        return MockPipeline(self)

    async def hset(self, name, mapping=None, **kwargs):
        if name not in self.store:
            self.store[name] = {}
        if mapping:
            self.store[name].update({str(k): str(v) for k, v in mapping.items()})
        if kwargs:
            self.store[name].update({str(k): str(v) for k, v in kwargs.items()})
        return len(mapping or {}) + len(kwargs)

    async def hgetall(self, name):
        return self.store.get(name, {})

    async def delete(self, *names):
        count = 0
        for name in names:
            for container in (self.store, self.sets, self.lists, self.kv, self.zsets):
                if name in container:
                    del container[name]
                    count += 1
        return count

    async def sadd(self, name, *values):
        if name not in self.sets:
            self.sets[name] = set()
        added = 0
        for v in values:
            if str(v) not in self.sets[name]:
                self.sets[name].add(str(v))
                added += 1
        return added

    async def srem(self, name, *values):
        if name not in self.sets:
            return 0
        removed = 0
        for v in values:
            if str(v) in self.sets[name]:
                self.sets[name].remove(str(v))
                removed += 1
        return removed

    async def smembers(self, name):
        return self.sets.get(name, set())

    async def set(self, name, value):
        self.kv[name] = str(value)
        return True

    async def get(self, name):
        return self.kv.get(name)

    async def rpush(self, name, *values):
        if name not in self.lists:
            self.lists[name] = []
        for v in values:
            self.lists[name].append(v)
        return len(self.lists[name])

    async def lrange(self, name, start, end):
        lst = self.lists.get(name, [])
        if end == -1:
            return lst[start:]
        return lst[start:end + 1]

    # --- Поддержка Sorted Sets для очередей матчмейкинга ---
    async def zadd(self, name, mapping):
        if name not in self.zsets:
            self.zsets[name] = {}
        added = 0
        for k, v in mapping.items():
            if k not in self.zsets[name] or self.zsets[name][k] != float(v):
                added += 1
            self.zsets[name][k] = float(v)
        return added

    async def zrange(self, name, start, end):
        if name not in self.zsets:
            return []
        sorted_items = sorted(self.zsets[name].items(), key=lambda x: x[1])
        keys = [k for k, v in sorted_items]
        if end == -1:
            return keys[start:]
        return keys[start:end + 1]

    async def zrem(self, name, *values):
        if name not in self.zsets:
            return 0
        removed = 0
        for v in values:
            if v in self.zsets[name]:
                del self.zsets[name][v]
                removed += 1
        return removed

def test_calculate_accuracy_bounds_and_monotonicity():
    perfect = calculate_accuracy([0, 0, 5, 0])
    decent = calculate_accuracy([15, 35, 60, 20])
    blunder_heavy = calculate_accuracy([250, 400, 310, 500])

    assert 95.0 <= perfect <= 100.0
    assert decent < perfect
    assert blunder_heavy < decent
    assert 0.0 <= blunder_heavy <= 100.0
