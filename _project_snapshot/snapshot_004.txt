def evaluate_zukertort_style_bonus(board_before: chess.Board, move: chess.Move) -> tuple[float, str | None]:
    piece = board_before.piece_at(move.from_square)
    if not piece:
        return 0.0, None

    bonus = 0.0
    note = None

    if board_before.is_castling(move):
        bonus += 40.0
        note = "Рокировка! Завершаем развитие и готовим тактический прорыв."

    if piece.piece_type in {chess.KNIGHT, chess.BISHOP} and board_before.fullmove_number <= 14:
        back_rank = 0 if board_before.turn == chess.WHITE else 7
        if chess.square_rank(move.from_square) == back_rank:
            bonus += 28.0
            note = "Гибкое развитие фигур по системе Цукерторта."
        if move.to_square in EXTENDED_CENTER:
            bonus += 18.0

    if move.to_square in CENTER_SQUARES:
        bonus += 22.0
        if not note:
            note = "Энергичный захват ключевых центральных полей!"

    if board_before.gives_check(move):
        bonus += 26.0
        note = "Шах! Переходим к решительной комбинационной атаке в стиле Лондонского турнира 1883 года!"
    elif board_before.is_capture(move):
        bonus += 16.0

    return bonus, note

"""
Менеджер WebSocket-соединений, корректный при нескольких воркерах (процессах) и вкладках.

Состояние, которое должно быть общим для всех воркеров, хранится в Redis:
- присутствие: каждое соединение — элемент sorted set `presence:user:{uid}` с отметкой
  последнего пульса. Игрок считается офлайн, только когда живых соединений нет ни на одном
  воркере; записи упавшего воркера устаревают через PRESENCE_STALE_MS;
- зрители партий: `game:{gid}:spectators` и обратный индекс `user:{uid}:spectating`.

Доставка: сообщение сразу отправляется в локальные сокеты пользователя и всегда публикуется
в канал `user:{uid}` с id воркера-отправителя. Остальные воркеры доставляют его в свои
вкладки этого пользователя, отправитель своё сообщение пропускает (без дублей).
"""

import asyncio
import json
import logging
import time
import uuid

from fastapi import WebSocket

from src.core.instance import WORKER_ID
from src.db.redis import redis_client

logger = logging.getLogger(__name__)

PRESENCE_HEARTBEAT_S = 10
PRESENCE_STALE_MS = 35_000
SPECTATING_TTL_S = 6 * 60 * 60


def _presence_key(uid: int) -> str:
    return f"presence:user:{uid}"


def _spectators_key(gid: int) -> str:
    return f"game:{gid}:spectators"


def _spectating_key(uid: int) -> str:
    return f"user:{uid}:spectating"


def _now_ms() -> int:
    return int(time.time() * 1000)


class ConnectionManager:
    def __init__(self):
        self.worker_id = WORKER_ID
        self.active_connections: dict[int, WebSocket] = {}
        self._user_sockets: dict[int, set[WebSocket]] = {}
        # websocket -> (user_id, connection_id в Redis-присутствии)
        self._connections: dict[WebSocket, tuple[int, str]] = {}
        self.redis = redis_client
        self.pubsub = self.redis.pubsub()
        self._listener_task: asyncio.Task | None = None
        self._heartbeat_task: asyncio.Task | None = None

    # ------------------------------------------------------------------ фоновые задачи

    def _ensure_background_tasks(self) -> None:
        try:
            if self._listener_task is None or self._listener_task.done():
                self._listener_task = asyncio.create_task(self._listen_pubsub())
            if self._heartbeat_task is None or self._heartbeat_task.done():
                self._heartbeat_task = asyncio.create_task(self._heartbeat_loop())
        except RuntimeError:
            pass

    async def _heartbeat_loop(self):
        """Обновляет отметки присутствия всех локальных соединений, пока они открыты."""
        while True:
            try:
                await asyncio.sleep(PRESENCE_HEARTBEAT_S)
                now = _now_ms()
                for uid, conn_id in list(self._connections.values()):
                    await self.redis.zadd(_presence_key(uid), {conn_id: now})
                    await self.redis.expire(_presence_key(uid), PRESENCE_STALE_MS // 1000 * 2)
            except asyncio.CancelledError:
                raise
            except Exception as exc:
                logger.warning("Presence heartbeat failed: %s", exc)

    async def _listen_pubsub(self):
        """Доставляет в локальные сокеты сообщения, опубликованные другими воркерами."""
        while True:
            try:
                if not getattr(self.pubsub, "subscribed", False):
                    await asyncio.sleep(0.25)
                    continue

                message = await self.pubsub.get_message(ignore_subscribe_messages=True, timeout=1.0)
                if not message or message.get("type") != "message":
                    continue
                channel = str(message.get("channel", ""))
                if not channel.startswith("user:"):
                    continue
                raw = self._unwrap_envelope(str(message.get("data", "")))
                if raw is not None:
                    await self._deliver_local(int(channel.split(":")[1]), raw)
            except asyncio.CancelledError:
                raise
            except Exception as e:
                logger.debug("PubSub listener loop warning: %s", e)
                await asyncio.sleep(0.5)

    def _unwrap_envelope(self, data: str) -> str | None:
        """Возвращает полезную нагрузку или None, если сообщение опубликовал этот же воркер."""
        try:
            envelope = json.loads(data)
        except ValueError:
            return data
        if isinstance(envelope, dict) and "origin" in envelope and "data" in envelope:
            if envelope["origin"] == self.worker_id:
                return None
            return str(envelope["data"])
        return data

    # ------------------------------------------------------------------ присутствие

    def _has_open_sockets(self, uid: int) -> bool:
        return bool(self._user_sockets.get(uid))

    async def _presence_add(self, uid: int, conn_id: str) -> None:
        try:
            await self.redis.zadd(_presence_key(uid), {conn_id: _now_ms()})
            await self.redis.expire(_presence_key(uid), PRESENCE_STALE_MS // 1000 * 2)
        except Exception as exc:
            logger.warning("Presence add failed for user %s: %s", uid, exc)

    async def _presence_remove(self, uid: int, conn_id: str) -> None:
        try:
            await self.redis.zrem(_presence_key(uid), conn_id)
        except Exception as exc:
            logger.warning("Presence remove failed for user %s: %s", uid, exc)

    async def is_user_online(self, user_id: int) -> bool:
        """Есть ли у пользователя живое соединение на любом воркере."""
        uid = int(user_id)
        if self._has_open_sockets(uid):
            return True
        try:
            key = _presence_key(uid)
            await self.redis.zremrangebyscore(key, 0, _now_ms() - PRESENCE_STALE_MS)
            return int(await self.redis.zcard(key)) > 0
        except Exception as exc:
            logger.warning("Presence check failed for user %s: %s", uid, exc)
            return False

    # ------------------------------------------------------------------ соединения

    async def connect(self, websocket: WebSocket, user_id: int):
        uid = int(user_id)
        await websocket.accept()
        conn_id = uuid.uuid4().hex
        self._connections[websocket] = (uid, conn_id)
        self.active_connections[uid] = websocket
        self._user_sockets.setdefault(uid, set()).add(websocket)
        await self._presence_add(uid, conn_id)

        try:
            await self.pubsub.subscribe(f"user:{uid}")
        except Exception as exc:
            logger.warning("PubSub subscribe warning for user %s: %s", uid, exc)
        self._ensure_background_tasks()

        from src.services.game_state import state_manager
        await state_manager.handle_player_reconnect(uid, self)

        logger.info("User %s connected. Active users on worker: %s", uid, len(self.active_connections))

    async def disconnect(self, user_id: int, websocket: WebSocket | None = None):
        """
        Закрывает одно соединение пользователя. Пока у него открыта хотя бы одна вкладка на
        любом воркере, он считается онлайн: очередь, зрительство и партия не затрагиваются.
        websocket=None закрывает все локальные соединения пользователя.

        Между await-ами пользователь может переподключиться (перезагрузка страницы, в том числе
        на другой воркер), поэтому присутствие перепроверяется. Новое соединение сначала
        регистрируется в Redis и только потом вызывает handle_player_reconnect, поэтому при
        любом порядке событий флаг отключения либо не ставится, либо снимается.
        """
        uid = int(user_id)

        closing = list(self._user_sockets.pop(uid, set())) if websocket is None else [websocket]
        for ws in closing:
            sockets = self._user_sockets.get(uid)
            if sockets is not None:
                sockets.discard(ws)
            conn = self._connections.pop(ws, None)
            if conn is not None:
                await self._presence_remove(uid, conn[1])

        if self._has_open_sockets(uid):
            self.active_connections[uid] = next(iter(self._user_sockets[uid]))
            return

        self._user_sockets.pop(uid, None)
        self.active_connections.pop(uid, None)

        try:
            await self.pubsub.unsubscribe(f"user:{uid}")
        except Exception:
            pass

        if self._has_open_sockets(uid):
            # Переподключился на этом воркере во время отписки — восстанавливаем подписку.
            try:
                await self.pubsub.subscribe(f"user:{uid}")
            except Exception as exc:
                logger.warning("PubSub resubscribe warning for user %s: %s", uid, exc)
            return

        if await self.is_user_online(uid):
            return  # открыта вкладка на другом воркере

        await self.remove_user_from_all_spectating(uid)

        from src.services.matchmaker import matchmaker
        await matchmaker.leave_queue(uid)

        if await self.is_user_online(uid):
            return

        from src.services.game_state import state_manager
        await state_manager.handle_player_disconnect(uid, self)

        if await self.is_user_online(uid):
            await state_manager.handle_player_reconnect(uid, self)
            return

        logger.info("User %s disconnected.", uid)

    # ------------------------------------------------------------------ доставка

    async def _deliver_local(self, user_id: int, raw_text: str) -> bool:
        """Мгновенно отправляет JSON во все открытые на этом воркере вкладки пользователя."""
        uid = int(user_id)
        sockets = list(self._user_sockets.get(uid, set()))

        delivered = False
        dead_sockets: list[WebSocket] = []

        for ws in sockets:
            try:
                await ws.send_text(raw_text)
                delivered = True
            except Exception:
                dead_sockets.append(ws)

        for dead_ws in dead_sockets:
            if uid in self._user_sockets:
                self._user_sockets[uid].discard(dead_ws)
            if self.active_connections.get(uid) is dead_ws:
                self.active_connections.pop(uid, None)

        return delivered

    async def send_personal_message(self, message: dict, user_id: int):
        """
        Доставляет сообщение во вкладки пользователя на этом воркере и публикует его для
        остальных воркеров (у пользователя могут быть вкладки на нескольких процессах).
        """
        uid = int(user_id)
        payload_str = json.dumps(message)
        await self._deliver_local(uid, payload_str)

        envelope = json.dumps({"origin": self.worker_id, "data": payload_str})
        try:
            await self.redis.publish(f"user:{uid}", envelope)
        except Exception as exc:
            logger.error("Failed to publish message for user %s: %s", uid, exc)

    # ------------------------------------------------------------------ зрители

    async def add_spectator(self, game_id: int, user_id: int) -> None:
        gid, uid = int(game_id), int(user_id)
        await self.remove_user_from_all_spectating(uid)
        await self.redis.sadd(_spectators_key(gid), str(uid))
        await self.redis.expire(_spectators_key(gid), SPECTATING_TTL_S)
        await self.redis.set(_spectating_key(uid), str(gid), ex=SPECTATING_TTL_S)

    async def remove_spectator(self, game_id: int, user_id: int) -> None:
        gid, uid = int(game_id), int(user_id)
        await self.redis.srem(_spectators_key(gid), str(uid))
        if await self.redis.get(_spectating_key(uid)) == str(gid):
            await self.redis.delete(_spectating_key(uid))

    async def remove_user_from_all_spectating(self, user_id: int) -> None:
        uid = int(user_id)
        gid = await self.redis.get(_spectating_key(uid))
        if gid:
            await self.redis.srem(_spectators_key(int(gid)), str(uid))
            await self.redis.delete(_spectating_key(uid))

    async def get_spectator_ids(self, game_id: int) -> set[int]:
        members = await self.redis.smembers(_spectators_key(int(game_id)))
        return {int(m) for m in members}

    async def broadcast_to_spectators(self, game_id: int, message: dict) -> None:
        for uid in await self.get_spectator_ids(game_id):
            await self.send_personal_message(message, uid)


manager = ConnectionManager()

"""
Пул долгоживущих процессов Stockfish.

Раньше каждый запрос к боту или анализу запускал новый процесс движка (popen_uci) — это
сотни миллисекунд на старт и неограниченное число параллельных процессов под нагрузкой.
Пул держит не более ENGINE_POOL_SIZE процессов на один бинарник, переиспользует их между
запросами и ограничивает число одновременных расчётов: при исчерпании пула ожидающий
вызов получает EngineBusyError по таймауту, а не порождает новый процесс.

Пул синхронный и потокобезопасный: им пользуются из asyncio.to_thread (API) и из Celery.
"""

import atexit
import logging
import os
import queue
import threading
from contextlib import contextmanager
from typing import Iterator

import chess.engine

from src.core.config import settings

logger = logging.getLogger(__name__)

# Опции, которые боты меняют под свой стиль. Перед каждой выдачей движка они сбрасываются
# к значениям полной силы, чтобы настройки одного вызова не «протекали» в следующий.
_DEFAULT_OPTIONS: dict[str, int | bool] = {
    "Skill Level": 20,
    "UCI_LimitStrength": False,
}


class EngineUnavailableError(RuntimeError):
    """Бинарник Stockfish отсутствует или не запускается."""


class EngineBusyError(RuntimeError):
    """Все движки пула заняты дольше допустимого времени ожидания."""


class EnginePool:
    def __init__(self, engine_path: str, size: int, acquire_timeout_s: float):
        self.engine_path = engine_path
        self.size = max(1, int(size))
        self.acquire_timeout_s = acquire_timeout_s
        self._slots = threading.BoundedSemaphore(self.size)
        self._idle: queue.LifoQueue[chess.engine.SimpleEngine] = queue.LifoQueue()
        self._closed = False

    def _spawn(self) -> chess.engine.SimpleEngine:
        if not os.path.exists(self.engine_path):
            raise EngineUnavailableError(f"Stockfish not found: {self.engine_path}")
        try:
            return chess.engine.SimpleEngine.popen_uci(self.engine_path)
        except Exception as exc:
            raise EngineUnavailableError(f"Cannot start Stockfish ({self.engine_path}): {exc}") from exc

    @staticmethod
    def _quit(engine: chess.engine.SimpleEngine) -> None:
        try:
            engine.quit()
        except Exception:
            try:
                engine.close()
            except Exception:
                pass

    @staticmethod
    def _is_alive(engine: chess.engine.SimpleEngine) -> bool:
        """Процесс движка ещё работает (вызывающий код мог проглотить EngineTerminatedError)."""
        try:
            return not engine.protocol.returncode.done()
        except Exception:
            return False

    @staticmethod
    def _configure(engine: chess.engine.SimpleEngine, options: dict) -> None:
        merged = {**_DEFAULT_OPTIONS, **options}
        supported: dict = {}
        for name, value in merged.items():
            option = engine.options.get(name)
            if option is None:
                continue
            # Числовые (spin) опции приводим к диапазону конкретной сборки движка.
            if option.type == "spin" and isinstance(value, (int, float)) and not isinstance(value, bool):
                if option.min is not None:
                    value = max(option.min, value)
                if option.max is not None:
                    value = min(option.max, value)
                value = int(value)
            supported[name] = value
        if supported:
            engine.configure(supported)

    @contextmanager
    def acquire(self, options: dict | None = None, timeout_s: float | None = None) -> Iterator[chess.engine.SimpleEngine]:
        """
        Выдаёт движок в монопольное пользование на время блока with.

        options — UCI-опции поверх значений по умолчанию (полная сила).
        Движок, упавший с ошибкой протокола или по таймауту, не возвращается в пул, а
        завершается; вместо него при следующем запросе будет запущен новый процесс.
        """
        if self._closed:
            raise EngineUnavailableError("Engine pool is closed")

        wait = self.acquire_timeout_s if timeout_s is None else timeout_s
        if not self._slots.acquire(timeout=wait):
            raise EngineBusyError("All chess engines are busy")

        engine: chess.engine.SimpleEngine | None = None
        healthy = True
        try:
            try:
                engine = self._idle.get_nowait()
            except queue.Empty:
                engine = self._spawn()

            self._configure(engine, options or {})
            yield engine
        except (chess.engine.EngineError, TimeoutError):
            healthy = False
            raise
        finally:
            if engine is not None:
                if healthy and not self._closed and self._is_alive(engine):
                    self._idle.put(engine)
                else:
                    self._quit(engine)
            self._slots.release()

    def close(self) -> None:
        self._closed = True
        while True:
            try:
                engine = self._idle.get_nowait()
            except queue.Empty:
                break
            self._quit(engine)


_pools: dict[str, EnginePool] = {}
_pools_lock = threading.Lock()


def get_engine_pool(engine_path: str | None) -> EnginePool:
    """Возвращает пул для указанного бинарника (по одному пулу на путь, создаётся лениво)."""
    if not engine_path:
        raise EngineUnavailableError("Stockfish path is not configured")
    key = os.path.abspath(engine_path)
    with _pools_lock:
        pool = _pools.get(key)
        if pool is None or pool._closed:
            pool = EnginePool(key, settings.ENGINE_POOL_SIZE, settings.ENGINE_ACQUIRE_TIMEOUT_S)
            _pools[key] = pool
        return pool


def close_all_engine_pools() -> None:
    with _pools_lock:
        pools = list(_pools.values())
        _pools.clear()
    for pool in pools:
        pool.close()


# python-chess держит для каждого SimpleEngine фоновый НЕ-daemon поток. Интерпретатор ждёт
# такие потоки до вызова обычных atexit-обработчиков, поэтому процесс (Celery-воркер, pytest)
# с открытым пулом зависал бы на выходе. threading._register_atexit срабатывает раньше этого
# ожидания (так же завершается concurrent.futures); atexit — запасной вариант.
_register_early_exit_hook = getattr(threading, "_register_atexit", None)
if _register_early_exit_hook is not None:
    _register_early_exit_hook(close_all_engine_pools)
else:
    atexit.register(close_all_engine_pools)

import logging

import chess

logger = logging.getLogger(__name__)

STARTING_FEN = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"


def restore_board(fen: str, pgn_movetext: str = "") -> chess.Board:
    """
    Восстанавливает доску вместе с историей ходов.

    История нужна для правил повторения позиции (троекратное / пятикратное), которые
    невозможно проверить по одному FEN. Ходы переигрываются из PGN-строки партии
    («1. e4 e5 2. Nf3 ...») от стартовой позиции. Если PGN пуст, повреждён или не
    приводит к переданному FEN, используется доска из FEN без истории.
    """
    safe_fen = fen.strip() if fen and fen.strip() else STARTING_FEN
    movetext = (pgn_movetext or "").strip()

    if movetext:
        board = chess.Board()
        try:
            for token in movetext.split():
                # Номера ходов («12.», «12...») и результат («1-0») начинаются с цифры, SAN — никогда.
                if token[0].isdigit() or token == "*":
                    continue
                board.push_san(token)
        except ValueError as exc:
            logger.warning("Cannot replay PGN history (%s), falling back to FEN-only board", exc)
        else:
            if board.fen() == safe_fen:
                return board
            logger.warning("Replayed PGN does not match stored FEN, falling back to FEN-only board")

    return chess.Board(safe_fen)


def terminal_outcome(board: chess.Board) -> tuple[str, str] | None:
    """
    Итог партии по позиции на доске (с историей ходов) или None, если партия продолжается.

    Ничьи по повторению и правилу 50 ходов фиксируются автоматически, но только по фактически
    сложившейся позиции. board.is_game_over(claim_draw=True) здесь не подходит: он засчитывает
    ничью и тогда, когда повторение лишь достижимо следующим ходом.
    """
    if not (board.is_game_over() or board.is_repetition(3) or board.is_fifty_moves()):
        return None
    if board.is_checkmate():
        return ("0-1" if board.turn == chess.WHITE else "1-0"), "checkmate"
    if board.is_stalemate():
        return "1/2-1/2", "stalemate"
    if board.is_insufficient_material():
        return "1/2-1/2", "insufficient_material"
    if board.is_fifty_moves():
        return "1/2-1/2", "50_move_rule"
    if board.is_repetition(3):
        return "1/2-1/2", "repetition"
    return "1/2-1/2", "draw"


def side_has_mating_material(board_fen: str | None, color: str) -> bool:
    """
    Может ли color в принципе поставить мат какой-либо последовательностью легальных ходов
    (по материалу, с учётом цвета полей слонов). Если FEN не разбирается — считаем, что может.
    """
    try:
        board = chess.Board(board_fen or STARTING_FEN)
    except ValueError:
        return True
    return not board.has_insufficient_material(chess.WHITE if color == "white" else chess.BLACK)


FLAG_DRAW_TERMINATION = "timeout_vs_insufficient_material"


def flag_outcome(board_fen: str | None, flagged_color: str) -> tuple[str, str]:
    """
    Итог партии, когда у flagged_color кончилось время (правило ФИДЕ 6.9).

    Поражение, если только у соперника нет материала, которым мат невозможен никакой
    последовательностью легальных ходов, — тогда ничья. has_insufficient_material учитывает
    только материал (и цвета полей слонов) и гарантированно не объявит ничью там, где мат
    возможен: например, конь против ладьи — победа (кооперативный мат существует).
    Возвращает (result, termination).
    """
    winner = "black" if flagged_color == "white" else "white"
    if not side_has_mating_material(board_fen, winner):
        return "1/2-1/2", FLAG_DRAW_TERMINATION
    return ("0-1" if flagged_color == "white" else "1-0"), "timeout"


class GameEngine:
    @staticmethod
    def process_move(fen: str, move_str: str, current_pgn: str = "") -> tuple[bool, dict | None]:
        """
        Проверяет ход, применяет его к текущей позиции (FEN) и возвращает новые данные.

        Возвращает: (is_valid, move_data)
        Где move_data содержит:
        - new_fen: str (Новое состояние доски)
        - new_pgn: str (Обновленная строка ходов PGN)
        - san: str (Красивая нотация, например "Nf3")
        - uci: str (Универсальная нотация, например "g1f3")
        - fullmove_number: int (Номер хода)
        - is_obvious: bool (Флаг единственного легального хода для античита)
        - is_game_over: bool
        - result: str | None ("1-0", "0-1", "1/2-1/2")
        - termination: str | None ("checkmate", "stalemate", "repetition", etc.)
        """
        try:
            board = restore_board(fen, current_pgn)
        except ValueError:
            return False, None

        # 1. Пытаемся распарсить ход (UCI или SAN)
        move = None
        try:
            candidate = chess.Move.from_uci(move_str)
            if candidate in board.legal_moves:
                move = candidate
        except ValueError:
            pass

        if move is None:
            try:
                candidate = board.parse_san(move_str)
                if candidate in board.legal_moves:
                    move = candidate
            except ValueError:
                return False, None

        if move is None or move not in board.legal_moves:
            return False, None

        # 2. Собираем метаданные ДО выполнения хода
        legal_moves_count = board.legal_moves.count()
        is_obvious = legal_moves_count == 1
        is_white_turn = board.turn == chess.WHITE
        fullmove_number = board.fullmove_number

        san = board.san(move)
        uci = move.uci()

        # Формируем PGN-строку за O(1) без перестроения всего дерева
        pgn_base = (current_pgn or "").strip()
        if is_white_turn:
            move_token = f"{fullmove_number}. {san}"
        else:
            move_token = san if pgn_base else f"{fullmove_number}... {san}"
        new_pgn = f"{pgn_base} {move_token}".strip()

        # 3. Применяем ход на доске
        board.push(move)
        new_fen = board.fen()

        # 4. Проверяем условия завершения партии
        outcome = terminal_outcome(board)
        is_game_over = outcome is not None
        result, termination = outcome if outcome else (None, None)

        return True, {
            "new_fen": new_fen,
            "new_pgn": new_pgn,
            "san": san,
            "uci": uci,
            "fullmove_number": fullmove_number,
            "is_obvious": is_obvious,
            "is_game_over": is_game_over,
            "result": result,
            "termination": termination
        }

_INT_STATE_KEYS = (
    "white_id", "black_id", "white_time_ms", "black_time_ms",
    "base_time_ms", "increment_ms", "last_move_at",
    "white_disconnect_started_at", "black_disconnect_started_at",
    "state_version",
)

ABANDON_TIMEOUT_MS = 3 * 60 * 1000
ABANDON_DRAW_TERMINATION = "abandoned_vs_insufficient_material"

TIMEOUTS_LEADER_KEY = "timeouts:leader"
TIMEOUTS_LEADER_TTL_MS = 3000


def _coerce_state(raw: dict) -> dict:
    """Приводит числовые поля сырого Redis-хэша к int (возвращает новый словарь)."""
    state = dict(raw)
    for key in _INT_STATE_KEYS:
        value = state.get(key)
        if value is not None and str(value).lstrip("-").isdigit():
            state[key] = int(value)
    if not state.get("board_fen"):
        state["board_fen"] = STARTING_FEN
    return state


def side_to_move(state: dict) -> str:
    return "white" if state.get("turn", "white") in ("white", "w") else "black"


def live_clock_ms(state: dict, color: str, now_ms: int) -> int:
    """
    Фактический остаток времени игрока. Часы идут только у стороны, которая ходит, и всегда
    отсчитываются от last_move_at — разрыв соединения на них никак не влияет.
    """
    stored = int(state[f"{color}_time_ms"])
    if color != side_to_move(state):
        return stored
    return stored - max(0, now_ms - int(state["last_move_at"]))


def abandon_wait_ms(state: dict, color: str, now_ms: int) -> int:
    """Сколько ещё отключённый игрок может отсутствовать до поражения (для отсчёта у соперника)."""
    wait = ABANDON_TIMEOUT_MS
    if color == side_to_move(state):
        wait = min(wait, max(0, live_clock_ms(state, color, now_ms)))
    return wait


def evaluate_clock_expiry(state: dict, now_ms: int) -> tuple[str, int | None, str] | None:
    """
    Чистая проверка флага / брошенной партии по состоянию (уже приведённому _coerce_state).
    Возвращает (result, winner_id, reason) или None, если партия продолжается.

    1. Флаг: у стороны, которая ходит, кончилось время — независимо от того, подключена ли она.
    2. Брошенная партия: игрок отключён дольше ABANDON_TIMEOUT_MS (его часы при этом могут
       и не идти, если сейчас ход соперника).
    """
    white_id, black_id = int(state["white_id"]), int(state["black_id"])

    mover = side_to_move(state)
    if live_clock_ms(state, mover, now_ms) <= 0:
        result, termination = flag_outcome(state.get("board_fen"), mover)
        if result == "1/2-1/2":
            return result, None, termination
        return result, (black_id if mover == "white" else white_id), termination

    for color in ("white", "black"):
        if state.get(f"{color}_disconnected") != "true":
            continue
        started_at = int(state.get(f"{color}_disconnect_started_at") or 0)
        if started_at > 0 and now_ms - started_at >= ABANDON_TIMEOUT_MS:
            stayed = "black" if color == "white" else "white"
            # Как и при флаге: если оставшийся игрок не может поставить мат никакой
            # последовательностью ходов, победу ему не засчитываем — ничья.
            if not side_has_mating_material(state.get("board_fen"), stayed):
                return "1/2-1/2", None, ABANDON_DRAW_TERMINATION
            if color == "white":
                return "0-1", black_id, "abandoned"
            return "1-0", white_id, "abandoned"
    return None


class GameStateManager:
    def __init__(self):
        self.redis = redis_client

    def _get_key(self, game_id: int) -> str:
        return f"game:{game_id}:state"

    async def initialize_game(self, game_id: int, white_id: int, black_id: int, base_time_ms: int, increment_ms: int):
        now_ms = int(time.time() * 1000)
        state = {
            "white_id": str(white_id),
            "black_id": str(black_id),
            "board_fen": STARTING_FEN,
            "pgn": "",
            "state_version": "0",
            "white_time_ms": str(base_time_ms),
            "black_time_ms": str(base_time_ms),
            "base_time_ms": str(base_time_ms),
            "increment_ms": str(increment_ms),
            "last_move_at": str(now_ms),
            "turn": "white",
            "status": "in_progress",
            "white_disconnected": "false",
            "black_disconnected": "false",
            "white_disconnect_started_at": "0",
            "black_disconnect_started_at": "0",
            "draw_offer": "",
            "rematch_offer": "",
        }
        await self.redis.hset(self._get_key(game_id), mapping=state)
        await self.redis.sadd("active_games", str(game_id))
        await self.redis.set(f"user:{white_id}:current_game", str(game_id))
        await self.redis.set(f"user:{black_id}:current_game", str(game_id))

    init_game = initialize_game

    async def get_active_game_id(self, user_id: int) -> int | None:
        """
        id партии, которую игрок сейчас играет, или None. Устаревшую ссылку
        user:{id}:current_game (партия уже завершена) заодно удаляет.
        """
        key = f"user:{int(user_id)}:current_game"
        raw = await self.redis.get(key)
        if not raw:
            return None
        game_id = int(raw)
        state = await self.get_game_state(game_id)
        if state and state.get("status") == "in_progress" and int(user_id) in (
                int(state["white_id"]), int(state["black_id"])):
            return game_id
        if await self.redis.get(key) == raw:
            await self.redis.delete(key)
        return None

    async def get_game_state(self, game_id: int) -> dict | None:
        state = await self.redis.hgetall(self._get_key(game_id))
        if not state:
            return await self._recover_state_from_db(game_id)
        return _coerce_state(state)

    async def claim_game_end(
            self,
            game_id: int,
            result: str,
            termination: str,
            status: str = "completed",
            guard=None,
    ) -> dict | None:
        """
        Атомарно переводит партию из in_progress в финальный статус (WATCH/MULTI).

        Это единственная точка «захвата» завершения партии: из нескольких конкурентных
        претендентов (сдача, ничья, флаг, отмена, брошенная партия) успех получает ровно один,
        и только он должен вызывать finish_game / abort_game и рассылать game_over.

        guard(state) -> bool — дополнительное условие, перепроверяемое на свежем состоянии
        внутри транзакции (например, что флаг всё ещё упал, а предложение ничьей ещё в силе).

        Возвращает состояние партии на момент захвата или None, если партию уже завершил
        кто-то другой либо guard не выполнен.
        """
        game_key = self._get_key(game_id)
        async with self.redis.pipeline(transaction=True) as pipe:
            while True:
                try:
                    await pipe.watch(game_key)
                    raw = await pipe.hgetall(game_key)
                    if not raw or raw.get("status") != "in_progress":
                        await pipe.unwatch()
                        return None

                    state = _coerce_state(raw)
                    if guard is not None and not guard(state):
                        await pipe.unwatch()
                        return None

                    pipe.multi()
                    pipe.hset(game_key, mapping={
                        "status": status,
                        "result": result,
                        "termination": termination,
                    })
                    await pipe.execute()
                    return state
                except WatchError:
                    continue

    async def _recover_state_from_db(self, game_id: int) -> dict | None:
        """Восстанавливает состояние партии в Redis из PostgreSQL."""
        try:
            async with AsyncSessionLocal() as db:
                db_game = await db.get(Game, game_id)
                if not db_game:
                    return None

                now_ms = int(time.time() * 1000)
                turn_str = "white" if db_game.turn in ("w", "white") else "black"
                white_time = db_game.white_time_ms if db_game.white_time_ms is not None else db_game.base_time_ms
                black_time = db_game.black_time_ms if db_game.black_time_ms is not None else db_game.base_time_ms
                last_move = db_game.last_move_at if db_game.last_move_at is not None else now_ms

                status_map = {
                    GameStatus.IN_PROGRESS: "in_progress",
                    GameStatus.COMPLETED: "completed",
                    GameStatus.ABORTED: "aborted",
                    GameStatus.PENDING: "pending",
                }
                status_str = status_map.get(db_game.status, "completed")

                state = {
                    "white_id": str(db_game.white_player_id),
                    "black_id": str(db_game.black_player_id),
                    "board_fen": db_game.board_fen or STARTING_FEN,
                    "pgn": db_game.pgn or "",
                    "state_version": str(db_game.state_version or 0),
                    "white_time_ms": str(white_time),
                    "black_time_ms": str(black_time),
                    "base_time_ms": str(db_game.base_time_ms),
                    "increment_ms": str(db_game.increment_ms),
                    "last_move_at": str(last_move),
                    "turn": turn_str,
                    "status": status_str,
                    "white_disconnected": "false",
                    "black_disconnected": "false",
                    "white_disconnect_started_at": "0",
                    "black_disconnect_started_at": "0",
                }
                await self.redis.hset(self._get_key(game_id), mapping=state)
                if status_str == "in_progress":
                    await self.redis.sadd("active_games", str(game_id))
                    await self.redis.set(f"user:{db_game.white_player_id}:current_game", str(game_id))
                    await self.redis.set(f"user:{db_game.black_player_id}:current_game", str(game_id))

        except Exception as e:

    async def update_game_state(self, game_id: int, updates: dict):

        await self.redis.hset(self._get_key(game_id), mapping=str_updates)

        await self.redis.delete(f"game:{game_id}:commands")

    async def save_move_telemetry(self, game_id: int, move_data: dict):

        moves_raw = await self.redis.lrange(key, 0, -1)
        await self.redis.delete(key)
        return [json.loads(m) for m in moves_raw]

    async def process_atomic_move(
            self,
            game_id: int,
            user_id: int,
            command_id: str,
            move_str: str,
            expected_version: int | None = None,
    ) -> dict:
        game_key = self._get_key(game_id)
        cmd_key = f"game:{game_id}:commands"

        async with self.redis.pipeline(transaction=True) as pipe:
            while True:
                try:
                    await pipe.watch(game_key, cmd_key)

                    is_processed = await pipe.sismember(cmd_key, command_id)
                    if is_processed:
                        await pipe.unwatch()
                        return {"status": "ignored", "reason": "already_processed"}

                    state = await pipe.hgetall(game_key)
                    if not state or state.get("status") != "in_progress":
                        await pipe.unwatch()
                        return {"status": "error", "reason": "game_inactive"}

                    current_version = int(state.get("state_version", 0))
                    if expected_version is not None and int(expected_version) != current_version:
                        await pipe.unwatch()
                        return {
                            "status": "error",
                            "reason": "version_mismatch",
                            "state_version": current_version,
                            "fen": state.get("board_fen", STARTING_FEN),
                            "pgn": state.get("pgn", ""),
                        }

                    turn = state.get("turn", "white")
                    white_id = int(state["white_id"])
                    black_id = int(state["black_id"])

                    is_white_turn = turn in ("white", "w")
                    expected_player_id = white_id if is_white_turn else black_id

                    if user_id != expected_player_id:
                        await pipe.unwatch()
                        return {"status": "error", "reason": "not_your_turn"}

                    now_ms = int(time.time() * 1000)
                    last_move_at = int(state["last_move_at"])
                    elapsed_ms = max(0, now_ms - last_move_at)

                    increment = int(state["increment_ms"])
                    white_time = int(state["white_time_ms"])
                    black_time = int(state["black_time_ms"])

                    remaining_after_move = (white_time if is_white_turn else black_time) - elapsed_ms
                    if remaining_after_move <= 0:
                        # Флаг упал до хода: завершаем партию в той же транзакции, чтобы
                        # check_timeouts_loop или другой обработчик не завершили её повторно.
                        flag_result, flag_termination = flag_outcome(
                            state.get("board_fen"), "white" if is_white_turn else "black",
                        )
                        final_white = 0 if is_white_turn else white_time
                        final_black = black_time if is_white_turn else 0
                        pipe.multi()
                        pipe.hset(game_key, mapping={
                            "status": "completed",
                            "result": flag_result,
                            "termination": flag_termination,
                            "white_time_ms": str(final_white),
                            "black_time_ms": str(final_black),
                        })
                        pipe.sadd(cmd_key, command_id)
                        await pipe.execute()
                        return {
                            "status": "game_over",
                            "result": flag_result,
                            "termination": flag_termination,
                            "white_id": white_id,
                            "black_id": black_id,
                            "white_time_ms": final_white,
                            "black_time_ms": final_black,
                            "elapsed_ms": elapsed_ms,
                        }

                    if is_white_turn:
                        new_white_time = remaining_after_move + increment
                        new_black_time = black_time
                    else:
                        new_black_time = remaining_after_move + increment
                        new_white_time = white_time

                    current_fen = state.get("board_fen") or STARTING_FEN
                    current_pgn = state.get("pgn") or ""

                    is_valid, move_data = engine.process_move(current_fen, move_str, current_pgn)
                    if not is_valid or not move_data:
                        await pipe.unwatch()
                        return {"status": "error", "reason": "illegal_move"}

                    new_version = current_version + 1
                    new_turn = "black" if is_white_turn else "white"

                    pipe.multi()

                    if move_data["is_game_over"]:
                        updates = {
                            "status": "completed",
                            "result": move_data["result"],
                            "termination": move_data["termination"],
                            "state_version": str(new_version),
                            "board_fen": move_data["new_fen"],
                            "pgn": move_data["new_pgn"],
                            "last_move_at": str(now_ms),
                            "white_time_ms": str(new_white_time),
                            "black_time_ms": str(new_black_time),
                        }
                        pipe.hset(game_key, mapping=updates)
                        pipe.sadd(cmd_key, command_id)
                        await pipe.execute()
                        return {
                            "status": "game_over",
                            "result": move_data["result"],
                            "termination": move_data["termination"],
                            "move_data": move_data,
                            "state_version": new_version,
                            "white_time_ms": new_white_time,
                            "black_time_ms": new_black_time,
                            "white_id": white_id,
                            "black_id": black_id,
                            "elapsed_ms": elapsed_ms,
                            "now_ms": now_ms,
                        }

                    updates = {
                        "state_version": str(new_version),
                        "turn": new_turn,
                        "board_fen": move_data["new_fen"],
                        "pgn": move_data["new_pgn"],
                        "last_move_at": str(now_ms),
                        "white_time_ms": str(new_white_time),
                        "black_time_ms": str(new_black_time),
                    }
                    # Предложение ничьей действует, пока соперник не ответил ходом (ФИДЕ 9.1.2):
                    # ход того, кому предложили, его отклоняет; свой ход предложившего — нет.
                    draw_offer = str(state.get("draw_offer") or "")
                    if draw_offer and draw_offer != str(user_id):
                        updates["draw_offer"] = ""

                    pipe.hset(game_key, mapping=updates)
                    pipe.sadd(cmd_key, command_id)
                    pipe.expire(cmd_key, 86400)

                    await pipe.execute()

                    return {
                        "status": "success",
                        "move_data": move_data,
                        "state_version": new_version,
                        "turn": new_turn,
                        "white_time_ms": new_white_time,
                        "black_time_ms": new_black_time,
                        "white_id": white_id,
                        "black_id": black_id,
                        "elapsed_ms": elapsed_ms,
                        "now_ms": now_ms,
                    }

                except WatchError:
                    continue

    async def finish_game(self, db, game_id: int, result: str, termination: str = "unknown",
                          final_pgn: str = None) -> dict:
        """
        Фиксирует итог партии в БД и пересчитывает рейтинги.

        Вызывать только после успешного claim_game_end (или из ветки game_over process_atomic_move,
        которая захватывает завершение в своей транзакции). Дополнительно, на случай гонки между
        процессами или потери состояния Redis, строки партии и игроков блокируются
        (SELECT ... FOR UPDATE) и перечитываются из БД мимо identity map сессии, поэтому
        повторный вызов не начислит рейтинг второй раз.
        """
        db_game = await db.get(Game, game_id, with_for_update=True, populate_existing=True)
        if not db_game:
            return {}

        if db_game.status in (GameStatus.COMPLETED, GameStatus.ABORTED):
            await db.commit()  # снимаем блокировку
            return {
                "white_delta": db_game.white_rating_delta or 0,
                "black_delta": db_game.black_rating_delta or 0
            }

        # Блокируем игроков в фиксированном порядке (по id), чтобы параллельно завершающиеся
        # партии одного игрока не теряли обновление рейтинга и не взаимоблокировались.
        locked_users: dict[int, User] = {}
        for uid in sorted({db_game.white_player_id, db_game.black_player_id}):
            locked_users[uid] = await db.get(User, uid, with_for_update=True, populate_existing=True)
        white_user = locked_users[db_game.white_player_id]
        black_user = locked_users[db_game.black_player_id]

        white_initial = white_user.rating
        black_initial = black_user.rating
        delta_w = delta_b = 0
        new_w = white_initial
        new_b = black_initial

        if db_game.is_rated:
            w_games_q = select(func.count(Game.id)).where(
                or_(Game.white_player_id == white_user.id, Game.black_player_id == white_user.id),
                Game.status == GameStatus.COMPLETED,
                Game.is_rated == True,  # noqa: E712
            )
            b_games_q = select(func.count(Game.id)).where(
                or_(Game.white_player_id == black_user.id, Game.black_player_id == black_user.id),
                Game.status == GameStatus.COMPLETED,
                Game.is_rated == True,  # noqa: E712
            )
            white_games_count = (await db.execute(w_games_q)).scalar() or 0
            black_games_count = (await db.execute(b_games_q)).scalar() or 0

            new_w, delta_w, new_b, delta_b = elo_calculator.calculate_new_ratings(
                white_initial,
                black_initial,
                result,
                white_games=white_games_count,
                black_games=black_games_count,
            )
            white_user.rating = new_w
            black_user.rating = new_b

        db_game.result = result
        db_game.termination = termination

        if result == "1-0":
            db_game.winner_id = white_user.id
        elif result == "0-1":
            db_game.winner_id = black_user.id

        db_game.white_initial_rating = white_initial
        db_game.black_initial_rating = black_initial
        db_game.white_rating_delta = delta_w
        db_game.black_rating_delta = delta_b

        state = await self.redis.hgetall(self._get_key(game_id))
        if state:
            if state.get("state_version"):
                db_game.state_version = int(state["state_version"])
            if state.get("board_fen"):
                db_game.board_fen = state["board_fen"]
            if state.get("white_time_ms"):
                db_game.white_time_ms = int(state["white_time_ms"])
            if state.get("black_time_ms"):
                db_game.black_time_ms = int(state["black_time_ms"])
            if not final_pgn:
                final_pgn = state.get("pgn")

        today_str = datetime.datetime.now(datetime.timezone.utc).strftime("%Y.%m.%d")
        db_game.status = GameStatus.COMPLETED
        db_game.finished_at = func.now()
        db_game.pgn_date = today_str
        db_game.site = "MiChess"

        moves_telemetry = await self.extract_all_moves(game_id)
        if moves_telemetry:
            db_moves = [Move(game_id=game_id, **m) for m in moves_telemetry]
            db.add_all(db_moves)

            uci_seq = [m.get("notation", "") for m in moves_telemetry if m.get("notation")]
            eco_code = _detect_eco_from_moves(uci_seq)
            db_game.eco = eco_code

            pgn_game = chess.pgn.Game()
            pgn_game.headers["Event"] = "MiChess Rated Game" if db_game.is_rated else "MiChess Casual Game"
            pgn_game.headers["Site"] = "MiChess"
            pgn_game.headers["Date"] = today_str
            pgn_game.headers["White"] = white_user.username
            pgn_game.headers["Black"] = black_user.username
            pgn_game.headers["WhiteElo"] = str(white_initial)
            pgn_game.headers["BlackElo"] = str(black_initial)
            pgn_game.headers["Result"] = result
            pgn_game.headers["ECO"] = eco_code
            pgn_game.headers["TimeControl"] = f"{db_game.base_time_ms // 1000}+{db_game.increment_ms // 1000}"
            pgn_game.headers["Termination"] = termination

            node = pgn_game
            board_for_pgn = chess.Board()
            for m in moves_telemetry:
                try:
                    move_obj = chess.Move.from_uci(m["notation"])
                    if move_obj in board_for_pgn.legal_moves:
                        board_for_pgn.push(move_obj)
                        node = node.add_variation(move_obj)
                    else:
                        break
                except ValueError:
                    break
            db_game.pgn = str(pgn_game)
        elif final_pgn:
            db_game.pgn = final_pgn
            from src.db.seed import extract_pgn_metadata
            meta = extract_pgn_metadata(final_pgn, fallback_result=result)
            if meta["pgn_date"]:
                db_game.pgn_date = meta["pgn_date"]
            if meta["site"]:
                db_game.site = meta["site"]
            if meta["eco"]:
                db_game.eco = meta["eco"]

        await db.commit()

        if db_game.is_rated:
            try:
                await evaluate_game_telemetry(db, game_id)
            except Exception as ac_err:
                logger.error("Telemetry anti-cheat evaluation failed for game %s: %s", game_id, ac_err)

        await self.clear_game_state(game_id)

        try:
            analyze_game_task.delay(game_id)
        except Exception as e:
            logger.error("Could not schedule Celery analysis task for game %s: %s", game_id, e)

        return {
            "white_delta": delta_w,
            "black_delta": delta_b,
            "white_new_rating": new_w,
            "black_new_rating": new_b
        }

    async def handle_player_disconnect(self, user_id: int, manager):
        game_id_str = await self.redis.get(f"user:{user_id}:current_game")
        if not game_id_str:
            return
        game_id = int(game_id_str)
        state = await self.get_game_state(game_id)
        if not state or state["status"] != "in_progress":
            return

        white_id, black_id = int(state["white_id"]), int(state["black_id"])
        color = "white" if user_id == white_id else "black"
        opponent_id = black_id if color == "white" else white_id

        now_ms = int(time.time() * 1000)
        fresh = await self._update_if_in_progress(game_id, {
            f"{color}_disconnected": "true",
            f"{color}_disconnect_started_at": str(now_ms),
        })
        if fresh is None:
            return

        await manager.send_personal_message(
            {
                "event": "opponent_disconnected",
                "game_id": game_id,
                "max_wait_ms": abandon_wait_ms(fresh, color, now_ms),
            },
            opponent_id,
        )

    async def handle_player_reconnect(self, user_id: int, manager):
        game_id_str = await self.redis.get(f"user:{user_id}:current_game")
        if not game_id_str:
            return
        game_id = int(game_id_str)
        state = await self.get_game_state(game_id)
        if not state or state["status"] != "in_progress":
            return

        white_id, black_id = int(state["white_id"]), int(state["black_id"])
        color = "white" if user_id == white_id else "black"
        opponent_id = black_id if color == "white" else white_id

        if state.get(f"{color}_disconnected") == "true":
            # Часы не трогаем: они и так шли от last_move_at, пока игрок был офлайн.
            # Раньше здесь сдвигался last_move_at, и если был ход соперника, его часы
            # фактически стояли всё время отключения.
            fresh = await self._update_if_in_progress(game_id, {
                f"{color}_disconnected": "false",
                f"{color}_disconnect_started_at": "0",
            })
            if fresh is not None:
                await manager.send_personal_message(
                    {"event": "opponent_reconnected", "game_id": game_id}, opponent_id,
                )

    async def _update_if_in_progress(self, game_id: int, updates: dict) -> dict | None:
        """
        Атомарно (WATCH/MULTI) обновляет поля партии, только пока она in_progress.
        Обычный hset после clear_game_state воссоздал бы «огрызок» хэша без white_id и т.п.
        Возвращает состояние до обновления или None, если партия уже не идёт.
        """
        game_key = self._get_key(game_id)
        async with self.redis.pipeline(transaction=True) as pipe:
            while True:
                try:
                    await pipe.watch(game_key)
                    raw = await pipe.hgetall(game_key)
                    if not raw or raw.get("status") != "in_progress":
                        await pipe.unwatch()
                        return None
                    pipe.multi()
                    pipe.hset(game_key, mapping={k: str(v) for k, v in updates.items()})
                    await pipe.execute()
                    return _coerce_state(raw)
                except WatchError:
                    continue

    async def _hold_timeouts_leadership(self) -> bool:
        """
        Цикл таймаутов выполняет только один воркер (лидер). Лидерство — ключ с TTL; лидер
        продлевает его каждую итерацию, при падении лидера ключ истекает и его берёт другой.
        Даже если на мгновение лидеров окажется двое, двойного завершения партии не будет:
        оно защищено claim_game_end.
        """
        try:
            if await self.redis.set(TIMEOUTS_LEADER_KEY, WORKER_ID, nx=True, px=TIMEOUTS_LEADER_TTL_MS):
                return True
            if await self.redis.get(TIMEOUTS_LEADER_KEY) == WORKER_ID:
                await self.redis.pexpire(TIMEOUTS_LEADER_KEY, TIMEOUTS_LEADER_TTL_MS)
                return True
            return False
        except Exception as exc:
            logger.warning("Timeouts leadership check failed, running loop anyway: %s", exc)
            return True

    async def check_timeouts_loop(self, manager):
        while True:
            try:
                if not await self._hold_timeouts_leadership():
                    await asyncio.sleep(0.5)
                    continue

                active_game_ids = list(await self.redis.smembers("active_games"))
                now_ms = int(time.time() * 1000)

                for game_id_str in active_game_ids:
                    game_id = int(game_id_str)
                    state = await self.get_game_state(game_id)
                    if not state or state["status"] != "in_progress":
                        await self.redis.srem("active_games", game_id_str)
                        continue

                    if int(state.get("state_version", 0)) < 2 and now_ms - int(
                            state.get("last_move_at", now_ms)) >= 30000:
                        async with AsyncSessionLocal() as db:
                            await self.check_first_move_abort_timeout(db, game_id, manager=manager, timeout_ms=30000)
                        continue

                    white_id, black_id = int(state["white_id"]), int(state["black_id"])
                    decision = evaluate_clock_expiry(state, now_ms)

                    if decision:
                        game_result, winner_id, reason = decision

                        # Решение перепроверяется на свежем состоянии внутри транзакции: игрок мог
                        # успеть сходить, переподключиться или сдаться, пока мы читали состояние.
                        claimed = await self.claim_game_end(
                            game_id, game_result, reason,
                            guard=lambda fresh, expected=decision: evaluate_clock_expiry(fresh, now_ms) == expected,
                        )
                        if not claimed:
                            continue

                        logger.info("Game %s ended. Result: %s, Reason: %s", game_id, game_result, reason)

                        async with AsyncSessionLocal() as db:
                            rating_updates = await self.finish_game(db, game_id, game_result, termination=reason)

                        game_over_payload = {
                            "event": "game_over",
                            "game_id": game_id,
                            "reason": reason,
                            "winner_id": winner_id,
                            "result": game_result,
                        }
                        game_over_payload.update(rating_updates)

                        await manager.send_personal_message(game_over_payload, white_id)
                        await manager.send_personal_message(game_over_payload, black_id)
                        if hasattr(manager, "broadcast_to_spectators"):
                            await manager.broadcast_to_spectators(game_id, game_over_payload)

            except Exception as e:
                logger.error("Error in check_timeouts_loop: %s", e)

            await asyncio.sleep(0.5)

    async def abort_game(self, db, game_id: int, guard=None) -> dict:
        """
        Отменяет партию на 1-м ходу (до 2 полуходов) без изменения рейтинга.
        guard(state) -> bool — дополнительное условие, атомарно перепроверяемое при захвате.
        """
        state = await self.get_game_state(game_id)
        if not state or state.get("status") != "in_progress":
            return {"status": "error", "reason": "Game is not in progress"}

        state_version = int(state.get("state_version", 0))
        if state_version >= 2:
            return {"status": "error", "reason": "Cannot abort game after move 1"}

        white_id = int(state["white_id"])
        black_id = int(state["black_id"])

        def _abort_allowed(fresh: dict) -> bool:
            if int(fresh.get("state_version", 0)) >= 2:
                return False
            return guard(fresh) if guard is not None else True

        claimed = await self.claim_game_end(
            game_id, "*", "aborted", status="aborted", guard=_abort_allowed,
        )
        if not claimed:
            return {"status": "error", "reason": "Game is not in progress"}

        await self.redis.srem("active_games", str(game_id))
        await self.redis.delete(f"user:{white_id}:current_game")
        await self.redis.delete(f"user:{black_id}:current_game")

        from datetime import datetime, timezone

        db_game = await db.get(Game, game_id, with_for_update=True, populate_existing=True)
        white_user = await db.get(User, white_id)
        black_user = await db.get(User, black_id)

        w_rating = white_user.rating if white_user else 1200
        b_rating = black_user.rating if black_user else 1200

        if db_game and db_game.status not in (GameStatus.COMPLETED, GameStatus.ABORTED):
            db_game.status = GameStatus.ABORTED
            db_game.result = "*"
            db_game.termination = "aborted"
            db_game.white_rating_delta = 0
            db_game.black_rating_delta = 0
            is_tz = getattr(Game.__table__.c.finished_at.type, "timezone", False)
            now_utc = datetime.now(timezone.utc)
            db_game.finished_at = now_utc if is_tz else now_utc.replace(tzinfo=None)
            await db.commit()

        return {
            "status": "aborted",
            "result": "*",
            "termination": "aborted",
            "white_id": white_id,
            "black_id": black_id,
            "white_delta": 0,
            "black_delta": 0,
            "white_new_rating": w_rating,
            "black_new_rating": b_rating,
        }

    async def check_first_move_abort_timeout(self, db, game_id: int, manager=None, timeout_ms: int = 30000) -> bool:
        """Проверяет, не истекли ли 30 секунд на первые 2 полухода партии. Если истекли — отменяет игру."""
        state = await self.get_game_state(game_id)
        if not state or state.get("status") != "in_progress":
            return False

        if int(state.get("state_version", 0)) >= 2:
            return False

        now_ms = int(time.time() * 1000)
        last_move_at = int(state.get("last_move_at", now_ms))
        if now_ms - last_move_at < timeout_ms:
            return False

        def _still_idle(fresh: dict) -> bool:
            return int(time.time() * 1000) - int(fresh.get("last_move_at", 0)) >= timeout_ms

        res = await self.abort_game(db, game_id, guard=_still_idle)
        if res.get("status") == "aborted" and manager is not None:
            payload = {
                "event": "game_over",
                "game_id": game_id,
                "reason": "aborted",
                "result": "*",
                "white_delta": 0,
                "black_delta": 0,
                "white_new_rating": res["white_new_rating"],
                "black_new_rating": res["black_new_rating"],
            }
            await manager.send_personal_message(payload, res["white_id"])
            await manager.send_personal_message(payload, res["black_id"])
            if hasattr(manager, "broadcast_to_spectators"):
                await manager.broadcast_to_spectators(game_id, payload)
        return res.get("status") == "aborted"
