@pytest.mark.parametrize(
    "fen, flagged, expected",
    [
        # У чёрных (победителя по флагу) только конь против голого короля — мат невозможен.
        ("8/8/8/4k3/8/8/2n5/K7 w - - 0 60", "white", ("1/2-1/2", FLAG_DRAW_TERMINATION)),
        # Голый король у победителя.
        ("8/8/8/4k3/8/8/2q5/K7 b - - 0 60", "black", ("1/2-1/2", FLAG_DRAW_TERMINATION)),
        # Ладья у победителя — обычное поражение по времени.

        # Конь против ладьи: кооперативный мат существует, значит, победа.
        ("8/8/8/4k3/8/8/2n5/KR6 w - - 0 60", "white", ("0-1", "timeout")),
        # Одноцветные слоны у обоих: мат невозможен.

def _worker(client, worker_id: str) -> ConnectionManager:
    cm = ConnectionManager()
    cm.worker_id = worker_id
    cm.redis = client
    cm.pubsub = client.pubsub()
    return cm


async def _stop(*managers: ConnectionManager):
    for cm in managers:
        for task in (cm._listener_task, cm._heartbeat_task):
            if task is not None:
                task.cancel()
        await asyncio.gather(
            *(t for t in (cm._listener_task, cm._heartbeat_task) if t is not None), return_exceptions=True,
        )
        await cm.pubsub.aclose()


async def _wait_for(predicate, timeout_s: float = 3.0):
    deadline = time.monotonic() + timeout_s
    while time.monotonic() < deadline:
        if predicate():
            return True
        await asyncio.sleep(0.02)
    return False


async def test_tab_on_other_worker_keeps_player_online(shared_redis):

"""
Постановка Celery-задач из async-кода не должна блокировать event loop: раньше
analyze_game_task.delay() шёл прямо в loop и при недоступном брокере подвешивал весь воркер.
"""

import asyncio
import threading
import time

from src.services import task_dispatch
from src.worker.tasks import analyze_game_task


async def _max_loop_stall(coro) -> tuple[object, float]:
    """Выполняет coro и меряет самую длинную паузу, на которую останавливался event loop."""
    ticks: list[float] = []
    done = asyncio.Event()

    async def ticker():
        while not done.is_set():
            ticks.append(time.monotonic())
            await asyncio.sleep(0.01)

    ticker_task = asyncio.create_task(ticker())
    try:
        result = await coro
    finally:
        done.set()
        await ticker_task
    gaps = [b - a for a, b in zip(ticks, ticks[1:], strict=False)] or [0.0]
    return result, max(gaps)


async def test_publish_runs_in_a_thread(no_celery_publish):
    assert await task_dispatch.schedule_game_analysis(7) is True
    assert no_celery_publish == [7]


async def test_slow_broker_does_not_block_event_loop(monkeypatch):
    release = threading.Event()

    def hanging_publish(game_id):
        release.wait(5)  # брокер «висит» дольше таймаута публикации

    monkeypatch.setattr(task_dispatch, "_publish_analysis", hanging_publish)
    monkeypatch.setattr(task_dispatch.settings, "CELERY_PUBLISH_TIMEOUT_S", 0.3)

    published, stall = await _max_loop_stall(task_dispatch.schedule_game_analysis(8))
    release.set()

    assert published is False
    assert stall < 0.2  # loop продолжал работать, пока поток ждал брокер


async def test_broker_failure_starts_cooldown(monkeypatch):
    calls: list[int] = []

    def failing_publish(game_id):
        calls.append(game_id)
        raise ConnectionError("broker is down")

    monkeypatch.setattr(task_dispatch, "_publish_analysis", failing_publish)

    assert await task_dispatch.schedule_game_analysis(1) is False
    # Пока длится пауза, новые публикации не пытаются достучаться до брокера и не плодят потоков.
    assert await task_dispatch.schedule_game_analysis(2) is False
    assert calls == [1]

    task_dispatch.reset_broker_cooldown()
    assert await task_dispatch.schedule_game_analysis(3) is False
    assert calls == [1, 3]


async def test_background_scheduling_returns_immediately(monkeypatch):

    monkeypatch.setattr(task_dispatch, "_publish_analysis", lambda game_id: release.wait(5))

Проверка таймаутов изолирует ошибки по партиям: одна «битая» партия не должна останавливать
флаги и отмены во всех остальных (раньше весь проход шёл в одном try и обрывался на ней).

import pytest

from src.services import game_state as game_state_module
from src.services.game_state import (
    TIMEOUT_RETRY_BASE_MS,
    TIMEOUT_RETRY_MAX_MS,
    state_manager,

NOW = 10_000_000


class DummyManager:
    def __init__(self):
        self.sent: list[tuple[int, dict]] = []

    async def send_personal_message(self, message: dict, user_id: int):
        self.sent.append((user_id, message))

    async def broadcast_to_spectators(self, game_id: int, message: dict):
        pass


class _NoDbSession:
    async def __aenter__(self):
        return self

    async def __aexit__(self, *exc):
        return False


@pytest.fixture
def finished(monkeypatch) -> list[tuple[int, str, str]]:
    """Подменяет запись итога в БД: тест проверяет цикл, а не finish_game."""
    calls: list[tuple[int, str, str]] = []

    async def fake_finish_game(db, game_id, result, termination="unknown", final_pgn=None):
        calls.append((game_id, result, termination))
        return {"white_delta": 0, "black_delta": 0}

    monkeypatch.setattr(state_manager, "finish_game", fake_finish_game)
    monkeypatch.setattr(game_state_module, "AsyncSessionLocal", _NoDbSession)
    monkeypatch.setattr(state_manager, "_timeout_failures", {})
    return calls


async def _flagged_game(game_id: int, white_id: int, black_id: int) -> None:
    """Партия, где у белых (их ход) кончилось время."""
    await state_manager.initialize_game(game_id, white_id, black_id, base_time_ms=60000, increment_ms=0)
    await state_manager.update_game_state(game_id, {
        "state_version": "4", "white_time_ms": "1000", "last_move_at": str(NOW - 5000),
    })


async def _broken_game(game_id: int) -> None:
    await state_manager.initialize_game(game_id, 1, 2, base_time_ms=60000, increment_ms=0)
    await state_manager.update_game_state(game_id, {"state_version": "4", "white_id": "broken"})


async def test_broken_game_does_not_block_flags_in_other_games(finished):
    await _broken_game(91)
    await _flagged_game(92, 10, 20)

    manager = DummyManager()
    await state_manager.check_timeouts_once(manager, now_ms=NOW)

    assert finished == [(92, "0-1", "timeout")]
    assert {uid for uid, msg in manager.sent if msg["event"] == "game_over"} == {10, 20}
    assert state_manager._timeout_failures[91][0] == 1


async def test_failing_game_is_retried_with_backoff_and_recovers(finished):
    await _broken_game(93)
    manager = DummyManager()

    await state_manager.check_timeouts_once(manager, now_ms=NOW)
    assert state_manager._timeout_failures[93] == (1, NOW + TIMEOUT_RETRY_BASE_MS)

    # Состояние починили (или сбой БД прошёл), и у белых тем временем упал флаг.
    await state_manager.update_game_state(93, {
        "white_id": "1", "white_time_ms": "1000", "last_move_at": str(NOW - 5000),
    })

    await state_manager.check_timeouts_once(manager, now_ms=NOW + TIMEOUT_RETRY_BASE_MS // 2)
    assert finished == []  # пауза ещё не прошла — партию не трогаем

    await state_manager.check_timeouts_once(manager, now_ms=NOW + TIMEOUT_RETRY_BASE_MS)
    assert finished == [(93, "0-1", "timeout")]
    assert 93 not in state_manager._timeout_failures


async def test_backoff_grows_and_is_capped(finished):
    for _ in range(12):
        state_manager._record_timeout_failure(94, NOW)
    count, retry_at = state_manager._timeout_failures[94]
    assert count == 12
    assert retry_at == NOW + TIMEOUT_RETRY_MAX_MS


async def test_broken_game_stays_active_for_later_retries(finished):
    """Партия с ошибкой не выбрасывается: при временном сбое БД иначе потерялись бы все партии."""
    await _broken_game(95)
    await state_manager.check_timeouts_once(DummyManager(), now_ms=NOW)
    assert "95" in await state_manager.redis.smembers("active_games")


async def test_malformed_active_game_id_is_removed(finished):
    await state_manager.redis.sadd("active_games", "not-a-number")
    await _flagged_game(96, 30, 40)

    await state_manager.check_timeouts_once(DummyManager(), now_ms=NOW)

    assert "not-a-number" not in await state_manager.redis.smembers("active_games")
    assert finished == [(96, "0-1", "timeout")]


async def test_failure_counters_of_finished_games_are_dropped(finished):
    state_manager._timeout_failures[97] = (3, NOW + 5000)  # партии 97 среди активных уже нет
    await state_manager.check_timeouts_once(DummyManager(), now_ms=NOW)
    assert 97 not in state_manager._timeout_failures

def _auth(token: str) -> dict:

def _register_and_login(client, username: str = "tokenuser") -> str:
    assert client.post("/api/v1/auth/register", json={"username": username, "password": PASSWORD}).status_code == 201
    res = client.post("/api/v1/auth/login", data={"username": username, "password": PASSWORD})

    return client.get("/api/v1/profile/me", headers=_auth(token))


class _NoDbFactory:
    """Фабрика сессий для get_current_user_ws: отдаёт заранее известного пользователя."""

        self.user = user

        return self

        return self

    async def __aexit__(self, *exc):
        return False

    async def get(self, model, pk):
        return self.user if self.user is not None and self.user.id == pk else None

    token = _register_and_login(client)
    second_device = client.post("/api/v1/auth/login", data={"username": "tokenuser", "password": PASSWORD})
    other_token = second_device.json()["access_token"]

"""
Celery-задача анализа: каждая задача идёт в собственном asyncio.run, поэтому глобальный
пул соединений использовать нельзя — со второй задачи он ломался («different loop»).
"""

import asyncio
import json

import pytest
from sqlalchemy.ext.asyncio import async_sessionmaker, create_async_engine

from src.models.base import Base
from src.models.game import Game, GameStatus
from src.models.user import User
from src.services.bots.base import resolve_stockfish_path
from src.worker import tasks

PGN = '[Event "t"]\n[Result "1-0"]\n\n1. e4 e5 2. Qh5 Nc6 3. Bc4 Nf6 4. Qxf7# 1-0'


async def _prepare_db(url: str) -> list[int]:
    engine = create_async_engine(url)
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
    session_factory = async_sessionmaker(engine, expire_on_commit=False)
    async with session_factory() as db:
        white = User(username="worker_w", hashed_password="pw", rating=1500)
        black = User(username="worker_b", hashed_password="pw", rating=1500)
        db.add_all([white, black])
        await db.commit()
        games = [
            Game(white_player_id=white.id, black_player_id=black.id, status=GameStatus.COMPLETED,
                 base_time_ms=180000, increment_ms=0, is_rated=True, pgn=PGN)
            for _ in range(2)
        ]
        db.add_all(games)
        await db.commit()
        ids = [g.id for g in games]
    await engine.dispose()
    return ids


async def _cached_analysis(url: str, game_id: int) -> str | None:
    engine = create_async_engine(url)
    async with async_sessionmaker(engine)() as db:
        game = await db.get(Game, game_id)
        value = game.analysis_json
    await engine.dispose()
    return value

"""
WebSocket не должен держать соединение пула БД, пока игрок просто подключён.

Раньше обработчик брал одну сессию на всё соединение: первый запрос (проверка токена) открывал
транзакцию, и соединение пула оставалось занятым до отключения игрока. При пуле на N соединений
(N+1)-й подключившийся игрок ждал pool_timeout и получал ошибку.
"""

import asyncio
import json

import pytest
from fastapi import WebSocketDisconnect
from sqlalchemy.ext.asyncio import async_sessionmaker, create_async_engine
from sqlalchemy.pool import AsyncAdaptedQueuePool, NullPool

from src.api.deps import get_current_user_ws
from src.api.websockets import matchmaking_endpoint
from src.db import redis as redis_module
from src.models.base import Base
from src.models.user import User
from src.services.game_state import state_manager

_CLOSE = object()


class QueueSocket:
    """WebSocket-заглушка: входящие сообщения — из очереди, исходящие складываются в список."""

    def __init__(self):
        self.incoming: asyncio.Queue = asyncio.Queue()
        self.sent: list[dict] = []

    async def accept(self):
        pass

    async def send_text(self, text: str):
        self.sent.append(json.loads(text))

    async def receive_text(self):
        item = await self.incoming.get()
        if item is _CLOSE:
            raise WebSocketDisconnect(code=1000)
        return json.dumps(item)

    async def wait_for(self, event: str, wait_s: float = 5.0) -> dict:
        async def _poll():
            while True:
                for message in self.sent:
                    if message.get("event") == event:
                        return message
                await asyncio.sleep(0.01)
        return await asyncio.wait_for(_poll(), wait_s)


async def _make_users(url: str, count: int) -> list[tuple[int, str]]:
    setup_engine = create_async_engine(url, poolclass=NullPool)
    async with setup_engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
    async with async_sessionmaker(setup_engine, expire_on_commit=False)() as db:
        names = [f"pool_user_{i}" for i in range(count)]
        users = [User(username=name, hashed_password="pw", rating=1500) for name in names]
        db.add_all(users)
        await db.commit()
        created = [(u.id, u.username) for u in users]
    await setup_engine.dispose()

поток мусора закрывает соединение, а отклонённый по лимиту ход возвращает клиенту позицию.

_CLOSE = object()


class _NoDbSession:
    async def __aenter__(self):
        return self

    async def __aexit__(self, *exc):
        return False

    async def get(self, *args, **kwargs):
        return None


class QueueSocket:
    def __init__(self):
        self.incoming: asyncio.Queue = asyncio.Queue()
        self.sent: list[dict] = []
        self.closed_with: int | None = None

    async def accept(self):
        pass

    async def send_text(self, text: str):
        self.sent.append(json.loads(text))

    async def receive_text(self):
        item = await self.incoming.get()
        if item is _CLOSE:
            raise WebSocketDisconnect(code=1000)
        return item if isinstance(item, str) else json.dumps(item)

    async def close(self, code: int = 1000):
        self.closed_with = code

    def events(self, event: str) -> list[dict]:
        return [m for m in self.sent if m.get("event") == event]


async def _run(user: User, messages: list, *, close: bool = True) -> QueueSocket:
    """Прогоняет сообщения через эндпоинт и дожидается, пока он их обработает."""
    socket = QueueSocket()
    for message in messages:
        socket.incoming.put_nowait(message)
    if close:
        socket.incoming.put_nowait(_CLOSE)
    await asyncio.wait_for(matchmaking_endpoint(socket, user, _NoDbSession), timeout=5)
    return socket

def matches_player(header_name: str, surname: str, exclude: tuple[str, ...]) -> bool:
    name_lower = (header_name or "").lower()
    if any(ex in name_lower for ex in exclude):
        return False
    return surname in name_lower


def process_pgn_stream(pgn_stream, surname: str, exclude: tuple[str, ...], book_additions: dict) -> int:
    games_processed = 0
    while True:
        try:
            game = chess.pgn.read_game(pgn_stream)
            if game is None:
                break

            is_w = matches_player(game.headers.get("White", ""), surname, exclude)
            is_b = matches_player(game.headers.get("Black", ""), surname, exclude)
            track_both = not (is_w or is_b)

            board = game.board()
            ply = 0
            for move in game.mainline_moves():
                if ply >= MAX_BOOK_PLY:
                    break
                uci = move.uci()
                if uci == "0000" or not bool(move):
                    break

                white_turn = board.turn == chess.WHITE
                if track_both or (white_turn and is_w) or (not white_turn and is_b):
                    parts = board.fen().split(" ")
                    fen_key = f"{parts[0]} {parts[1]} {parts[2]} -"
                    book_additions.setdefault(fen_key, {})
                    book_additions[fen_key][uci] = book_additions[fen_key].get(uci, 0) + 1

                try:
                    board.push(move)
                except ValueError:
                    break
                ply += 1  # noqa: SIM113 — считаем только успешно сыгранные ходы

            games_processed += 1
        except Exception:
            continue
    return games_processed


def find_pgn_files_for_bot(identifiers: tuple[str, ...]) -> list[Path]:
    """Ищет PGN-файлы чемпиона во всех возможных директориях проекта."""
    search_dirs = [
        PROJECT_ROOT / "data",
        PROJECT_ROOT / "downloaded_pgns",
        PROJECT_ROOT / "src" / "db" / "data",
        PROJECT_ROOT,
    ]
    found: list[Path] = []
    seen: set[Path] = set()

    for directory in search_dirs:
        if not directory.exists():
            continue
        iterator = directory.glob("*.pgn") if directory == PROJECT_ROOT else directory.rglob("*.pgn")
        for pgn_path in iterator:
            resolved = pgn_path.resolve()
            if resolved in seen or not resolved.is_file():
                continue
            path_lower = str(resolved).lower()
            if any(ident in path_lower for ident in identifiers):
                seen.add(resolved)
                found.append(resolved)

    # Если есть как объединённый файл (*_all_*.pgn), так и мелкие файлы из папки —
    # отдаём приоритет объединённому файлу, чтобы не считать одни и те же партии дважды
    combined = [p for p in found if "all" in p.name.lower() or "все партии" in p.name.lower()]
    if combined:
        return [max(combined, key=lambda p: p.stat().st_size)]
    return found


def build_book_for_bot(bot_id: str) -> None:
    cfg = BOT_BOOK_CONFIG[bot_id]
    target_json = BOOKS_DIR / f"{bot_id}.json"

    existing_book: dict = {}
    if target_json.exists():
        try:
            with open(target_json, encoding="utf-8") as f:
                loaded = json.load(f)
                if isinstance(loaded, dict):
                    existing_book = loaded
        except Exception:
            existing_book = {}

    matched_files = find_pgn_files_for_bot(cfg["identifiers"])
    book_additions: dict[str, dict[str, int]] = {}
    total_games = 0

    for filepath in matched_files:
        with open(filepath, encoding="utf-8", errors="ignore") as f:
            total_games += process_pgn_stream(f, cfg["surname_filter"], cfg["exclude"], book_additions)

    if total_games > 0:
        formatted_book: dict[str, list] = dict(existing_book)
        for fen, moves_dict in book_additions.items():
            sorted_moves = sorted(moves_dict.items(), key=lambda x: x[1], reverse=True)
            total_count = sum(moves_dict.values())
            formatted_book[fen] = [
                [
                    uci,
                    max(int((count / total_count) * 100), 1),
                    cfg["comment_tpl"].format(count=count),
                ]
                for uci, count in sorted_moves
            ]

        with open(target_json, "w", encoding="utf-8") as out_f:
            json.dump(formatted_book, out_f, ensure_ascii=False, indent=2)
        print(
            f"[+] {bot_id:13s}: спарсено партий из PGN: {total_games:4d} | "
            f"Позиций в книге: {len(formatted_book):4d} -> {target_json.relative_to(PROJECT_ROOT)}"
        )
        return

    # Если сырых PGN-файлов нет, но JSON-книга уже перенесена и лежит в src/services/bots/books/
    if existing_book:
        print(
            f"[+] {bot_id:13s}: книга активна и проверена | "
            f"Позиций в книге: {len(existing_book):4d} -> {target_json.relative_to(PROJECT_ROOT)}"
        )
        return

    # Если нет ни PGN, ни JSON-книги — инициализируем JSON-файл базовым репертуаром из модуля бота
    try:
        module = __import__(f"src.services.bots.{bot_id}", fromlist=["*"])
        default_attr = f"DEFAULT_{bot_id.upper()}_BOOK"
        fallback_data = getattr(module, default_attr, {})
        if fallback_data:
            serializable = {
                fen: [[m[0], int(m[1]), str(m[2])] for m in moves]
                for fen, moves in fallback_data.items()
            }
            with open(target_json, "w", encoding="utf-8") as out_f:
                json.dump(serializable, out_f, ensure_ascii=False, indent=2)
            print(
                f"[~] {bot_id:13s}: создана базовая книга | "
                f"Позиций в книге: {len(serializable):4d} -> {target_json.relative_to(PROJECT_ROOT)}"
            )
            return
    except Exception as exc:
        print(f"[!] Ошибка инициализации базовой книги для {bot_id}: {exc}")

async def fetch_gids_from_page(
        client: httpx.AsyncClient, pid: int, page: int
) -> tuple[list[str], int | None, int | None]:
    url = (
        f"{BASE_URL}/perl/chessplayer?pid={pid}"
        if page == 1
        else f"{BASE_URL}/perl/chess.pl?page={page}&pid={pid}"
    )

    for attempt in range(1, 7):
        try:
            response = await client.get(url)
            if response.status_code in (429, 403, 503):
                wait_sec = 10 * attempt
                print(f"[!] Лимит ({response.status_code}) на стр. {page}, пауза {wait_sec} сек...")
                await asyncio.sleep(wait_sec)
                continue

            response.raise_for_status()
            gids = list(dict.fromkeys(TABLE_GAME_REGEX.findall(response.text)))

            detected_pages, detected_total = None, None
            match = PAGINATION_INFO_REGEX.search(response.text)
            if match:
                detected_pages, detected_total = int(match.group(1)), int(match.group(2))

            if not gids:
                wait_sec = 8 * attempt
                print(f"[!] Стр. {page} вернула 0 партий (попытка {attempt}/6), пауза {wait_sec} сек...")
                await asyncio.sleep(wait_sec)
                continue

            return gids, detected_pages, detected_total
        except httpx.HTTPError as exc:
            wait_sec = 4 * attempt
            print(f"[!] Ошибка сети на стр. {page} (попытка {attempt}): {exc}")
            await asyncio.sleep(wait_sec)

    return [], None, None


async def download_pgn(
        client: httpx.AsyncClient,
        player_key: str,
        player_dir: Path,
        gid: str,
        semaphore: asyncio.Semaphore,
        index: int,
        total: int,
) -> tuple[int, str, str]:
    file_path = player_dir / f"{index:03d}_{player_key}_{gid}.pgn"
    if file_path.exists():
        cached = file_path.read_text(encoding="utf-8", errors="ignore").strip()
        if "[Event " in cached:
            return index, gid, cached

    url = f"{BASE_URL}/njs/api/game/downloadPGN/{gid}"
    async with semaphore:
        for attempt in range(1, 7):
            try:
                resp = await client.get(url)
                text = resp.text.strip()
                if resp.status_code == 200 and "[Event " in text:
                    file_path.write_text(text + "\n", encoding="utf-8")
                    if index % 25 == 0 or index == total:
                        print(f"[{index:03d}/{total}] Скачано (gid={gid})...")
                    await asyncio.sleep(0.35)
                    return index, gid, text
                await asyncio.sleep(15 * attempt)
            except httpx.RequestError:
                await asyncio.sleep(3 * attempt)

    return index, gid, ""


async def download_player_games(player_key: str) -> None:
    cfg = PLAYERS_CONFIG[player_key]
    pid = cfg["pid"]
    total_pages = cfg["pages"]
    player_dir = DATA_PGNS_DIR / f"{player_key}_pgns"
    player_dir.mkdir(parents=True, exist_ok=True)
    combined_file = DATA_PGNS_DIR / f"{player_key}_all_games.pgn"

    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/130.0.0.0 Safari/537.36",
        "Referer": f"{BASE_URL}/perl/chessplayer?pid={pid}",
    }
    semaphore = asyncio.Semaphore(3)

    async with httpx.AsyncClient(headers=headers, timeout=25.0, follow_redirects=True) as client:
        print(f"=== Скачивание базы: {cfg['name']} (pid={pid}) ===")
        all_gids: list[str] = []
        page = 1
        while page <= total_pages:
            gids, det_pages, det_total = await fetch_gids_from_page(client, pid, page)
            if det_pages:
                total_pages = det_pages
            for gid in gids:
                if gid not in all_gids:
                    all_gids.append(gid)
            total_info = f"/{det_total}" if det_total else ""
            print(
                f"[Страница {page:02d}/{total_pages:02d}] Найдено: {len(gids)} | Всего ID: {len(all_gids)}{total_info}")
            page += 1
            await asyncio.sleep(0.35)

        if not all_gids:
            print(f"[-] Партии для {player_key} не найдены.")
            return

        tasks = [
            download_pgn(client, player_key, player_dir, gid, semaphore, idx, len(all_gids))
            for idx, gid in enumerate(all_gids, start=1)
        ]
        results = await asyncio.gather(*tasks)
        results.sort(key=lambda x: x[0])
        valid_pgns = [pgn for _, _, pgn in results if pgn]

        combined_file.write_text("\n\n".join(valid_pgns) + "\n", encoding="utf-8")
        print(f"[+] Сохранено {len(valid_pgns)} партий в {combined_file}")

    build_book_for_bot(player_key)


async def main() -> None:
    parser = argparse.ArgumentParser(description="Единый загрузчик PGN исторических чемпионов с Chessgames.")
    parser.add_argument(
        "player",
        nargs="?",
        default="zukertort",
        choices=list(PLAYERS_CONFIG.keys()),
        help="Ключ шахматиста (по умолчанию: zukertort)",
    )
    parser.add_argument("--all", action="store_true", help="Скачать партии всех шахматистов")
    args = parser.parse_args()

    if args.all:
        for key in PLAYERS_CONFIG:
            await download_player_games(key)
    else:
        await download_player_games(args.player)


if __name__ == "__main__":
    asyncio.run(main())
